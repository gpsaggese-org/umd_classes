\begin{frame}{What is Knowledge Representation?}
\protect\phantomsection\label{what-is-knowledge-representation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Knowledge Representation (KR)}} is the study
  of how to formally encode information so that machines can reason with
  it

  \begin{itemize}
  \tightlist
  \item
    E.g., rules, logic, ontologies, semantic networks
  \item
    It is at the heart of symbolic AI and complements learning-based
    approaches
  \end{itemize}
\item
  Defines:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{structure}} (how knowledge is
    organized)
  \item
    \textbf{\textcolor{blue}{semantics}} (what it means)
  \end{itemize}
\item
  Serves as a bridge between perception (data) and reasoning (logic)

  \begin{itemize}
  \tightlist
  \item
    Essential for explainability and transparency in intelligent systems
  \end{itemize}
\item
  Enables machines to:

  \begin{itemize}
  \tightlist
  \item
    Draw conclusions
  \item
    Perform planning
  \item
    Answer queries
  \item
    \ldots{}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Expressiveness vs.~Tractability}
\protect\phantomsection\label{expressiveness-vs.-tractability}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Tradeoff in AI / ML}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Expressiveness}}: richness of concepts
    that can be captured
  \item
    \textbf{\textcolor{darkgreen}{Tractability}}: whether reasoning can
    be performed efficiently
  \item
    More expressive languages lead to harder computation
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  Choosing the right knowledge representation formalism depends on the
  application needs

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{teal}{Atomic}}

    \begin{itemize}
    \tightlist
    \item
      Treats each state as a single, indivisible entity
    \item
      E.g., depth-first search algorithms (e.g.,
      \textcolor{blue}{\texttt{E3}} in Chess)
    \item
      Simple and fast but limited in capturing complex relationships
    \end{itemize}
  \item
    \textbf{\textcolor{cyan}{Factored}}

    \begin{itemize}
    \tightlist
    \item
      E.g., propositional logic
    \item
      E.g., \(P_{1,1}\): \emph{``Pit in square (1,1)''},
      \(B_{1,2} \Leftrightarrow (P_{1,1} \lor P_{2,2} \lor P_{1,3})\)
    \item
      Captures relationships between variables but can't express complex
      structures
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Structured}}

    \begin{itemize}
    \tightlist
    \item
      E.g., first-order logic
    \item
      \(\forall x \, \forall y \, Father(x, y) \Rightarrow Parent(x, y)\)
      = \emph{``A father of a person is their parent''}
    \item
      More expressive but undecidable in general
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Symbolic vs.~Sub-symbolic Representation}

\begin{itemize}
\tightlist

  \textbf{\textcolor{red}{Symbolic knowledge}} representation uses

  \end{itemize}
\item
  \textbf{\textcolor{blue}{Neuro-symbolic approaches}} blends the two

\begin{frame}{Procedural vs Declarative Approaches}
\protect\phantomsection\label{procedural-vs-declarative-approaches}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Procedural approach}}

  \begin{itemize}
  \tightlist
  \item
    Focuses on \emph{how} a task is done
  \item
    Encodes desired behavior directly into the program
  \item
    E.g., a robot programmed with specific steps to navigate a maze
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Declarative approach}}

  \begin{itemize}
  \tightlist
  \item
    Specifies \emph{what} the goal is, \emph{not how} to achieve it
  \item
    Describes relationships between actions and goals
  \item
    Leaves solution search to the system
  \item
    E.g., describing the goal ``reach the exit'' and letting the system
    find the path
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Comparison}}

  \begin{itemize}
  \tightlist
  \item
    Procedural: more control, less flexibility
  \item
    Declarative: more abstraction, easier to modify or extend
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Integration of approaches}}

  \begin{itemize}
  \tightlist
  \item
    Many successful AI systems use a hybrid approach
  \item
    Declarative knowledge can be compiled into procedural code
  \item
    E.g., a planner generates procedures (plans) from declarative goals
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Programming Languages}
\protect\phantomsection\label{programming-languages}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{programming language}} (e.g., C++, Python)
  is a formal language

  \begin{itemize}
  \tightlist
  \item
    Data structures represent facts
  \item
    Code updates data structures in a domain-specific way
  \item
    Programming is procedural (vs declarative)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Limitations}}:

  \begin{itemize}
  \tightlist
  \item
    Programming languages lack:
  \end{itemize}

  \begin{enumerate}
  \tightlist
  \item
    A general mechanism for deriving facts from other facts

    \begin{itemize}
    \tightlist
    \item
      Code updates data structures based on programmer's domain
      knowledge
    \end{itemize}
  \item
    Expressiveness to handle partial information

    \begin{itemize}
    \tightlist
    \item
      A variable represents a single value or unknown
    \item
      Can't easily handle partial information or quantify uncertainty
    \item
      E.g., ``A white knight is in \textcolor{blue}{\texttt{b1}} or in
      \textcolor{blue}{\texttt{f6}}''
    \end{itemize}
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Declarative language}}

  \begin{itemize}
  \tightlist
  \item
    E.g., propositional logic, first order logic
  \item
    Knowledge and inference are separate:

    \begin{enumerate}
    \tightlist
    \item
      Knowledge represents the domain-specific problem
    \item
      Inference is domain independent
    \end{enumerate}
  \item
    Compositional semantics

    \begin{itemize}
    \tightlist
    \item
      The meaning of a sentence is a function of the meaning of its
      parts
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Propositional Logic}
\protect\phantomsection\label{propositional-logic}
\begin{itemize}
\tightlist
\item
  Uses atomic statements (propositions) and logical connectives

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Syntax}}

    \begin{itemize}
    \tightlist
    \item
      Atomic formulas: \(P\), \(Q\)
    \item
      Connectives: NOT (\(\lnot\)), AND (\(\land\)), OR (\(\lor\)),
      IMPLIES (\(\implies\))
    \end{itemize}
  \item
    \textbf{\textcolor{orange}{Semantics}}

    \begin{itemize}
    \tightlist
    \item
      Based on truth tables
    \item
      Each proposition has a binary truth value: true or false
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Inference mechanisms}}

    \begin{itemize}
    \tightlist
    \item
      Modus ponens: from \(P\) and \(P \implies Q\), infer \(Q\)
    \item
      Resolution: derive contradictions to infer conclusions
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Applications}}: best used in closed and
  well-defined environments

  \begin{itemize}
  \tightlist
  \item
    Digital circuit design
  \item
    Rule-based systems
  \item
    Simplified AI models
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Limitations}}

  \begin{itemize}
  \tightlist
  \item
    Cannot represent objects, relations, or quantifiers
  \item
    Not suitable for open or dynamic domains
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{First-Order Logic (FOL)}
\protect\phantomsection\label{first-order-logic-fol}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Extension of propositional logic}}

  \begin{itemize}
  \tightlist
  \item
    Introduces predicates, variables, and quantifiers

    \begin{itemize}
    \tightlist
    \item
      Variables \(x\)
    \item
      Predicate \(Human(x)\)
    \item
      Universal quantifier ``for all'' \(\forall\)
    \item
      Existential quantifier ``there exists'' \(\exists\)
    \end{itemize}
  \item
    E.g., \(\forall x (Human(x) \implies Mortal(x))\) = \emph{``All
    humans are mortal''}
  \item
    Represents more complex and structured knowledge than propositional
    logic
  \item
    Can model properties, relationships, and quantification over objects
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inference mechanisms}}

  \begin{itemize}
  \tightlist
  \item
    Unification: matches predicates with variables
  \item
    Resolution: deduces new facts from known statements
  \item
    Model checking: verifies truth of statements under specific
    interpretations
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Computational properties}}

  \begin{itemize}
  \tightlist
  \item
    Inference is semi-decidable: valid conclusions may require infinite
    time
  \item
    More powerful but computationally more complex than propositional
    logic
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Knowledge representation
  \item
    Automated theorem proving
  \item
    Semantic web and ontologies
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Reasoning and Inference in Logic}
\protect\phantomsection\label{reasoning-and-inference-in-logic}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Logical inference}} is the process of deriving
  new facts from known ones using formal rules

  \begin{itemize}
  \tightlist
  \item
    Used to make decisions and answer questions based on a Knowledge
    Base
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Knowledge base (KB)}}:

  \begin{itemize}
  \tightlist
  \item
    A structured set of facts and rules used for logical reasoning
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inference engine}}:

  \begin{itemize}
  \tightlist
  \item
    Mechanism that applies logical rules to a KB to derive conclusions
    or answer queries
  \item
    \textbf{\textcolor{teal}{Forward chaining}}:

    \begin{itemize}
    \tightlist
    \item
      Starts with known facts and applies inference rules to extract
      more data
    \item
      E.g., given \(A \rightarrow B\) and \(A\), infer \(B\)
    \end{itemize}
  \item
    \textbf{\textcolor{cyan}{Backward chaining}}:

    \begin{itemize}
    \tightlist
    \item
      Begins with a goal and works backward to find supporting facts
    \item
      E.g., to prove \(B\), check if \(A \rightarrow B\) and then prove
      \(A\)
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Resolution}}:

    \begin{itemize}
    \tightlist
    \item
      A complete inference rule for propositional and first-order logic
    \item
      Useful in automated theorem proving
    \end{itemize}
  \item
    \textbf{\textcolor{violet}{Entailment ($KB \models \alpha$)}}:

    \begin{itemize}
    \tightlist
    \item
      Sentence \(\alpha\) is entailed by KB if it is true in all models
      where KB is true
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Rule-Based Systems (1/2)}
\protect\phantomsection\label{rule-based-systems-12}
\begin{itemize}
\tightlist
\item
  A rule-based system uses ``if-then'' rules to derive conclusions or
  make decisions

  \begin{itemize}
  \tightlist
  \item
    It mimics human decision-making by applying logical rules to a set
    of facts
  \end{itemize}
\item
  \textbf{\textcolor{red}{Key components}}

  \begin{itemize}
  \tightlist
  \item
    Knowledge base: stores facts and rules
  \item
    Inference engine: applies rules to known facts to infer new facts or
    take actions
  \item
    Working memory: holds current facts being considered
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{How it works}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Match}: find rules whose conditions match current facts
  \item
    \emph{Conflict resolution}: decide which rule to apply if multiple
    rules match
  \item
    \emph{Act}: apply the chosen rule to modify facts or trigger actions
  \item
    \emph{Repeat}: continue until no more rules can be applied
  \item
    \textbf{\textcolor{blue}{E.g.,}}

    \begin{itemize}
    \tightlist
    \item
      \emph{Rule}: If a patient has a fever and a rash, then suggest
      measles
    \item
      \emph{Fact}: Patient has a fever and a rash
    \item
      \emph{Conclusion}: Suggest measles
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Reflex Agents}

  \textbf{\textcolor{red}{Reflex agents}} act based on the current

  \item
    Struggle with complex or partially observable environments
  \item
    Cannot plan ahead or learn from experience
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Simple or fully observable environments where quick reactions are
    sufficient

\begin{frame}{Logic / Knowledge Base (1/2)}
\protect\phantomsection\label{logic-knowledge-base-12}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Knowledge base (KB)}} is a set of:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Sentences}} \(\alpha\) expressing
    assertions (observed, assumed or derived) about the world

    \begin{itemize}
    \tightlist
    \item
      E.g., ``it rains'', ``the ground is dry'', ``the ground is wet''
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Rules}}

    \begin{itemize}
    \tightlist
    \item
      E.g, ``If it rains, the ground gets wet''
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Knowledge representation language}} is a
  formal way of creating sentences about the world
\item
  \textbf{\textcolor{cyan}{Syntax}} specifies all the sentences
  \(\alpha\) that are well-formed in a logic / knowledge base

  \begin{itemize}
  \tightlist
  \item
    E.g., in arithmetic the sentence:

    \begin{itemize}
    \tightlist
    \item
      ``\(x + y = 4\)'' is well-formed
    \item
      ``\(x4y+ =\)'' is not well-formed
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Semantics}} is the meaning of sentences
  (i.e., their truth) with respect to each possible world

  \begin{itemize}
  \tightlist
  \item
    E.g., the sentence \(x + y = 4\)

    \begin{itemize}
    \tightlist
    \item
      Is true in the world (model) in which \(x = 2, y = 2\)
    \item
      Is false in the world \(x = 1, y = 1\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Logic / Knowledge Base (2/2)}
\protect\phantomsection\label{logic-knowledge-base-22}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Axiom}} is a sentence taken as given

  \begin{itemize}
  \tightlist
  \item
    Not derived from other sentences
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Inference}} is the process of deriving new
  sentences from old ones

  \begin{itemize}
  \tightlist
  \item
    It should be done in a ``logical'' way
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Truth values}} of a sentence

  \begin{itemize}
  \tightlist
  \item
    In most logics every sentence is either true or false
  \item
    \textbf{\textcolor{teal}{Fuzzy logic}} allows sentences to have
    different degrees of truth

    \begin{itemize}
    \tightlist
    \item
      \(Belief(\alpha) = 0.5\)
    \end{itemize}
  \item
    \textbf{\textcolor{cyan}{Probabilistic logic}} allows sentences to
    have different probability of being true

    \begin{itemize}
    \tightlist
    \item
      \(\Pr(\alpha) = 0.3\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Satisfaction of a Sentence in a Model}
\protect\phantomsection\label{satisfaction-of-a-sentence-in-a-model}
\begin{itemize}
\tightlist
\item
  A model \(m\) fixes all the variables \(x_1, ..., x_n\) used in
  sentences

  \begin{itemize}
  \tightlist
  \item
    E.g., \((Rain = T, WetGround = T)\)
  \end{itemize}
\item
  If a sentence \(\alpha\) is true in model \(m\), we say \emph{``the
  model \(m\) \textbf{\textcolor{red}{satisfies}} the sentence
  \(\alpha\)''}

  \begin{itemize}
  \tightlist
  \item
    E.g., the model \((Rain = T, WetGround = F)\) satisfies
    \(\alpha: Rain = T\)
  \item
    Note: this seems backwards, since in our common way of reasoning,
    the world is fixed and sentences are evaluated as true or false
  \end{itemize}
\item
  \(M(\alpha)\) is the set of all the models in which \(\alpha\) is true

  \begin{itemize}
  \tightlist
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      \(\alpha: Rain = T\)
    \item
      \(M(Rain = T) = \{(Rain, WetGround), (Rain, \lnot WetGround)\}\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Logical Entailment}
\protect\phantomsection\label{logical-entailment}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Logical entailment}} between sentences is the
  fact that a sentence follows logically from another sentence in a KB
\item
  ``\(\alpha\) \textbf{\textcolor{darkgreen}{entails}} \(\beta\)''
  (written \(\alpha \models \beta\)) iff (by def) in every model in
  which \(\alpha\) is true, \(\beta\) is also true

  \begin{itemize}
  \tightlist
  \item
    Equivalent to \(M(\alpha) \subseteq M(\beta)\)
  \end{itemize}
\item
  E.g., in the ``rain and wet ground'' world

  \begin{itemize}
  \tightlist
  \item
    \(\alpha: "Rain \implies WetGround"\) entails
    \(\beta: "(Rain = T, WetGround = T)"\)
  \end{itemize}
\item
  E.g., in the ``sitting table'' world

  \begin{itemize}
  \tightlist
  \item
    \(\alpha\): ``\(x = 0\)'', \(\beta\): ``\(x \cdot y = 0\)''
  \item
    \(\alpha\) entails \(\beta\) since in any model in which \(x = 0\)
    is true, also \(x \cdot y = 0\) is true, regardless of the value of
    \(y\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Intuition}}:

  \begin{itemize}
  \tightlist
  \item
    Entailment is not related to a proof, it just ``preserves truth''
    across all models
  \item
    \emph{``If you believe your KB, you must believe the entailed
    sentences''}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Logical Entailment vs Implication}
\protect\phantomsection\label{logical-entailment-vs-implication}
\begin{itemize}
\tightlist
\item
  Entailment and implication are related but distinct

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Logical entailment}} is about truth
    following from known facts
  \item
    \textbf{\textcolor{orange}{Implication}} is about a relationship
    between two statements
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Logical entailment}}
  (\(KB \models \alpha\)):

  \begin{itemize}
  \tightlist
  \item
    Means \(\alpha\) is always true in any world where KB is true
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      KB: ``It is raining'', ``If it rains, the ground is wet''
    \item
      Entailed: ``The ground is wet''
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Implication}} (\(A \implies B\)):

  \begin{itemize}
  \tightlist
  \item
    A statement in logic that says: ``If A is true, then B is true''

    \begin{itemize}
    \tightlist
    \item
      Doesn't guarantee A or B is true by itself
    \item
      Implication is true unless A is true and B is false
    \end{itemize}
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      \(A\): ``It is raining'', \(B\): ``The ground is wet''
    \item
      \(A \implies B\) is the statement ``If it is raining, then the
      ground is wet''
    \item
      This statement can be true even if it's not raining
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Intuition}}:

  \begin{itemize}
  \tightlist
  \item
    Entailment is ``meta-level truth-following''
  \item
    Implication is ``within the logic''
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Model Checking Procedure}
\protect\phantomsection\label{model-checking-procedure}
\begin{itemize}
\item
  \(M(KB)\) represents all the models / possible worlds that are true
  given our \(KB\)
\item
  \textbf{\textcolor{red}{Problem}}:

  \begin{itemize}
  \tightlist
  \item
    We want to verify whether ``a sentence \(\alpha\) is entailed by
    \(KB\)'' (\(KB \models \alpha\))
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    According to the definition, we need to verify that \(\alpha\) is
    true in all the models in which \(KB\) is true

    \begin{itemize}
    \tightlist
    \item
      I.e., \(M(KB) \subseteq M(\alpha)\)
    \end{itemize}
  \end{itemize}
\item
  E.g., \textbf{\textcolor{blue}{model checking procedure}} (brute
  force)

  \begin{enumerate}
  \tightlist
  \item
    Enumerate all the models / possible worlds
  \item
    Find which models are possible given the \(KB\), i.e., \(M(KB)\)
  \item
    Check whether the sentence \(\alpha\) is true in all the models that
    are compatible with the \(KB\)
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Sound and Complete Inference Algorithm}
\protect\phantomsection\label{sound-and-complete-inference-algorithm}
\begin{itemize}
\item
  Inference: a syntactic process of deriving new sentences from others,
  using formal rules of a proof system (e.g., modus ponens, resolution,
  etc.)

  \begin{itemize}
  \tightlist
  \item
    You know: ``If it rains, the ground gets wet.''
  \item
    You see: ``It is raining.''
  \item
    You infer: ``The ground must be wet.''
  \end{itemize}
\item
  The ideal inference algorithm is both sound and complete
\item
  \textbf{\textcolor{red}{Sound}} inference algorithm

  \begin{itemize}
  \tightlist
  \item
    Derives only sentences entailed from \(KB\)
  \item
    ``Whatever the inference algorithm finds, it's correct'', i.e., no
    false positives
  \item
    E.g., model checking is sound

    \begin{itemize}
    \tightlist
    \item
      It works only when the space of models is finite
    \item
      When it works, it is truth preserving
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Complete}} inference algorithm

  \begin{itemize}
  \tightlist
  \item
    Can derive any sentence entailed from \(KB\)
  \item
    ``The inference algorithm doesn't miss anything,'' i.e., no false
    negatives
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Entailment vs Inference vs Implication}
\protect\phantomsection\label{entailment-vs-inference-vs-implication}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Logical entailment}}

  \begin{itemize}
  \tightlist
  \item
    \(A\) entails \(B\): if the fact \(A\) is true and that
    automatically guarantees that fact \(B\) must also be true
  \item
    E.g., \(Rain\) entails \(WetGround\) iff in every possible world
    where \(Rain\) is true, \(WetGround\) is also true

    \begin{itemize}
    \tightlist
    \item
      \(Rain = T, WetGround = F \to\) violation
    \item
      Since there is at least one counterexample, \(Rain\) does not
      entail \(WetGround\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inference}}

  \begin{itemize}
  \tightlist
  \item
    This is what you (a person or a computer) figure out based on what
    you know
  \item
    You start with some truths, then reason your way to new truths
  \item
    It's ``reasoning inside the logic system''
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Implication}}

  \begin{itemize}
  \tightlist
  \item
    ``If \(A\), then \(B\)''
  \item
    It doesn't say whether A is true; it just says, if it happens, then
    B follows
  \item
    It's a ``statement inside the logic system''
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Grounding}
\protect\phantomsection\label{grounding-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Grounding}} is the operation of linking
  abstract symbols to reality

  \begin{itemize}
  \tightlist
  \item
    E.g., words, variables in the representation \ldots{} \(\to\)
    objects, entities, or situations in the real world
  \item
    It is the bridge between representation in a KB and the world
  \end{itemize}
\item
  How \textbf{\textcolor{darkgreen}{can we know}} that a KB accurately
  reflects the real world?

  \begin{itemize}
  \tightlist
  \item
    We can't be sure!
  \item
    Do we live in a simulation? What is reality?
  \end{itemize}
\item
  We \textbf{\textcolor{blue}{assume}} that is correct

  \begin{itemize}
  \item
    Agent's sensors create a sentence in the KB when something happens
    in the real world

\begin{verbatim}
IF smell = burning THEN food_is_burning
\end{verbatim}
  \item
    Agent learns rules and acts

\begin{verbatim}
IF food_is_burning THEN turn_off_stove
\end{verbatim}
  \end{itemize}
\item
  We assume that ``learning'' (going from particular cases to general
  cases) is typically correct

  \begin{itemize}
  \tightlist
  \item
    Learning is still fallible
  \item
    E.g., \textcolor{blue}{\texttt{smell = burning}} because maybe
    somebody is cooking on a grill
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ontologies (in computer science)}
\protect\phantomsection\label{ontologies-in-computer-science}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Ontology:}}

  \begin{itemize}
  \tightlist
  \item
    Is a formal, explicit representation of a domain
  \item
    Describes the types of things that exist and how they relate to each
    other

    \begin{itemize}
    \tightlist
    \item
      Classes: types of things
    \item
      Individuals: specific objects
    \item
      Properties: how things are related
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Examples}}:

  \begin{itemize}
  \tightlist
  \item
    A medical ontology defines relationships between diseases, symptoms,
    and treatments
  \item
    A geographical ontology describes cities, states, and countries
  \item
    Semantic web (an extension of the current web to give meaning to
    information)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Goal}}

  \begin{itemize}
  \tightlist
  \item
    Provide a vocabulary for a domain of knowledge
  \item
    Enable machines and humans to understand and share information
    consistently
  \item
    Enable reasoning about entities and their relationships
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Related Concepts}}

  \begin{itemize}
  \tightlist
  \item
    Schema: database-oriented structure, often more rigid than
    ontologies
  \item
    Taxonomy: simpler hierarchical tree-like classification
  \item
    Knowledge base: a collection of facts and rules, sometimes built
    from an ontology
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ontologies: Components}
\protect\phantomsection\label{ontologies-components}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Classes / Concepts}}:

  \begin{itemize}
  \tightlist
  \item
    Represent general concepts in a domain
  \item
    E.g., \textcolor{blue}{\texttt{Person}},
    \textcolor{blue}{\texttt{City}}, \textcolor{blue}{\texttt{Car}}
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Individuals / Instances}}:

  \begin{itemize}
  \tightlist
  \item
    Specific, concrete examples of classes
  \item
    E.g., \textcolor{blue}{\texttt{GP}} (an instance of
    \textcolor{blue}{\texttt{Person}}), \textcolor{blue}{\texttt{Rome}},
    \textcolor{blue}{\texttt{Ferrari 458}}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Properties / Relations}}:

  \begin{itemize}
  \tightlist
  \item
    Describe interactions or associations between classes or instances
  \item
    E.g., \textcolor{blue}{\texttt{isMortal}},
    \textcolor{blue}{\texttt{locatedIn}},
    \textcolor{blue}{\texttt{hasAge}}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Attributes / Data values}}

  \begin{itemize}
  \tightlist
  \item
    Specify data associated with instances
  \item
    E.g., (\textcolor{blue}{\texttt{GP}},
    \textcolor{blue}{\texttt{hasAge}},
    \textcolor{blue}{\texttt{<your\_guess>}})
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Constraints}}

  \begin{itemize}
  \tightlist
  \item
    Rules that restrict the kinds of values a property can take
  \item
    E.g., (\textcolor{blue}{\texttt{Ferrari 458}},
    \textcolor{blue}{\texttt{mustBe}}, \textcolor{blue}{\texttt{red}})
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Axioms}}:

  \begin{itemize}
  \tightlist
  \item
    Logical statements that define rules and constraints
  \item
    E.g., all humans are mortal:
    \(\forall x (Person(x) \implies Mortal(x))\)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Hierarchies}}:

  \begin{itemize}
  \tightlist
  \item
    Organize classes and properties into parent-child relationships
  \item
    E.g., \textcolor{blue}{\texttt{Student}} is a subclass of
    \textcolor{blue}{\texttt{Person}}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Example of Reasoning Tasks (1/4)}
\protect\phantomsection\label{example-of-reasoning-tasks-14}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Subsumption}}

  \begin{itemize}
  \tightlist
  \item
    ``Is class \textcolor{blue}{\texttt{A}} a subclass of
    \textcolor{blue}{\texttt{B}}?''
  \item
    Check whether one concept is more general than another
  \item
    E.g., if \textcolor{blue}{\texttt{Person}} subsumes
    \textcolor{blue}{\texttt{Student}}, every
    \textcolor{blue}{\texttt{Student}} is necessarily a
    \textcolor{blue}{\texttt{Person}}
  \item
    Important for building taxonomies and ontologies
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Satisfiability}}

  \begin{itemize}
  \tightlist
  \item
    ``Can an instance of a concept exist?''
  \item
    Test if a concept is logically consistent (i.e., without
    contradiction)
  \item

    requires flying but is also defined as a penguin (which cannot fly),
    it might be not satisfiable
  \end{itemize}

  \begin{itemize}
  \tightlist

  \item

    subsumption relationships
  \item
    E.g., given definitions of \textcolor{blue}{\texttt{Animal}},
    \textcolor{blue}{\texttt{Bird}}, and
    \textcolor{blue}{\texttt{Penguin}}, classification places
    \textcolor{blue}{\texttt{Penguin}} under
    \textcolor{blue}{\texttt{Bird}}, and \textcolor{blue}{\texttt{Bird}}
    under \textcolor{blue}{\texttt{Animal}}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Propositional Logic}
\protect\phantomsection\label{propositional-logic-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Propositional logic}} is a formal system for
  reasoning about statements that can be true or false

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Syntax}} defines the allowable sentences

    \begin{itemize}
    \tightlist
    \item
      Consists of proposition symbol and logical connectives
    \item
      E.g., \(P \land Q\)
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Semantics}} is the way in which the
    truth of sentences is determined

    \begin{itemize}
    \tightlist
    \item
      Truth tables or deduction rules evaluate the truth value of
      complex sentences
    \item
      E.g., if \(P\) is true and \(Q\) is false then \(P \land Q\) is
      false
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Atomic representation}}

  \begin{itemize}
  \tightlist
  \item
    No internal structure within atomic propositions
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Uses}}

  \begin{itemize}
  \tightlist
  \item
    SAT solvers

    \begin{itemize}
    \tightlist
    \item
      Tools for determining if a propositional logic formula can be
      satisfied
    \item
      E.g., used in hardware verification and scheduling problems
    \end{itemize}
  \item
    Expert systems

    \begin{itemize}
    \tightlist
    \item
      Use logic rules to mimic human decision-making
    \item
      E.g., medical diagnosis systems
    \end{itemize}
  \item
    Rule-based agents

    \begin{itemize}
    \tightlist
    \item
      Agents that operate based on a set of predefined rules
    \item
      E.g., automated customer service chatbots
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Proposition Symbol}
\protect\phantomsection\label{proposition-symbol}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Proposition symbol}}

  \begin{itemize}
  \tightlist
  \item
    Is an atomic sentence consisting of a single symbol

    \begin{itemize}
    \tightlist
    \item
      E.g., \(P\), \(Q\), \(North\)
    \end{itemize}
  \item
    Doesn't have truth value, it is just a symbol for a real-world
    statement

    \begin{itemize}
    \tightlist
    \item
      Needs grounding
    \end{itemize}
  \item
    Stands for a proposition that can be true or false

    \begin{itemize}
    \tightlist
    \item
      E.g., \(K_{E,5}\) = ``the Knight is in E5''
    \item
      \(K_{E,5}\) is not composed of any other symbol, it is an atomic
      symbol
    \end{itemize}
  \end{itemize}
\item
  \(True\) and \(False\) are proposition symbols with inherent truth
  values
\end{itemize}
\end{frame}

\begin{frame}{Sentences}
\protect\phantomsection\label{sentences}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Atomic sentence}}

  \begin{itemize}
  \tightlist
  \item
    Is a sentence composed of a single proposition symbol
  \item
    E.g., \(P\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Complex sentence}}

  \begin{itemize}
  \tightlist
  \item
    Is constructed from simpler (sentences) using parentheses and
    logical connectives
  \item
    E.g., \((P \land Q) \lor R\)
  \item
    It is a recursive definition that allow to build more complex
    sentences
  \end{itemize}
\item
  Common logical connectives

  \begin{itemize}
  \tightlist
  \item
    Not: \(\lnot\)
  \item
    And: \(\land\) (looks like an ``A'' for ``and'')
  \item
    Or: \(\lor\) (comes from Latin ``vel'' which means ``or'')
  \item
    Implies: \(\implies\)
  \item
    If and only if: \(\iff\)
  \end{itemize}
\item
  Each sentence (atomic or complex) can be only true or false
\end{itemize}
\end{frame}

\begin{frame}{Proposition Logic: Weather Example}
\protect\phantomsection\label{proposition-logic-weather-example}
\begin{itemize}
\tightlist
\item
  Proposition symbols are:

  \begin{itemize}
  \tightlist
  \item
    \(Rain\) = ``it's raining''
  \item
    \(Cold\) = ``it's cold''
  \item
    \(Sunny\) = ``it's sunny''
  \item
    \(Snow\) = ``it's snowing''
  \item
    \(Cloudy\) = ``it's cloudy''
  \end{itemize}
\item
  \textbf{\textcolor{red}{Atomic sentence}} can be positive or negated

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Rain\), \(\lnot Rain\) = ``it's not raining''
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Negation}}

  \begin{itemize}
  \tightlist
  \item
    E.g., \(\lnot (Rain \lor Cloudy)\) = ``it's not the case that it's
    raining or cloudy''
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Conjunction / Disjunction}}

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Rain \land Cold\) = ``it's raining and it's cold''
  \item
    E.g., \(Rain \lor Snow\) = ``it's either raining or snowing''
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Implication}} is a sentence containing a
  premise \(\implies\) conclusion

  \begin{itemize}
  \tightlist
  \item
    Aka ``if-then statements'', ``rules''
  \item
    E.g., \(Rain \implies \lnot Snow\) = ``if it's raining, it's not
    snowing''
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Biconditional}}:
  \(A \implies B \land B \implies A\)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Sunny \iff \lnot Cloudy\) = ``it's sunny if and only if it's
    not cloudy''
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Semantics of Propositional Logic}
\protect\phantomsection\label{semantics-of-propositional-logic}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Semantics}} are rules for determining the
  truth of a sentence \(\alpha\) with respect to a model \(m\)

  \begin{itemize}
  \tightlist
  \item
    Determine if a sentence is true or false, given a possible world
  \end{itemize}
\item
  In propositional logic, a model \(m\) fixes the truth value (true or
  false) for every proposition symbol/atomic sentence, e.g.,
\item
  The models are \textbf{\textcolor{blue}{abstractions}} of the real
  world and have no a-priori connection to a specific world, e.g.,

  \begin{itemize}
  \tightlist
  \item
    \(P_{1,2}\) is just a symbol and can mean:

    \begin{itemize}
    \tightlist
    \item
      ``There is a pit in {[}1, 2{]}'' or
    \item
      ``I'm in Paris today and tomorrow''
    \end{itemize}
  \item
    Need grounding
  \end{itemize}
\end{itemize}

\begin{frame}[fragile]{Computing the Truth Value of a Sentence}
\protect\phantomsection\label{computing-the-truth-value-of-a-sentence}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{truth value of a sentence}} is derived
  from the truth of the proposition symbols (recursively from the model
  \(m\)) \[
  m = \{ P_{1,2}=F, P_{2,2}=F, P_{3,1}=T \}
  \]
\item
  All sentences \(\alpha\) are constructed from atomic sentences
  (assigned by model \(m\)) and connectives:

  \begin{itemize}
  \tightlist
  \item
    \(\lnot P\) is T iff \(P\) is F in \(m\)
  \item
    \(P \land Q\) is T iff \(P\) and \(Q\) are both true in \(m\)
  \item
    \(P \lor Q\) is T iff \(P\) or \(Q\) are true in \(m\)
  \item
    \(P \implies Q\) is true unless \(P\) is true and \(Q\) is false in
    \(m\)
  \item
    \(P \iff Q\) is true iff \(P\) and \(Q\) are both true or both false
    in \(m\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Truth table}} contains truth value of a
  sentence for each possible model \(m\)

  \begin{itemize}
  \item
    E.g., \(X = A \land B \lor C\)

\begin{verbatim}
A   B   C   X
F   F   F   F
F   F   F   T
...
\end{verbatim}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of Implication}
\protect\phantomsection\label{interpretation-of-implication}
\begin{itemize}
\item
  In a \textbf{\textcolor{red}{logical implication}}
  ``\(P \implies Q\)'' there is
  \textbf{\textcolor{darkgreen}{no causation}} between \(P\) and \(Q\)
\item
  ``\(P \implies Q\)'' says \emph{``If \(P\) is true, I claim that \(Q\)
  is true. Otherwise I am making no claim''}

  \begin{itemize}
  \tightlist
  \item
    E.g., \emph{``5 is odd implies that Tokyo is the capital of Japan''}
    is a true sentence in propositional logic (although very odd)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pathological cases}} for implication

  \begin{itemize}
  \tightlist
  \item
    An implication is true whenever the antecedent is false
  \item
    E.g., \emph{``5 is even implies pigs fly''} is true
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Model Checking is Sound and Complete}
\protect\phantomsection\label{model-checking-is-sound-and-complete}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Model checking algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Enumerate all the models (truth tables)
  \item
    Check if \(\alpha\) is true for every model where \(KB\) is true
  \end{itemize}
\item
  The model checking algorithm is:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Sound}}

    \begin{itemize}
    \tightlist
    \item
      ``Any inference made by the algorithm is correct''
    \item
      Implements the definition of entailment
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Complete}}

    \begin{itemize}
    \tightlist
    \item
      ``Any true sentence is inferred correctly by the algorithm''
    \item
      Works for any \(KB\) and \(\alpha\)
    \item
      Always terminates (finite number of models)
    \end{itemize}
  \end{itemize}
\item
  Complexity of model checking with \(n\) variables

  \begin{itemize}
  \tightlist
  \item
    Time complexity is \(O(2^n)\) (NP-complete)

    \begin{itemize}
    \tightlist
    \item
      Worst case is exponential
    \item
      Average case is better than exponential
    \end{itemize}
  \item
    Space complexity is \(O(n)\) since enumeration is depth-first
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Inference in Propositional Logic}
\protect\phantomsection\label{inference-in-propositional-logic}
\begin{itemize}
\tightlist
\item
  Inference are the rules of reasoning

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Modus Ponens}}: if \(p \implies q\) and
    \(p\), infer \(q\)

    \begin{itemize}
    \tightlist
    \item
      If it rains, the ground will be wet. It rains.
    \item
      Therefore, the ground is wet.
    \end{itemize}
  \item
    \textbf{\textcolor{orange}{Modus Tollens}}: if \(p \implies q\) and
    \(\lnot q\), infer \(\lnot p\)

    \begin{itemize}
    \tightlist
    \item
      If it rains, the ground will be wet. The ground is not wet.
    \item
      Therefore, it did not rain.
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Syllogism (Transitivity)}}

    \begin{itemize}
    \tightlist
    \item
      If \(p \implies q\) and \(q \implies r\), then \(p \implies r\)
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Disjunctive Syllogism}}

    \begin{itemize}
    \tightlist
    \item
      If \(p \lor q\) and \(\lnot p\), infer \(q\)
    \end{itemize}
  \item
    \textbf{\textcolor{cyan}{Addition}}: if \(p\), then \(p \lor q\)
  \item
    \textbf{\textcolor{blue}{Simplification}}: from \(p \land q\), infer
    \(p\) (or \(q\))
  \item
    \textbf{\textcolor{violet}{Conjunction}}: from \(p\) and k\(q\),
    infer \(p \land q\)
  \item
    \textbf{\textcolor{magenta}{Resolution Rule}}

    \begin{itemize}
    \tightlist
    \item
      From \((p \lor q)\) and \((\lnot p \lor r)\), infer \((q \lor r)\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Propositional Theorem Proving}
\protect\phantomsection\label{propositional-theorem-proving}
\begin{itemize}
\tightlist
\item
  To \textbf{\textcolor{red}{prove}} a desired sentence \(\alpha\) under
  a knowledge base KB

  \begin{itemize}
  \tightlist
  \item
    Apply rules of inference to construct a proof of \(\alpha\)
  \item
    Any sentence can have only one of the following truth values:

    \begin{enumerate}
    \tightlist
    \item
      True
    \item
      False
    \item
      Undecidable under the KB
    \end{enumerate}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Theorem proving vs model checking}}

  \begin{itemize}
  \tightlist
  \item
    Model checking involves enumerating all models to show the sentence
    is true/false in all models where KB is true
  \item
    Propositional theorem proving builds a proof
  \item
    If the proof is short, theorem proving can be more efficient than
    model checking
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Logical Equivalence of Sentences}
\protect\phantomsection\label{logical-equivalence-of-sentences}
\begin{itemize}
\tightlist
\item
  Two sentences \(\alpha\) and \(\beta\) are logically equivalent
  \(\alpha \equiv \beta\)

  \begin{itemize}
  \tightlist
  \item
    Iff they are true in the same set of models: \[
    M(\alpha) = M(\beta)
    \]
  \item
    Iff they entail each other: \[
    \alpha \models \beta \land \beta \models \alpha
    \]
  \end{itemize}
\item
  E.g., \(P \lor Q \equiv Q \lor P\)
\end{itemize}
\end{frame}

\begin{frame}{Logical Equivalences (1/2)}
\protect\phantomsection\label{logical-equivalences-12}
\begin{itemize}
\tightlist
\item
  Commutativity of \(\land\) and \(\lor\) \[
  (\alpha \land \beta) \equiv (\beta \land \alpha)
  \] \[
  (\alpha \lor \beta) \equiv (\beta \lor \alpha)
  \]
\item
  Associativity of \(\land\) and \(\lor\) \[
  (\alpha \land \beta) \land \gamma
  \equiv \alpha \land (\beta \land \gamma)
  \equiv \alpha \land \beta \land \gamma
  \] \[
  (\alpha \lor \beta) \lor \gamma
  \equiv \alpha \lor (\beta \lor \gamma)
  \equiv \alpha \lor \beta \lor \gamma
  \]
\item
  Distributivity of \(\land\) over \(\lor\) \[
  \alpha \land (\beta \lor \gamma) \equiv
    (\alpha \land \beta) \lor (\alpha \land \gamma)
  \]
\item
  Distributivity of \(\lor\) over \(\land\) \[
  \alpha \lor (\beta \land \gamma) \equiv
    (\alpha \lor \beta) \land (\alpha \lor \gamma)
  \]
\item
  Double negation elimination: \[
  \lnot (\lnot \alpha) \equiv \alpha
  \]
\end{itemize}
\end{frame}

\begin{frame}{Logical Equivalences (2/2)}
\protect\phantomsection\label{logical-equivalences-22}
\begin{itemize}
\tightlist
\item
  Contraposition: \[
  (\alpha \implies \beta) \equiv (\lnot \beta \implies \lnot \alpha)
  \]
\item
  Implication elimination: \[
  (\alpha \implies \beta) \equiv (\lnot \alpha \lor \beta)
  \]
\item
  Biconditional elimination: \[
  (\alpha \iff \beta)
  \equiv (\alpha \implies \beta) \land (\beta \implies \alpha)
  \]
\item
  De Morgan equivalence: \[
  \lnot (\alpha \land \beta) \equiv (\lnot \alpha \lor \lnot \beta)
  \] \[
  \lnot (\alpha \lor \beta) \equiv (\lnot \alpha \land \lnot \beta)
  \]
\end{itemize}
\end{frame}

\begin{frame}{Deduction theorem}
\protect\phantomsection\label{deduction-theorem}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{valid sentence}} \(\alpha\) is true for all
  the models

  \begin{itemize}
  \tightlist
  \item
    E.g., \(P \lor \lnot P\)
  \item
    Aka ``tautology''
  \item
    Every tautology is equivalent to the sentence \(True\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Contradiction}} is a sentence \(\alpha\)
  that is false for all the models

  \begin{itemize}
  \tightlist
  \item
    E.g., \(P \land \lnot P\)
  \item
    Every contradiction is equivalent to the sentence \(False\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Deduction theorem}}

  \begin{itemize}
  \tightlist
  \item
    The sentence \(\alpha\) entails \(\beta\) (written
    \(\alpha \models \beta\)) iff the sentence \(\alpha \implies \beta\)
    is a tautology
  \end{itemize}
\item
  The deduction theorem is like a \textbf{\textcolor{violet}{bridge}}
  between two different but closely related ideas:

  \begin{itemize}
  \tightlist
  \item
    Entailment \(\models\) is a semantic notion

    \begin{itemize}
    \tightlist
    \item
      It's about truth across all models
    \item
      \(\alpha entails \beta\) means in every possible world where
      \(\alpha\) is true, \(\beta\) is also true
    \end{itemize}
  \item
    Implication \(\implies\) is a syntactic notion

    \begin{itemize}
    \tightlist
    \item
      \(\alpha \implies \beta\) is just another formula inside the logic
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Satisfiability}
\protect\phantomsection\label{satisfiability}
\begin{itemize}
\item
  A sentence \(\alpha\) is \textbf{\textcolor{red}{satisfiable}} iff
  \(\alpha\) is true for some model
\item
  \textbf{\textcolor{darkgreen}{SAT problem}} is about determining
  satisfiability of sentence in propositional logic

  \begin{itemize}
  \tightlist
  \item
    One can enumerate all the possible models until one is found to
    satisfy the sentence \(\alpha\)
  \item
    It is NP-complete
  \end{itemize}
\item
  A sentence \(\alpha\) is \textbf{\textcolor{blue}{un-satisfiable}} iff
  \(\alpha\) is never true (i.e., a contradiction)
\item
  \textbf{\textcolor{violet}{Validity and satisfiability}}

  \begin{itemize}
  \tightlist
  \item
    \(\alpha\) is valid (i.e., a tautology) iff \(\lnot \alpha\) is
    un-satisfiable
  \item
    By contrapositive, \(\alpha\) is satisfiable iff \(\lnot \alpha\) is
    not valid (\(\lnot \alpha\) is not a tautology)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Proof by Contraction}
\protect\phantomsection\label{proof-by-contraction}
\begin{itemize}
\item
  The sentence \(\alpha \models \beta\) is true iff the sentence
  ``\(\alpha \lor \lnot \beta)\)'' is un-satisfiable (i.e., a
  contradiction)
\item
  In other words in a proof by contradiction:

  \begin{itemize}
  \tightlist
  \item
    Assume \(\alpha\)
  \item
    Assume that the sentence \(\beta\) is false
  \item
    Prove that this leads to a contradiction
  \item
    Thus \(\beta\) must be true
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Propositional Logic}
\protect\phantomsection\label{propositional-logic-2}
\begin{itemize}
\tightlist
\item
  E.g., \(P \land Q\)
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Declarative

    \begin{itemize}
    \tightlist
    \item
      Semantics is based on relation between sentences and possible
      worlds
    \end{itemize}
  \item
    Can deal with partial information

    \begin{itemize}
    \tightlist
    \item
      E.g., ``A white knight is in \textcolor{blue}{\texttt{b1}} or in
      \textcolor{blue}{\texttt{f6}}'' is represented with
      \(WK1_{b1} \lor WK2_{f6}\)
    \end{itemize}
  \item
    Compositional semantics

    \begin{itemize}
    \tightlist
    \item
      The meaning of a sentence is a function of the meaning of its
      parts
    \end{itemize}
  \item
    Context independent
  \item
    Unambiguous
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Can't concisely describe environment with many objects, e.g.,

    \begin{itemize}
    \tightlist
    \item
      In English \emph{``The pawn is in a cell around
      \textcolor{blue}{\texttt{b1}}''} requires all the possible states
      to be enumerated
    \end{itemize}
  \item
    Can't represent uncertainty

    \begin{itemize}
    \tightlist
    \item
      \emph{``There is 50\% of probability that the pawn is in
      \textcolor{blue}{\texttt{b1}}''}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{First-Order Logic (FOL)}
\protect\phantomsection\label{first-order-logic-fol-1}
\begin{itemize}
\tightlist
\item
  First-order logic (FOL) extends propositional logic by:

  \begin{itemize}
  \tightlist
  \item
    Introducing \textbf{\textcolor{red}{quantifiers}} (e.g.,
    \(\forall\), \(\exists\))
  \item
    Using \textbf{\textcolor{blue}{predicates}} to represent properties
    and relations
  \end{itemize}
\item
  Combines pros of propositional logic with pros of natural language

  \begin{itemize}
  \tightlist
  \item
    Built around objects and relations
  \item
    Allows to express facts about some or all objects, e.g.,

    \begin{itemize}
    \tightlist
    \item
      \emph{``Some humans have green eyes''}
    \item
      \emph{``Chess pieces around the Queen are at risk''}
    \end{itemize}
  \end{itemize}
\item
  FOL provides expressive power to represent structured, relational
  knowledge
\end{itemize}
\end{frame}

\begin{frame}{First-Order Logic: Syntax}
\protect\phantomsection\label{first-order-logic-syntax}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Constants}}: represent specific objects

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Socrates\)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Predicates}}: describe properties or
  relations

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Human(x)\) = \emph{``x is human''}

\item
  \textbf{\textcolor{darkgreen}{Functions}}: map tuples of objects to

  \tightlist
  \item
    E.g., \(x, y\)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Quantifiers}}

  \begin{itemize}
  \tightlist
  \item
    E.g., \(\forall x\) (for all \(x\)), \(\exists x\) (there exists an
    \(x\))
  \end{itemize}
\end{itemize}

\begin{frame}{Sentences}
\protect\phantomsection\label{sentences-1}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Term}} is a logical expression that refers to
  an object

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Richard\)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Atomic sentence}} is a predicate symbol
  (i.e., corresponds to relations) followed by a list of terms in
  parenthesis \(Predicate(Term1, Term2, ...)\)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Brother(Richard, John)\) means \emph{``Richard is the
    brother of John''}, under a given grounding / interpretation,\\
  \item
    E.g., \(Married(Father(Richard), Mother(John))\) means \emph{``the
    father of Richard and the mother of John are married''}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Complex sentence}} is a sentence using
  logical connectives complex, with the same syntax and semantics as in
  propositional logic
\item
  \textbf{\textcolor{teal}{Variable}} is a term that represents a
  possible object

  \begin{itemize}
  \tightlist
  \item
    Typically represented as lowercase letter (e.g., \(x\), \(y\),
    \(z\))
  \item
    Can be used as argument of a function, e.g., \(LeftLeg(x)\)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Equality symbol}} signifies that two terms
  refer to the same object

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Father(John) = Henry\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Quantifiers}
\protect\phantomsection\label{quantifiers}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Quantifiers}} express properties of entire
  collections of objects, instead of enumerating objects by name (like
  in propositional logic)
\item
  \textbf{\textcolor{orange}{Universal quantifier}}:
  \(\forall x\, P(x)\)

  \begin{itemize}
  \tightlist
  \item
    Universal quantifier makes a statement about \emph{every} object
  \item
    Statement is true if \(P(x)\) is true for all \(x\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Existential quantifier}}:
  \(\exists x\, P(x)\)

  \begin{itemize}
  \tightlist
  \item
    Existential quantifier makes a statement about \emph{some} object
    (without naming it)
  \item
    Statement is true if \(P(x)\) is true for at least one \(x\)
  \end{itemize}
\item
  Variables are \textbf{\textcolor{teal}{bound}} by quantifiers or
  \textbf{\textcolor{cyan}{free}} (unbound)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(\forall x (Cat(x) \rightarrow Mammal(x))\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Nested Quantifiers}
\protect\phantomsection\label{nested-quantifiers}
\begin{itemize}
\tightlist
\item
  More complex sentences can be expressed using
  \textbf{\textcolor{red}{multiple quantifiers}}

  \begin{itemize}
  \tightlist
  \item
    The order of quantifiers is important, so one can use parentheses to
    clarify
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{``Brothers are siblings''}:
    \(\forall x, y \; Brother(x, y) \implies Sibling(x, y)\)
  \item
    Siblings is a symmetric relationship:
    \(\forall x, y \; Sibling(x, y) \iff Sibling(y, x)\)
  \item
    \emph{``Everybody loves somebody''}:
    \(\forall x \; \exists y \; Loves(x, y)\)
  \item
    \emph{``There is someone loved by everyone''}:
    \(\exists y \; \forall x \; Loves(x, y)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Connection between \(\forall\) and \(\exists\)}
\protect\phantomsection\label{connection-between-forall-and-exists}
\begin{itemize}
\tightlist
\item
  The universal and existential quantifiers are
  \textbf{\textcolor{red}{connected}} through negation and De Morgan
  rules \[
  \forall x \; \lnot P(x) \iff \lnot \exists x: P(x)
  \] \[
  \lnot (\forall x \; P(x)) \iff \exists x \lnot P(x)
  \] \[
  \forall x \; P(x) \iff \lnot \exists x \lnot P(x)
  \] \[
  \exists x \; P(x) \iff \lnot (\forall x \; \lnot P(x))
  \]
\end{itemize}
\end{frame}

\begin{frame}{First-order Logic: Semantics}
\protect\phantomsection\label{first-order-logic-semantics}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Semantics}} define how sentences are
  interpreted in a domain
\item
  Symbols represent entities, relationships, and functions in the domain

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Constant symbols}} represent specific
    objects

    \begin{itemize}
    \tightlist
    \item
      E.g., \(Alice\), \(GP\), \(CS101\)
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Predicate symbols}} represent
    relationships among objects

    \begin{itemize}
    \tightlist
    \item
      E.g., \(EnrolledIn(Student, Class)\),
      \(Teaches(Professor, Class)\), \(IsStudent(x)\),
      \(IsProfessor(x)\)
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Function symbols}} represent mappings
    between objects

    \begin{itemize}
    \tightlist
    \item
      E.g., \(AdvisorOf(Student)\), \(DepartmentOf(Professor)\)
    \end{itemize}
  \end{itemize}
\item
  An \textbf{\textcolor{cyan}{interpretation}} maps the world to its
  mathematical description, and vice versa

  \begin{itemize}
  \tightlist
  \item
    There are many possible interpretations
  \item
    The intended interpretation is the one that is the most natural
  \item
    E.g., map the symbol \(GP\) \(\to\) me
  \item
    \(\forall x \, (Human(x) \rightarrow Mortal(x))\) means that every
    human is mortal
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Representing Knowledge in FOL}
\protect\phantomsection\label{representing-knowledge-in-fol}
\begin{itemize}
\item
  FOL enables representation of:

  \begin{itemize}
  \tightlist
  \item
    General rules: \(\forall x\, (Bird(x) \rightarrow CanFly(x))\)
  \item
    Specific facts: \(Bird(Tweety)\)
  \end{itemize}
\item
  Complex relations captured through predicates:

  \begin{itemize}
  \tightlist
  \item
    \(Loves(Romeo, Juliet)\), \(GreaterThan(3, 2)\)
  \end{itemize}
\item
  Functions express object construction:

  \begin{itemize}
  \tightlist
  \item
    \(FatherOf(John)\)
  \end{itemize}
\item
  Knowledge base is built from axioms and facts
\item
  Enables reasoning about objects, properties, and their relationships
\end{itemize}
\end{frame}

\begin{frame}{First-Order Logic: Inference}
\protect\phantomsection\label{first-order-logic-inference}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: derive new sentences from existing
  ones using sound rules

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Universal Instantiation}}:

    \begin{itemize}
    \tightlist
    \item
      From \(\forall x\, P(x)\) infer \(P(c)\) for any constant \(c\)
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Existential Instantiation}}:

    \begin{itemize}
    \tightlist
    \item
      From \(\exists x\, P(x)\) infer \(P(c)\) with a new constant \(c\)
    \end{itemize}
  \item
    Standard propositional logic rules such as Modus Ponens, Modus
    Tollens, Resolution, etc., also apply within FOL

    \begin{itemize}
    \tightlist
    \item
      Modus ponens: from \(P\) and \(P \implies Q\), infer \(Q\)
    \end{itemize}
  \end{itemize}
\item
  FOL inference is semi-decidable:

  \begin{itemize}
  \tightlist
  \item
    If a sentence is entailed, a proof can be found
  \item
    If not entailed, proof search may not terminate
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ontological Commitment}
\protect\phantomsection\label{ontological-commitment}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Ontological commitments}} are assumptions
  about reality made by a language
\item
  Different formal models make
  \textbf{\textcolor{blue}{different assumptions}} on how the truth of
  sentences is defined

  \begin{itemize}
  \tightlist
  \item
    Propositional logic:

    \begin{itemize}
    \tightlist
    \item
      The world consists of facts that are either true or false
    \item
      E.g., \(P \lor Q\)
    \end{itemize}
  \item
    First-order logic:

    \begin{itemize}
    \tightlist
    \item
      The world consists of objects with relations among them that hold
      or do not hold
    \item
      \(\forall x: Human(x) \implies Mortal(x)\)
    \end{itemize}
  \item
    Temporal logic:

    \begin{itemize}
    \tightlist
    \item
      Facts about objects and relations hold at particular times or
      intervals
    \end{itemize}
  \item
    Higher-order logic:

    \begin{itemize}
    \tightlist
    \item
      Relations of first-order logic are objects themselves
    \item
      E.g., can make assertions about relations (e.g., ``all relations
      are transitive'')
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Epistemological Commitment}
\protect\phantomsection\label{epistemological-commitment}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Epistemological commitment}} is a possible
  states of knowledge by an agent with respect to each fact (i.e.,
  belief), e.g.,

  \begin{itemize}
  \tightlist
  \item
    Propositional logic, first-order logic

    \begin{itemize}
    \tightlist
    \item
      3 possible states of belief regarding any sentence: true, false,
      or unknown
    \end{itemize}
  \item
    Probability theory

    \begin{itemize}
    \tightlist
    \item
      There is a degree of belief in {[}0, 1{]} about each sentence
    \item
      \(\Pr(X = 6) = 0.3\)
    \end{itemize}
  \end{itemize}
\item
  Ontological commitment = what exists in the world
\item
  Epistemological commitment = what an agent believes about facts
\end{itemize}
\end{frame}

\begin{frame}{Non-monotonic Logic}
\protect\phantomsection\label{non-monotonic-logic}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Non-monotonic logic}} is a type of logic where
  adding new information can invalidate previous conclusions
\item
  Monotonic (classical) vs non-monotonic logic

  \begin{itemize}
  \tightlist
  \item
    In classical logic, once something is proven, it stays proven even
    if more information is added
  \item
    In non-monotonic logic, conclusions can change as new facts are
    learned
  \end{itemize}
\item
  \textbf{\textcolor{blue}{E.g.,}}

  \begin{itemize}
  \tightlist
  \item
    Initial knowledge: \emph{``Birds typically fly''}
  \item
    Fact: \emph{``Tweety is a bird''}
  \item
    Conclusion: \emph{``Tweety can fly''}
  \item
    New fact: \emph{``Tweety is a penguin''}
  \item
    Fact: \emph{``penguins are birds that can't fly''}
  \item
    Revised conclusion: \emph{``Tweety cannot fly''}
  \end{itemize}
\item
  E.g., ``Swans are white birds'' and then black swans are discovered
\item
  Real-world situations often involve incomplete or evolving knowledge

  \begin{itemize}
  \tightlist
  \item
    Non-monotonic logic allows systems to reason flexibly and adapt to
    new circumstances
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Default Reasoning}
\protect\phantomsection\label{default-reasoning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Default reasoning}} is reasoning where
  assumptions are made by default in the absence of contrary evidence

  \begin{itemize}
  \tightlist
  \item
    It allows conclusions based on typical situations unless exceptions
    are found
  \item
    Relates to ``common sense''
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Key idea}}

  \begin{itemize}
  \tightlist
  \item
    Assume the most likely case unless specified otherwise
  \item
    If new information contradicts the assumption, revise the conclusion
  \end{itemize}
\item
  \textbf{\textcolor{blue}{E.g.,}}

  \begin{itemize}
  \tightlist
  \item
    Default rule: \emph{``Typically, birds can fly''}
  \item
    Fact: \emph{``Tweety is a bird''}
  \item
    Conclusion: \emph{``Tweety can fly''}
  \item
    New fact: \emph{``Tweety is a penguin''}
  \item
    Revised conclusion: ``Tweety cannot fly''
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Why it is useful}}

  \begin{itemize}
  \tightlist
  \item
    Default reasoning allows systems to function reasonably without
    knowing everything
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Common Sense Reasoning}
\protect\phantomsection\label{common-sense-reasoning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Common sense reasoning}} is the ability to
  make assumptions, draw conclusions based on everyday knowledge about
  the world

  \begin{itemize}
  \tightlist
  \item
    Involves typical, unstated knowledge that humans take for granted,
    e.g.,

    \begin{itemize}
    \tightlist
    \item
      \emph{``If you drop a glass, it will likely break''}
    \item
      Knowing that \emph{``people eat food when they are hungry''}
      without being explicitly told
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Characteristics}}

  \begin{itemize}
  \tightlist
  \item
    Deals with incomplete, uncertain, or ambiguous information
  \item
    Relies on defaults, heuristics, and typical patterns rather than
    strict logical proofs
  \item
    Often flexible and tolerant of exceptions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Challenges}}

  \begin{itemize}
  \tightlist
  \item
    Common sense knowledge is vast, informal, and often not precisely
    defined
  \item
    Difficult to encode all of it explicitly in a machine-readable form
  \item
    Handling exceptions and contradictions is complex
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Techniques}}

  \begin{itemize}
  \tightlist
  \item
    Knowledge graphs
  \item
    Non-monotonic logic
  \item
    Probabilistic reasoning
  \item
    Machine learning models trained on large, diverse data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Open World vs Closed World Assumptions}
\protect\phantomsection\label{open-world-vs-closed-world-assumptions}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Closed World Assumption (CWA)}}

  \begin{itemize}
  \tightlist
  \item
    Missing information is false
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Fact: \emph{``Alice takes CS101''} is known
    \item
      Nothing is said about Bob
    \item
      Under CWA: Conclude Bob does not take CS101
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Open World Assumption (OWA)}}

  \begin{itemize}
  \tightlist
  \item
    Missing information is unknown, not false
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Fact: \emph{``Alice takes CS101''} is known
    \item
      Nothing is said about Bob
    \item
      Under OWA: Can't conclude if Bob takes CS101 or not it is unknown
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    CWA

    \begin{itemize}
    \tightlist
    \item
      Traditional relational databases (SQL)
    \item
      Logic programming
    \item
      Business rules and systems requiring complete data
    \end{itemize}
  \item
    OWA

    \begin{itemize}
    \tightlist
    \item
      Semantic Web (RDF, OWL)
    \item
      Knowledge representation with incomplete or growing data
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

  \item
    Produces human-readable logical rules
  \item
    Integrates learning with symbolic reasoning
  \item
    Supports background knowledge integration
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Challenges}}

  \begin{itemize}
  \tightlist
  \item
    Computational complexity with large datasets
  \item
    Handling noisy, incomplete, or ambiguous data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Description Logic}
\protect\phantomsection\label{description-logic}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Description logic}}

  \begin{itemize}
  \tightlist
  \item
    Represents structured knowledge about a domain
  \item
    Balances expressivity and computational efficiency
  \item
    More expressive than propositional logic, less than first-order
    logic
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Core building blocks:}}

  \begin{itemize}
  \tightlist
  \item
    Classes: abstract groups

    \begin{itemize}
    \tightlist
    \item
      E.g., \(Person\), \(Animal\)
    \end{itemize}
  \item
    Properties: binary relations between individuals

    \begin{itemize}
    \tightlist
    \item
      E.g., \(hasChild\), \(ownsPet\)
    \end{itemize}
  \item
    Instances: specific objects

    \begin{itemize}
    \tightlist
    \item
      E.g., \(GP\), \(Nuvolo\)
    \end{itemize}
  \end{itemize}
\item
  Supports reasoning tasks such as:

  \begin{itemize}
  \tightlist
  \item
    Concept subsumption: \emph{``Is class \(A\) a subset of class
    \(B\)?''}
  \item
    Instance checking: \emph{``Does instance \(a\) belong to class
    \(A\)?''}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Syntax often combines:}}

  \begin{itemize}
  \tightlist
  \item
    Atomic concepts and roles
  \item
    Logical constructors (\(\sqcap\), \(\sqcup\), \(\neg\), \(\forall\),
    \(\exists\))
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      \(Father \equiv Man \sqcap \exists hasChild.Person\)
    \end{itemize}
  \end{itemize}
\item
  Widely used in ontologies, e.g., OWL (Web Ontology Language)
\end{itemize}
\end{frame}

\begin{frame}{ALC}
\protect\phantomsection\label{alc}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Attributive Concept Language with Complements (ALC)}}
  is a basic but expressive description logic

  \begin{itemize}
  \tightlist
  \item
    Concepts can be combined using logical operators, e.g.,

    \begin{itemize}
    \tightlist
    \item
      \(\sqcap\) means ``and''
    \item
      \(\sqcup\) means ``or''
    \item
      \(\neg\) means ``not''
    \end{itemize}
  \item
    Allows for existential and universal quantification, e.g.,
    \(\exists R.C\), \(\forall R.C\)
  \item
    Interpretation is set-theoretic

    \begin{itemize}
    \tightlist
    \item
      Classes as sets, properties as binary relations
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``All students take some course''}:
    \(\text{Student} \equiv \exists \text{takes}.\text{Course}\)
  \item
    \emph{``A mother is a woman who has at least one child''}:
    \(\text{Mother} \equiv \text{Woman} \sqcap \exists \text{hasChild}.\top\)
  \end{itemize}
\item
  ALC:

  \begin{itemize}
  \tightlist
  \item
    Is decidable
  \item
    Balances expressiveness and computational complexity
  \item
    Is the basis for more complex logics used in OWL
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{SHOIN}
\protect\phantomsection\label{shoin}
\begin{itemize}
\item
  \textbf{\textcolor{red}{SHOIN}} is a description logic more expressive
  than ALC
\item
  Components:

  \begin{itemize}
  \tightlist
  \item
    \(\mathcal{S}\): Allows transitive properties

    \begin{itemize}
    \tightlist
    \item
      E.g., \(ancestorOf\) is transitive
    \end{itemize}
  \item
    \(\mathcal{H}\): Supports role hierarchies

    \begin{itemize}
    \tightlist
    \item
      E.g., \(hasSon \sqsubseteq hasChild\)
    \end{itemize}
  \item
    \(\mathcal{O}\): Introduces specific individuals

    \begin{itemize}
    \tightlist
    \item
      E.g., \(John\) is a nominal class
    \end{itemize}
  \item
    \(\mathcal{I}\): Enables roles to be navigated backward

    \begin{itemize}
    \tightlist
    \item
      E.g., \(isChildOf\) is inverse of \(hasChild\)
    \end{itemize}
  \item
    \(\mathcal{N}\): Sets cardinality constraints

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``has exactly 1 children''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``Exactly two children''}:
    \(Person \sqsubseteq (=2\ hasChild.\top)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Characteristics}}

  \begin{itemize}
  \tightlist
  \item
    More powerful but reasoning is harder (exponential complexity)
  \item
    Model richer real-world scenarios
  \item
    Foundation for OWL DL reasoning capabilities
  \end{itemize}
\end{itemize}
\end{frame}

  \textbf{\textcolor{red}{Web Ontology Language (OWL)}}:

  \begin{itemize}

    \begin{itemize}

\item
  \textbf{\textcolor{blue}{OWL variants}}:

  \begin{itemize}
  \tightlist
  \item
    OWL Lite: simpler, for classification hierarchies
  \item
    OWL DL: full expressiveness with decidable reasoning
  \item
    OWL Full: maximum expressiveness, but undecidable
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Semantic search
  \item
    Biomedical data
  \end{itemize}
\end{itemize}

\begin{frame}[fragile]{SPARQL}

  \textbf{\textcolor{red}{Key Concepts}}:

    graph
  \item
    \emph{Basic Graph Pattern}: A set of triple patterns combined
  \item
    \emph{Variables}: Stand in for unknown parts of the triples

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{?person}},
      \textcolor{blue}{\texttt{?animal}}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Main Query Types}}:

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{SELECT}}: Retrieve specific variables from
    the data
  \item
    \textcolor{blue}{\texttt{CONSTRUCT}}: Create new RDF triples based
    on query results
  \item
    \textcolor{blue}{\texttt{ASK}}: Return a boolean indicating whether
    a pattern exists
  \item
    \textcolor{blue}{\texttt{DESCRIBE}}: Return an RDF graph describing
    resources
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \item
    \emph{``Find all resources that are of type Bird''}

\begin{verbatim}
SELECT ?animal WHERE { ?animal rdf:type ex:Bird }
\end{verbatim}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Semantic Web}
\protect\phantomsection\label{semantic-web}
\begin{itemize}
\tightlist
\item
  The \textbf{\textcolor{red}{Semantic Web}} extends the current Web by
  enabling machines to understand and interpret data

  \begin{itemize}
  \tightlist
  \item
    HTML is human-readable but lacks semantic structure for computers
  \item
    The Semantic Web adds meaning / semantics to data
  \item
    Allow better data integration, automation, and discovery across
    sites
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Key Technologies}}

  \begin{itemize}
  \tightlist
  \item
    RDF (Resource Description Framework): base data model
  \item
    SPARQL: query language for RDF data
  \item
    OWL (Web Ontology Language): define rich ontologies
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Current Status}}

  \begin{itemize}
  \tightlist
  \item
    Some core ideas (e.g., structured data and ontologies) are widely
    adopted
  \item
    Full vision remains only partially realized
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Challenges}}

  \begin{itemize}
  \tightlist
  \item
    Complexity of widespread adoption
  \item
    Issues around privacy, data ownership, and feasibility
  \item
    Need for standardization and tools
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Criticism}}

  \begin{itemize}
  \tightlist
  \item
    Skepticism about practicality and scalability
  \item
    Concerns about centralization and censorship
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{WikiData}
\protect\phantomsection\label{wikidata}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{WikiData}} is a free, open, collaborative
  knowledge base

  \begin{itemize}
  \tightlist
  \item
    Stores structured data for Wikipedia
  \item
    Accessible via APIs using SPARQL queries
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Graph-based data model}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Item}: represents an entity or concept

    \begin{itemize}
    \tightlist
    \item
      \textcolor{blue}{\texttt{Q42}} \(\rightarrow\) Douglas Adams
    \end{itemize}
  \item
    \emph{Property}: describes a relationship or attribute

    \begin{itemize}
    \tightlist
    \item
      \textcolor{blue}{\texttt{P31}} (instance of),
      \textcolor{blue}{\texttt{P27}} (country of citizenship)
    \end{itemize}
  \item
    \emph{Value}: specific data linked to an item via a property

    \begin{itemize}
    \tightlist
    \item
      \textcolor{blue}{\texttt{Q42}} (Douglas Adams) \(\to\)
      \textcolor{blue}{\texttt{P31}} (instance of) \(\to\)
      \textcolor{blue}{\texttt{Q5}} (human)
    \item
      \textcolor{blue}{\texttt{Q42}} \(\to\)
      \textcolor{blue}{\texttt{P106}} (occupation) \(\to\)
      \textcolor{blue}{\texttt{Q36180}} (science fiction writer)
    \end{itemize}
  \item
    \emph{Reference}: supports a claim by citing a source

    \begin{itemize}
    \tightlist
    \item
      Stating Douglas Adams's citizenship with a reference to a
      biography
    \end{itemize}
  \item
    \emph{Qualifier}: adds context or additional information to a
    statement

    \begin{itemize}
    \tightlist
    \item
      \textcolor{blue}{\texttt{Q90}} (Paris) \(\to\)
      \textcolor{blue}{\texttt{P1082}} (population) \(\to\)
      ``2,165,423''
    \item
      With qualifier: \textcolor{blue}{\texttt{P585}} (point in time)
      \(\to\) ``2021''
    \item
      Meaning: ``The population of Paris was 2,165,423 in the year
      2021''
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications:}}

  \begin{itemize}
  \tightlist
  \item
    Knowledge graph
  \item
    Semantic search
  \item
    AI reasoning
  \item
    Data enrichment
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Semantic Networks}

\begin{itemize}

  \textbf{\textcolor{red}{Semantic Networks}} represent knowledge as

  \item

    \begin{itemize}

  \end{itemize}

  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Easy to visualize and traverse
  \item
    Support reasoning
  \item
    Common in early AI systems and current KG applications
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{WordNet}
\protect\phantomsection\label{wordnet}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{WordNet}} is a large lexical database of
  English words

  \begin{itemize}
  \tightlist
  \item
    Models semantic relationships between words
  \item
    Manually curated for high-quality semantic relations
  \item
    Incomplete for domain-specific language
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L03.WordNet.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Graph}} with synsets as nodes and
  relations as edges

  \begin{itemize}
  \tightlist
  \item
    \emph{Synsets} are sets of synonyms expressing a distinct concepts

    \begin{itemize}
    \tightlist
    \item
      E.g., \{\textcolor{blue}{\texttt{car}},
      \textcolor{blue}{\texttt{automobile}}\} share the same synset
    \end{itemize}
  \item
    \emph{Relations} between synsets

    \begin{itemize}
    \tightlist
    \item
      Is-a relationships (e.g., \textcolor{blue}{\texttt{Dog}} is a type
      of \textcolor{blue}{\texttt{Animal}})
    \item
      Part-whole relationships (e.g., \textcolor{blue}{\texttt{Wheel}}
      is a part of \textcolor{blue}{\texttt{Car}})
    \item
      Opposite meanings
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Word sense disambiguation: choose correct word meaning in context
  \item
    Semantic similarity measures: assess concept closeness
  \item
    Information retrieval and question answering systems
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Knowledge Graphs}
\protect\phantomsection\label{knowledge-graphs}
\begin{itemize}
\tightlist
\item
  Knowledge Graphs (KGs) represent entities and their relationships as a
  graph structure

  \begin{itemize}
  \tightlist
  \item
    Nodes = entities
  \item
    Edges = relations
  \item
    E.g., ``Paris \(\to\) isCapitalOf \(\to\) France''
  \end{itemize}
\item
  Query languages like SPARQL allow expressive information retrieval
\item
  KGs support reasoning via path traversal and schema inference
\item
  Applications:

  \begin{itemize}
  \tightlist
  \item
    Question answering
  \item
    Recommendation
  \item
    Semantic search
  \end{itemize}
\item
  Widely used by Google, Facebook, and academic search engines
\end{itemize}
\end{frame}
