\subsubsection{Simple Is Better}\label{simple-is-better}

\begin{frame}{Occam's Razor}
\protect\phantomsection\label{occams-razor}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Simple is better}}

  \begin{itemize}
  \item
    \emph{``The simplest model that fits the data is also the most
    plausible''} (Occam)
  \item
    \emph{``An explanation of the data should be as simple as possible,
    but not simpler''} (Einstein?)
  \end{itemize}
\item
  Corollary:

  \begin{itemize}
  \tightlist
  \item
    Trim the model to the bare minimum necessary to explain the data
  \end{itemize}
\item
  What does \textbf{\textcolor{darkgreen}{simple}} mean?

  \begin{itemize}
  \tightlist
  \item
    A model is simple when it is one of few possible objects
  \end{itemize}
\item
  What does \textbf{\textcolor{blue}{better}} mean?

  \begin{itemize}
  \tightlist
  \item
    A model is better means better out of sample performance
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Why it's true}}?

  \begin{itemize}
  \tightlist
  \item
    Less likely to fit a given data by coincidence
  \item
    An event is more significant if it is unlikely (formalized in terms
    entropy)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What is a ``simple model''?}
\protect\phantomsection\label{what-is-a-simple-model}
\begin{itemize}
\item
  An object is \textbf{\textcolor{red}{simple}} when it is one of few
  possible objects
\item
  \textbf{\textcolor{darkgreen}{Examples}}:

  \begin{itemize}
  \tightlist
  \item
    Polynomial of order 2 is simpler than order 17

    \begin{itemize}
    \tightlist
    \item
      More polynomials of order 17 than order 2, though both are
      infinite sets
    \end{itemize}
  \item
    SVM (Support Vector Machine) characteristics:

    \begin{itemize}
    \tightlist
    \item
      Separating hyperplane appears wiggly, but defined by few support
      vectors
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Measures of complexity}}

  \begin{itemize}
  \tightlist
  \item
    Complexity of a single hypothesis \(h\)

    \begin{itemize}
    \tightlist
    \item
      E.g., polynomial order, MDL (describe hypothesis in bits),
      Kolmogorov complexity
    \end{itemize}
  \item
    Complexity of a hypothesis set \(\calH\)

    \begin{itemize}
    \tightlist
    \item
      E.g., VC dimension of the model
    \end{itemize}
  \item
    Complexity of \(h\) and \(\calH\) related by counting:

    \begin{itemize}
    \tightlist
    \item
      If you need \(l\) bits to specify \(h\), then \(h\) is one of
      \(2^l\) elements of set \(\calH\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Model Soundness}
\protect\phantomsection\label{model-soundness}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Model should tell a story}}

  \begin{itemize}
  \tightlist
  \item
    You cannot blindly accept results of modeling
  \item
    Ask: \emph{``What criticisms would you give if the model was
    presented for the first time?''}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Benchmark models}}: compute performance
  if model outputs:

  \begin{itemize}
  \tightlist
  \item
    Always 0 or 1

    \begin{itemize}
    \tightlist
    \item
      E.g., long-only model for stock predictions
    \end{itemize}
  \item
    Random results

    \begin{itemize}
    \tightlist
    \item
      E.g., bootstrap of null hypothesis ``no prediction power''
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Perfect fit can mean nothing}}, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Get 2 data points on a plane

    \begin{itemize}
    \tightlist
    \item
      Fit data with a linear relationship
    \item
      Perfect fit
    \end{itemize}
  \item
    Means nothing since:

    \begin{itemize}
    \tightlist
    \item
      Always a line between 2 points
    \item
      Data cannot falsify hypothesis
    \end{itemize}
  \item
    The model (line) is too complex for data set (only 2 points)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sampling Bias (1/2)}
\protect\phantomsection\label{sampling-bias-12}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Sampling bias}} occurs when data is not
  representative of the intended population

  \begin{itemize}
  \tightlist
  \item
    Biased sampling leads to biased outcomes
  \item
    Model views world through training data
  \item
    Hoeffding's hypothesis: training and testing distributions are the
    same
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causes of Sampling Bias}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Non-random sampling}: Favors certain outcomes or groups
  \item
    \emph{Undercoverage}: Inadequate representation of some population
    members
  \item
    \emph{Survivorship bias}: Includes only successful subjects
  \item
    \emph{Self-selection bias}: Participants choose to participate,
    introducing bias
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Consequences}}

  \begin{itemize}
  \tightlist
  \item
    Systematic errors distort relationships or predictions
  \item
    Models trained on biased samples perform poorly on real-world data
  \item
    Leads to incorrect or unfair decisions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sampling Bias (2/2)}
\protect\phantomsection\label{sampling-bias-22}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    Training facial recognition on lighter-skinned faces reduces
    accuracy on darker-skinned faces
  \item
    Predicting income using data only from employed individuals omits
    unemployed people
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Mitigation}}

  \begin{itemize}
  \tightlist
  \item
    Compare sample statistics with population statistics
  \item
    Use stratified sampling or resampling to balance data
  \item
    Apply re-weighting or bias correction techniques
  \item
    If data points have zero probability (\(\Pr = 0\)), no remedies
    possible
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Data Snooping (1/2)}
\protect\phantomsection\label{data-snooping-12}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Data snooping}} occurs when test set
  information improperly influences model-building
\item
  \textbf{\textcolor{darkgreen}{Typical Sources}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Train-test contamination}: Test data leaks into training

    \begin{itemize}
    \tightlist
    \item
      E.g., Using test data during feature engineering, model selection,
      or tuning
    \end{itemize}
  \item
    \emph{Multiple hypothesis testing}: Trying many models inflates
    performance
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Consequences}}

  \begin{itemize}
  \tightlist
  \item
    Overly optimistic evaluations
  \item
    Models perform well in validation but fail on unseen data
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Why It's Dangerous}}

  \begin{itemize}
  \tightlist
  \item
    Misleading results
  \item
    False confidence in model quality
  \item
    Common pitfall in machine learning workflows
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Data Snooping (2/2)}
\protect\phantomsection\label{data-snooping-22}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    Choosing features based on the full dataset before splitting gives
    foresight not present in deployment
  \item
    E.g., demean the entire data and then split in train / test
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Preventive Strategies}}

  \begin{itemize}
  \tightlist
  \item
    Keep training, validation, and test sets separate
  \item
    Use cross-validation properly
  \item
    Fit preprocessing steps only on training data, then apply to test
    data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{``Burning the Test Set''}
\protect\phantomsection\label{burning-the-test-set}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Repeated test set use during development leaks into training
  \item
    Overfitting to test data
  \item
    \emph{``If you torture the data long enough, it will confess
    whatever you want''} (Coase, 1982)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Symptoms}}

  \begin{itemize}
  \tightlist
  \item
    Test accuracy improves; real-world performance stagnates or degrades
  \item
    Model evaluation becomes untrustworthy
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solutions}}

  \begin{itemize}
  \tightlist
  \item
    \emph{One-time use principle}: Evaluate on test set once,
    post-tuning
  \item
    \emph{Use a validation set}: For model development and tuning
  \item
    \emph{Statistical adjustments}:

    \begin{itemize}
    \tightlist
    \item
      Adjust p-values for multiple testing
    \item
      Use Bonferroni or False Discovery Rate corrections
    \end{itemize}
  \item
    \emph{Theoretical tools}:

    \begin{itemize}
    \tightlist
    \item
      VC dimension reflects learning capacity, including repeated trials
    \item
      Minimum Description Length (MDL) penalizes complex models and
      repeated fitting
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How to Achieve Out-Of-Sample Fit}
\protect\phantomsection\label{how-to-achieve-out-of-sample-fit}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}: Choose an hypothesis \(g \in \calH\)
  approximates the unknown target function \(f\)

  \begin{itemize}
  \tightlist
  \item
    What does \(g \approx f\) mean? \[
    g \approx f \iff E_{out}(g) \approx 0
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    Achieve:

    \begin{itemize}
    \tightlist
    \item
      Good in-sample performance \(E_{in}(g) \approx 0\)
    \item
      Good generalization \(E_{out}(g) \approx E_{in}(g)\)
    \end{itemize}
  \item
    Together \(\implies\) good out-of-sample performance
    \(E_{out}(g) \approx 0\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What If Out-Of-Sample Fit Is Poor?}
\protect\phantomsection\label{what-if-out-of-sample-fit-is-poor}
\begin{itemize}
\item
  The model performs well in sample (\(E_{in} \approx 0\)) but poorly
  out of sample (\(E_{out} \gg E_{in}\))
\item
  \textbf{\textcolor{red}{What does it mean?}}

  \begin{itemize}
  \tightlist
  \item
    In-sample performance is optimistic
  \item
    Model is overfitted and fails to generalize
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{What do you do?}}

  \begin{itemize}
  \tightlist
  \item
    Run diagnostics before long-term projects
  \item
    Gain insight on what works/doesn't to improve performance

    \begin{itemize}
    \tightlist
    \item
      E.g., bias-variance curves and learning curves
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{How to fix?}}

  \begin{enumerate}
  \tightlist
  \item
    Training data

    \begin{itemize}
    \tightlist
    \item
      Get more training data \(\iff\) fixes high variance
    \end{itemize}
  \item
    Features

    \begin{itemize}
    \tightlist
    \item
      Remove features \(\iff\) fixes high variance
    \item
      Add features \(\iff\) fixes high bias
    \item
      Add derived features (e.g., polynomial features) \(\iff\) fixes
      high bias
    \end{itemize}
  \item
    Regularization

    \begin{itemize}
    \tightlist
    \item
      Decrease regularization \(\lambda\) \(\iff\) fixes high bias
    \item
      Increase regularization \(\lambda\) \(\iff\) fixes high variance
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Why Using a Lot of Data?}
\protect\phantomsection\label{why-using-a-lot-of-data}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Several studies}} show that:

  \begin{itemize}
  \tightlist
  \item
    Different algorithms/models have remarkably similar performance
  \item
    Increasing training set improves performance
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Consequence}}

  \begin{itemize}
  \tightlist
  \item
    High capacity model + massive training set = good performance
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Why}}

  \begin{itemize}
  \tightlist
  \item
    Using a high capacity model with many parameters (e.g., neural
    network) \[
    E_{in} \approx 0
    \] due to low bias (and high variance)
  \item
    A massive data set helps avoid overfitting \[
    E_{out} \approx E_{in}
    \]
  \item
    These two conditions together \[
    E_{out} \approx E_{in} \approx 0 \implies E_{out} \approx 0
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What to Do When You Have Lots of Data?}
\protect\phantomsection\label{what-to-do-when-you-have-lots-of-data}
\begin{itemize}
\item
  You have \(m\) = 100M examples in data set: great!
\item
  \textbf{\textcolor{red}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Scalability issue (slow, lots of compute)
  \item
    Require work on infrastructure
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{First step}}:

  \begin{itemize}
  \tightlist
  \item
    Plot learning curves (in-sample and out-of-sample performance) for
    increasing amount of data \(m = 1k, 10k, 100k, 1M, ...\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Next step}}:

  \begin{itemize}
  \tightlist
  \item
    If model has large bias

    \begin{itemize}
    \tightlist
    \item
      Training and validation performance are similar at \(1M\)
    \item
      Next step: use more complex model rather than training on \(100M\)
      instances
    \end{itemize}
  \item
    If model has large variance

    \begin{itemize}
    \tightlist
    \item
      Next step: use all \(100M\) instances to train the model
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Why We Do Things?}
\protect\phantomsection\label{why-we-do-things}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Always, always, always:}}

  \begin{itemize}
  \tightlist
  \item
    Understand the purpose of the task

    \begin{itemize}
    \tightlist
    \item
      Ask: \emph{``Why are we doing something?''}
    \end{itemize}
  \item
    Clarify goals and outcomes of the task

    \begin{itemize}
    \tightlist
    \item
      Ask: \emph{``What do we hope to determine by performing the
      task?''}
    \end{itemize}
  \item
    Think about actions with the bigger picture in mind
  \item
    Avoid going through the motions
  \item
    Prioritize tasks by importance and impact
  \end{itemize}

  E.g., when conducting a customer survey:

  \begin{itemize}

    \end{itemize}
  \item
    Q: \emph{``What is the desired outcome?''}

    \begin{itemize}
    \tightlist
    \item
      A: To identify areas for improvement and innovation
    \end{itemize}
  \end{itemize}
\item
  E.g., before starting a marketing campaign:

  \begin{itemize}
  \tightlist
  \item
    Q: \emph{``Why is this campaign run?''}

    \begin{itemize}
    \tightlist
    \item
      A: To increase brand awareness or drive sales
    \end{itemize}
  \item
    Q: \emph{``What are the specific goals?''}

    \begin{itemize}
    \tightlist

    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{frame}{Right and Wrong Approach to Research}
\protect\phantomsection\label{right-and-wrong-approach-to-research}
\begin{itemize}
\item
  It is not clear how to prioritize the different possible tasks
\item
  \textbf{\textcolor{red}{Bad}}

  \begin{enumerate}
  \tightlist
  \item
    Use gut feeling to choose a task
  \item
    Complete task
  \item
    Re-evaluate performance
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Good}}

  \begin{enumerate}
  \tightlist
  \item
    Build a simple algorithm

    \begin{itemize}
    \tightlist
    \item
      Within 1 day
    \end{itemize}
  \item
    Set up performance evaluation

    \begin{itemize}
    \tightlist
    \item
      Single number and bounds to improve
    \item
      Use cross-validation
    \end{itemize}
  \item
    Set up diagnostic tools

    \begin{itemize}
    \tightlist
    \item
      Compute learning and bias-variance curves
    \item
      Understand issues before fixing (avoid premature optimization)
    \end{itemize}
  \item
    Review misclassified emails manually

    \begin{itemize}
    \tightlist
    \item
      Identify features to improve performance
    \item
      E.g., what are types of misclassified emails?
    \end{itemize}
  \end{enumerate}
\item
  Sometimes you just need to try approaches to see if they work

  \begin{itemize}
  \tightlist
  \item
    E.g., use stemming software to equate words
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Incremental vs Iterative}
\protect\phantomsection\label{incremental-vs-iterative}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Incremental Development}}

  \begin{itemize}
  \tightlist
  \item
    Each increment adds functional components
  \item
    Require upfront planning to divide features meaningfully
  \item
    Integration of increments can be complex
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Iterative Development}}

  \begin{itemize}
  \tightlist
  \item
    Each increment delivers usable system
  \item
    Refine and improve product through repeated cycles
  \item
    Get feedback
  \item
    Uncover and adjust for unknown requirements
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Incremental $\gg$ Iterative}}
\end{itemize}
\end{column}
