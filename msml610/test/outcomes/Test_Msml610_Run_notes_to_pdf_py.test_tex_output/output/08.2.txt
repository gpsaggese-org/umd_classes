\begin{frame}{(Non-Causal) Bayesian Networks}
\protect\phantomsection\label{non-causal-bayesian-networks}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Bayesian networks}} represent a joint
  distribution function

  \begin{itemize}
  \tightlist
  \item
    The direction of the arrow represent \emph{conditional dependence}
    (not causality)
  \item
    \(A \to B\) requires to estimate \(\Pr(A | B)\)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Many possible Bayesian networks}} with same
  nodes, different edges to explain the same phenomenon
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Example}}

  \begin{itemize}
  \tightlist
  \item
    A Bayesian network with \(Fire\) and \(Smoke\), which are dependent
  \item
    \(Fire \to Smoke\)

    \begin{itemize}
    \tightlist
    \item
      Need \(\Pr(Fire)\) and \(\Pr(Smoke | Fire)\) to compute
      \(\Pr(Fire, Smoke)\)
    \end{itemize}
  \item
    \(Smoke \to Fire\)

    \begin{itemize}
    \tightlist
    \item
      Need \(\Pr(Smoke)\) and \(\Pr(Fire | Smoke)\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{teal}{Different Bayesian networks}}:

  \begin{itemize}
  \tightlist
  \item
    Are equivalent and convey the same information
  \item
    Have different difficulties to be estimated
  \end{itemize}
\item
  There is an \textbf{\textcolor{cyan}{asymmetry in nature}}

  \begin{itemize}
  \tightlist
  \item
    Extinguishing fire stops smoke
  \item
    Clearing smoke doesn't affect fire
  \end{itemize}
\end{itemize}

\begin{frame}{Causal (Bayesian) Networks}
\protect\phantomsection\label{causal-bayesian-networks}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Causal networks are Bayesian networks with only causal edges}}

  \begin{itemize}
  \tightlist
  \item
    Use judgment based on nature instead of just statistics
  \item
    E.g., you need to go from

    \begin{itemize}
    \tightlist
    \item
      \emph{``Are random variables \(Smoke\) and \(Fire\) correlated?''}
      to
    \item
      \emph{``What causes what, \(Smoke\) or \(Fire\)?''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{"Dependency in nature"}} is like
  assignment in programming

  \begin{itemize}
  \tightlist
  \item
    E.g., nature assigns \(Smoke\) based on \(Fire\):

    \begin{itemize}
    \tightlist
    \item
      \includegraphics[width=0.125in,height=\textheight,keepaspectratio]{assets/emoji/check_mark.png}
      \(Smoke := f(Fire)\)
    \item
      \includegraphics[width=0.125in,height=\textheight,keepaspectratio]{assets/emoji/wrong.png}
      \(Fire := f(Smoke)\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Structural equations}} describe ``assignment
  mechanism'' in causal graphs \[
  X_i := f(X_j) \iff X_j \to X_i
  \]
\end{itemize}
\end{frame}

\begin{frame}{Causal DAG}
\protect\phantomsection\label{causal-dag}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Causal DAG}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Directed}: Arrows show cause \(\rightarrow\) effect
  \item
    \emph{Acyclic}: No feedback loops

    \begin{itemize}
    \tightlist
    \item
      Causal relationships assume temporal order: cause before effect
    \item
      A cycle implies a variable is both cause and effect of itself
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Benefits}}

  \begin{itemize}
  \tightlist
  \item
    DAGs makes explicit \emph{causal} links
  \item
    Support explainable AI models
  \item
    Stability in conditional probability estimation
  \item
    Reason about interventions and counterfactuals
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Limitations}}

  \begin{itemize}
  \tightlist
  \item
    Requires domain knowledge for structure
  \item
    Assumes all relevant variables included (no hidden confounders)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Edges are Stable}
\protect\phantomsection\label{causal-edges-are-stable}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Causal edges reflect stable relationship}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Mechanistic stability}

    \begin{itemize}
    \tightlist
    \item
      Causal relationships show system function, not just behavior in
      one dataset
    \item
      E.g., \emph{``Temperature \(\rightarrow\) ice melting rate''}
      holds true in Alaska and Arizona
    \end{itemize}
  \item
    \emph{Invariance under interventions}

    \begin{itemize}
    \tightlist
    \item
      If \(X\) causes \(Y\), intervening on \(X\) affects \(Y\)
      consistently, despite confounders or context changes
    \end{itemize}
  \item
    \emph{Easier estimation through causal modeling}

    \begin{itemize}
    \tightlist
    \item
      Identifying causal direction focuses estimation on effect size
      (e.g., regression of \(Y\) on \(X\) under intervention)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: study
  \(Exercise \rightarrow Health\):

  \begin{itemize}
  \tightlist
  \item
    Correlation may differ in young or elderly populations
  \item
    Causal effect remains stable, as physiological mechanism doesn't
    change
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Structural Causal Model}
\protect\phantomsection\label{structural-causal-model}
\begin{itemize}
\item
  A \textbf{\textcolor{red}{Structural Causal Model}} (SCM) translates a
  causal DAG into mathematical equations to define how variables
  interact
\item
  \textbf{\textcolor{darkgreen}{Structure of SCMs}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Variables} \(X_1, X_2, ..., X_n\) represent quantities in the
    system
  \item
    \emph{Equations} model each variable as a function of its direct
    causes
  \item
    Formally, \(X_i\) is modeled as:
    \[X_i = f_i(Parents(X_i), \varepsilon_i)\] where:

    \begin{itemize}
    \tightlist
    \item
      \(Parents(X_i)\) are direct causes of \(X_i\)
    \item
      \(\varepsilon_i\) is an exogenous (external, unobserved) noise
      term
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Properties}}

  \begin{itemize}
  \tightlist
  \item
    Same properties of causal networks

    \begin{itemize}
    \tightlist
    \item
      Explain causal relationships between variables
    \item
      Provide a foundation for causal reasoning and simulation
    \item
      \ldots{}
    \end{itemize}
  \item
    Quantify effect
  \end{itemize}
\item
  Used in econometrics and genetics for a long time (even before theory
  of causality)
\end{itemize}
\end{frame}

\begin{frame}{Structural Causal Model: Sprinkler Example}
\protect\phantomsection\label{structural-causal-model-sprinkler-example}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Structural equations}} for this causal DAG:
  \begingroup \small
  $$
  \left\{
  \begin{aligned}
    & C := f_C(\varepsilon_C) \\
    & R := f_R(C, \varepsilon_R) \\
    & S := f_S(C, \varepsilon_S) \\
    & W := f_W(R, S, \varepsilon_W) \\
    & G := f_G(W, \varepsilon_G) \\
  \end{aligned}
  \right.
  $$
  \endgroup
\item
  \textbf{\textcolor{darkgreen}{Unmodeled variables}} \(\varepsilon_x\)
  represent error terms

  \begin{itemize}
  \tightlist
  \item
    E.g., \(\varepsilon_W\) is another source of wetness (e.g.,
    \(MorningDew\)) besides \(Sprinkler\) and \(Rain\)
  \item
    Assume unmodeled variables are exogenous, independent, with a
    certain distribution (prior)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.25\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.8.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  Express \textbf{\textcolor{blue}{joint distribution}} of all variables
  as a product of conditional distributions using causal DAG topology:
  \begingroup \small
  $$
  \Pr(C, R, S, W, G) = \Pr(W|R,S) \Pr(G|W) \Pr(S|C) \Pr(R|C) \Pr(C)
  $$
  \endgroup
\end{itemize}
\end{frame}

\begin{frame}{Mediator Variable: Example}
\protect\phantomsection\label{mediator-variable-example}
\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Research question}}: \emph{``Does a training
  program increase employee productivity?''}
\item
  Causal effect may be indirect, through a
  \textbf{\textcolor{darkgreen}{mediator}}

  \begin{itemize}

  \item

  \item

  \end{itemize}

  \begin{itemize}

  \item
    \(Y\): Employee Productivity (effect)
  \item

  \end{itemize}
\item

  \begin{itemize}
  \tightlist
  \item
    \emph{Indirect effect}: \(X\) affects \(Y\) through \(M\)
  \item
    \emph{Direct effect}: \(X\) affects \(Y\) not through \(M\)
  \item
    Controlling for \(M\) separates effects, clarifying training impact
  \end{itemize}

\begin{frame}{Collider Bias}
\protect\phantomsection\label{collider-bias}
\begin{itemize}
\item
  Aka ``Berkson's paradox''
\item
  \textbf{\textcolor{red}{Conditioning on a collider}} can introduce a
  spurious association between its parents by \emph{``opening a path
  that is blocked''}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Example}}

  \begin{itemize}
  \tightlist
  \item
    \(Diet\) (D)
  \item
    \(Exercise\) (E)
  \item
    \(BodyWeight\) (W)
  \item
    \(HeartDisease\) (D)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Without conditioning on $W$}}

  \begin{itemize}
  \tightlist
  \item
    \(E\) and \(D\) are independent

    \begin{itemize}
    \tightlist
    \item
      E.g., knowing exercise level \(E\) doesn't inform about diet
      \(D\), and vice versa
    \end{itemize}
  \item
    Collider \(W\) blocks association between \(E\) and \(D\)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.25\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.26.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{violet}{After conditioning on $W$}}

  \begin{itemize}
  \tightlist
  \item
    E.g., individuals with specific body weight
  \item
    Introduce dependency between \(E\) and \(D\)
  \item
    With \(W\) fixed, changes in \(E\) balanced by changes in \(D\),
    inducing spurious correlation between \(E\) and \(D\)
  \item
    In Bayesian network it was called \emph{``explaining away''}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Fork Structure}
\protect\phantomsection\label{fork-structure}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{fork}} \(D \leftarrow X \rightarrow C\)
  occurs when a single variable causally influences two or more
  variables

  \begin{itemize}
  \tightlist
  \item
    \(X\) is a \textbf{\textcolor{darkgreen}{confounder}} (common cause)
    of \(C\) and \(D\)
  \item
    Forks induce statistical dependence between \(C\) and \(D\) even if
    \(C\) and \(D\) are not causally linked
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Conditioning}} on \(X\) \emph{blocks the
  path} and removes spurious correlation
\end{itemize}
\end{column}

\begin{frame}{Inverted Fork}
\protect\phantomsection\label{inverted-fork}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  An \textbf{\textcolor{red}{inverted fork}} occurs when two or more
  arrows converge on a common node

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Colliders}} block associations unless
    the collider or its descendants are conditioned on
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Conditioning on a collider}} \emph{opens a
  path}, inducing spurious correlations

  \begin{itemize}
  \tightlist
  \item
    This is the basis of selection bias
  \end{itemize}
\end{itemize}
\end{column}
