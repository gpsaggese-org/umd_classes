\begin{frame}{Random Variables and Index Sets}
\protect\phantomsection\label{random-variables-and-index-sets}
\begin{itemize}
\tightlist
\item
  A stochastic process is a collection of random variables indexed by
  time or space
\item
  Indexed set: \(\{X_t : t \in T\}\) where \(T\) can be discrete or
  continuous
\item
  Describes evolving random phenomena
\item
  Example: daily temperature, stock prices
\end{itemize}
\end{frame}

\begin{frame}{Markov Chains}
\protect\phantomsection\label{markov-chains}
\begin{itemize}
\tightlist
\item
  Memoryless stochastic process:
  \(P(X_{t+1} | X_t, ..., X_0) = P(X_{t+1} | X_t)\)
\item
  Characterized by a transition matrix \(P\)
\item
  Types: discrete-time, continuous-time
\item
  Applications: web ranking, genetic modeling, board games
\end{itemize}
\end{frame}

\begin{frame}{Stationarity}
\protect\phantomsection\label{stationarity}
\begin{itemize}
\tightlist
\item
  A process is stationary if its statistical properties do not change
  over time
\item
  Strict stationarity: joint distribution invariant under time shift
\item
  Weak stationarity: constant mean and autocovariance depend only on lag
\item
  Important in time-series modeling and signal processing
\end{itemize}
\end{frame}

\begin{frame}{Martingales}
\protect\phantomsection\label{martingales}
\begin{itemize}
\tightlist
\item
  Process with conditional expectation:
  \(\mathbb{E}[X_{t+1} | X_1, ..., X_t] = X_t\)
\item
  Models fair games and conservative estimates
\item
  Used in financial modeling and online learning
\end{itemize}
\end{frame}

\begin{frame}{Poisson Process}
\protect\phantomsection\label{poisson-process}
\begin{itemize}
\tightlist
\item
  Models arrival of random events in continuous time
\item
  Events occur independently with constant average rate \(\lambda\)
\item
  Interarrival times are exponentially distributed
\item
  Applications: queuing theory, rare event modeling
\end{itemize}
\end{frame}

\begin{frame}{Brownian Motion (Wiener Process)}
\protect\phantomsection\label{brownian-motion-wiener-process}
\begin{itemize}
\tightlist
\item
  Continuous-time stochastic process with stationary, independent
  Gaussian increments
\item
  Starts at 0: \(B_0 = 0\)
\item
  Used in modeling diffusion, stock prices, reinforcement learning
\item
  Foundation for stochastic differential equations
\end{itemize}
\end{frame}

\begin{frame}{Autoregressive (AR) Processes}
\protect\phantomsection\label{autoregressive-ar-processes}
\begin{itemize}
\tightlist
\item
  \(X_t\) depends linearly on its past values:
  \(X_t = \sum_{i=1}^p \phi_i X_{t-i} + \epsilon_t\)
\item
  \(\epsilon_t\) is white noise
\item
  Common in time-series forecasting (e.g., ARIMA)
\end{itemize}
\end{frame}

\begin{frame}{Moving Average (MA) Processes}
\protect\phantomsection\label{moving-average-ma-processes}
\begin{itemize}
\tightlist
\item
  \(X_t\) is a linear function of current and past noise:
  \(X_t = \sum_{i=0}^q \theta_i \epsilon_{t-i}\)
\item
  Used to model short-term dependencies
\item
  Often combined with AR in ARMA/ARIMA models
\end{itemize}
\end{frame}

\begin{frame}{Hidden Markov Models (HMMs)}
\protect\phantomsection\label{hidden-markov-models-hmms}
\begin{itemize}
\tightlist
\item
  Markov chain with unobserved (hidden) states
\item
  Observations are generated from state-dependent distributions
\item
  Used in speech recognition, bioinformatics, and NLP
\end{itemize}
\end{frame}

\begin{frame}{Gaussian Processes}
\protect\phantomsection\label{gaussian-processes}
\begin{itemize}
\tightlist
\item
  A collection of random variables, any finite number of which are
  jointly Gaussian
\item
  Fully specified by a mean function and covariance kernel
\item
  Used in Bayesian regression, spatial modeling, and active learning
\end{itemize}
\end{frame}

\begin{frame}{Stochastic Differential Equations (SDEs)}
\protect\phantomsection\label{stochastic-differential-equations-sdes}
\begin{itemize}
\tightlist
\item
  Differential equations with random noise
\item
  General form: \(dX_t = \mu(X_t, t)dt + \sigma(X_t, t)dB_t\)
\item
  Models continuous-time random phenomena
\item
  Applications: physics, finance, control theory
\end{itemize}
\end{frame}

\begin{frame}{Ergodicity}
\protect\phantomsection\label{ergodicity}
\begin{itemize}
\tightlist
\item
  Time averages equal ensemble averages under certain conditions
\item
  Ensures statistical estimation is feasible from a single realization
\item
  Crucial for learning from time-series data
\end{itemize}
\end{frame}

\begin{frame}{Renewal Processes}
\protect\phantomsection\label{renewal-processes}
\begin{itemize}
\tightlist
\item
  Generalization of Poisson processes
\item
  Interarrival times are i.i.d. but not necessarily exponential
\item
  Used in reliability theory and system maintenance
\end{itemize}
\end{frame}

\begin{frame}{Birth-Death Processes}
\protect\phantomsection\label{birth-death-processes}
\begin{itemize}
\tightlist
\item
  Special case of Markov chains with transitions to neighboring states
\item
  Models population dynamics, queue lengths
\item
  Characterized by birth rate \(\lambda_n\) and death rate \(\mu_n\)
\end{itemize}
\end{frame}

\begin{frame}{Queueing Models}
\protect\phantomsection\label{queueing-models}
\begin{itemize}
\tightlist
\item
  Systems where entities wait in line for service
\item
  Described using stochastic processes (e.g., M/M/1 queue)
\item
  Analyzed via arrival and service rate distributions
\item
  Applied in networks, servers, and traffic modeling
\end{itemize}
\end{frame}

\begin{frame}{Random Walks}
\protect\phantomsection\label{random-walks}
\begin{itemize}
\tightlist
\item
  Discrete stochastic process: \(X_{t+1} = X_t + \epsilon_t\)
\item
  \(\epsilon_t\) is typically i.i.d. and symmetric
\item
  Central to modeling cumulative processes and diffusion
\end{itemize}
\end{frame}

\begin{frame}{Law of Large Numbers and Central Limit Theorem}
\protect\phantomsection\label{law-of-large-numbers-and-central-limit-theorem}
\begin{itemize}
\tightlist
\item
  Justify convergence and Gaussian approximations of stochastic
  processes
\item
  Law of large numbers: sample mean converges to expected value
\item
  CLT: sum of i.i.d. variables approximates a normal distribution
\end{itemize}
\end{frame}
