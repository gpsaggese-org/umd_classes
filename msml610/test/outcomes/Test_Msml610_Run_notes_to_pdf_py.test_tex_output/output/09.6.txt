\begin{frame}{Dynamic Bayesian Networks (DBNs)}
\protect\phantomsection\label{dynamic-bayesian-networks-dbns}
\begin{itemize}
\item
  DBNs extend Bayesian networks to model temporal processes
\item
  Main idea

  \begin{itemize}
  \tightlist
  \item
    ``Unroll'' the model over time
  \item
    Capture intra-slice (within time) and inter-slice (across time)
    dependencies
  \end{itemize}
\item
  Each time slice includes:

  \begin{itemize}
  \tightlist
  \item
    State variables \(X_t\)
  \item
    Evidence variables \(E_t\)
  \end{itemize}
\item
  Assumptions

  \begin{itemize}
  \tightlist
  \item
    First-order Markov process: current state depends only on the
    previous state
  \item
    First-order sensor Markov process: evidence depends only on current
    state
  \item
    Stationarity: each time slice is the same, both structure and
    parameters do not change over time

    \begin{itemize}
    \tightlist
    \item
      Structure and CPTs (Conditional Probability Tables) are the same
      across slices (time-homogeneous model)
    \end{itemize}
  \item
    No Gaussian distribution
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DBNs vs HMMs}
\protect\phantomsection\label{dbns-vs-hmms}
\begin{itemize}
\item
  DBNs generalize Hidden Markov Models (HMMs)
\item
  HMMs are a special case with a single hidden and evidence variable per
  time step
\item
  DBNs model more complex systems than HMMs by:

  \begin{itemize}
  \tightlist
  \item
    Using multiple state variables

    \begin{itemize}
    \tightlist
    \item
      Enables modeling large systems like robot localization with many
      state components
    \end{itemize}
  \item
    Exploiting sparse connections among variables yielding compact model

    \begin{itemize}
    \tightlist
    \item
      HMM: transition matrix of size \(O(d^{2n})\)
    \item
      DBN: size \(O(nd^k)\) with \(k\) bounded parents per variable
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DBNs vs Kalman Filters}
\protect\phantomsection\label{dbns-vs-kalman-filters}
\begin{itemize}
\item
  DBNs generalize Kalman filters
\item
  Every Kalman filter can be represented in a DBN with:

  \begin{itemize}
  \tightlist
  \item
    Continuous variables
  \item
    Linear / Gaussian conditional distributions
  \end{itemize}
\item
  Not every DBN can be represented by a Kalman filter, since:

  \begin{itemize}
  \tightlist
  \item
    DBN variables can mix discrete/continuous and non-Gaussian
  \item
    Allow arbitrary conditional dependencies among variables
  \end{itemize}
\item
  \textbf{\textcolor{red}{Pros of DBNs}}

  \begin{itemize}
  \tightlist
  \item
    DBNs are applicable to broader domains including:

    \begin{itemize}
    \tightlist
    \item
      Fault diagnosis in networks
    \item
      Complex system monitoring
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros of Kalman filters:}}

  \begin{itemize}
  \tightlist
  \item
    Optimal for linear systems with Gaussian noise
  \item
    Support exact inference, DBNs often require approximate methods
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Constructing a DBN}
\protect\phantomsection\label{constructing-a-dbn}
\begin{itemize}
\tightlist
\item
  Key components of a DBN

  \begin{itemize}
  \tightlist
  \item
    Prior distribution of state \(\Pr(X_0)\)
  \item
    Transition model \(\Pr(X_{t+1} | X_t)\)
  \item
    Sensor model \(\Pr(E_t | X_t)\)
  \item
    Transition and sensor models are time-homogeneous
  \end{itemize}
\item
  Network topology includes:

  \begin{itemize}
  \tightlist
  \item
    Intra-slice topology
  \item
    Inter-slice links
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DBN Example: Tracking a Robot (1/3)}
\protect\phantomsection\label{dbn-example-tracking-a-robot-13}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem:}}

  \begin{itemize}
  \tightlist
  \item
    Tracking a robot moving randomly on a line \(X\) over time
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Initial model:}}

  \begin{itemize}
  \tightlist
  \item
    Position \(X_t\) and velocity \(\dot{X_t}\) as state variables
  \item
    Update via Newton's laws
  \item
    Easy to generalize for 2d or 3d by using a \(\vX_t\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Issue:}}

  \begin{itemize}
  \tightlist
  \item
    Velocity changes over time
  \item
    Battery exhaustion affects velocity systematically
  \item
    Effect depends on cumulative energy use
  \item
    Violates the Markov property (future depends on full history)
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Solution:}}

  \begin{itemize}
  \tightlist
  \item
    Include battery level \(Battery_t\) in the state \(X_t\)
  \item
    Restores the Markov assumption
  \item
    Allows motion prediction considering energy constraints
  \item
    Enables coherent reasoning about motion and power consumption over
    time
  \end{itemize}
\item

\begin{frame}{Inference in DBNs}
\protect\phantomsection\label{inference-in-dbns}
\begin{itemize}
\tightlist
\item
  DBNs are Bayesian networks and we can use the same inference
  algorithms

  \begin{itemize}
  \tightlist
  \item
    ``Unroll'' the DBN over time (i.e., replicate slices for each time
    step) and apply standard BN inference
  \item
    We can't unroll ``forever'', but we limit to a certain number of
    slices to approximate a fixed amount of time dependency
  \end{itemize}
\item
  Use recursive methods to get a constant time and space update
  complexity

  \begin{itemize}
  \tightlist
  \item
    Variable elimination with temporal ordering
  \item
    At time step \(t + 1\) add slice \(t + 2\) and remove slice \(t\) so
    one has always two slices to do inference
  \item
    Maintains constant memory by keeping only two slices at a time
  \end{itemize}
\item
  Complexity:

  \begin{itemize}
  \tightlist
  \item
    Exponential in number of state variables (\(O(nd^{n+k})\))
  \item
    More efficient than full HMM representation (\(O(d^{2n})\))
  \end{itemize}
\item
  Even though we can use DBNs to represent very complex temporal
  processes with many sparsely connected variables, we cannot reason
  efficiently and exactly about those processes

  \begin{itemize}
  \tightlist
  \item
    The prior joint distribution over all the variables is factorizable
    into its constituents CPTs
  \item
    The posterior joint distribution conditioned on observation sequence
    is not factorizable
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Approximate Inference in DBNs}
\protect\phantomsection\label{approximate-inference-in-dbns}
\begin{itemize}
\tightlist
\item
  Particle Filtering:

  \begin{itemize}
  \tightlist
  \item
    Represent belief state with weighted samples (particles)
  \item
    Steps: propagate, weight, resample
  \end{itemize}
\item
  Benefits:

  \begin{itemize}
  \tightlist
  \item
    Focuses computation on high-probability regions
  \item
    Maintains manageable memory and time per step
  \end{itemize}
\item
  Challenges:

  \begin{itemize}
  \tightlist
  \item
    Approximation error
  \item
    Sensitive to transition and observation model assumptions
  \end{itemize}
\item
  Used when exact inference is computationally impractical
\item
  Real-world application: robot localization, speech recognition
\end{itemize}
\end{frame}

\begin{frame}{DBN to Represent Changing Model}
\protect\phantomsection\label{dbn-to-represent-changing-model}
\begin{itemize}
\tightlist
\item
  We can model the fact that the system can change over time

  \begin{itemize}
  \tightlist
  \item
    Transient failure: a sensor reads wrong measures
  \item
    Persistent failure model: we can model it with additional variables
    (e.g., \(Sensor Broken\))
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DBN: Inference}
\protect\phantomsection\label{dbn-inference}
\begin{itemize}
\tightlist
\item
  We can unroll the DBN and get a BayesNet and then perform exact or
  approximate inference with the known methods (e.g., MCMC)
\end{itemize}
\end{frame}

  \begin{itemize}

    posterior distribution
  \end{itemize}
