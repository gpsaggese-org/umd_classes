\begin{frame}{Estimating Out-Of-Sample Error with One Point}
\protect\phantomsection\label{estimating-out-of-sample-error-with-one-point}
\begin{itemize}
\item
  Pick a \textbf{\textcolor{red}{single out-of-sample point}}
  \((\vx', y)\)
\item
  The error of the model \(h\) is: \[
  E_{val}(h) = e(h(\vx'), y)
  \] where the error can be:

  \begin{itemize}
  \tightlist
  \item
    Squared error \((h(\vx) - y)^2\)
  \item
    Binary error \(I[h(\vx) - y]\)
  \item
    \ldots{}
  \end{itemize}
\item
  The \textbf{\textcolor{darkgreen}{error on an out-of-sample point}} is
  an \textbf{\textcolor{blue}{unbiased estimate}} of \(E_{out}\), since:
  \[
  \EE[E_{val}(h)] = \EE[e(h(\vx), y)] = E_{out}
  \]
\item
  The quality of the estimate depends on the standard error
  \(\VV[e(h(\vx), y)]\), which in an unknown value
\end{itemize}
\end{frame}

\begin{frame}{Estimating Out-Of-Sample Error with \(K\) Points}
\protect\phantomsection\label{estimating-out-of-sample-error-with-k-points}
\begin{itemize}
\item
  To improve the estimate, use a validation set with \(K\) points
  \((\vx_1, y_1), ... , (\vx_K, y_K)\) drawn IID
\item
  Compute the \textbf{\textcolor{red}{error on the validation set}} as:
  \[
  E_{val}(h) = \frac{1}{K} \sum_{i=1}^K e(h(\vx_i), y_i)
  \]
\item
  The validation error is an
  \textbf{\textcolor{blue}{unbiased estimate}} of out-of-sample error
  since: \[
  \EE[E_{val}(h)] = E_{out}(h)
  \]
\item
  The standard error is \(\sqrt{K}\) smaller: \begingroup \small
  \begin{alignat*}{2}
  \VV[E_{val}(h)]
  & = \frac{1}{K^2} \sum_i \VV[e(h(\vx_i), y_i)] + \text{covariances}
  \\
  &\text{(covariances are 0 because $\vx_i$ independent)}
  \\
  &= \frac{1}{K^2} \sum_i \VV[e(h(\vx_i), y_i)]
  \\
  &= \frac{K \sigma^2}{K^2} = \frac{\sigma^2}{K}
  \\
  \end{alignat*}
  \endgroup
\end{itemize}
\end{frame}

\begin{frame}{Trade-Off Between Training and Validation Set}
\protect\phantomsection\label{trade-off-between-training-and-validation-set}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: you have finite amount of data
  points \(N\)
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\[
  \begin{aligned}
  & D_{val} = \{K \text{ points}\} \\
  & D_{train} = \{N - K \text{ points}\} \\
  \end{aligned}
  \]

\begin{itemize}
\tightlist
\item
  You know that: \[
  \begin{aligned}
  \uparrow K
  & \implies |E_{val} - E_{out}| \downarrow \text{ (most reliable estimate)}
  \\
  & \implies \downarrow N - K \implies E_{in}, E_{out} \uparrow \text{ (worse model)}
  \\
  \end{aligned}
  \]

  \begin{itemize}
  \tightlist
  \item
    If \(K\) is too big, you get a
    \textbf{\textcolor{darkgreen}{reliable estimate of a bad number}}!
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L05.3.Train_Validation_Set_Trade_Off.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    Rule of thumb: 70-30 or 80-20 split between train and validation
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Error From VC Analysis vs Learn-Validation Approach}
\protect\phantomsection\label{error-from-vc-analysis-vs-learn-validation-approach}
\begin{itemize}
\item
  In general: \[
  E_{out}(h) = E_{in}(h) + \text{generalization error}
  \]
\item
  \textbf{\textcolor{red}{VC analysis}}

  \begin{itemize}
  \tightlist
  \item
    Estimates generalization error as ``overfit penalty'' in terms of
    hypothesis set complexity
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Learn-validation}}

  \begin{itemize}
  \tightlist
  \item
    Estimates \(E_{out}\) directly by holding out data as validation
    set: \[
    E_{val} \approx E_{out}
    \]
  \item
    Use learn-validation approach at different points of the modeling
    flow:

    \begin{itemize}
    \tightlist
    \item
      For selecting hyperparameters (validation set)
    \item
      To estimate final performance (test set)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Reusing Validation / Test Set for Training}
\protect\phantomsection\label{reusing-validation-test-set-for-training}
\begin{itemize}
\item
  Any data used for learning (e.g., model training or model selection)

  \begin{itemize}
  \tightlist
  \item
    Is biased and optimistic
  \item
    Cannot used to assess the learned model
  \end{itemize}
\item
  \textbf{\textcolor{red}{Never use}} \(D_{val}\) / \(D_{test}\) for
  training or \(D_{train}\) for evaluation
\item
  After research is done, all available data can be used to learn
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Algorithm}}

  \begin{enumerate}
  \tightlist
  \item
    Train with \(N - K\) points to learn \(g^-\)
  \item
    Use \(K\) points to compute \(E_{val}[g^-]\) estimating
    \(E_{out}[g^-]\)
  \item
    Once the model form is finalized, use all \(N\) data points
    (including validation, test set) to re-train to get \(g\)

    \begin{itemize}
    \tightlist
    \item
      \(E_{val}[g] < E_{val}[g^-]\) since \(g\) is learned on a larger
      data set than \(g^-\) and is thus better
    \end{itemize}
  \item
    Deliver to customers:

    \begin{itemize}
    \tightlist
    \item
      Final hypothesis \(g\)
    \item
      Upper bound of out-of-sample performance \(E_{val}[g^-]\)
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{column}

\begin{frame}{Cross-validation}
\protect\phantomsection\label{cross-validation}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Divide dataset}} into \(K\) folds, each with
  \(\frac{N}{K}\) samples

  \begin{itemize}
  \tightlist
  \item
    Each fold should reflect dataset statistics
  \item
    E.g., stratified sampling
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Do $K$ iterations}} \(i = 1, ..., K\)
\item
  In \(i\)-th iteration

  \begin{itemize}
  \tightlist
  \item
    Train on all folds except \(i\) using \(\frac{K - 1}{K}N\) points
    and get \(g^{(-i)}(\vx)\)
  \item
    Validate on \(i\)-th fold using \(\frac{N}{K}\) points to compute \[
    E_{val}^{(i)} = E_{val}[g^{(-i)}(\vx)]
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Average}} \(K\) error rates and compute
  bounds: \[
  E_{val} = \frac{1}{K} \sum_i E_{val}^{(i)}
  \]
\end{itemize}
\end{column}

\begin{frame}{Repeated Cross-Validation}

  \item

  \item

  \end{itemize}

\begin{frame}{Leave-One-Out Cross-Validation}
\protect\phantomsection\label{leave-one-out-cross-validation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Leave-one-out}} (LOO) cross-validation

  \begin{itemize}
  \tightlist
  \item
    There are \(N\) training sessions
  \item
    Each session trains on \(N - 1\) points and validates on 1 point
  \item
    Equivalent to ``\(N\)-fold cross-validation'' where \(N\) is the
    number of examples in the dataset
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Train}}

  \begin{itemize}
  \tightlist
  \item
    For \(i\)-th fold, \(N - 1\) samples for training
    \(\implies g^{(-i)} \approx g\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Validate}}

  \begin{itemize}
  \tightlist
  \item
    Estimate the validation error on a single point (bad): \[
    E_{val}[g^{(-i)}] = e(g^{(-i)}(\vx_i), y_i) \not\approx E_{out}[g^{(-i)}]
    \]
  \item
    Average \(E_{val}\) over the points as estimate of \(E_{out}\)
    (good): \[
    E_{val}
    = \frac{1}{N} \sum_{i=1}^K E_{val}[g^{(-i)}]
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Leave-One-Out Cross-Validation: Pros and Cons}
\protect\phantomsection\label{leave-one-out-cross-validation-pros-and-cons}
\begin{itemize}

  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item

  \item

    \begin{itemize}
    \tightlist
    \item

\begin{frame}{Bootstrap}
\protect\phantomsection\label{bootstrap}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Pick \(N\) samples with replacement from a data set with \(N\)
    instances to build training set
  \item
    Pick the elements never chosen to build the test set (``out-of-bag''
    samples)
  \item
    Training set contains 63.2\% of all the samples, 36.8\% in the test
    set
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Works for small data sets since it ``expands'' the data
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Not flexible
  \item
    Smaller percentage of instances are used for training set than
    10-fold cross validation
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap: Problem}
\protect\phantomsection\label{bootstrap-problem}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Assumption}}

  \begin{itemize}
  \tightlist
  \item
    Given \(X \sim F\)
  \item
    Draw \(n\) IID samples \(X_i\) from \(F\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Goal}}:

  \begin{itemize}
  \tightlist
  \item
    Consider a sampling statistic \(T = g(X_1, ..., X_n)\)

    \begin{itemize}
    \tightlist
    \item
      E.g., mean, median, OLS coefficients, \ldots{}
    \end{itemize}
  \item
    You want to:

    \begin{itemize}
    \tightlist
    \item
      Estimate distribution of \(T\), e.g., \(F_T(x), f_T(x)\)
    \item
      Estimate a statistic of \(T\), e.g., mean, std err
    \item
      Construct confidence intervals, e.g., \(\mu \pm \epsilon\)
    \item
      Calculate standard errors, e.g., \(\sigma(\hat{T}_n)\)
    \item
      Hypothesis testing
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap Procedure: Algorithm}
\protect\phantomsection\label{bootstrap-procedure-algorithm}
\begin{itemize}
\item
  Use observed data \(X_1, ..., X_n\) to construct estimated population
  distribution \(\hat{F}_n\)
\item
  Repeat \(B\) times:

  \begin{itemize}
  \tightlist
  \item
    Draw \(n\) samples with replacement from estimated population
    distribution \(\hat{F}_n\)
  \item
    Sampling with replacement from \(X_i\) equals sampling from
    \(\hat{F}_n\)
  \item
    Compute sample statistics from \(n\) samples:
    \(T^{(i)} = g(X^{(i)}_1, ..., X^{(i)}_n)\)
  \end{itemize}
\item
  Use \(B\) samples of \(T^{(i)}\) to estimate empirical distribution
  \(\hat{T}\)
\item
  Compute statistics (e.g., confidence interval, standard error) of
  \(T\) from empirical distribution of \(\hat{T}\)
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap: Pros}
\protect\phantomsection\label{bootstrap-pros}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Tremendously useful tool}}

  \begin{itemize}
  \tightlist
  \item
    Fewer assumptions

    \begin{itemize}
    \tightlist
    \item
      No need for simplifying assumptions for closed formulas
    \item
      Data doesn't need to be Gaussian
    \end{itemize}
  \item
    Generality

    \begin{itemize}
    \tightlist
    \item
      Applies to any sample statistics, even non-linear ones (e.g.,
      median)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Math $\to$ simulation}}

  \begin{itemize}
  \tightlist
  \item
    Bootstrap is a non-parametric method
  \item
    Does not rely on large sample sizes, e.g., Central Limit Theorem
    (CLT) / Law Large Numbers (LLN)
  \item
    Bootstrap frees data scientists from complex math, approximations,
    and asymptotics
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap: Example of Die Rolls}
\protect\phantomsection\label{bootstrap-example-of-die-rolls}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Compute distribution of sum of rolling a die 50 times
    \[Y = \sum_{i=1}^{50} X_i = g(X_1, ..., X_{50})\]
  \item
    Sample statistics similar to sample mean, but can be anything
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Solution}}

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{By math}}

    \begin{itemize}
    \tightlist
    \item
      If you know PMF of the die, compute distribution using mathematics
    \item
      Theorem of lazy statistician for mean, variance
    \item
      Compute PDF of \(Y\) by convolving PDFs
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{By sampling}} (real or simulated)

    \begin{itemize}
    \tightlist
    \item
      Roll die 50 times
    \item
      Compute sample statistic
    \item
      Repeat procedure
    \item
      Plot approximate distribution of \(Y\)
    \end{itemize}
  \item
    If only 50 samples of die are known \(\to\)
    \textbf{\textcolor{cyan}{bootstrap}}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Bootstrap of the Median: Pseudo-Code}
\protect\phantomsection\label{bootstrap-of-the-median-pseudo-code}
\begingroup \small

\begin{verbatim}
def bootstrap_median(x, n_boot):
    # Compute n_boot sample statistics.
    median_boot = [0.0] * n_boot
    for i in range(n_boot):
        # Sample with replacement.
        x_star = sample with replacements from x
        # Compute median for bootstrapped samples.
        median_boot[i] = median(x_star)
    # Compute mean and std err from approximation of sample statistics.
    m_median = numpy.mean(median_boot)
    se_median = numpy.std(median_boot)
    return m_median, se_median
\end{verbatim}

\endgroup
\end{frame}

\begin{frame}{Bootstrap for variance of sample statistics: explanation}
\protect\phantomsection\label{bootstrap-for-variance-of-sample-statistics-explanation}
\begin{itemize}
\tightlist
\item
  Under the hypotheses of bootstrap:

  \begin{itemize}
  \tightlist
  \item
    \(X \sim F\)
  \item
    \(n\) samples \(X_i\) IID from \(F\)
  \item
    Statistic \(T = g(X_1, ..., X_n)\)
  \item
    Compute \(\VV_F[T]\) (variance of sample statistics), where \(F\) is
    unknown
  \end{itemize}
\item
  \textbf{\textcolor{red}{First approximation}}

  \begin{itemize}
  \tightlist
  \item
    Only samples from \(F\) available
  \item
    Approximate \(F\) with \(\hat{F}\), as empirical CDF \(\hat{F}\)
    converges to \(F\) \[\VV_F[T] \approx \VV_{\hat{F}}[T]\]
  \item
    Approximation

    \begin{itemize}
    \tightlist
    \item
      Not small
    \item
      Depends on sample size and shape of \(F\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Second approximation}}

  \begin{itemize}
  \tightlist
  \item
    Exact \(\hat{F}\) may lack closed formula for \(\VV_{\hat{F}}[T]\)
  \item
    Use LLN and simulation to approximate variance: \[
    \VV_{\hat{F}}[T]
    \approx v_{boot}
    = \frac{1}{B} \sum_i (T_i - \overline{T}) ^ 2
    \]
  \item
    Approximation size reduced by increasing \(B\)
  \item
    \(v_{boot} \to \VV_{\hat{F}}[T]\) as \(B \to \infty\)
  \end{itemize}
\end{itemize}
\end{frame}
