\begin{frame}{Bayesian Networks: Semantics}
\protect\phantomsection\label{bayesian-networks-semantics}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: Two equivalent semantic
  interpretations of a Bayesian Network:

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Joint Distribution View}}

    \begin{itemize}
    \tightlist
    \item
      The network encodes the \emph{joint probability distribution} over
      all variables
    \item
      Computed as the product of local conditional probabilities: \[
      P(X_1, ..., X_n) = \prod_{i=1}^{n} P(X_i \mid \text{Parents}(X_i))
      \]
    \item
      Useful for constructing models and understanding overall behavior
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Conditional Independence View}}

    \begin{itemize}
    \tightlist
    \item
      The structure encodes \emph{conditional independency} between
      variables
    \item
      Useful for inference and reasoning
    \item
      A variable is conditionally independent of its non-descendants
      given its parents
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Chain Rule for a Joint Distribution}
\protect\phantomsection\label{chain-rule-for-a-joint-distribution}
\begin{itemize}
\tightlist
\item
  \textbf{Theorem}: A joint distribution can always be expressed using
  the chain rule for any:

  \begin{itemize}
  \tightlist
  \item
    Subset of its random variables
  \item
    Ordering of the random variables
  \end{itemize}
\item
  \textbf{Proof}
\end{itemize}

\begin{enumerate}
\item
  \textbf{Express one variable} conditionally to the remaining ones \[
  \Pr(\textcolor{blue}{x_1, ..., x_{n-1}}, \textcolor{red}{x_n})
  = \Pr(\textcolor{red}{x_n} | \textcolor{blue}{x_{n-1}, ..., x_1}) \Pr(\textcolor{blue}{x_{n-1}, ..., x_1})
  \]
\item
  Apply the same formula \textbf{recursively}, until you get an
  unconditional probability \begingroup \small
  \begin{align*}
  & \Pr(\gray{x_1}, \violet{x_2}, ..., \teal{x_{n-2}}, \olive{x_{n-1}}, \orange{x_n}) \\
  & = \Pr(\orange{x_n} | x_{n-1}, ..., x_1) \Pr(x_{n-1}, ..., x_1) \\
  & = \Pr(\orange{x_n} | x_{n-1}, ..., x_1)
  \Pr(\olive{x_{n-1}} | x_{n-2}, ..., x_1) \Pr(x_{n-2}, ..., x_1) \\
  & ... \\
  & = \Pr(\orange{x_n} | x_{n-1}, ..., x_1)
  \Pr(\olive{x_{n-1}} | x_{n-2}, ..., x_1) \Pr(\teal{x_{n-2}} | x_{n-3}, ..., x_1)
  ...
  \Pr(\violet{x_2} | x_1) \Pr(\gray{x_1}) \\
  & = \prod_{i=1}^n \Pr(x_i | x_{i-1}, ..., x_1) \\
  \end{align*}
  \endgroup
\end{enumerate}
\end{frame}

\begin{frame}{Evaluate a Bayesian Network}
\protect\phantomsection\label{evaluate-a-bayesian-network}
\begin{enumerate}
\tightlist
\item
  Sort the nodes in topological order

  \begin{itemize}
  \tightlist
  \item
    There are several orderings consistent with the directed graph
    structure
  \end{itemize}
\item
  Use the chain rule with the topological ordering: \[
  \Pr(X_1, ..., X_n) = \prod_{i=1}^n \Pr(X_i | X_{i-1}, ..., X_1)
  \]
\item
  Since the probability of each node is conditionally independent of all
  its predecessors given its parents \[
  \Pr(X_i | X_{i-1}, ..., X_1) = \Pr(X_i | Parents(X_i))
  \]
\item
  Express the joint probability in terms of the Conditional Probability
  Tables (CPTs): \[
  \Pr(X_1, ..., X_n) = \prod_{i=1}^n \Pr(X_i | Parents(X_i))
  \]
\end{enumerate}
\end{frame}

\tightlist
\item
  \textbf{\textcolor{blue}{Solution}}
\item
  Compute the probability as a product of conditional probabilities from
  the Bayesian Network: \begin{align*}
  & \Pr(JohnCalls, MaryCalls, Alarm, \lnot Burglary, \lnot Earthquake) \\
  & = \Pr(JohnCalls|Alarm) \cdot \\
  & \hspace{1cm} \Pr(MaryCalls|Alarm) \cdot \\
  & \hspace{1cm} \Pr(Alarm|\lnot Burglary \land \lnot Earthquake) \cdot \\
  & \hspace{1cm} \Pr(\lnot Burglary) \cdot \Pr(\lnot Earthquake) \\
  \end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Constructing a Bayesian Network}
\protect\phantomsection\label{constructing-a-bayesian-network}
\begin{enumerate}
\tightlist
\item
  \textbf{\textcolor{red}{Gather domain knowledge}}

  \begin{itemize}
  \tightlist
  \item
    List all relevant random variables necessary to describe the system
  \item
    Identify key variables and their potential interactions
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Order the nodes}} according to
  cause-effects dependencies

  \begin{itemize}
  \tightlist
  \item
    The Bayesian network will be minimal
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pick the minimum set of parents}}
  \(Parents(X_i)\) for each node

  \begin{itemize}
  \tightlist
  \item
    Add edges to represent the dependencies
  \item
    Avoid redundant connections
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Estimate the conditional probability}}
  \(\Pr(X_i | Parents(X_i))\) for each node

  \begin{itemize}
  \tightlist
  \item
    Gather data or expert opinion
  \item
    Use statistical techniques if necessary
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Validate the model}}

  \begin{itemize}
  \tightlist
  \item
    Have domain experts review it
  \item
    Ensure that the network is a Directed Acyclic Graph (DAG)
  \item
    Test the network by predicting known outcomes and comparing with
    actual data
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Bayesian Networks: Properties}
\protect\phantomsection\label{bayesian-networks-properties}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Fact}}: Bayesian networks are a representation
  with several interesting properties

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Complete}}

    \begin{itemize}
    \tightlist
    \item
      Encode all information in a joint probability
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Consistent}} (non-redundant)

    \begin{itemize}
    \tightlist
    \item
      In a Bayesian network, there are no redundant probability values
    \item
      One (e.g., a domain expert) can't create a Bayesian network
      violating probability axioms
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Compact}} (locally structured, sparse)

    \begin{itemize}
    \tightlist
    \item
      Each subcomponent interacts directly with a limited number of
      other components
    \item
      Typically yields linear (not exponential) growth in complexity
    \item
      Sometimes we ignore real-world dependency to keep the graph simple
    \end{itemize}
  \end{itemize}
\item
  In \textbf{\textcolor{cyan}{fully connected systems}} each variable is
  influenced by all others

  \begin{itemize}
  \tightlist
  \item
    The Bayesian network is sparse but has the same complexity as the
    joint probability
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How Can a Node Be Influenced by Its Children?}
\protect\phantomsection\label{how-can-a-node-be-influenced-by-its-children}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{descendant can influence its ancestor}}
  indirectly through \emph{``explaining away''}

  \begin{itemize}
  \tightlist
  \item
    Evidence about the descendant can change what you believe about the
    ancestor through dependent paths
  \item
    Information flows both ways in Bayesian networks
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.75\linewidth}
\begin{itemize}
\tightlist
\item

  \item

  \item
    This evidence increases the probability of either causes \(Rain\) or
    \(Sprinkler\)
  \item
    If you find out that the \(Sprinkler\) was on, this ``explains
    away'' the \(WetGrass\), and the probability of \(Rain\) goes down
  \item
    The evidence from a descendant \(WetGrass\) can update your belief
    about an ancestor \(Rain\)
  \end{itemize}
\end{itemize}

\begin{frame}{Deterministic Nodes}
\protect\phantomsection\label{deterministic-nodes}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: \emph{deterministic nodes} have
  values specified by their parents, without uncertainty, e.g.,

  \begin{itemize}
  \tightlist
  \item
    A logical relationship:

    \begin{itemize}
    \tightlist
    \item
      Useful when a condition is met if any of the sub-conditions are
      true
    \item
      \(IsNorthAmerican = IsCanadian \lor IsUS \lor IsMexican\)
    \end{itemize}
  \item
    A numerical relationship:

    \begin{itemize}
    \tightlist
    \item
      \(BestPrice = \min(Price_i)\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Note}}

  \begin{itemize}
  \tightlist
  \item
    Deterministic nodes do not involve randomness or probability
  \item
    Often used in models to simplify relationships and computations
  \end{itemize}
\item
  Noisy Logical Relationships
\end{itemize}
\end{frame}

\begin{frame}{Noisy Logic Relationships}
\protect\phantomsection\label{noisy-logic-relationships}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: \emph{noisy logical
  relationships} (e.g., noisy-OR, noisy-MAX):

  \begin{itemize}
  \tightlist
  \item
    Are a probabilistic version of a logical relationship
  \item
    Can be simpler to describe given \(k\) parents
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}
\item
  In \emph{propositional logic}, relationships are expressed like
  \(Fever \iff Cold \lor Flu \lor Malaria\)
\item
  In \emph{Bayesian networks}, the assumptions are:

  \begin{enumerate}
  \tightlist
  \item
    All the possible causes are listed (you can use a leak node for
    ``miscellaneous causes'')
  \item
    There is uncertainty about the parents causing the child node
  \item
    The probabilities of the parents are independent
  \end{enumerate}

  \begin{itemize}
  \tightlist
  \item
    Under these assumptions: \begin{align*}
    &\Pr(Fever | parents(Fever)) \\
    & \hspace{1cm} = 1 - \Pr(\lnot Fever | Cold, \lnot Flu, \lnot Malaria) \cdot \\
    & \hspace{1cm} \Pr(\lnot Fever | \lnot Cold, Flu, \lnot Malaria) \cdot \\
    & \hspace{1cm} \Pr(\lnot Fever | \lnot Cold, \lnot Flu, Malaria)
    \end{align*}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Context-specific Independence}
\protect\phantomsection\label{context-specific-independence}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: a variable exhibits
  \emph{context-specific independence} if it is conditionally
  independent of its parents given certain values of others
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \tightlist
  \item
    \(Damage\) occurs depending on the \(Ruggedness\) of your car and
    whether an \(Accident\) occurred in that period:
    \begingroup \scriptsize
    $$
    \Pr(Damage \mid Ruggedness, Accident) =
    \begin{cases}
    d_1 & \text{if } Accident = True \\
    d_2(Ruggedness) & \text{if } Accident = False
    \end{cases}
    $$
    \endgroup where \(d1\) and \(d2\) are distributions
  \end{itemize}
\end{itemize}
\end{frame}

  \tightlist
  \item
    E.g., height, mass, temperature, money
  \end{itemize}

  \tightlist

    \end{itemize}

    Continuous variables

    \begin{itemize}
    \tightlist
    \item
      Families of probability density functions (e.g., Gaussian

    \item
      Non-parametric PDFs
    \end{itemize}
  \end{enumerate}

  \textbf{\textcolor{blue}{Definition}}: \emph{hybrid Bayesian networks}
  mix discrete and continuous variables, e.g.,

  \begin{itemize}
  \tightlist
  \item
    \_``A customer buys a number of apples (discrete) depending on its

  \item
    \emph{``Decide annual premium to charge to insure a vehicle based on

\begin{frame}{Exact Inference in Bayesian Networks}
\protect\phantomsection\label{exact-inference-in-bayesian-networks}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}

  \begin{itemize}
  \tightlist
  \item
    Compute the posterior \(P(X|\vE=\ve)\) for the query variable \(X\)
    given evidence \(\ve\), where:

    \begin{itemize}
    \tightlist
    \item
      Query variable \(X\)
    \item
      Evidence variables \(\vE = \{E_1, \dots, E_m\}\)
    \item
      Hidden variables \(\vY = \{Y_1, \dots, Y_\ell\}\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inference by Enumeration}}

  \begin{itemize}
  \tightlist
  \item
    Use the full joint distribution and sum over all hidden variables:
    \[P(X|e) = \alpha \sum_Y P(X,e,Y)\]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Variable Elimination}}

  \begin{itemize}
  \tightlist
  \item
    Improves efficiency by caching intermediate results
  \item
    Eliminates variables systematically to avoid redundant sums
  \item
    Removes irrelevant variables

    \begin{itemize}
    \tightlist
    \item
      Variables that are not ancestors of the query or evidence can be
      ignored
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Problems}}

  \begin{itemize}
  \tightlist
  \item
    Exact inference is efficient \(O(n)\) for trees, but intractable
    \(O(2^n)\) in general
  \item
    It does not work for continuous variables
  \item
    Basis for approximate methods when exact methods are impractical
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Exact Inference in Bayesian Networks: Example}
\protect\phantomsection\label{exact-inference-in-bayesian-networks-example}
\begin{itemize}
\item
  You get a call from both John and Mary, what is the probability of the
  burglary? \[P(Burglary | JohnCalls=True, MaryCalls=True)\]
\item
  A conditional probability can be computed summing terms from the full
  joint distribution
  \[\Pr(X | \ve) = \alpha \Pr(X, \ve) = \alpha \sum_y \Pr(X, \ve, \vy)\]
\item
  Terms of the joint distribution can be written as products of
  conditional probabilities from the Bayesian network \[
  \Pr(b | j, m)
  = \alpha \Pr(B, j, m)
  = \alpha \sum_e \sum_a \Pr(B, j, m, e, a)
  \]
\item
  Then the joint probability is written in terms of CPTs of the Bayesian
  network \[
  \Pr(b | j, m)
  = \alpha \sum_e \sum_a \Pr(b) \Pr(e) \Pr(a | b, e) \Pr(j | a) \Pr(m | a)
  \]
\end{itemize}
\end{frame}

  \textbf{\textcolor{red}{Monte Carlo algorithms}} are randomized
  sampling algorithms used to estimate quantities that are difficult to
  calculate exactly

  \begin{itemize}
  \tightlist
  \item
    E.g., samples from the posterior probability of a Bayes network
  \end{itemize}

  \textbf{\textcolor{darkgreen}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    The accuracy of the approximation depends on the number of samples

  \item

    with enough samples
  \item
    Is used in many branches of science
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Difficult to understand how the variables interact
  \item
    Computationally intensive
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sampling from Arbitrary Distributions}
\protect\phantomsection\label{sampling-from-arbitrary-distributions}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: Sample from a discrete or continuous
  probability distribution
\item
  \textbf{\textcolor{blue}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Start with uniform random numbers \(r \in [0, 1]\)
  \item
    Construct CDF (cumulative distribution function) \(F(x)\)

    \begin{itemize}
    \tightlist
    \item
      \(F(x) = \Pr(X \leq x)\)
    \end{itemize}
  \item
    For discrete distributions:

    \begin{itemize}
    \tightlist
    \item
      Create table of outcomes and cumulative probabilities
    \item
      Find smallest outcome where \(F(x) > r\)
    \end{itemize}
  \item
    For continuous distributions:

    \begin{itemize}
    \tightlist
    \item
      Use inverse transform: \(x = F^{-1}(r)\), e.g., \[
      F(x) = 1 - e^{-\lambda x} \to x = F^{-1}(r) = -\frac{1}{\lambda} \ln(1 - r)
      \]
    \item
      If \(F^{-1}\) has no closed form, use numerical methods
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Consistency of Sampling}
\protect\phantomsection\label{consistency-of-sampling}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Consistency of estimation}}: distribution from
  prior sampling converges to true probability as \(N \to \infty\)
\item
  If \(N_{PS}\) is the number of times event \(x_1, ..., x_n\) occurs:
  \[
  \lim_{N \to \infty} \frac{N_{PS}(x_1, ..., x_n)}{N}
  = \Pr(x_1, ..., x_n)
  \]
\item
  Estimate probability using: \[
  \Pr(x_1, ..., x_n) \approx \frac{N_{PS}(x_1, ..., x_n)}{N}
  \] converges with rate \(\frac{1}{\sqrt{N}}\)
\end{itemize}
\end{frame}

\begin{frame}{Rejection Sampling}
\protect\phantomsection\label{rejection-sampling}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Rejection sampling}} is a method for sampling
  from a hard-to-sample distribution
\item
  \textbf{\textcolor{darkgreen}{Goal}}: Compute \(\Pr(X=x| E=e)\) when
  evidence \(e\) is rare
\item
  \textbf{\textcolor{blue}{Algorithm}}:

  \begin{enumerate}
  \tightlist
  \item
    Generate samples from the prior distribution

    \begin{itemize}
    \tightlist
    \item
      Estimate \(\Pr(x, e)\)
    \end{itemize}
  \item
    Reject samples not matching evidence, i.e., \(X \land E \neq e\)

    \begin{itemize}
    \tightlist
    \item
      Remaining samples \(X \land E=e\) estimate \(\Pr(X, E=e)\)
    \end{itemize}
  \item
    Count occurrences of \(X=x\) in remaining samples \(X \land E=e\)

    \begin{itemize}
    \tightlist
    \item
      Estimate \(\Pr(X=x | E=e)\)
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Rejection Sampling: Garden World Example}
\protect\phantomsection\label{rejection-sampling-garden-world-example}
\begin{itemize}
\tightlist
\item
  You want to estimate \(\Pr(Rain | Sprinkler = T)\)
\item
  Sample 100 times

  \begin{itemize}
  \tightlist
  \item
    You get 73 samples with \(\lnot Sprinkler\) and they are rejected
  \item
    You are left with 27 samples with \(Sprinkler\)
  \item
    Out of them only 8 have \(Rain\) and 19 have \(\lnot Rain\)
  \end{itemize}

  \]

\begin{frame}{Rejection Sampling: Pros and Cons}
\protect\phantomsection\label{rejection-sampling-pros-and-cons}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Consistent estimate

    \begin{itemize}
    \tightlist
    \item
      I.e., converges to true value as number of samples increases
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Many samples are rejected, depending on rarity of \(\Pr(E=e)\)
  \item
    Fraction of samples matching evidence \(e\) decreases exponentially
    with more evidence variables

    \begin{itemize}
    \tightlist
    \item
      Curse of dimensionality
    \item
      Not suitable for complex systems
    \end{itemize}
  \item
    Difficult with continuous variables

    \begin{itemize}
    \tightlist
    \item
      E.g., \(\Pr(E=e)\) is theoretically 0 due to limited
      floating-point precision
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Importance Sampling}
\protect\phantomsection\label{importance-sampling}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Importance sampling}}

  \begin{itemize}
  \tightlist
  \item
    Draw samples from ``easier'' distribution \(Q(X)\)
  \item
    Weight each sample by importance weight \(w = \frac{\Pr(X)}{Q(X)}\)
  \item
    Estimate probability by averaging weighted samples: \[
    E[f(X)] \approx \frac{1}{N} \sum_{i=1}^N w_i f(X_i)
    \]
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L06.2.Importance_Sampling.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Estimate \(\Pr(A | E=e)\) where \(E\) is rare
  \item
    Standard sampling might miss \(E=e\)
  \item
    Importance sampling focuses samples near \(E=e\) with reweighting
  \item
    Analogy: Rebalance biased survey by giving underrepresented groups
    higher weights
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Improves inference efficiency over rejection sampling
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Markov Chain Construction}
\protect\phantomsection\label{markov-chain-construction}
\begin{itemize}
\item
  A \textbf{\textcolor{red}{Markov chain}} is a ``random walk'' through
  states, where the future depends only on the present

  \begin{itemize}
  \tightlist
  \item
    \emph{Sequence of states}:
    \(\vx^{(0)}, \vx^{(1)}, \vx^{(2)}, \dots\)
  \item
    \emph{Initial state}: starting configuration
  \item
    \emph{Transition probabilities}: \(\Pr(\vx \to \vx')\)
  \item
    After \(t\) steps, distribution is \(\pi_t(\vx)\)
  \item
    When \(\pi_t(\vx) = \pi_{t+1}(\vx)\), chain reaches stationary
    distribution
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Transition operator}}: moves from one
  state to another:

  \begin{itemize}
  \tightlist
  \item
    Gibbs sampling: resample one variable given its Markov blanket
  \item
    Metropolis--Hastings: propose a new state, then accept/reject based
    on a probability ratio
  \end{itemize}
\item
  There are algorithms generate a Markov Chain from a Bayesian network
\item
  Under certain conditions:

  \begin{itemize}
  \tightlist
  \item
    Ergodicity: chain can reach any state
  \item
    Aperiodicity: chain does not get stuck in cycles stationary
    distribution equals the
    \textbf{\textcolor{blue}{posterior distribution}} over non-evidence
    variables given evidence
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Markov Chain Monte Carlo: Mixing}
\protect\phantomsection\label{markov-chain-monte-carlo-mixing}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Mixing}} describes how quickly a Markov chain
  forgets its starting point and explores the whole state space
  efficiently

  \begin{itemize}
  \tightlist
  \item
    A well-mixed chain:

    \begin{itemize}
    \tightlist
    \item
      Moves between different high-probability regions often
    \item
      Has low correlation between successive samples
    \end{itemize}
  \item
    Poor mixing:

    \begin{itemize}
    \tightlist
    \item
      Chain gets stuck in one mode for a long time
    \item
      Leads to biased estimates and high variance
    \end{itemize}
  \end{itemize}
\item
  In practice:

  \begin{itemize}
  \tightlist
  \item
    Discard initial samples as a burn-in period (before convergence)
  \item
    After convergence, collected samples approximate the true posterior
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \tightlist
  \item
    If sampling from a bimodal distribution:

    \begin{itemize}
    \tightlist
    \item
      ``Poor mixing'' means the chain stays in one peak
    \item
      ``Good mixing'' jumps between both peaks to reflect the true
      posterior
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Gibbs Sampling in Bayesian Networks}
\protect\phantomsection\label{gibbs-sampling-in-bayesian-networks}
\begin{itemize}
\item
  Special case of Markov Chain Monte Carlo (MCMC) method that samples
  one variable at a time
\item
  \textbf{\textcolor{red}{Algorithm}}:

  \begin{itemize}
  \tightlist
  \item
    Start with an initial complete assignment to all non-evidence
    variables
  \item
    Keep evidence variables fixed at observed values
  \item
    For each non-evidence variable \(X_i\):

    \begin{itemize}
    \tightlist
    \item
      Sample \(X_i\) from \(P(X_i | \text{MB}(X_i))\) where
      \(\text{MB}(X_i)\) is the Markov blanket, i.e., parents, children,
      spouse of a node
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Weather network: \(P(Cloudy | Sprinkler, Rain, WetGrass)\)
  \item
    Fix \(WetGrass = true\), \(Sprinkler = true\)
  \item
    Sample \(Cloudy\) and \(Rain\) iteratively
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Simple to implement for any Bayesian network
  \item
    Handles large, complex graphs with local updates
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Can mix slowly if variables are highly correlated
  \item
    May need many samples for accurate estimates
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Metropolis--Hastings Sampling}
\protect\phantomsection\label{metropolishastings-sampling}
\begin{itemize}
\item
  More general Markov Chain Monte Carlo method than Gibbs sampling
\item
  \textbf{\textcolor{red}{Algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Start at a current state \(\vx\)
  \item
    Propose a new state \(\vx'\) from a proposal distribution
    \(q(x' | x)\), e.g.,

    \begin{itemize}
    \tightlist
    \item
      With 95\% probability Gibbs sampling
    \item
      Otherwise use importance sampling
    \end{itemize}
  \item
    Compute the acceptance probability:

    \begin{itemize}
    \tightlist
    \item
      \(A(x, x') = \min(1, \frac{\pi(x')q(x|x')}{\pi(x)q(x'|x)})\)
    \end{itemize}
  \item
    Move to \(x'\) with probability \(A(x,x')\), otherwise stay at \(x\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Intuition}}

  \begin{itemize}
  \tightlist
  \item
    Propose local moves
  \item
    Accept if they lead to higher probability, or sometimes accept
    lower-probability states to explore
  \item
    Balances exploration and exploitation to avoid getting stuck in
    local modes
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Very flexible: works with any proposal distribution
  \item
    Can handle high-dimensional spaces
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    May ``mix'' slowly if proposal steps are poorly chosen
  \item
    Requires careful tuning of proposal distribution for efficiency
  \end{itemize}
\end{itemize}
\end{frame}
