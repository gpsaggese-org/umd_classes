\begin{frame}{Causal Discovery: Motivation}
\protect\phantomsection\label{causal-discovery-motivation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: So far we have assumed the causal
  DAG was known

  \begin{itemize}
  \tightlist
  \item
    E.g., given by domain experts
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Questions}}:

  \begin{itemize}
  \tightlist
  \item
    How do we discover the DAG when it is not specified?
  \item
    Can we learn causal structure directly from observational data?
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Why it matters}}

  \begin{itemize}
  \tightlist
  \item
    Observational data is plentiful but ambiguous

    \begin{itemize}
    \tightlist
    \item
      E.g., due to confounding
    \end{itemize}
  \item
    Experiments are expensive, slow, or unethical

    \begin{itemize}
    \tightlist
    \item
      E.g., cannot randomize exposing humans to asbestos
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Solution}}: \emph{Causal discovery} (aka
  structure learning)

  \begin{itemize}
  \tightlist
  \item
    Recover the causal DAG from observational and/or interventional data
  \item
    Extract causal hypotheses that can be tested and refined
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Discovery: Formal Definition}
\protect\phantomsection\label{causal-discovery-formal-definition}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: Given a dataset
  \(\mathcal{D} \sim \Pr(X_1, \dots, X_n)\), \emph{causal discovery} is
  the process of inferring a causal DAG \(G\) (or equivalence class)
  such that:
  \[\Pr(X_1, \dots, X_n) = \prod_{i=1}^n \Pr(X_i | Pa_G(X_i))\] where
  \(Pa_G(X_i)\) are the parents of \(X_i\) in \(G\)
\item
  \textbf{\textcolor{blue}{Examples}}:

  \begin{itemize}
  \tightlist
  \item
    Macroeconomic drivers: from time-series data, infer causal links
    between inflation, unemployment, and interest rates
  \item
    Education and income: estimate whether education causally affects
    earnings (vs pure confounding)
  \item
    Social networks: infer whether friends influence each other's
    purchases or opinions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Markov Equivalence}
\protect\phantomsection\label{markov-equivalence}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Question}}: Can observational data distinguish
  every Directed Acyclic Graph (DAG)?

  \begin{itemize}
  \tightlist
  \item
    No, it cannot
  \item
    Different DAGs can encode identical conditional independencies,
    forming an equivalence class
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Definition}}: \emph{CPDAG} (Completed
  Partially Directed Acyclic Graph) represents a Markov equivalence
  class of Bayesian network DAGs

  \begin{itemize}
  \tightlist
  \item
    A CPDAG includes both directed and undirected edges to capture the
    equivalence class
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Markov equivalent DAGs}} share the same:

  \begin{itemize}
  \tightlist
  \item
    Skeleton (undirected edges between nodes)
  \item
    v-structures (e.g., \(X \to Z \leftarrow Y\))
  \item
    E.g., consider two DAGs, \(A \to B \to C\) and
    \(A \leftarrow B \to C\). Both have the same skeleton and no
    v-structures, hence they are Markov equivalent
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Markov Equivalence: Examples}
\protect\phantomsection\label{markov-equivalence-examples}
\begin{itemize}
\tightlist
\item
  Chain: \(X \to Y \to Z\)

  \begin{itemize}
  \tightlist
  \item
    \emph{``If we intervene on X, Z changes''}
  \end{itemize}
\item
  Reverse chain: \(X \leftarrow Y \leftarrow Z\)

  \begin{itemize}
  \tightlist
  \item
    \emph{``Intervening on X has no effect on Z''}
  \end{itemize}
\item
  Confounder: \(X \leftarrow Y \to Z\)

  \begin{itemize}
  \tightlist
  \item
    \emph{``Intervening on X has no effect on Z''}
  \end{itemize}
\item
  Same observational correlation, different causal implications

  \begin{itemize}
  \tightlist
  \item
    All imply \(X \perp Z | Y\)
  \end{itemize}
\item
  No amount of data can distinguish these structures

  \begin{itemize}
  \tightlist
  \item
    Without intervention data or additional assumptions, we cannot
    distinguish these structures from correlation alone
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Typical Assumptions for Causal Discovery}
\protect\phantomsection\label{typical-assumptions-for-causal-discovery}
\begin{itemize}
\item
  When performing causal discovery, certain assumptions are made:

  \begin{itemize}
  \item
    \textbf{\textcolor{red}{Causal Markov condition}}: each variable is
    independent of its non-descendants given its parents

    \begin{itemize}
    \tightlist
    \item
      Equivalent to: \(\Pr(X) = \prod_i \Pr(X_i | Pa(X_i))\)
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Faithfulness}}: all conditional
    independencies in the data stem from DAG structure, not from
    numerical coincidences
  \item
    \textbf{\textcolor{blue}{Causal sufficiency}}: no hidden common
    causes (confounders)

    \begin{itemize}
    \tightlist
    \item
      All relevant variables are measured
    \end{itemize}
  \end{itemize}
\item
  Violating any assumption leads to incorrect recovered DAGs
\end{itemize}
\end{frame}

\begin{frame}{Identifiability}
\protect\phantomsection\label{identifiability}
\begin{itemize}
\tightlist
\item
  From observational data alone:

  \begin{itemize}
  \tightlist
  \item
    Only the Markov equivalence class (CPDAG) can be identified
  \item
    Edge directions remain ambiguous without functional assumptions
  \end{itemize}
\item
  With stronger functional assumptions, full DAG recovery is possible

  \begin{itemize}
  \tightlist
  \item
    Linear non-Gaussian noise (LiNGAM): asymmetry in noise breaks
    direction ambiguity
  \item
    Additive noise models with nonlinear functions
  \item
    Post-nonlinear causal models
  \end{itemize}
\item
  With interventional data, each intervention cuts incoming edges

  \begin{itemize}
  \tightlist
  \item
    Sufficient interventions can recover the full directed DAG
  \end{itemize}
\item
  \textbf{\textcolor{red}{Rule of Thumb}}: More structural information
  leads to more identifiability
\end{itemize}
\end{frame}

\begin{frame}{Faithfulness}
\protect\phantomsection\label{faithfulness}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Faithfulness assumption}}:

  \begin{itemize}
  \tightlist
  \item
    Independent variables in data should be independent in causal graph,
    and vice versa
  \item
    The statistical relationships are assumed to ``faithfully'' reflect
    the graph structure
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Problem}}: path cancellation leads to
  violation of faithfulness
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Two paths: \(X \to Y\) and \(X \to Z \to Y\)
  \item
    Equal but opposite effects cancel
  \item
    Algorithm incorrectly concludes: no edge between \(X\) and \(Y\)
    (i.e., \(X \perp Y\) from data) though \(X\) causes \(Y\)
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Practical Challenges for Causal Discovery}
\protect\phantomsection\label{practical-challenges-for-causal-discovery}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Sample complexity}}:

  \begin{itemize}
  \tightlist
  \item
    Causal discovery from data is sample-intensive
  \item
    Testing all pairs of variables is \(O(n^2)\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Hidden confounders}}:

  \begin{itemize}
  \tightlist
  \item
    Algorithms assume causal sufficiency (all confounders are measured),
    rarely true
  \item
    Unmeasured variables break core assumptions
  \item
    Create spurious edges (false associations) in the learned graph or
    incorrect edge directions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Selection bias}}:

  \begin{itemize}
  \tightlist
  \item
    Sampling mechanism depends on studied variables, creating artificial
    associations
  \item
    E.g., hospitals see sicker patients (selection bias toward severe
    cases), so treatment and outcome appear more associated than in the
    general population
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Finite-sample faithfulness}}:

  \begin{itemize}
  \tightlist
  \item
    Weak dependencies appear as independencies
  \item
    Statistical power drops with larger conditioning sets
  \item
    ``Absence of evidence'' seen as ``evidence of absence''
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Practical Challenges for Causal Discovery}
\protect\phantomsection\label{practical-challenges-for-causal-discovery-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Temporal information often ignored}}

  \begin{itemize}
  \tightlist
  \item
    Time is one of the strongest causal signals, but most discovery
    algorithms ignore temporal ordering
  \item
    Some algorithms (e.g., Granger-causality) exploit temporal order
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Non-stationarity and heterogeneity}}

  \begin{itemize}
  \tightlist
  \item
    Real-world data rarely satisfies the i.i.d. assumption across
    populations and time
  \item
    Non-stationarity: statistical properties change over time
  \item
    Heterogeneity: different subpopulations have different causal
    structures
  \item
    Both violate the foundational assumptions most discovery algorithms
    rely on
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Computational complexity}}:

  \begin{itemize}
  \tightlist
  \item
    DAG space grows super-exponentially
    \[|\{\text{DAGs over } n \text{ nodes}\}| \approx 2^{O(n^2)}\]
  \item
    PC algorithm: \(O(n^q)\) where \(q\) is maximum graph degree
  \item
    Score-based search is NP-hard in general
  \item
    Practical implication: exact discovery is infeasible beyond small
    graphs
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Using Domain Knowledge as Constraints}
\protect\phantomsection\label{using-domain-knowledge-as-constraints}
\begin{itemize}
\item
  Partial domain knowledge can \emph{drastically} reduce the search
  space for causal discovery algorithms
\item
  Common forms of prior knowledge:

  \begin{itemize}
  \tightlist
  \item
    Forbidden edges: ``variable \(X\) cannot cause variable \(Y\)''

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``age cannot be caused by purchasing behavior''}
    \end{itemize}
  \item
    Required edges: ``we know \(X \to Y\) from physics''
  \item
    Tiered ordering:

    \begin{itemize}
    \tightlist
    \item
      Organize variables into causal levels where lower tiers cannot
      affect higher ones
    \item
      E.g., \emph{demographics \(\to\) behavior \(\to\) outcome}
    \end{itemize}
  \item
    Temporal order: past cannot depend on future
  \end{itemize}
\item
  These constraints are passed to the discovery algorithm as a
  ``knowledge'' object
\item
  \textbf{\textcolor{red}{Pros}}: faster, more accurate, more
  interpretable
\item
  \textbf{\textcolor{blue}{Cons}}: wrong priors inject wrong edges
\end{itemize}
\end{frame}

\begin{frame}{Metrics}
\protect\phantomsection\label{metrics}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Motivation}}:

  \begin{itemize}
  \tightlist
  \item
    Causal discovery algorithms estimate graphs
  \item
    Quantitative metrics assess match with true structure
  \item
    Different errors have different consequences
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Three Categories of Metrics}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Edge-level metrics}: TPR (recall), FDR (false discovery rate),
    F1 score
  \item
    \emph{Structure-level metrics}: Structural Hamming Distance (SHD)
  \item
    \emph{Complexity metrics}: Number of Non-Zero edges (NNZ)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key Distinction}}: Structure vs Parameters

  \begin{itemize}
  \tightlist
  \item
    Structure learning: Right edges and directions?
  \item
    Parameter learning: Correct edge weights?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Edge-Level Metrics}
\protect\phantomsection\label{edge-level-metrics}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{True Positive Rate (TPR)}}:
  \(\text{TPR} = \frac{\text{TP}}{\text{TP} + \text{FN}}\) = fraction of
  true edges recovered

  \begin{itemize}
  \tightlist
  \item
    High TPR: Find most causal relationships
  \item
    Low TPR: Miss many real causal effects
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{False Discovery Rate (FDR)}}:
  \(\text{FDR} = \frac{\text{FP}}{\text{FP} + \text{TP}}\) = fraction of
  predicted edges that are false

  \begin{itemize}
  \tightlist
  \item
    Low FDR: Predictions mostly correct
  \item
    High FDR: Many false positives
  \end{itemize}
\item
  \textbf{\textcolor{blue}{F1 Score}}:
  \(\text{F1} = 2 \cdot \frac{\text{TPR} \cdot (1 - \text{FDR})}{\text{TPR} + (1 - \text{FDR})}\)
  = harmonic mean balancing both
\end{itemize}
\end{frame}

\begin{frame}{Structure-Level and Complexity Metrics}
\protect\phantomsection\label{structure-level-and-complexity-metrics}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Structural Hamming Distance}}:
  \(\text{SHD} = |\text{missing}| + |\text{extra}| + |\text{reversed}|\)

  \begin{itemize}
  \tightlist
  \item
    Counts missing edges, false positives, reversed edge directions
  \item
    Lower SHD is better (0 = perfect recovery)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Complexity metrics}}: Number of Non-Zero
  edges (NNZ)
\item
  \textbf{\textcolor{blue}{Best Practice}}: Report TPR, FDR, F1, SHD
  together

  \begin{itemize}
  \tightlist
  \item
    Different stakeholders prioritize different metrics
  \item
    Scientists want high TPR (find all mechanisms)
  \item
    Practitioners want low FDR (avoid false claims)
  \item
    Report NNZ to contextualize conservative vs liberal predictions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Discovery Algorithm Families: Overview}
\protect\phantomsection\label{discovery-algorithm-families-overview}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Question}}: \emph{``How do we discover causal
  relationships from data?''}

  \begin{itemize}
  \tightlist
  \item
    Different data types and assumptions enable different approaches
  \item
    No single algorithm works for all settings
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Key Insight}}: Causal discovery
  algorithms fall into distinct families based on their assumptions and
  data requirements

  \begin{itemize}
  \tightlist
  \item
    Constraint-based: Use conditional independence tests
  \item
    Score-based: Optimize likelihood with regularization
  \item
    Functional: Exploit non-linear or non-Gaussian properties
  \item
    Interventional: Leverage experimental data from interventions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Trade-off}}: More assumptions about the data
  \(\to\) stronger causal guarantees but narrower applicability
\end{itemize}
\end{frame}

\begin{frame}{Constraint-Based Discovery}
\protect\phantomsection\label{constraint-based-discovery}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Intuition}}: \emph{``If two variables are
  independent given a set of others, they cannot have a direct causal
  link''}

  \begin{itemize}
  \tightlist
  \item
    Conditional independence (CI) is a fingerprint of the causal graph
    structure
  \item
    Different graph structures imply different independence patterns
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Working Principle}}:

  \begin{itemize}
  \tightlist
  \item
    Start with a complete undirected graph (all variables connected)
  \item
    Test conditional independence: are \(X \perp Y | Z\)?
  \item
    Remove edges where CI holds
  \item
    Orient edges using rules (e.g., detect colliders)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: In a causal graph
  \(X \to Z \to Y\), we expect \(X \perp Y | Z\)

  \begin{itemize}
  \tightlist
  \item
    If this CI holds empirically, we know \(Z\) must be a mediator
  \item
    If \(X \perp Y | Z\) fails, \(Z\) might not be the only path
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Limitation}}: Many different graphs can
  have the same CI pattern

  \begin{itemize}
  \tightlist
  \item
    Result is partially directed (undirected edges remain)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Score-Based Discovery}
\protect\phantomsection\label{score-based-discovery}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Intuition}}: \emph{``The true causal graph
  should have high likelihood and be simple''}

  \begin{itemize}
  \tightlist
  \item
    Good models explain data well without overfitting
  \item
    Regularization penalizes complexity
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Working Principle}}:

  \begin{itemize}
  \tightlist
  \item
    Define a score combining fit and complexity:
    Score\((G) = \text{Likelihood}(G) -
    \lambda \cdot \text{Complexity}(G)\)
  \item
    Search over possible DAGs to maximize the score
  \item
    Use greedy algorithms (e.g., hill climbing) to avoid exponential
    search
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: GES (Greedy Equivalence Search)

  \begin{itemize}
  \tightlist
  \item
    Forward phase: Add edges that most improve score
  \item
    Backward phase: Remove edges if they don't decrease score
  \item
    Find a local optimum in the space of DAGs
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Limitation}}: Search is NP-hard; greedy
  algorithm finds local optima
\end{itemize}
\end{frame}

\begin{frame}{Functional Discovery}
\protect\phantomsection\label{functional-discovery}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Intuition}}: \emph{``Causal mechanisms have
  special statistical properties that differ from confounding''}

  \begin{itemize}
  \tightlist
  \item
    True causal relationships exhibit asymmetry in the residuals
  \item
    Fitting \(Y = f(X) + \varepsilon\) works better than
    \(X = f(Y) + \varepsilon'\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Working Principle}} (example: LiNGAM):

  \begin{itemize}
  \tightlist
  \item
    Assume linear model: \(X_i = \sum_j w_{ij} X_j + e_i\)
  \item
    Assume non-Gaussian noise: \(e_i\) are independent and non-Gaussian
  \item
    Non-Gaussianity breaks symmetry: fitting in reverse direction leaves
    dependent residuals
  \item
    Identify the causal order using this asymmetry
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: \(Y = 2X + e\) (causal) vs
  \(X = 0.4Y + e'\) (spurious)

  \begin{itemize}
  \tightlist
  \item
    Non-Gaussian noise \(e\) makes the forward model identifiable
  \item
    Backward model would require dependent residuals \(e'\)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Limitation}}: Strong assumptions (linear,
  non-Gaussian); breaks under Gaussianity
\end{itemize}
\end{frame}

\begin{frame}{Interventional Discovery}
\protect\phantomsection\label{interventional-discovery}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Intuition}}: \emph{``Intervening on a variable
  breaks confounding''}

  \begin{itemize}
  \tightlist
  \item
    Randomized experiments eliminate confounders by design
  \item
    Multiple interventions can resolve ambiguities from observational
    data
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Working Principle}}:

  \begin{itemize}
  \tightlist
  \item
    Perform interventions: set variable \(X\) to value \(x\) and observe
    effects
  \item
    Interventional data follows different distribution than
    observational
  \item
    Combine with constraint-based or other methods on interventional
    data
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Effect of price on sales

  \begin{itemize}
  \tightlist
  \item
    Observational: price and sales are correlated (confounded by demand)
  \item
    Interventional: set price \(= 5\) regardless of demand, observe
    sales
  \item
    Causal effect is identified under randomization assumption
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Advantage}}: Can identify full DAG with
  sufficient interventions

  \begin{itemize}
  \tightlist
  \item
    Gold standard in experimental science
  \item
    Limitation: Cost and feasibility of experiments
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Choosing an Algorithm Family}
\protect\phantomsection\label{choosing-an-algorithm-family}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Constraint-based}} (PC, FCI)

  \begin{itemize}
  \tightlist
  \item
    Use when: Observational data with large sample size
  \item
    Advantage: Minimal distributional assumptions
  \item
    Limitation: Many edges may remain undirected; high sample
    requirement
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Score-based}} (GES, NOTEARS)

  \begin{itemize}
  \tightlist
  \item
    Use when: Enough data to reliably estimate likelihood
  \item
    Advantage: Flexible to different model specifications
  \item
    Limitation: NP-hard search problem; local optima common
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Functional}} (LiNGAM, ANM)

  \begin{itemize}
  \tightlist
  \item
    Use when: Strong domain knowledge about the functional form
  \item
    Advantage: Can identify full DAG from observational data
  \item
    Limitation: Requires assumptions (linear, non-Gaussian; or additive
    noise)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Interventional}} (IGSP, JCI)

  \begin{itemize}
  \tightlist
  \item
    Use when: Experimental data available from targeted interventions
  \item
    Advantage: Strongest causal guarantees; can resolve ambiguities
  \item
    Limitation: Expensive; requires designing good experiments
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Validating a Discovered DAG}
\protect\phantomsection\label{validating-a-discovered-dag}
\begin{itemize}
\item
  A recovered DAG is a \emph{hypothesis} until validated
\item
  \textbf{\textcolor{red}{Validation strategies}}:

  \begin{enumerate}
  \tightlist
  \item
    \emph{Expert review}: do edges match known mechanisms?
  \item
    \emph{Refutation tests}: does data contradict implied conditional
    independencies?
  \item
    \emph{Sensitivity analysis}: do results change under perturbations
    of the assumptions?
  \item
    \emph{Interventional check}: do targeted experiments confirm
    predicted effects?
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Refutation Testing}
\protect\phantomsection\label{refutation-testing}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: given a discovered DAG \(G\),
  check each implied independence against data
\item
  \textbf{\textcolor{orange}{Procedure}}:

  \begin{enumerate}
  \tightlist
  \item
    Enumerate CI statements implied by \(G\) (via d-separation)
  \item
    Test each on held-out data using an appropriate CI test
  \item
    Count violations; low count \(\Rightarrow\) DAG is \emph{consistent}
  \end{enumerate}
\item
  \textbf{\textcolor{darkgreen}{Placebo tests}}: introduce a known-null
  edge; discovery should assign it low weight
\item
  \textbf{\textcolor{teal}{Example}}

  \begin{itemize}
  \tightlist
  \item
    Replace treatment with a random variable; effect should vanish
  \item
    Add an unobserved confounder; estimate sensitivity of conclusions
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Remark}}: if the conditional independence was
  used to discover the causal DAG, then a refutation test is redundant
\item
  \textbf{\textcolor{blue}{Question}}: what if the DAG passes all
  refutation tests?

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{violet}{Answer}}: it is \emph{consistent}, not
    \emph{proven}
  \item
    Causality is never established by data alone; always requires
    domain-informed interpretation
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{When Discovery Should Change Your DAG}
\protect\phantomsection\label{when-discovery-should-change-your-dag}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Update your DAG when}}:

  \begin{itemize}
  \tightlist
  \item
    Multiple independent algorithms converge on the same edge
  \item
    The new edge is \emph{mechanistically plausible}: the discovered
    edge must be theoretically coherent with domain knowledge, not just
    statistically present
  \item
    Refutation tests fail your current DAG
  \item
    Interventional experiments confirm the discovered structure
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Do not update your DAG when}}:

  \begin{itemize}
  \tightlist
  \item
    Edge contradicts well-established mechanism (e.g., time order,
    physics)
  \item
    Result hinges on weak faithfulness near independence boundary
  \item
    Sample size is too small for reliable CI tests
  \item
    Hidden confounders could explain the edge
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Rule of thumb}}: Discovery

  \begin{itemize}
  \tightlist
  \item
    Shifts the \emph{prior probability} of an edge
  \item
    Does not provide a \emph{posterior certainty}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Discovery: Key Takeaways}
\protect\phantomsection\label{causal-discovery-key-takeaways}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Causal discovery}} infers DAG structure from
  data, but only under strong assumptions (Markov, faithfulness,
  sufficiency)
\item
  \textbf{\textcolor{darkgreen}{Identifiability is limited}}

  \begin{itemize}
  \tightlist
  \item
    Observational data recover only the Markov equivalence class
  \item
    Interventions or functional assumptions are needed for full
    identification
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Three algorithm families}}:

  \begin{itemize}
  \tightlist
  \item
    Constraint-based (PC, FCI)
  \item
    Score-based (GES, NOTEARS)
  \item
    Functional / non-Gaussian (LiNGAM)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Discovery $\neq$ truth}}

  \begin{itemize}
  \tightlist
  \item
    Use it to \emph{generate} hypotheses
  \item
    Combine with domain knowledge and interventions
  \item
    Validate via refutation and expert review
  \item
    Automated discovery is a powerful exploratory tool, not a
    replacement for scientific reasoning
  \item
    Treat discovery as informative but fallible
  \end{itemize}
\end{itemize}
\end{frame}
