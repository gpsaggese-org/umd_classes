\begin{frame}{What are Multi-Armed Bandits?}
\protect\phantomsection\label{what-are-multi-armed-bandits}
\begin{itemize}
\item
  Imagine you're in a casino facing \(K\) slot machines (aka ``one-armed
  bandits'')

  \begin{itemize}
  \tightlist
  \item
    Each machine has a different unknown probability of winning
  \item
    Limited budget of plays
  \item
    \emph{Goal}: Maximize total winnings
  \end{itemize}
\item
  \textbf{\textcolor{red}{Multi-Armed Bandit (MAB)}} is a sequential
  decision-making problem where:

  \begin{itemize}
  \tightlist
  \item
    At each time step \(t = 1, 2, \ldots, T\), choose one action (arm)
    from \(K\) options
  \item
    After choosing arm \(i\), receive a random reward \(r_t \sim P_i\)
  \item
    Reward distributions \(P_1, \ldots, P_K\) are unknown
  \item
    Learn which arms are best while collecting rewards
  \end{itemize}
\item
  Sequential decisions with uncertain outcomes and limited feedback
\item
  \textbf{\textcolor{blue}{Key Challenge}}: Balance:

  \begin{itemize}
  \tightlist
  \item
    \emph{Exploration}: Try different arms to learn their reward
    distributions
  \item
    \emph{Exploitation}: Play the arm you believe is best
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Why Study Bandits?}
\protect\phantomsection\label{why-study-bandits}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Fundamental Problem in Sequential Decision-Making}}

  \begin{itemize}
  \tightlist
  \item
    Core model for learning under uncertainty
  \item
    Simplest non-trivial reinforcement learning setting
  \item
    Rich theoretical framework with practical algorithms
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Tractable Analysis}}

  \begin{itemize}
  \tightlist
  \item
    Stateless: No long-term consequences of actions
  \item
    Clean mathematical formulation
  \item
    Provable regret bounds and optimality results
  \item
    Serves as building block for more complex RL problems
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Wide Applicability}}

  \begin{itemize}
  \tightlist
  \item
    Online advertising (which ad to show?)
  \item
    Recommendation systems (which movie to suggest?)
  \item
    Clinical trials (which treatment to test?)
  \item
    Website optimization (A/B testing)
  \item
    Resource allocation (which server to query?)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Applications: Real-World Decision-Making}
\protect\phantomsection\label{applications-real-world-decision-making}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Online Advertising}}

  \begin{itemize}
  \tightlist
  \item
    Platform must choose which ad to display
  \item
    Each ad click generates revenue
  \item
    Must learn which ads perform best
  \item
    Trade-off: Show proven ads vs.~test new ones
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Clinical Trials}}

  \begin{itemize}
  \tightlist
  \item
    Allocate patients to treatment arms
  \item
    Each treatment has unknown efficacy
  \item
    \emph{Ethical imperative}: Minimize patients receiving inferior
    treatments
  \item
    \emph{Exploration-exploitation} is life-or-death
  \end{itemize}
\item
  \textbf{\textcolor{blue}{A/B Testing}}

  \begin{itemize}
  \tightlist
  \item
    Test website designs, features, UI changes
  \item
    Each variant has unknown conversion rate
  \item
    Want to identify best variant quickly
  \item
    Minimize opportunity cost during testing
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Exploration--Exploitation Tradeoff}
\protect\phantomsection\label{the-explorationexploitation-tradeoff}
\begin{itemize}
\item
  The ``Exploration--Exploitation Tradeoff'' is the central dilemma of
  sequential learning
\item
  \textbf{\textcolor{red}{Why This Tradeoff is Unavoidable}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Pure exploitation}: Never try suboptimal-looking arms \(\to\)
    may stick with suboptimal choice forever
  \item
    \emph{Pure exploration}: Always try random arms \(\to\) collect lots
    of information but low total reward
  \item
    \emph{Must do both}: Explore enough to find good arms, exploit
    enough to gain high reward
  \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{The K-Armed Bandit Problem: Formal Definition}
\protect\phantomsection\label{the-k-armed-bandit-problem-formal-definition}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Setting}}:

  \begin{itemize}
  \tightlist
  \item
    \(K\) arms (actions), indexed \(i = 1, \ldots, K\)
  \item
    Time horizon: \(T\) rounds (sometimes \(T = \infty\))
  \item
    At round \(t = 1, \ldots, T\):

    \begin{enumerate}
    \tightlist
    \item
      Agent selects arm \(A_t \in \{1, \ldots, K\}\)
    \item
      Environment reveals reward \(R_t \sim P_{A_t}\)
    \item
      Agent observes only \(R_t\) (not rewards from other arms)
    \end{enumerate}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Stochastic Bandit Assumption}}:

  \begin{itemize}
  \tightlist
  \item
    Each arm \(i\) has a fixed reward distribution \(P_i\) with mean
    \(\mu_i = \EE[R | A = i]\)
  \item
    Rewards are independent across rounds: \(R_t\) iid
  \item
    The means \(\mu_1, \ldots, \mu_K\) are unknown to the agent
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Goal}}: Design a policy (algorithm) that
  selects \(A_1, A_2, \ldots, A_T\) to maximize
  \[\text{Total Reward} = \sum_{t=1}^{T} R_t\]
\item
  \textbf{\textcolor{violet}{Key Insight}}: Can't maximize reward with
  certainty (rewards are random), but can minimize \emph{regret}
\end{itemize}
\end{frame}

\begin{frame}{Reward Distributions}
\protect\phantomsection\label{reward-distributions}
\begin{itemize}
\tightlist
\item
  Each arm \(i\) has an associated
  \textbf{\textcolor{red}{reward distribution}} \(P_i\)

  \begin{itemize}
  \tightlist
  \item
    When arm \(i\) is pulled, reward \(R \sim P_i\) is sampled
  \item
    Common distributions: Bernoulli, Gaussian, bounded support
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Expected Reward}} (Mean):

  \begin{itemize}
  \tightlist
  \item
    Each arm has true mean reward: \(\mu_i = \EE[R | A = i]\)
  \item
    This is the quantity we care about for maximizing total reward
  \item
    The agent does not know \(\mu_i\) initially
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Unknown Parameters}}: The Challenge

  \begin{itemize}
  \tightlist
  \item
    The means \(\mu_1, \ldots, \mu_K\) are unknown
  \item
    Also variance, shape of distribution may be unknown
  \item
    Agent must estimate these from observed rewards
  \item
    Must balance learning (exploration) and earning (exploitation)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}: Bernoulli Bandits

  \begin{itemize}
  \tightlist
  \item
    Each arm \(i\) returns reward \(R_t = 1\) with probability
    \(\mu_i\), else \(R_t = 0\)
  \item
    \(\mu_i\) is the ``success probability'' or ``conversion rate''
  \item
    Common in A/B testing: \(\mu_i\) = click-through rate of ad \(i\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Performance Metrics}
\protect\phantomsection\label{performance-metrics}
\begin{itemize}
\item
  How do we measure algorithm performance when rewards are random?
\item
  \textbf{\textcolor{red}{Optimal Arm}}: Define
  \(i^* = \argmax_i \mu_i\)

  \begin{itemize}
  \tightlist
  \item
    The arm with highest expected reward
  \item
    If we knew \(\mu_1, \ldots, \mu_K\), we would always pull arm
    \(i^*\)
  \item
    Optimal expected reward per round: \(\mu^* = \mu_{i^*}\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Instantaneous Regret}} at time \(t\):
  \[\ell_t = \mu^* - \mu_{A_t}\]

  \begin{itemize}
  \tightlist
  \item
    The difference between optimal mean and mean of chosen arm
  \item
    How much expected reward we ``lose'' by not playing \(i^*\)
  \item
    Note: \(\ell_t \geq 0\) always
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cumulative Regret}} over \(T\) rounds:
  \[L_T = \sum_{t=1}^{T} \ell_t = T \mu^* - \sum_{t=1}^{T} \mu_{A_t}\]

  \begin{itemize}
  \tightlist
  \item
    Total expected loss from suboptimal decisions
  \item
    Measures how much worse the algorithm is vs.~always pulling \(i^*\)
  \item
    \emph{Goal}: Design algorithm with low regret \(L_T\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Performance Metrics: Intuition}
\protect\phantomsection\label{performance-metrics-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Why Regret?}}

  \begin{itemize}
  \tightlist
  \item
    Cannot maximize reward with certainty (randomness in rewards)
  \item
    But can minimize expected loss relative to optimal strategy
  \item
    Regret = ``opportunity cost'' of learning
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Key Property}}: Regret depends only on
  how many times we pull each arm

  \begin{itemize}
  \tightlist
  \item
    Let \(N_i(T) = \sum_{t=1}^{T} I[\{A_t = i\}\){]} = number of times
    arm \(i\) pulled
  \item
    Then:
    \[L_T = \sum_{i=1}^{K} (\mu^* - \mu_i) N_i(T) = \sum_{i=1}^{K} \Delta_i \, N_i(T)\]
  \item
    Where \(\Delta_i = \mu^* - \mu_i\) is the ``suboptimality gap'' of
    arm \(i\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Implications}}:

  \begin{itemize}
  \tightlist
  \item
    To minimize regret, pull suboptimal arms as few times as possible
  \item
    But we don't know which arms are suboptimal initially!
  \item
    Must explore to identify \(i^*\), but exploration increases regret
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Regret Lower Bounds}
\protect\phantomsection\label{regret-lower-bounds}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Why Lower Bounds Matter}}

  \begin{itemize}
  \tightlist
  \item
    Shows fundamental limits of \emph{any} algorithm
  \item
    No algorithm can do better (up to constants)
  \item
    Tells us if an algorithm is near-optimal
  \item
    Guides algorithm design: what's achievable?
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Intuition}}: Why Can't Regret Be Constant?

  \begin{itemize}
  \tightlist
  \item
    Need to distinguish between arms with close means
  \item
    If \(\mu_1 \approx \mu_2\), need many samples to tell which is
    better
  \item
    Must pull suboptimal arms enough to be confident they're suboptimal
  \item
    Trade-off: confidence vs.~regret
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Regret Lower Bounds: Lai--Robbins Bound}
\protect\phantomsection\label{regret-lower-bounds-lairobbins-bound}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Lai and Robbins (1985)}} proved:

  \begin{itemize}
  \tightlist
  \item
    For any ``consistent'' algorithm (one that identifies optimal arm
    eventually)
  \item
    The cumulative regret must satisfy:
    \[\liminf_{T \to \infty} \frac{L_T}{\log T} \geq \sum_{i: \Delta_i > 0} \frac{\Delta_i}{D(\mu_i \| \mu^*)}\]
  \item
    Where \(D(\mu_i \| \mu^*)\) is the KL divergence between reward
    distributions
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Simplified Version}} (for bounded
  rewards):

  \begin{itemize}
  \tightlist
  \item
    For any consistent algorithm:
    \[L_T = \Omega\left( \sum_{i: \Delta_i > 0} \frac{\log T}{\Delta_i} \right)\]
  \item
    Regret must grow at least logarithmically with \(T\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Interpretation}}:

  \begin{itemize}
  \tightlist
  \item
    Cannot achieve constant regret (must be \(\Omega(\log T)\))
  \item
    Smaller gap \(\Delta_i\) \(\to\) more pulls needed \(\to\) higher
    regret
  \item
    Best achievable: \(L_T = O(\log T)\) (logarithmic regret)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Regret Lower Bounds: Implications for Algorithm Design}
\protect\phantomsection\label{regret-lower-bounds-implications-for-algorithm-design}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Design Goal}}: Achieve \(O(\log T)\) regret

  \begin{itemize}
  \tightlist
  \item
    Matches lower bound (up to constants)
  \item
    Called ``order-optimal'' or ``asymptotically optimal''
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Key Insight}}: Must pull each suboptimal
  arm \(i\) roughly \(O(\frac{\log T}{\Delta_i^2})\) times

  \begin{itemize}
  \tightlist
  \item
    Arms with smaller gaps \(\Delta_i\) need more exploration
  \item
    Total regret:
    \(\sum_i \Delta_i \cdot O(\frac{\log T}{\Delta_i^2}) = O(\frac{\log T}{\Delta_i})\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{What Won't Work}}:

  \begin{itemize}
  \tightlist
  \item
    Fixed exploration schedule (e.g., pull each arm \(n\) times):
    \(\Omega(K)\) regret
  \item
    Random sampling: \(\Omega(\sqrt{T})\) regret
  \item
    Need adaptive exploration that concentrates on promising arms
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Baseline Strategies}
\protect\phantomsection\label{baseline-strategies}
\begin{itemize}
\item
  Before studying sophisticated algorithms, consider naive approaches
\item
  \textbf{\textcolor{red}{Greedy Algorithm}}:

  \begin{itemize}
  \tightlist
  \item
    At each time \(t\):

    \begin{enumerate}
    \tightlist
    \item
      Compute empirical mean reward for each arm:
      \(\hat{\mu}_i(t) = \frac{1}{N_i(t)} \sum_{\tau: A_\tau = i} R_\tau\)
    \item
      Pull arm with highest empirical mean:
      \(A_t = \argmax_i \hat{\mu}_i(t)\)
    \end{enumerate}
  \item
    Pure exploitation: always pull current best arm
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Random Selection}}:

  \begin{itemize}
  \tightlist
  \item
    At each time \(t\): Choose \(A_t\) uniformly at random from
    \(\{1, \ldots, K\}\)
  \item
    Pure exploration: no use of observed rewards
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Baseline Strategies: Limitations}
\protect\phantomsection\label{baseline-strategies-limitations}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Greedy Algorithm Fails}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Problem}: May get ``stuck'' on suboptimal arm
  \item
    \emph{Example}: Suppose arm 1 has \(\mu_1 = 0.6\), arm 2 has
    \(\mu_2 = 0.7\)

    \begin{itemize}
    \tightlist
    \item
      If we happen to pull arm 1 first and get reward 1
    \item
      Then \(\hat{\mu}_1(1) = 1\), and we'll keep pulling arm 1
    \item
      Never try arm 2, which is actually better
    \end{itemize}
  \item
    \emph{Regret}: \(L_T = \Omega(T)\) (linear regret, very bad!)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Random Selection Fails}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Problem}: Never uses information from rewards
  \item
    Pulls all arms equally, even after learning which are bad
  \item
    \emph{Regret}: \(L_T = \Theta(\sqrt{KT})\) (grows with \(T\),
    suboptimal)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key Lesson}}: Need to balance exploration and
  exploitation

  \begin{itemize}
  \tightlist
  \item
    Greedy: too much exploitation
  \item
    Random: too much exploration
  \item
    Need adaptive strategy that explores less over time
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{\(\epsilon\)-Greedy Algorithm}
\protect\phantomsection\label{epsilon-greedy-algorithm}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Simple Idea}}: With small probability
  \(\epsilon\), explore; otherwise exploit
\item
  \textbf{\textcolor{darkgreen}{Algorithm}}:

  \begin{itemize}
  \tightlist
  \item
    At each time \(t\):

    \begin{enumerate}
    \tightlist
    \item
      Compute empirical means:
      \(\hat{\mu}_i(t) = \frac{\sum_{\tau < t: A_\tau = i} R_\tau}{N_i(t)}\)
    \item
      With probability \(\epsilon\): Choose arm uniformly at random
      (explore)
    \item
      With probability \(1 - \epsilon\): Choose arm
      \(\argmax_i \hat{\mu}_i(t)\) (exploit)
    \end{enumerate}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Hyperparameter}}: \(\epsilon \in (0, 1)\)

  \begin{itemize}
  \tightlist
  \item
    Small \(\epsilon\) (e.g., 0.01): mostly exploit, little exploration
  \item
    Large \(\epsilon\) (e.g., 0.5): explore half the time
  \item
    Typical choice: \(\epsilon = 0.1\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{\(\epsilon\)-Greedy: Exploration Schedules}
\protect\phantomsection\label{epsilon-greedy-exploration-schedules}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Fixed $\epsilon$}}:

  \begin{itemize}
  \tightlist
  \item
    Use same \(\epsilon\) for all \(t\)
  \item
    \emph{Problem}: Keeps exploring even after identifying best arm
  \item
    \emph{Regret}: \(L_T = \Omega(T)\) (linear, not optimal)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Decaying $\epsilon\_t$}}:

  \begin{itemize}
  \tightlist
  \item
    Use time-varying \(\epsilon_t\) that decreases with \(t\)
  \item
    Example: \(\epsilon_t = \frac{1}{t}\) or
    \(\epsilon_t = \min(1, \frac{K}{t})\)
  \item
    \emph{Intuition}: Explore a lot early, less over time
  \item
    \emph{Regret}: Can achieve \(L_T = O(T^{2/3})\) with right schedule
    (still not \(O(\log T)\))
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Limitations}}:

  \begin{itemize}
  \tightlist
  \item
    Hard to choose optimal schedule without knowing \(\Delta_i\)'s
  \item
    Explores uniformly (doesn't focus on promising arms)
  \item
    Better algorithms exist (UCB, Thompson Sampling)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{\(\epsilon\)-Greedy: Regret Behavior}
\protect\phantomsection\label{epsilon-greedy-regret-behavior}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Fixed $\epsilon$}}: \(L_T = \Theta(T)\)

  \begin{itemize}
  \tightlist
  \item
    Explores \(\epsilon\) fraction of time forever
  \item
    Linear regret: \(\epsilon T \sum_i \Delta_i / K\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Decaying $\epsilon\_t = O(1/t^\alpha)$}}:

  \begin{itemize}
  \tightlist
  \item
    If \(\alpha\) too small: not enough exploration \(\to\) may miss
    \(i^*\)
  \item
    If \(\alpha\) too large: too much exploration \(\to\) high regret
  \item
    Best tuned: \(L_T = O(T^{2/3})\) (worse than \(O(\log T)\) lower
    bound)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Practical Performance}}:

  \begin{itemize}
  \tightlist
  \item
    Simple, easy to implement
  \item
    Works reasonably well in practice
  \item
    Often used as baseline
  \item
    But theoretically suboptimal (cannot achieve \(O(\log T)\))
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Optimism in the Face of Uncertainty}
\protect\phantomsection\label{optimism-in-the-face-of-uncertainty}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Core Idea}}: ``Be optimistic about uncertain
  options''

  \begin{itemize}
  \tightlist
  \item
    If unsure about arm's value, assume it could be the best
  \item
    Optimism drives exploration of uncertain arms
  \item
    As arms are explored, optimism diminishes (uncertainty shrinks)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Why Optimism Works}}:

  \begin{itemize}
  \tightlist
  \item
    If an arm \emph{is} optimal: Optimism ensures we keep trying it
  \item
    If an arm \emph{is not} optimal: We'll eventually learn and stop
    pulling it
  \item
    Natural exploration-exploitation balance without explicit
    randomization
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Contrast with $\epsilon$-Greedy}}:

  \begin{itemize}
  \tightlist
  \item
    \(\epsilon\)-greedy: Random exploration (ignores uncertainty)
  \item
    Optimism: Directed exploration toward uncertain arms
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Optimism: Confidence Intervals}
\protect\phantomsection\label{optimism-confidence-intervals}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Confidence Bound}} for arm \(i\) after \(n\)
  pulls:

  \begin{itemize}
  \tightlist
  \item
    Empirical mean:
    \(\hat{\mu}_i = \frac{1}{n} \sum_{\tau: A_\tau = i} R_\tau\)
  \item
    Uncertainty radius: \(c_i(n) = \sqrt{\frac{2 \log T}{n}}\) (example)
  \item
    Upper confidence bound: \(\text{UCB}_i = \hat{\mu}_i + c_i(n)\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Interpretation}}:

  \begin{itemize}
  \tightlist
  \item
    With high probability, true mean \(\mu_i \leq \text{UCB}_i\)
  \item
    Fewer samples \(n\) \(\to\) larger \(c_i(n)\) \(\to\) more
    optimistic
  \item
    More samples \(n\) \(\to\) smaller \(c_i(n)\) \(\to\) less
    optimistic
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Optimistic Strategy}}:

  \begin{itemize}
  \tightlist
  \item
    Pull arm with highest upper confidence bound:
    \(A_t = \argmax_i \text{UCB}_i(t)\)
  \item
    Automatically explores uncertain arms and exploits promising arms
  \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{Upper Confidence Bound (UCB)}
\protect\phantomsection\label{upper-confidence-bound-ucb}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{UCB1 Algorithm}} (Auer, Cesa-Bianchi, Fischer,
  2002):

  \begin{itemize}
  \tightlist
  \item
    For first \(K\) rounds: Pull each arm once
  \item
    For round \(t > K\):

    \begin{enumerate}
    \tightlist
    \item
      For each arm \(i\), compute UCB:
      \[\text{UCB}_i(t) = \hat{\mu}_i(t) + \sqrt{\frac{2 \log t}{N_i(t)}}\]
    \item
      Pull arm: \(A_t = \argmax_i \text{UCB}_i(t)\)
    \end{enumerate}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Components}}:

  \begin{itemize}
  \tightlist
  \item
    \(\hat{\mu}_i(t)\): Empirical mean (exploitation term)
  \item
    \(\sqrt{\frac{2 \log t}{N_i(t)}}\): Exploration bonus (uncertainty
    term)
  \item
    \(N_i(t)\): Number of times arm \(i\) pulled so far
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Upper Confidence Bound: Arm Selection Rule}
\protect\phantomsection\label{upper-confidence-bound-arm-selection-rule}
\begin{itemize}
\item
  \textbf{\textcolor{red}{UCB Index}}:
  \[\text{UCB}_i(t) = \underbrace{\hat{\mu}_i(t)}_{\text{exploitation}} + \underbrace{\sqrt{\frac{2 \log t}{N_i(t)}}}_{\text{exploration bonus}}\]
\item
  \textbf{\textcolor{darkgreen}{How It Works}}:

  \begin{itemize}
  \tightlist
  \item
    Arms with high empirical mean \(\hat{\mu}_i\) have high UCB
    (exploitation)
  \item
    Arms pulled few times (\(N_i\) small) have high UCB (exploration)
  \item
    Bonus \(\propto \frac{1}{\sqrt{N_i}}\): Shrinks with pulls
  \item
    Bonus \(\propto \sqrt{\log t}\): Grows slowly with time
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Behavior Over Time}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Early}: All arms have few pulls \(\to\) large bonuses \(\to\)
    explores
  \item
    \emph{Middle}: Focuses on promising arms, occasionally revisits
    uncertain arms
  \item
    \emph{Late}: Mostly pulls optimal arm (bonus becomes negligible)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Upper Confidence Bound: Key Intuition}
\protect\phantomsection\label{upper-confidence-bound-key-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Why UCB Works}}:

  \begin{itemize}
  \tightlist
  \item
    The bonus \(\sqrt{\frac{2 \log t}{N_i(t)}}\) is carefully
    calibrated:

    \begin{itemize}
    \tightlist
    \item
      Derived from Hoeffding's concentration inequality
    \item
      Ensures with high probability: \(\mu_i \leq \text{UCB}_i(t)\)
    \item
      ``Optimism'': We act as if each arm is as good as its UCB
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Self-Correcting}}:

  \begin{itemize}
  \tightlist
  \item
    If we pull a suboptimal arm too much: Its UCB decreases (more data
    \(\to\) less uncertainty)
  \item
    If we neglect the optimal arm: Its UCB increases (\(\log t\) grows,
    while \(N_i\) stays fixed)
  \item
    Automatically balances exploration and exploitation
  \end{itemize}
\item
  \textbf{\textcolor{blue}{No Tuning Parameters}}:

  \begin{itemize}
  \tightlist
  \item
    Unlike \(\epsilon\)-greedy (requires choosing \(\epsilon\))
  \item
    UCB has no hyperparameters (besides confidence constant)
  \item
    Adapts automatically to problem structure
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Regret Analysis of UCB}
\protect\phantomsection\label{regret-analysis-of-ucb}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Theorem}} (Auer et al., 2002):

  \begin{itemize}
  \tightlist
  \item
    For UCB1 with rewards in \([0, 1]\), the cumulative regret
    satisfies:
    \[L_T \leq 8 \sum_{i: \Delta_i > 0} \frac{\log T}{\Delta_i} + \left( 1 + \frac{\pi^2}{3} \right) \sum_{i=1}^{K} \Delta_i\]
  \item
    \textbf{\textcolor{darkgreen}{Simplified}}:
    \(L_T = O\left( \sum_{i: \Delta_i > 0} \frac{\log T}{\Delta_i} \right)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Interpretation}}:

  \begin{itemize}
  \tightlist
  \item
    Logarithmic regret: \(L_T = O(\log T)\)
  \item
    Matches Lai--Robbins lower bound (up to constants)
  \item
    \emph{Order-optimal}: Cannot do better asymptotically
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Dependence on $\Delta\_i$}}:

  \begin{itemize}
  \tightlist
  \item
    Smaller gaps \(\Delta_i\) \(\to\) higher regret (harder to
    distinguish arms)
  \item
    Arms with \(\Delta_i = 0\) (i.e., optimal arms) don't contribute to
    regret bound
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Regret Analysis: Role of Concentration Inequalities}
\protect\phantomsection\label{regret-analysis-role-of-concentration-inequalities}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Key Tool}}: Hoeffding's Inequality

  \begin{itemize}
  \tightlist
  \item
    For rewards \(R_1, \ldots, R_n\) iid in \([0, 1]\) with mean
    \(\mu\):
    \[\Pr\left( \left|\hat{\mu} - \mu\right| \geq \epsilon \right) \leq 2 \exp(-2n\epsilon^2)\]
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Application to UCB}}:

  \begin{itemize}
  \tightlist
  \item
    Choose radius \(c = \sqrt{\frac{2 \log t}{n}}\) so that:
    \[\Pr\left( \mu_i > \hat{\mu}_i + \sqrt{\frac{2 \log t}{N_i(t)}} \right) \leq \frac{1}{t^4}\]
  \item
    With high probability, the true mean is below the UCB
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Union Bound}}: Across all arms and time
  steps:

  \begin{itemize}
  \tightlist
  \item
    Probability that any UCB is violated at any time
    \(\leq \sum_{t=1}^{T} \sum_{i=1}^{K} \frac{1}{t^4} < \infty\)
  \item
    UCB indices are ``optimistic'' with high probability
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Consequence}}: If optimal arm has high UCB,
  we'll pull it; if suboptimal arm has high UCB, we'll pull it until UCB
  decreases
\end{itemize}
\end{frame}

\begin{frame}{Regret Comparison: UCB vs.~\(\epsilon\)-Greedy}
\protect\phantomsection\label{regret-comparison-ucb-vs.-epsilon-greedy}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\textbf{\textcolor{red}{$\epsilon$-Greedy}}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{orange}{Regret}}: \(L_T = O(T)\) (fixed
  \(\epsilon\)) or \(O(T^{2/3})\) (decaying \(\epsilon\))
\item
  \textbf{\textcolor{darkgreen}{Exploration}}: Uniform random sampling
\item
  \textbf{\textcolor{teal}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    Simple to implement
  \item
    Easy to understand
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Cons}}:

  \begin{itemize}
  \tightlist
  \item
    Linear or polynomial regret
  \item
    Ignores uncertainty
  \item
    Requires tuning \(\epsilon\)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.5\linewidth}
\textbf{\textcolor{blue}{UCB}}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{violet}{Regret}}: \(L_T = O(\log T)\)
\item
  \textbf{\textcolor{magenta}{Exploration}}: Directed by uncertainty
\item
  \textbf{\textcolor{brown}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    Logarithmic regret (optimal)
  \item
    No hyperparameters
  \item
    Adaptive
  \end{itemize}
\item
  \textbf{\textcolor{olive}{Cons}}:

  \begin{itemize}
  \tightlist
  \item
    Slightly more complex
  \item
    Requires confidence bounds
  \end{itemize}
\end{itemize}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{gray}{Practical Takeaway}}: UCB is theoretically
  and empirically superior for stationary problems
\end{itemize}
\end{frame}

\begin{frame}{Variants of UCB}
\protect\phantomsection\label{variants-of-ucb}
\begin{itemize}
\item
  \textbf{\textcolor{red}{UCB1}}: Original algorithm (assumes bounded
  rewards in \([0, 1]\))
\item
  \textbf{\textcolor{orange}{UCB-V}} (Variance-Aware UCB):

  \begin{itemize}
  \tightlist
  \item
    Incorporates empirical variance of rewards
  \item
    Index:
    \[\text{UCB-V}_i(t) = \hat{\mu}_i(t) + \sqrt{\frac{2 \hat{\sigma}_i^2(t) \log t}{N_i(t)}} + \frac{3 \log t}{N_i(t)}\]
  \item
    Where \(\hat{\sigma}_i^2(t)\) is sample variance of arm \(i\)
  \item
    \textbf{\textcolor{darkgreen}{Benefit}}: Better for arms with low
    variance (tighter bounds)
  \end{itemize}
\item
  \textbf{\textcolor{teal}{KL-UCB}}:

  \begin{itemize}
  \tightlist
  \item
    Uses Kullback--Leibler divergence for tighter confidence bounds
  \item
    Index:
    \[\text{KL-UCB}_i(t) = \max \left\{ q : N_i(t) \, D(\hat{\mu}_i(t) \| q) \leq \log t \right\}\]
  \item
    Where \(D(p \| q)\) is KL divergence
  \item
    \textbf{\textcolor{cyan}{Benefit}}: Asymptotically optimal constants
    (better than UCB1)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Variants of UCB: When They Matter}
\protect\phantomsection\label{variants-of-ucb-when-they-matter}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{UCB1}}: Good default choice

  \begin{itemize}
  \tightlist
  \item
    Simple, no tuning, strong guarantees
  \item
    Works for any reward distribution with bounded support
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{UCB-V}}: Use when variance varies across
  arms

  \begin{itemize}
  \tightlist
  \item
    Example: Some ads have consistent CTR, others highly variable
  \item
    Can reduce regret if some arms have low variance
  \end{itemize}
\item
  \textbf{\textcolor{blue}{KL-UCB}}: Use when asymptotic performance
  matters

  \begin{itemize}
  \tightlist
  \item
    Better constants in regret bound
  \item
    More computationally expensive (requires solving equation for index)
  \item
    Recommended for Bernoulli or Gaussian bandits
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Practical Advice}}: Start with UCB1; switch
  to variants if:

  \begin{itemize}
  \tightlist
  \item
    You have domain knowledge about variance structure (UCB-V)
  \item
    You need best possible asymptotic performance (KL-UCB)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Bandits}
\protect\phantomsection\label{bayesian-bandits}
\begin{itemize}
\tightlist
\item
  So far: \textbf{\textcolor{red}{Frequentist}} approach (UCB,
  \(\epsilon\)-greedy)

  \begin{itemize}
  \tightlist
  \item
    Treat \(\mu_i\) as fixed unknown constants
  \item
    Use confidence bounds and empirical means
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Bayesian Approach}}: Treat \(\mu_i\) as
  random variables

  \begin{itemize}
  \tightlist
  \item
    Start with prior distribution \(p(\mu_i)\) for each arm
  \item
    Update beliefs using Bayes' rule after each observation
  \item
    Make decisions based on posterior distributions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bayesian Framework}}:

  \begin{itemize}
  \tightlist
  \item
    Prior: \(p(\mu_i)\) (e.g., \(\mu_i \sim \text{Beta}(1, 1)\) for
    Bernoulli arm)
  \item
    Likelihood: \(p(R | \mu_i)\) (e.g.,
    \(R \sim \text{Bernoulli}(\mu_i)\))
  \item
    Posterior:
    \(p(\mu_i | R_{1:t}) \propto p(R_{1:t} | \mu_i) p(\mu_i)\) (by
    Bayes' rule)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Bandits: Priors and Posteriors}
\protect\phantomsection\label{bayesian-bandits-priors-and-posteriors}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Prior Distribution}}: Encodes initial beliefs
  about \(\mu_i\)

  \begin{itemize}
  \tightlist
  \item
    Uniform prior: \(\mu_i \sim \text{Beta}(1, 1)\) (no prior knowledge)
  \item
    Informative prior: \(\mu_i \sim \text{Beta}(\alpha, \beta)\) (domain
    knowledge)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Posterior Update}}: After observing
  rewards from arm \(i\)

  \begin{itemize}
  \tightlist
  \item
    Suppose arm \(i\) has been pulled \(n\) times: \(s\) successes,
    \(f\) failures
  \item
    Posterior (for Bernoulli arm with Beta prior):
    \[p(\mu_i | \text{data}) = \text{Beta}(\alpha + s, \beta + f)\]
  \item
    Conjugate prior: Posterior has same form as prior (easy to update)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bayesian Inference}}: Posterior encodes all
  information about \(\mu_i\)

  \begin{itemize}
  \tightlist
  \item
    Mean:
    \(\EE[\mu_i | \text{data}] = \frac{\alpha + s}{\alpha + \beta + n}\)
  \item
    Uncertainty:
    \(\VV[\mu_i | \text{data}] = \frac{(\alpha + s)(\beta + f)}{(\alpha + \beta + n)^2 (\alpha + \beta + n + 1)}\)
  \item
    More data \(\to\) sharper posterior (less uncertainty)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Decision-Making}
\protect\phantomsection\label{bayesian-decision-making}
\begin{itemize}
\item
  \textbf{\textcolor{red}{How to Choose Arm?}} Several strategies:
\item
  \textbf{\textcolor{orange}{Greedy Bayesian}}: Choose arm with highest
  posterior mean \[A_t = \argmax_i \EE[\mu_i | \text{data}_t]\]

  \begin{itemize}
  \tightlist
  \item
    Problem: No exploration (same issue as frequentist greedy)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Bayes-UCB}}: Use posterior quantile as
  UCB \[A_t = \argmax_i Q_i(1 - \frac{1}{t})\]

  \begin{itemize}
  \tightlist
  \item
    Where \(Q_i(p)\) is the \(p\)-th quantile of posterior
    \(p(\mu_i | \text{data}_t)\)
  \item
    Similar to UCB, but uses Bayesian uncertainty
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Probability of Best Arm}}: Choose arm most
  likely to be optimal
  \[A_t = \argmax_i \Pr(\mu_i = \max_j \mu_j | \text{data}_t)\]
\item
  \textbf{\textcolor{cyan}{Thompson Sampling}}: Sample from posteriors
  and play best sample (next slide)
\end{itemize}
\end{frame}

\begin{frame}{Thompson Sampling}
\protect\phantomsection\label{thompson-sampling}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Posterior Sampling}} (Thompson, 1933; Agrawal
  \& Goyal, 2012)

  \begin{itemize}
  \tightlist
  \item
    At each time \(t\):

    \begin{enumerate}
    \tightlist
    \item
      Sample \(\tilde{\mu}_i \sim p(\mu_i | \text{data}_t)\) for each
      arm \(i\)
    \item
      Pull arm \(A_t = \argmax_i \tilde{\mu}_i\)
    \item
      Observe reward \(R_t\)
    \item
      Update posterior for arm \(A_t\)
    \end{enumerate}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Intuition}}: ``Probability matching''

  \begin{itemize}
  \tightlist
  \item
    Pull arm \(i\) with probability
    \(\approx \Pr(i \text{ is optimal} | \text{data})\)
  \item
    If arm likely to be best, sample will often be highest
  \item
    If arm uncertain, might still be sampled (exploration)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Bernoulli Bandits with Beta Priors

  \begin{itemize}
  \tightlist
  \item
    Posterior:
    \(p(\mu_i | \text{data}) = \text{Beta}(\alpha_i, \beta_i)\)
  \item
    Algorithm:

    \begin{enumerate}
    \tightlist
    \item
      Sample \(\tilde{\mu}_i \sim \text{Beta}(\alpha_i, \beta_i)\) for
      each \(i\)
    \item
      Pull \(A_t = \argmax_i \tilde{\mu}_i\)
    \item
      If reward \(R_t = 1\): \(\alpha_{A_t} \gets \alpha_{A_t} + 1\);
      else \(\beta_{A_t} \gets \beta_{A_t} + 1\)
    \end{enumerate}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Thompson Sampling: Algorithm Steps}
\protect\phantomsection\label{thompson-sampling-algorithm-steps}
\begingroup \small

\textbf{\textcolor{red}{Thompson Sampling for Bernoulli Bandits}}

\textbf{\textcolor{orange}{Input}}: Number of arms \(K\), time horizon
\(T\)

\textbf{\textcolor{darkgreen}{Initialize}}: For each arm
\(i = 1, \ldots, K\): - \(\alpha_i \gets 1, \beta_i \gets 1\) (Beta(1,1)
prior)

\textbf{\textcolor{teal}{For}} \(t = 1, 2, \ldots, T\):

\begin{enumerate}
\item
  \textbf{\textcolor{cyan}{Sample}} from posteriors:

  \begin{itemize}
  \tightlist
  \item
    For each arm \(i\):
    \(\tilde{\mu}_i \sim \text{Beta}(\alpha_i, \beta_i)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Select arm}}:
  \(A_t \gets \argmax_i \tilde{\mu}_i\)
\item
  \textbf{\textcolor{violet}{Observe reward}}: \(R_t\)
\item
  \textbf{\textcolor{magenta}{Update posterior}} for arm \(A_t\):

  \begin{itemize}
  \tightlist
  \item
    If \(R_t = 1\): \(\alpha_{A_t} \gets \alpha_{A_t} + 1\)
  \item
    If \(R_t = 0\): \(\beta_{A_t} \gets \beta_{A_t} + 1\)
  \end{itemize}
\end{enumerate}

\endgroup
\end{frame}

\begin{frame}{Thompson Sampling: Practical Performance}
\protect\phantomsection\label{thompson-sampling-practical-performance}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Theoretical Guarantees}}:

  \begin{itemize}
  \tightlist
  \item
    Achieves \(L_T = O(\log T)\) regret (optimal)
  \item
    Matches UCB asymptotically
  \item
    Proofs more involved than UCB (Agrawal \& Goyal, 2012)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Empirical Performance}}:

  \begin{itemize}
  \tightlist
  \item
    Often \emph{outperforms} UCB in practice (better constants)
  \item
    Faster convergence to optimal arm in many problems
  \item
    More robust to model misspecification
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Advantages}}:

  \begin{itemize}
  \tightlist
  \item
    Naturally incorporates prior knowledge (via priors)
  \item
    Flexible: Works with any reward distribution (if conjugate prior
    available)
  \item
    Simple randomized algorithm (no complex optimization)
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Disadvantages}}:

  \begin{itemize}
  \tightlist
  \item
    Requires choosing prior (though often uniform prior works well)
  \item
    Posterior updates may be non-trivial for complex distributions
  \item
    Theoretically less understood than UCB (until recent years)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Current Status}}: \emph{De facto} standard in
  practice (especially industry)
\end{itemize}
\end{frame}
