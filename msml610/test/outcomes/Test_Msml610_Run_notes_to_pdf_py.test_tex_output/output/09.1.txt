\begin{frame}{Static vs Dynamic Probabilistic Reasoning}
\protect\phantomsection\label{static-vs-dynamic-probabilistic-reasoning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Static probabilistic reasoning}}

  \begin{itemize}
  \tightlist
  \item
    Random variables have \emph{fixed} values
  \item
    Everything done so far
  \item
    E.g., repairing a car:

    \begin{itemize}
    \tightlist
    \item
      Broken parts remain broken during diagnosis
    \item
      Observed evidence stays fixed
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Dynamic probabilistic reasoning}}

  \begin{itemize}
  \tightlist
  \item
    Random variables \emph{change over time}

    \begin{itemize}
    \tightlist
    \item
      E.g., tracking a plane's location, economic activity
    \end{itemize}
  \item
    E.g., treating a diabetic patient

    \begin{itemize}
    \tightlist
    \item
      Assess patient state, decide insulin dose
    \item
      Evidence: insulin doses, food intake, blood sugar (change over
      time)
    \item
      Time dependency (e.g., metabolic activity, time of day)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Agents in Partially Observable Environments}
\protect\phantomsection\label{agents-in-partially-observable-environments}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.1.png}}

\begin{itemize}
\tightlist
\item
  Agents in \emph{partially observable environments} track current state
  using:

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{red}{Belief state}}

    \begin{itemize}
    \tightlist
    \item
      Store possible world states
    \item
      Use probability theory to quantify belief
    \item
      Posterior distribution of current state given all evidence
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Belief state + Transition model}}

    \begin{itemize}
    \tightlist
    \item
      Predict world evolution in next step
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Sensor model + Percepts}}

    \begin{itemize}
    \tightlist
    \item
      Update belief state
    \end{itemize}
  \end{enumerate}
\item
  \emph{Idea}: handle time by making each quantity a function of time

  \begin{itemize}
  \tightlist
  \item
    \(\vX_{a:b}\) represents variables in \([a, b]\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Agent: Model Components}
\protect\phantomsection\label{agent-model-components}
\begin{enumerate}
\tightlist
\item
  \textbf{\textcolor{red}{State of the world}}: \(\vX_t\)

  \begin{itemize}
  \tightlist
  \item
    Not directly observable
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Prior probability of the state}} at time 0:
  \(\vX_0\)
\item
  \textbf{\textcolor{darkgreen}{Evidence variables}}: \(\vE_t\)

  \begin{itemize}
  \tightlist
  \item
    Observable
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Transition model}}:
  \(\Pr(\vX_t | \vX_{0:t-1})\)

  \begin{itemize}
  \tightlist
  \item
    Models world evolution
  \item
    Specifies probability distribution of state \(\vX_t\) given previous
    values
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Sensor model}}:
  \(\Pr(\vE_t | \vX_{0:t}, \vE_{0:t-1})\)

  \begin{itemize}
  \tightlist
  \item
    Models generation of evidence variables \(\vE_t\)
  \end{itemize}
\end{enumerate}

\end{frame}

\begin{frame}{Discrete vs Continuous Time Models}
\protect\phantomsection\label{discrete-vs-continuous-time-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Discrete time models}}

  \begin{itemize}
  \tightlist
  \item
    View world as time slices (``snapshots'')

    \begin{itemize}
    \tightlist
    \item
      Equal time intervals, equispaced samples
    \item
      Label times \(t = 0, 1, 2, ...\)
    \end{itemize}
  \item
    Each slice contains random variables:

    \begin{itemize}
    \tightlist
    \item
      Hidden RVs (e.g., \(\vX_t\))
    \item
      Observable RVs (e.g., \(\vE_t\))
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Continuous time models}}

  \begin{itemize}
  \tightlist
  \item
    Model uncertainty over continuous time with stochastic differential
    equations (SDEs)
  \item
    Discrete time models approximate SDEs
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Markov Property}
\protect\phantomsection\label{markov-property}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{In general}}, current state \(\vX_t\) depends
  on a growing number of past states: \[
  \Pr(\vX_t | history)
  \defeq \Pr(\vX_t | \vX_{0:t-1})
  = \Pr(\vX_t | \vX_0, \vX_1, ..., \vX_{t-1})
  \]

  \begin{itemize}
  \tightlist
  \item
    Of course, there can't be dependency from the future \(\vX_{t + k}\)
    with \(k > 0\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Markov property}}: current state depends
  (conditionally) only on a finite fixed number of \(k\) previous
  states: \[
  \begin{aligned}
  \Pr(\vX_t | \vX_{0:t-1})
  & = \Pr(\vX_t | \textcolor{red}{\vX_0, \vX_1, ..., \vX_{t-k-1}},
    \textcolor{purple}{\vX_{t-k}, ..., \vX_{t-1}}) \\
  & = \Pr(\vX_t | \textcolor{purple}{\vX_{t-k:t-1}})
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Markov Process}
\protect\phantomsection\label{markov-process}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Markov processes}} (aka Markov chains) have
  the Markov property \[
  \Pr(\vX_t | history) = \Pr(\vX_t | \textcolor{purple}{\vX_{t-k:t-1}})
  \; \forall k, t
  \]
\item
  \textbf{\textcolor{darkgreen}{First-order Markov process}}: current
  state \(\vX_t\) depends only on previous state \(\vX_{t-1}\): \[
  \Pr(\vX_t | history) = \Pr(\vX_t | \textcolor{purple}{\vX_{t-1}})
  \; \forall k, t
  \]

  \begin{itemize}
  \tightlist
  \item
    Next state depends only on previous state, not full history
  \item
    \emph{Intuition}: system ``forgets'' everything except immediate
    last state
  \item
    Bayesian network for a first-order Markov process:
    \includegraphics[width=0.7\linewidth,height=\textheight,keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.3.png}
  \item
    E.g., probability of rain today depends only on yesterday,
    \(\Pr(R_t | R_{t-1}) \; \forall t\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Second-order Markov process}}: current state
  \(\vX_t\) depends only on \(\vX_{t-1}\) and \(\vX_{t-2}\)
\end{itemize}
\end{frame}

\begin{frame}{Time-Homogeneous Process}
\protect\phantomsection\label{time-homogeneous-process}
\begin{itemize}
\item
  Even with the Markov assumption, there is an infinite number of
  probability distributions, one for each \(t\) \[
  \Pr(\vX_t | \vX_{t-1:t-k})
  \]
\item
  \textbf{\textcolor{red}{Time-homogeneous}} (aka stationarity):
  probability remains constant by translation over \(t\) \[
  \Pr(\textcolor{purple}{\vX_t} | \vX_{0:t-1})
  = \Pr(\textcolor{purple}{\vX_{t-k}} | \vX_{0:t-k-1}) \; \forall k, t
  \]

  \begin{itemize}
  \tightlist
  \item
    \emph{Intuition}: even if process evolves, governing laws remain
    unchanged
  \item
    E.g., in the real-world, most physical laws are constant
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{First-Order Time-Homogeneous Process}
\protect\phantomsection\label{first-order-time-homogeneous-process}
\begin{itemize}
\item
  \textbf{\textcolor{red}{First-order Markov property}} \[
  \Pr(\vX_t | history) = \Pr(\vX_t | \textcolor{purple}{\vX_{t-1}})
  \]
\item
  \textbf{\textcolor{darkgreen}{Time-homogeneous}} \[
  \Pr(\textcolor{purple}{\vX_t} | \vX_{0:t-1}) = \Pr(\textcolor{purple}{\vX_{t-k}} | \vX_{0:t-k-1}) \;
  \forall k, t
  \]
\item
  \textbf{\textcolor{blue}{First-order time-homogeneous}}: putting both
  properties together, one conditional probability table suffices: \[
  \Pr(\vX_t | \vX_{t-1}) = \Pr(\vX_{t-k} | \vX_{t-k-1}) \; \forall k, t
  \]

  \begin{itemize}
  \tightlist
  \item
    E.g., rain probability for today depends only on yesterday and is
    constant: \(\Pr(R_t | R_{t-1}) = f(R_{t-1}) \; \forall t\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sensor Model}
\protect\phantomsection\label{sensor-model}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{In general}}, evidence variables \(\vE_t\)
  depend on:

  \begin{itemize}
  \tightlist
  \item
    Previous state of the world \(\vX_{0:t}\)
  \item
    Previous sensor values \(\vE_{0:t-1}\) \[
    \Pr(\vE_t | \vX_{0:t}, \vE_{0:t-1})
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Sensor Markov property}}

  \begin{itemize}
  \tightlist
  \item
    Assume sensor value \(\vE_t\) depends only on current state
    \(\vX_t\), not on previous sensor values \[
    \Pr(\vE_t | \vX_{0:t}, \textcolor{purple}{\vE_{0:t-1}})
    = \Pr(\vE_t | \vX_t)
    \]
  \item
    In a Bayesian network, even if \(\vX_t\) and \(\vE_t\) are
    contemporaneous, the arrow goes from \(\vX_t \to \vE_t\) since the
    world causes the sensor to take on particular values
  \end{itemize}
\end{itemize}
\end{frame}

\protect\phantomsection\label{sensor-model-rain-example}

  \(\vX_t \to \vE_t\)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(Rain_t \to Umbrella_t\), since rain ``causes'' the umbrella
    to appear
  \item

\begin{frame}{Prior Probability}
\protect\phantomsection\label{prior-probability}
\begin{itemize}
\tightlist
\item
  Complete system specification needs
  \textbf{\textcolor{red}{prior probability of state variables}} at
  initial time, \(\Pr(\vX_0)\)

  \begin{itemize}
  \tightlist
  \item
    Represents initial belief about system state before observations
  \item
    Crucial for initializing state estimation process
  \end{itemize}
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    \(\vX_0\) represents position and velocity of a moving object

  \end{itemize}

\begin{frame}{First-Order Markov Process: Joint Distribution}
\protect\phantomsection\label{first-order-markov-process-joint-distribution}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: model a sequence of states
  \(\vX_0, \vX_1, ..., \vX_t\) and observations \(\vE_1, ..., \vE_t\)
  over time, i.e., \(\Pr(\vX_{0:t}, \vE_{1:t})\)
\item
  \textbf{\textcolor{orange}{Express the joint distribution}} of \(n\)
  random variables using the chain rule: \[
  \Pr(\vX_1, ..., \vX_n) = \prod_{i=1}^n \Pr(\vX_i | \vX_{0:i-1})
  \]
\item
  Like for Bayesian networks
  \textbf{\textcolor{darkgreen}{factorize joint distribution}} according
  to graph dependencies: \[
  \Pr(\vX_1, ..., \vX_n) = \prod_{i=1}^n \Pr(\vX_i | \text{parents}(\vX_i))
  \]
\end{itemize}

\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{teal}{First-order Markov assumption}}:
  \[\Pr(\vX_i | \vX_{0:i-1}) = \Pr(\vX_i | \vX_{i-1})\]
\item
  \textbf{\textcolor{cyan}{First-order Markov sensor model}}:
  \[\Pr(\vE_i | \vX_{0:i}, \vE_{1:i-1}) = \Pr(\vE_i | \vX_i)\]
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.5.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{First-Order Markov Process: Joint Distribution}
\protect\phantomsection\label{first-order-markov-process-joint-distribution-1}
\begin{itemize}
\tightlist
\item
  Putting everything together, the joint distribution probability for a
  time-homogeneous first-order Markov process: \begin{align*}
  \Pr(\vX_{0:t}, \vE_{1:t})
  & = \red{\Pr(\vX_0)} 
    \prod_{i=1}^t \green{\Pr(\vX_i | \vX_{i-1})} \blue{\Pr(\vE_i | \vX_i)} \\
  & = \red{prior} \times \prod_{i} \green{transition model} \times \blue{sensor model} \\
  \end{align*}
\end{itemize}

\vspace{-0.5cm}

\begin{itemize}
\tightlist
\item
  \textbf{Remarks}:

  \begin{itemize}
  \tightlist
  \item
    The state evolves probabilistically from the previous state
    (transition model)
  \item
    This structure reduces complexity and enables tractable inference
  \item
    A Bayesian network can represent a temporal model by modeling time
    with indices \(t\), i.e., ``unrolling the model''
  \end{itemize}
\item
  \textbf{Problem}: infinite \(t\), even assuming the Markov property
\end{itemize}
\end{frame}

\begin{frame}{Improving Approximation of Real-World Systems}

  \textbf{\textcolor{red}{reasonable approximation of reality}}?

  \begin{itemize}
  \tightlist
  \item
    Particle following random walk is well represented by Markov process
  \item
    In umbrella example, rain depends only on previous day

  \item
    \emph{Increase number of sensor variables}

    \begin{itemize}
    \tightlist
    \item

\begin{frame}{Task 1: Filtering}
\protect\phantomsection\label{task-1-filtering}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Filtering}} (aka ``state estimation'')
  computes the posterior distribution of the \emph{current state} given
  \emph{all evidence to date}: \[
  \Pr(\vX_t | \vE_{1:t} = \ve_{1:t})
  \]

  \begin{itemize}
  \tightlist
  \item
    Estimate probability of rain today, given all umbrella observations
    so far \(\Pr(Rain_t | Umbrella_{1:t})\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Filtering needed by a rational agent to \emph{track current state}
  \item
    Agent believes current state \(\Pr(\vX_{t-1})\) at time \(t - 1\)
  \item
    New evidence \(\ve_t\) arrives for time \(t\)
  \item
    Agent updates belief about current state \(\Pr(\vX_t)\) at time
    \(t\)
  \end{itemize}
\item
  \emph{``Filtering''} refers to filtering out noise in a signal by
  estimating system parameters
\end{itemize}
\end{frame}

\begin{frame}{Task 2: Prediction}
\protect\phantomsection\label{task-2-prediction}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Prediction}} involves predicting the posterior
  distribution over a \emph{future state}, given \emph{all evidence to
  date}: \[
  \Pr(\vX_{t+k} | \vE_{1:t} = \ve_{1:t}) \text{ with } k > 0
  \]

  \begin{itemize}
  \tightlist
  \item
    E.g., compute the probability of rain three days from now: \[
    \Pr(Rain_{t+3} | Umbrella_{1:t})
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Prediction helps rational agents evaluate actions based on expected
    outcomes
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Task 3: Smoothing}
\protect\phantomsection\label{task-3-smoothing}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Smoothing}} compute posterior distribution
  over a \emph{past state} given \emph{all past, present, and future
  evidence}: \[
  \Pr(\vX_k | \vE_{1:t} = \ve_{1:t}) \text{ with } 0 \le k < t
  \]

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Important}}: you have information
    about the ``future'' of the evidence, but not the state
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Smoothing provides a better state estimate by incorporating more
    future evidence
  \item
    E.g., compute the probability it rained last Wednesday, given all
    observations up to today
  \end{itemize}
\item
  The term \emph{``smoothing''} refers to the state estimate being
  smoother than filtering
\end{itemize}
\end{frame}

\begin{frame}{Task 4: Most-Likely Explanation}
\protect\phantomsection\label{task-4-most-likely-explanation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Most-likely explanation}} finds the sequence
  of states \(\vX_{1:t}\) most likely to have generated observations
  \(\vE_{1:t}\): \[
  \argmax_{\vX_{1:t}} \Pr(\vX_{1:t} | \vE_{1:t})
  \]

  \begin{itemize}
  \tightlist
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Umbrella appeared on 3 days, not on the fourth
    \item
      Most likely explanation: rained for 3 days, then stopped
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Speech recognition: most likely sequence of words given sounds
  \item
    Digital processing: reconstruct bit strings over a noisy channel
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Task 5: Learning}
\protect\phantomsection\label{task-5-learning}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Learning}} involves estimating the transition
  model \(\Pr(\vX_t | \vX_{0:t-1})\) and the sensor model
  \(\Pr(\vE_i | \vX_i)\) from observations
\item
  Learning benefits from smoothing rather than filtering for better
  state estimates

  \begin{itemize}
  \tightlist
  \item
    Smoothing uses all data to estimate states, leading to more accurate
    models
  \item
    E.g., in weather prediction, smoothing uses past, present, and
    future data to better estimate current weather state
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Solving Task 1: Filtering}
\protect\phantomsection\label{solving-task-1-filtering}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Filtering}} computes the posterior
  distribution of the \emph{current state} given \emph{all evidence to
  date}, i.e., \(\Pr(\vX_t | \vE_{1:t} = \ve_{1:t})\)
\item
  A practical filtering algorithm updates the current state estimate
  \(\vX_{t+1}\) using the previous state \(\vX_t\) and the new evidence
  \(\ve_{t+1}\)

  \begin{itemize}
  \tightlist
  \item
    Instead of recomputing each state by going over the entire history
    of the percepts
  \item
    Aka ``recursive state estimation'' \begin{align*}
    & \Pr(\vX_{t+1} | \ve_{1:t+1}) = f(\Pr(\vX_t | \ve_{1:t}), \ve_{t+1}) \\
    & NextState = f(PreviousState, \ve_{t+1}) \\
    \end{align*}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Why?}}

  \begin{itemize}
  \tightlist
  \item
    Time and space requirements for updating must be constant for a
    (finite) agent to keep track of current state indefinitively
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Is it possible?}}

  \begin{itemize}
  \tightlist
  \item
    What is the formula \(f(...)\)?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Recursive Filtering: Update Formula}
\protect\phantomsection\label{recursive-filtering-update-formula}
\begin{itemize}
\item
  Compute the state at time \(t+1\) with all the evidence up to that
  time
\item
  Assume that state and evidence are scalar and not vector
  \begingroup \footnotesize
  \begin{alignat*}{2}
  & \Pr(X_{t+1} | e_{1:t+1})
    & & \\
  &= \Pr(X_{t+1} | \red{e_{1:t}, e_{t+1}})
    & \quad & \red{\text{Divide up the evidence}} \\
  &= \alpha \Pr(e_{t+1}|X_{t+1},\green{e_{1:t}}) \Pr(X_{t+1}|\green{e_{1:t}})
    & \quad & \green{\text{Bayes rule given }} \\
  &= \alpha \Pr(e_{t+1}|X_{t+1}) \Pr(X_{t+1}|e_{1:t})
    & \quad & \text{\cyan{Markov sensor assumption}} \\
  &= \alpha \Pr(e_{t+1}|X_{t+1}) \sum_{x_t} \Pr(X_{t+1}|\blue{x_t},\violet{e_{1:t}}) \Pr(\blue{x_t} | e_{1:t})
    & \quad & \text{\blue{Condition on current state}} \\
  &= \alpha \Pr(e_{t+1}|X_{t+1}) \sum_{x_t} \Pr(X_{t+1}|x_t) \Pr(x_t | e_{1:t})
    & \quad & \text{\violet{Markov assumption}} \\
  \end{alignat*}
  \endgroup
\item
  It has the expected form: \[
  \Pr(X_{t+1} | e_{1:t+1}) = f(\Pr(X_t | e_{1:t}), e_{t+1})
  \]
\end{itemize}
\end{frame}

\begin{frame}{Recursive Filtering: Update Formula}
\protect\phantomsection\label{recursive-filtering-update-formula-1}
\begin{itemize}
\tightlist
\item
  The update formula for the state is: \[
  \Pr(X_{t+1} | e_{1:t+1})
  = \alpha \Pr(e_{t+1}|X_{t+1}) \sum_{x_t} \Pr(X_{t+1}|x_t) \Pr(x_t | e_{1:t})
  \]
\item
  The next state is ``Sensor model x Transition model x Recursive
  state''

  \begin{itemize}
  \tightlist
  \item
    Sensor model: \(\Pr(e_{t+1}|X_{t+1})\)
  \item
    Transition model: \(\Pr(X_{t+1}|x_t)\)
  \item
    Recursive term: \(\Pr(x_t|e_{1:t})\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Recursive Filtering: Intuition}
\protect\phantomsection\label{recursive-filtering-intuition}
\begin{itemize}
\item
  Recursive state estimation updates the state belief as new evidence
  arrives \[
  \textcolor{red}{\Pr(X_{t+1} | e_{1:t+1})}
  = \textcolor{darkgreen}{\alpha \Pr(e_{t+1} | X_{t+1})}
  \textcolor{blue}{\sum_{x_t} \Pr(X_{t+1} | x_t) \textcolor{purple}{\Pr(x_t | e_{1:t})}}
  \] in \textbf{\black{two steps}}

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\black{Prediction step}}: Use the transition model to
    predict the next state based on the current belief \[
    \textcolor{blue}{\Pr(X_{t+1} | e_{1:t})
      = \sum_{x_t} \Pr(X_{t+1} | x_t) \textcolor{purple}{\Pr(x_t | e_{1:t})}}
    \]

    \begin{itemize}
    \tightlist
    \item
      Intuition: Project the current belief forward using the model of
      system evolution
    \end{itemize}
  \item
    \textbf{\black{Update step}}: Incorporate the new observation to
    refine the prediction \[
    \textcolor{red}{\Pr(X_{t+1} | e_{1:t+1})} =
    \textcolor{darkgreen}{\alpha \Pr(e_{t+1} | X_{t+1})} \textcolor{blue}{\Pr(X_{t+1} | e_{1:t})}
    \]

    \begin{itemize}
    \tightlist
    \item
      Intuition: Correct the prediction using the likelihood of the new
      evidence
    \end{itemize}
  \end{enumerate}
\item
  Maintain \purple{$\Pr(X_t | e_{1:t})$}, the probability of the current
  state given all past evidence

  \begin{itemize}
  \tightlist
  \item
    E.g., in a weather model, if it was likely to rain today and rain
    usually continues, the prediction leans toward rain tomorrow
  \item
    Seeing an umbrella supports this and updates the belief accordingly
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Forward update}
\protect\phantomsection\label{forward-update}
\begin{itemize}
\item
  We achieved: \begin{align*}
  \Pr(\vX_{t+1} | \ve_{1:t+1})
  &= \alpha \Pr(\ve_{t+1}|\vX_{t+1}) \sum_{x_t} \Pr(\vX_{t+1}|\vx_t) \Pr(\vx_t | \ve_{1:t}) \\
  &= f(\Pr(\vX_t | \ve_{1:t}), \ve_{t+1}) \\
  \end{align*}
\item
  The filtered estimate \(\vf_{1:t} = \Pr(\vX_t | \ve_{1:t})\) is
  propagated forward and updated by each transition and new observation
  \[
  \vf_{1:t+1} = Forward(\vf_{1:t}, \ve_{t+1})
  \] starting with the initial condition \(\vf_{1:0} = \Pr(\vX_0)\)

  \begin{itemize}
  \tightlist
  \item
    This is called ``forward update''
  \end{itemize}
\item
  This process allows efficient online inference without storing the
  full history

  \begin{itemize}
  \tightlist
  \item
    Time and space requirements for updating is constant
  \item
    A (finite) agent can keep track of current state indefinitively
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Solving Task 2: Prediction}
\protect\phantomsection\label{solving-task-2-prediction}
\begin{itemize}
\item
  Prediction is equivalent to filtering without updating the state with
  new evidence, since there is no evidence

  \begin{itemize}
  \tightlist
  \item
    Only the transition model is needed, not the sensor model
  \end{itemize}
\item
  The rule predicting state \(\vX_{t+k+1}\) given state \(\vX_{t+k}\)
  and evidence \(\vE_{1:t}\) is:

  \[
  \Pr(\vX_{t+k+1} | \ve_{1:t}) =
  \sum_{\vx_{t+k}} \Pr(\vX_{t+k+1} | \vx_{t+k}) \Pr(\vx_{t+k} | \ve_{1:t})
  \]
\item
  This equation can be used recursively to advance over time

  \begin{itemize}
  \tightlist
  \item
    Predicting even a few steps ahead generally incurs large uncertainty
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Solving Task 3: Smoothing}
\protect\phantomsection\label{solving-task-3-smoothing}
\begin{itemize}
\tightlist
\item
  You want to calculate the probability distribution over the hidden
  state at time \(k\), given all evidence up to time \(t\) (in the
  future!) \[
  \Pr(X_k | e_{1:t}) \text{  where } 0 \le k < t
  \]

  \begin{itemize}
  \tightlist
  \item
    Filtering gives \(\Pr(X_k | e_{1:k})\) using past and present
    evidence
  \item
    Smoothing refines the estimate of past states using later evidence
  \end{itemize}
\item
  \textbf{\textcolor{red}{Example}}

  \begin{itemize}
  \tightlist
  \item
    You're tracking whether it was raining yesterday
  \item
    You had some evidence up to yesterday (e.g., a cloudy sky)
  \item
    Today you see puddles on the ground
  \item
    That new observation supports the idea that yesterday was raining
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Task 3: Smoothing: Update Formula}
\protect\phantomsection\label{task-3-smoothing-update-formula}
\begin{itemize}
\item
  Using the same math as for filtering and the two key assumptions of
  Markov process and Markov sensor
\item
  \textbf{\textcolor{red}{Forward Pass (aka filtering):}}

  \begin{itemize}
  \tightlist
  \item
    Move forward through time, using the filtering algorithm to compute:
    \[
    f_{1:k} = \Pr(X_k | e_{1:k})
    \]
  \item
    This gives you a ``best guess'' of the state at time \(k\), based
    only on evidence up to \(k\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Backward Pass (aka smoothing):}}

  \begin{itemize}
  \tightlist
  \item
    Move backward through time from time \(t\), computing: \[
    b_{k+1:t} = \Pr(e_{k+1:t} | X_k)
    \]
  \item
    This captures how likely the future evidence is, given a particular
    value of \(X_k\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Combine them:}}

  \begin{itemize}
  \tightlist
  \item
    Multiply forward and backward messages to get: \[
    \Pr(X_k | e_{1:t}) \propto f_{1:k} \times b_{k+1:t}
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Task 4: Most Likely Explanation: Intuition 1/2}
\protect\phantomsection\label{task-4-most-likely-explanation-intuition-12}
\begin{itemize}
\tightlist
\item
  You are tracking the weather (sunny or rainy) based on whether someone
  carries an umbrella

  \begin{itemize}
  \tightlist
  \item
    You can't see \(Weather\) directly (hidden state), but you observe
    umbrellas (which is a noisy observation)
  \item
    You have 5 observations \(Umbrella = [T, T, F, T, T]\)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Question}}: what is the most likely sequence
  of \(Weather\) states that explains the \(Umbrella\) observations?

  \begin{itemize}
  \tightlist
  \item
    You know something about:

    \begin{itemize}
    \tightlist
    \item
      the transition model (i.e., ``it tends to rain several days in a
      row'')
    \item
      the sensor model (i.e., ``people often forget the umbrella'')
    \end{itemize}
  \end{itemize}
\item
  Mathematically: \[
  \argmax_{x_{1:t}} \Pr(x_{1:t} | e_{1:t}) =
  \argmax_{Weather_{1:t}} \Pr(Weather_{1:t} | Umbrella_{1:t})
  \]
\end{itemize}
\end{frame}

\begin{frame}{Task 4: Most Likely Explanation: Intuition 2/2}
\protect\phantomsection\label{task-4-most-likely-explanation-intuition-22}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Naive approach}}: Use smoothing to choose the
  most likely state at each time step

  \begin{itemize}
  \tightlist
  \item
    Cons

    \begin{itemize}
    \tightlist
    \item
      Might lead to an implausible overall path
    \item
      Suboptimal since the question addresses joint probability and we
      are not using all the information (only one step at the time!)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Viterbi algorithm}}:

  \begin{itemize}
  \tightlist
  \item
    Constructs a path through a state-time graph with states as nodes
    and transitions as edges
  \item
    Finds the most likely entire path through the hidden states
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key difference:}}

  \begin{itemize}
  \tightlist
  \item
    In speech recognition, find the most likely word sequence behind a
    noisy audio signal

    \begin{itemize}
    \tightlist
    \item
      Smoothing: Best guess per time step (may miss non-English words or
      suboptimal sequence)
    \item
      Viterbi: Best overall path (maximizes joint probability of the
      entire sequence)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Viterbi Algorithm: Intuition}
\protect\phantomsection\label{viterbi-algorithm-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}: Find the most likely sequence of
  hidden states given observations
\end{itemize}

\begin{enumerate}
\tightlist
\item
  Initialization

  \begin{itemize}
  \tightlist
  \item
    At \(t = 1\), estimate probability of starting in each state using
    initial state distribution and observation likelihood
  \end{itemize}
\item
  Recursion via dynamic programming

  \begin{itemize}
  \tightlist
  \item
    For each \(t > 1\), for each state \(x_t\):
  \item
    Compute maximum probability path to \(x_t\) from any previous state
  \item
    Use:

    \begin{itemize}
    \tightlist
    \item
      \(\Pr(x_t | x_{t-1})\): transition model
    \item
      \(\Pr(e_t | x_t)\): sensor model
    \item
      Best path probability to \(x_{t-1}\) from prior step
    \end{itemize}
  \item
    Store probability and corresponding back-pointer to \(x_{t-1}\)
  \end{itemize}
\item
  Termination and backtrace

  \begin{itemize}
  \tightlist
  \item
    At final time \(t = T\), identify state with highest final
    probability
  \item
    Trace back through stored pointers to reconstruct optimal path
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Viterbi Algorithm: Example 1/2}
\protect\phantomsection\label{viterbi-algorithm-example-12}
\begin{itemize}
\tightlist
\item
  You observe a friend carrying an umbrella over 3 days

  \begin{itemize}
  \tightlist
  \item
    \(Umbrella = [Yes, Yes, No]\)
  \end{itemize}
\item
  You want to infer the most likely sequence of hidden \(Weather\)
  states

  \begin{itemize}
  \item
    States: \(S = \{\text{Sunny}, \text{Rainy}\}\) (weather)
  \item
    Observations: \(O = \{\text{Yes}, \text{No}\}\) (umbrella)
  \item
    Initial Probabilities:

    \[
    \Pr(\text{Sunny}) = 0.6, \quad \Pr(\text{Rainy}) = 0.4
    \]
  \item
    Transition Probabilities:

    \[
    \begin{aligned}
    \Pr(\text{Sunny} \to \text{Sunny}) &= 0.7, \quad \Pr(\text{Sunny} \to \text{Rainy}) = 0.3 \\
    \Pr(\text{Rainy} \to \text{Sunny}) &= 0.4, \quad \Pr(\text{Rainy} \to \text{Rainy}) = 0.6
    \end{aligned}
    \]
  \item
    Observation (Emission) Probabilities:

    \[
    \begin{aligned}
    \Pr(\text{Yes} | \text{Sunny}) &= 0.1, \quad \Pr(\text{No} | \text{Sunny}) = 0.9 \\
    \Pr(\text{Yes} | \text{Rainy}) &= 0.8, \quad \Pr(\text{No} | \text{Rainy}) = 0.2
    \end{aligned}
    \]
  \end{itemize}
\end{itemize}
\end{frame}
