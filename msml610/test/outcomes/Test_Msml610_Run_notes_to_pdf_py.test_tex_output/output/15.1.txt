\begin{frame}{Where LLMs They Excel}
\protect\phantomsection\label{where-llms-they-excel}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{LLMs excel at pattern matching at scale}}

  \begin{itemize}
  \tightlist
  \item
    Trained on massive text corpora to capture statistical regularities
  \item
    Exceptional for prediction of the next word
    \[\Pr(\text{next token} | \text{history})\]
  \item
    Strong performance on language understanding, summarization,
    generation
  \item
    World knowledge encoded through pre-training

    \begin{itemize}
    \tightlist
    \item
      E.g., the next word in \emph{``The capital of France''} is
      \ldots{}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Strengths}}:

  \begin{itemize}
  \tightlist
  \item
    Massive scale enables capturing complex statistical patterns
  \item
    Transfer learning from pre-training to diverse downstream tasks
  \item
    Few-shot learning through in-context examples
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Critical LLMs Limitations}
\protect\phantomsection\label{critical-llms-limitations}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{LLMs struggle with explicit causal reasoning}}

  \begin{itemize}
  \tightlist
  \item
    Pattern matching cannot distinguish correlation from causation
  \item
    No built-in mechanism to reason about counterfactuals (what-if
    scenarios)
  \item
    Cannot reliably predict interventions vs.~observational data
  \item
    Confounding variables not explicitly handled
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Given \emph{``Ice cream sales and drowning deaths both spike in
    summer''},
  \item
    Current LLMs infer a causal link rather than recognizing temperature
    as a confounder
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Critical gaps}}:

  \begin{itemize}
  \tightlist
  \item
    Intervention reasoning: \emph{``What if we increase X? How does Y
    change?''}
  \item
    Counterfactual reasoning: \emph{``What would have happened if the
    past were different?''}
  \item
    Causal discovery: Learning causal structure from data
  \item
    Robustness to distribution shift: Generalizing under causal changes
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Pattern Recognition and Its Limits}
\protect\phantomsection\label{pattern-recognition-and-its-limits}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pattern-based reasoning}}: Learn
  \(f: X \to Y\) from observed associations

  \begin{itemize}
  \tightlist
  \item
    Extract statistical regularities: what features predict outcomes
  \item
    No understanding of \emph{why} these associations exist
  \item
    Works very well in stationary environments
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causal reasoning}}: Learn mechanisms,
  i.e., the underlying data-generating process

  \begin{itemize}
  \tightlist
  \item
    Understand \emph{why} \(X\) causes \(Y\) through explicit models
  \item
    Transfer knowledge to new interventions and environments
  \item
    Explain and debug failures
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.4\linewidth}
\textbf{\textcolor{blue}{Pattern Matching}}

\begin{itemize}
\tightlist
\item
  Pros

  \begin{itemize}
  \tightlist
  \item
    Quality scales with data quantity
  \item
    Captures complex associations
  \item
    Practical for stable settings
  \end{itemize}
\item
  Cons

  \begin{itemize}
  \tightlist
  \item
    Breaks under distribution shift
  \item
    Cannot reason about interventions
  \item
    Vulnerable to confounding
  \item
    Opaque causal mechanisms
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\textbf{\textcolor{violet}{Causal Reasoning}}

\begin{itemize}
\tightlist
\item
  Pros

  \begin{itemize}
  \tightlist
  \item
    Robust to distribution shift
  \item
    Generalizes to new interventions
  \item
    Interpretable mechanisms
  \item
    Handles confounding
  \end{itemize}
\item
  Cons

  \begin{itemize}
  \tightlist
  \item
    Requires causal model specification
  \item
    Sample complexity for learning
  \item
    Stronger assumptions
  \end{itemize}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Robustness: When Pattern Matching Fails}
\protect\phantomsection\label{robustness-when-pattern-matching-fails}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Distribution shift}}: Test data differs from
  training data

  \begin{itemize}
  \tightlist
  \item
    Spurious correlations don't hold in deployment
  \item
    LLMs confidently make wrong predictions due to pattern reliance
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}: Medical diagnosis

  \begin{itemize}
  \tightlist
  \item
    Training: \textcolor{blue}{\texttt{Hospital A}} uses test \(Y\) for
    condition \(X\)
  \item
    Faulty calibration: test \(Y\) predicts diagnosis, not disease
  \item
    Model learns: high Y \(\to\) condition X
  \item
    New \textcolor{blue}{\texttt{Hospital B}}: model fails with correct
    calibration
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Intervention robustness}}: Actions change the
  world

  \begin{itemize}
  \tightlist
  \item
    Feedback loops: recommending product \(A\) increases demand, alters
    patterns
  \item
    Reward hacking: optimizing proxy metric, not true objective
  \item
    Policy change: new rules invalidate historical patterns
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}: Content recommendation

  \begin{itemize}
  \tightlist
  \item
    Pattern: users engaging with trending topics get more
    recommendations
  \item
    Intervention: change algorithm to reduce trending bias
  \item
    Unintended consequence: recommendation patterns shift, breaking past
    associations
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Chain-of-Thought Prompting}
\protect\phantomsection\label{chain-of-thought-prompting}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Chain-of-thought (CoT)}} encourages model to
  break down complex reasoning

  \begin{itemize}
  \tightlist
  \item
    Instead of: \emph{``Does X cause Y?''} \(\to\) Direct answer
  \item
    CoT: \emph{``Let me think step by step\ldots{}''} \(\to\)
    Intermediate reasoning steps
  \end{itemize}
\item
  How it helps with causal reasoning:

  \begin{itemize}
  \tightlist
  \item
    Forces explicit consideration of mechanisms
  \item
    Enables backtracking and error correction
  \item
    Makes assumptions and reasoning transparent
  \item
    Allows verification of causal logic
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}: CoT for causal inference Q:
  ``If we increase advertising budget, will sales increase?'' CoT: ``Let
  me think about this:

  \begin{enumerate}
  \tightlist
  \item
    Advertising increases brand awareness (mechanism)
  \item
    Brand awareness increases purchase intent (mechanism)
  \item
    Purchase intent leads to sales (mechanism)
  \item
    Check for confounders (Are we only increasing budget in growing
    markets?)
  \item
    Feedback loops: Does higher sales justify more spending?''
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Limitations of plain CoT}}:

  \begin{itemize}
  \tightlist
  \item
    Pattern-based, not mechanistic
  \item
    Can rationalize incorrect causal claims
  \item
    Doesn't ground reasoning in data or formal causal models
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Causal Prompting Frameworks}
\protect\phantomsection\label{causal-prompting-frameworks}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Structured causal prompts}} guide explicit
  causal reasoning
\item
  \textbf{\textcolor{blue}{Example}}: Structured causal prompt template

\begin{verbatim}
Problem: [description]

Variables:
- Treatment: [what are we intervening on?]
- Outcome: [what do we care about?]
- Confounders: [what common causes exist?]

Causal structure: [draw or describe relationships]

Identification strategy: [how do we isolate the causal effect?]
  - Assumption 1: [e.g., no unmeasured confounding]
  - Assumption 2: [e.g., positivity/overlap]
  - Assumption 3: [e.g., no feedback loops]

Conclusion: [the causal effect is...]
\end{verbatim}
\end{itemize}
\end{frame}

\begin{frame}{Causal Agent Architecture (2/2)}
\protect\phantomsection\label{causal-agent-architecture-22}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Causal model}} represents assumptions about
  mechanisms

  \begin{itemize}
  \tightlist
  \item
    Structural Causal Model (SCM): \(Y := f_Y(PA_Y, U_Y)\) for each
    variable
  \item
    Parameters: effect sizes, functional forms
  \item
    Uncertainty: distributions over models or parameters
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inference module}} answers causal
  queries

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(Y | do(X = x))\) : effect of intervention X on Y
  \item
    \(Y_{x'} = f_Y(x', U_Y)\) : counterfactual outcomes
  \item
    Effect heterogeneity: effects differ across populations
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Planning module}} optimizes actions

  \begin{itemize}
  \tightlist
  \item
    Evaluate action \(a\): \(\EE[U | do(a)]\) where U is utility
  \item
    Consider robustness: What if causal model is wrong?
  \item
    Explore vs.~exploit: Balance testing assumptions vs.~maximizing
    reward
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Learning module}} updates model

  \begin{itemize}
  \tightlist
  \item
    Fit parameters to observed data
  \item
    Test causal assumptions (Markov, faithfulness)
  \item
    Detect model misspecification
  \end{itemize}
\end{itemize}

\begin{frame}{Causal Markov Decision Processes (MDPs)}
\protect\phantomsection\label{causal-markov-decision-processes-mdps}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Standard MDP}}:
  \(M = \langle \mathcal{S}, \mathcal{A}, P, R \rangle\)

  \begin{itemize}
  \tightlist
  \item
    State space, action space, transition dynamics, reward function
  \item
    Dynamics may embed causal structure implicitly
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causal MDP}}: explicitly model causal
  mechanisms

  \begin{itemize}
  \tightlist
  \item
    Variables \(X = (X_1, \ldots, X_n)\) that evolve over time
  \item
    Actions intervene on specific variables
  \item
    Rewards depend on achievable states
  \item
    Transition (causal functions):
    \(X_t' := f(X_t, A_t, \varepsilon_t)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Medical treatment sequential
  decision

  \begin{itemize}
  \tightlist
  \item
    Variables: \((Patient\_Severity, Treatment, Patient\_Outcome)\)
  \item
    Action: choose treatment type
  \item
    Outcome: health improvement (depends on severity + treatment)
  \item
    Causal structure: Treatment has different effects for different
    severity levels
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Value Functions and Causal Effects}
\protect\phantomsection\label{value-functions-and-causal-effects}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Standard RL}}: value function
  \(V(s) = \EE[\sum_t \gamma^t r_t | s_0 = s]\)
\item
  \textbf{\textcolor{darkgreen}{Causal RL}}: value function accounts for
  causal effects

  \begin{itemize}
  \tightlist
  \item
    \(V(s) = \EE[R | do(A = a^*), S = s]\) where \(a^*\) is optimal
    action
  \item
    Policies must account for how actions change world state
  \item
    Offline learning: learn from historical data with unknown
    confounding
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Q-function with causal effects}}:
  \[Q(s, a) = \Pr(\text{Reward} | do(A = a), S = s)\]

  \begin{itemize}
  \tightlist
  \item
    Standard RL assumes known transition:
    \[\Pr(\text{next state} | a, s) = \Pr(s' | a, s)\]
  \item
    Causal RL must infer: Does \(a\) actually cause \(s'\), or is \(s\)
    a confounder?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Policy Robustness: Worst-Case Planning}
\protect\phantomsection\label{policy-robustness-worst-case-planning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Robust planning}}: find policy that performs
  well under model uncertainty

  \begin{itemize}
  \tightlist
  \item
    Let \(\Theta\) be set of possible causal models
  \item
    Robust value: \[
    V_{\text{robust}}(s) = \max_a \min_{\theta \in \Theta} Q_\theta(s, a)
    \]
  \item
    Guarantees: policy works even if true model is worst-case
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Uncertainty sets}}:

  \begin{itemize}
  \tightlist
  \item
    Parameter uncertainty: effect sizes could be different
  \item
    Structural uncertainty: causal graph might be different
  \item
    Identification uncertainty: causal effects may be unidentified
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Drug dosage decision under causal
  uncertainty

  \begin{itemize}
  \tightlist
  \item
    Low dose vs.~high dose: which is safer?
  \item
    Risk: high dose might have nonlinear side effects we haven't modeled
  \item
    Robust policy: start with conservative low dose, observe response,
    adapt
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Making Reasoning Explicit and Interpretable}
\protect\phantomsection\label{making-reasoning-explicit-and-interpretable}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Opacity problem}}: Black-box AI systems hard
  to trust and debug

  \begin{itemize}
  \tightlist
  \item
    Neural networks: uninterpretable high-dimensional feature spaces
  \item
    LLMs: billions of parameters, unclear reasoning
  \item
    Consequences: hard to catch failures before deployment
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causal transparency}}: explicit
  mechanisms make reasoning visible

  \begin{itemize}
  \tightlist
  \item
    Causal graph: clear representation of relationships
  \item
    Causal effects: quantify impact of each decision
  \item
    Counterfactuals: explain decisions by contrasting what-ifs
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Loan approval decision

  \begin{itemize}
  \tightlist
  \item
    Black-box: Model says \emph{``Loan denied''}
  \item
    Causal: Model says \emph{``Loan denied because credit score drops
    effect on default rate by 15\%, controlling for income and
    employment''}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Explanations for Decisions}
\protect\phantomsection\label{causal-explanations-for-decisions}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Contrastive explanations}}: Why \(A\) rather
  than \(B\)?

  \begin{itemize}
  \tightlist
  \item
    Not just why \(A\) happened, but what would need to change for \(B\)
    instead
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Recourse}}: How can someone affected by
  decision change the outcome?

  \begin{itemize}
  \tightlist
  \item
    Requires causal model: what actions lead to decision reversal?
  \item
    Fairness concern: recourse may be impossible if requirements are
    correlated with protected attributes
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Hiring decision

  \begin{itemize}
  \tightlist
  \item
    Denial reason (non-causal): \emph{``You have fewer years of
    experience''}
  \item
    Causal recourse: \emph{``Adding 2 years of experience would likely
    change decision, but experience is correlated with age, raising
    fairness concerns''}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Feature importance from causality}}:

  \begin{itemize}
  \tightlist
  \item
    Not correlation (feature co-varies with outcome) \(\to\) confounding
  \item
    True causal importance: direct effect on outcome under intervention
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Finding Brittle Decisions Through Causal Analysis}
\protect\phantomsection\label{finding-brittle-decisions-through-causal-analysis}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Brittleness}}: decisions that break under
  small perturbations

  \begin{itemize}
  \tightlist
  \item
    Spurious correlations learned by pattern-based models
  \item
    Vulnerable to adversarial examples or distribution shift
  \item
    Often due to unrecognized confounding in training data
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causal analysis uncovers brittleness}}:

  \begin{itemize}
  \tightlist
  \item
    Compare learned patterns to causal ground truth
  \item
    Identify decisions driven by spurious correlations
  \item
    Test robustness under causal interventions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Recidivism prediction (criminal
  justice)

  \begin{itemize}
  \tightlist
  \item
    Pattern learned: arrest history strongly predicts re-offense
  \item
    Causal truth: arrest history may be proxy for policing patterns, not
    criminality
  \item
    Brittleness: model fails if policing patterns change (different
    neighborhood, different police policy)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Constraints on Model Predictions}
\protect\phantomsection\label{causal-constraints-on-model-predictions}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Constraint approach}}: restrict model to
  predictions consistent with causal knowledge

  \begin{itemize}
  \tightlist
  \item
    Hard constraints: must enforce causal assumptions (e.g., no effect
    if no mechanism)
  \item
    Soft constraints: penalize predictions violating causal knowledge
  \item
    Helps prevent spurious patterns from being learned
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Monotonicity constraints}}: if treatment
  increases, outcome can't decrease

  \begin{itemize}
  \tightlist
  \item
    E.g., more education should not decrease earnings
  \item
    Helps prevent absurd predictions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Structural constraints}}: encode causal DAG

  \begin{itemize}
  \tightlist
  \item
    Variable A cannot affect B if no directed path A \(\to\) B
  \item
    Reduces model flexibility but increases robustness
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Adversarial Robustness with Causal Models}
\protect\phantomsection\label{adversarial-robustness-with-causal-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Adversarial examples}}: small input
  perturbations that fool models

  \begin{itemize}
  \tightlist
  \item
    Pattern-based models: vulnerable because they rely on associations
  \item
    Causal models: more robust because mechanisms are explicit
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Causal adversarial robustness}}:

  \begin{itemize}
  \tightlist
  \item
    Only perturbations along causal edges can meaningfully affect
    outcome
  \item
    Confounded perturbations (correlating with true causes) are benign
  \item
    Helps distinguish real threats from spurious ones
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Approaches to Bias and Discrimination}
\protect\phantomsection\label{causal-approaches-to-bias-and-discrimination}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Fairness challenge}}: what does it mean to
  treat people fairly?

  \begin{itemize}
  \tightlist
  \item
    Statistical parity: outcomes equally distributed across groups
  \item
    Causal fairness: no discrimination through causal mechanisms
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Three sources of group differences}}:

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Direct discrimination}}: causal effect
    of protected attribute (illegal)
  \item
    \textbf{\textcolor{teal}{Indirect discrimination}}: protected
    attribute causes unprotected predictor which causes outcome
  \item
    \textbf{\textcolor{cyan}{Structural discrimination}}: system
    perpetuates historical inequities
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Example}}: Hiring fairness

  \begin{itemize}
  \tightlist
  \item
    Protected: gender
  \item
    Potential discriminator: field of study (may be correlated with
    gender)
  \item
    Question: can we use field of study if it correlates with gender?
  \item
    Causal answer: depends on whether field of study is a barrier to
    entry or just different productivity
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Definitions of Fairness}
\protect\phantomsection\label{causal-definitions-of-fairness}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Counterfactual fairness}} (Kusner et al.):

  \begin{itemize}
  \tightlist
  \item
    Decisions would be same if protected attribute were different
  \item
    Requires model of how protected attribute influences other variables
  \item
    Strong requirement: may be too restrictive in practice
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Path-specific effects}} (decompositional
  approach):

  \begin{itemize}
  \tightlist
  \item
    Decompose total effect into direct (discriminatory) and indirect
  \item
    Allow indirect effects if not caused by discrimination
  \item
    More nuanced: separates correlation from causation
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Gender wage gap

  \begin{itemize}
  \tightlist
  \item
    Total gap: women earn 20\% less
  \item
    Causal decomposition:

    \begin{itemize}
    \tightlist
    \item
      Direct effect: 5\% unexplained by occupation/experience
      (discrimination)
    \item
      Indirect effect: 15\% via occupational segregation (structural
      inequity)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Constraints for Fair Models}
\protect\phantomsection\label{causal-constraints-for-fair-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Fairness through constraints}}:

  \begin{itemize}
  \tightlist
  \item
    Control for mediators: adjust for variables on causal path
  \item
    Balance confounders: ensure treatment and control groups similar on
    confounders
  \item
    Remove selection bias: correct for how samples were selected
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: College admissions fairness

  \begin{itemize}
  \tightlist
  \item
    Constraint: admission decision should not depend on applicant's
    gender
  \item
    But gender may have influenced GPA (indirect path)
  \item
    Solution: directly adjust for GPA (mediator) if you believe it's
    fair predictor
  \item
    Or: use causal effect of gender on success (counterfactual score)
    instead of observed score
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Constraints on Harmful Outcomes}

  \textbf{\textcolor{red}{Safety challenge}}: prevent AI systems from

  \begin{itemize}
  \tightlist
  \item

  \item

  \item

  \textbf{\textcolor{blue}{Example}}: Content recommendation safety

  \item
    Harm: radicalization through information bubbles
  \item
    Causal path: algorithm optimizes for engagement \(\to\) recommends
    extreme content \(\to\) radicalizes users
  \item
    Safeguard: constraint that algorithm cannot increase polarization
    score
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Monitoring and Adaptation}
\protect\phantomsection\label{causal-monitoring-and-adaptation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Deployment robustness}}: system must work even
  when deployed

  \begin{itemize}
  \tightlist
  \item
    Distribution shift: world changes from training environment
  \item
    Causal monitoring: detect when causal assumptions are violated
  \item
    Adapt: update model or trigger human oversight
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causal anomaly detection}}:

  \begin{itemize}
  \tightlist
  \item
    Expected relationship: \(Y = f(X) + \varepsilon\)
  \item
    Observe: \(Y\) deviates from expectation
  \item
    Diagnosis: Is it random noise, confounding, or violation of causal
    mechanism?
  \item
    Response: retrain, debug, or alert human
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Medical treatment effectiveness
  monitoring

  \begin{itemize}
  \tightlist
  \item
    Deployed model: drug X increases recovery by 30\%
  \item
    Observation: new patient cohort shows only 10\% improvement
  \item
    Diagnosis: new cohort is older, has comorbidities
  \item
    Causal monitoring: heterogeneous effect analysis reveals drug less
    effective for elderly
  \item
    Adaptation: patient stratification in treatment recommendations
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Reasoning for Trustworthy AI}
\protect\phantomsection\label{causal-reasoning-for-trustworthy-ai}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Core insight}}: LLMs excel at pattern matching
  but fail at causal reasoning

  \begin{itemize}
  \tightlist
  \item
    Pattern-based decisions break under distribution shift and
    interventions
  \item
    Causal models provide robustness through explicit mechanisms
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Integration strategy}}: combine LLM
  strengths with causal frameworks

  \begin{itemize}
  \tightlist
  \item
    LLM reasoning + causal tools + explicit models = trustworthy AI
  \item
    Tool use enables grounding claims in data and formal inference
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causal agent architectures}}:

  \begin{itemize}
  \tightlist
  \item
    Explicit causal models in reasoning and planning
  \item
    Planning under uncertainty with robustness
  \item
    Adaptive learning from experience
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Four pillars of trustworthy AI through causality}}:

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{cyan}{Transparency}}: explicit mechanisms are
    interpretable
  \item
    \textbf{\textcolor{blue}{Robustness}}: causal constraints prevent
    brittle decisions
  \item
    \textbf{\textcolor{violet}{Fairness}}: causal decomposition

\end{itemize}
