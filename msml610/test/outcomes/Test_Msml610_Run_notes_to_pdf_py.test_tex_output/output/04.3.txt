\begin{frame}{Similarity-Based Models: Intuition}
\protect\phantomsection\label{similarity-based-models-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Idea}}: the model evaluated in one point
  \(h(\vx)\) is affected by:

  \begin{itemize}
  \tightlist
  \item
    Other data points in the training set \((\vx_n, y_n) \in \calD\)
  \item
    The effect is based on the distance
    \(d(\vx, \vx_n) = \|\vx - \vx_n\|\)
  \end{itemize}
\item
  The model is a \textbf{\textcolor{blue}{superposition of effects}}

  \begin{itemize}
  \tightlist
  \item
    Sum of the effect of each point in the training set, scaled down by
    the distance \[
    h(\vx)
    = \sum_i \text{ effect of } h(\vx_i) \text{ scaled by } d(\vx, \vx_i)
    \]
  \end{itemize}
\item
  This approach allows to define complex decision boundaries
\end{itemize}
\end{frame}

\begin{frame}{Similarity-Based Models: Gaussian Kernels}
\protect\phantomsection\label{similarity-based-models-gaussian-kernels}
\begin{itemize}
\item
  Consider a Gaussian kernel with a ``landmark'' point \(\vx_i\) and a
  similarity distance defined as: \[
  K(\vx, \vx_i) = \exp(- \frac{\| \vx - \vx_i \| ^ 2}{2 \sigma^2})
  \]
\item
  E.g., the hypothesis model has the form: \[
  h(\vx)
  = \sum_{i=1}^3 y_i K(\vx, \vx_i)
  = y_1 K(\vx, \vx_1) + y_2 K(\vx, \vx_2) + y_3 K(\vx, \vx_3)
  \]

  \begin{itemize}
  \tightlist
  \item
    The response is weighting the responses \(y_i = \{0, 1\}\) through
    the similarity of \(\vx\) from the landmark points
  \item
    This can be seen by plotting \(h(\vx)\) on a plane
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Radial Basis Function Model for Regression}
\protect\phantomsection\label{radial-basis-function-model-for-regression}
\begin{itemize}
\item
  Aka RBF
\item
  The model form for \textbf{\textcolor{red}{regression}} is: \[
  h(\vx) = \sum_{i=1}^{N} w_i \exp(-\gamma \|\vx - \vx_i\| ^2)
  \] where:

  \begin{itemize}
  \tightlist
  \item
    If \(\gamma\) is small \(\implies\) the exponential falls off
    slowly, and multiple training points affect a point between them
  \item
    If \(\gamma\) is large \(\implies\) there are spikes centered in the
    training points and nothing outside
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Radial Basis Function Model for Classification}
\protect\phantomsection\label{radial-basis-function-model-for-classification}
\begin{itemize}
\tightlist
\item
  For \textbf{\textcolor{red}{classification}} use a similar approach to
  ``linear regression for classification''

  \begin{itemize}
  \tightlist
  \item
    Fit a regression model: \[
    s(\vx) = \sum_{i=1}^{N} w_i \exp(-\gamma \|\vx - \vx_i\| ^2)
    \]
  \item
    Take the sign to make predictions: \[
    h(\vx) = \sign(s(\vx))
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\protect\phantomsection\label{rbf-number-of-parameters}
\begin{itemize}
\tightlist
\item
  Some variants for RBF:

  \begin{itemize}
  \tightlist
  \item
    Add bias term
  \item
    Use different \(\gamma_i\) for each point
  \item
    Increase degrees of freedom
  \end{itemize}
\item
  RBF has lots of parameters

  \begin{itemize}
  \tightlist
  \item
    Parameters \(\vw\) equal data points \(N\) (e.g., \(N = 10^{9}\))
  \item
    One parameter \(w_i\) per training point (e.g., \(N = 10^6\))
  \item
    \textbf{\textcolor{red}{Cons}}: Negative impact on generalization
    error
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RBF: Reducing Model VC Dimension}
\protect\phantomsection\label{rbf-reducing-model-vc-dimension}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{To reduce number of parameters}}

  \begin{itemize}
  \tightlist
  \item
    Pick \(K \ll N\) centers \(\vmu_1, ... , \vmu_K\) instead of
    \(\vx_1, ... , \vx_N\)

    \begin{itemize}
    \tightlist
    \item
      Use \(k\)-means clustering to find centers
    \item
      Unsupervised learning; doesn't use labels
    \end{itemize}
  \item
    Same as RBF model using distances from cluster centers: \[
    h(\vx) = \sum_{i=1}^{K} w_i \exp (-\gamma \|\vx - \vmu_i\|^2)
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Still many parameters}} because:

  \begin{itemize}
  \tightlist
  \item
    \(K\) (scalar) weights \(w_k\)
  \item
    \(K\) reference points \(\vmu_k\) (\(d\)-dimensional vectors)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RBF: Learning Models (1/2)}
\protect\phantomsection\label{rbf-learning-models-12}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Learn}} \(w_i, \gamma\), with fixed centers
  \(\vmu_i\), for an RBF model: \[
  h(\vx) = \sum_{i=1}^{K} w_i \exp (-\gamma \|\vx - \vmu_i\|^2)
  \]
\item
  \textbf{\textcolor{blue}{Minimize:}} \[
  E_{in} = \sum_i (h_{\vw, \gamma}(\vx_i) - y_i)^2 = f(\vw, \gamma)
  \]

  \begin{itemize}
  \tightlist
  \item
    Use an iterative approach (e.g., EM, coordinate descent)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RBF: Learning Models (2/2)}
\protect\phantomsection\label{rbf-learning-models-22}
\begin{itemize}
\tightlist
\item
  Use iterative approach (similar to EM algorithm):

  \begin{itemize}
  \tightlist
  \item
    Fix \(\gamma\), solve for \(\vw\) (one-step learning)
  \item
    Fix \(\vw\), solve for \(\gamma\) (gradient descent)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Step 1}}

  \begin{itemize}
  \tightlist
  \item
    Assume \(\gamma\) is known and fixed
  \item
    Learn \(\vw\)
  \end{itemize}
\item
  Impose perfect interpolation:
  \[E_{in} = \frac{1}{n} \sum (h(\vx_i) - y_i)^2 = 0\]
\item
  \textbf{\textcolor{blue}{Problem}}: \[
  h(\vx_j)
  = \sum_i w_i \exp(-\gamma \|\vx_i - \vx_j\| ^2)
  = \sum_i w_i \phi_{i,j}
  = \vphi_j^T \vw = y_i
  \]

  \begin{itemize}
  \tightlist
  \item
    \(N\) equations (one per point) and \(N\) unknowns \(\vw\)
  \item
    \(\mat{\Phi}\) is known, function of data set and \(\gamma\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Learning RBF Models}
\protect\phantomsection\label{learning-rbf-models}
\begin{itemize}
\tightlist
\item
  The problem in matrix form is: \[\mat{\Phi} \cdot \vw = \vy\]

  \begin{itemize}
  \tightlist
  \item
    If \(\mat{\Phi}\) is invertible, then \(\vw = \mat{\Phi}^{-1} \vy\)

    \begin{itemize}
    \tightlist
    \item
      Desired values on training points
    \item
      Exponential interpolates other points
    \end{itemize}
  \item
    If \(\mat{\Phi}\) is not invertible, optimize in least square sense:
    \[\argmin_{\vw} E_{in} = \sum_i (h(\vx_j) - y_i)^2\]

    \begin{itemize}
    \tightlist
    \item
      Compute pseudo-inverse (assuming \(\mat{\Phi}^T \mat{\Phi}\) is
      invertible)
    \item
      Assign weights:
      \[\vw = (\mat{\Phi}^T \mat{\Phi})^{-1} \mat{\Phi}^T \vy\]
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{red}{Step 2}}

  \begin{itemize}
  \tightlist
  \item
    Assume \(\vw\) is known and fixed
  \item
    Learn \(\gamma\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RBF Network vs Neural Networks}
\protect\phantomsection\label{rbf-network-vs-neural-networks}
\begin{itemize}
\tightlist
\item
  The regression model for Neural Networks and RBF model is
  \textbf{\textcolor{red}{similar}}:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{RBF}}: \[
    h(\vx)
    = \sum_i w_i e^{-\gamma \|\vx - \vx_i\|^2}
    = \vw^T \vv{\vphi}
    \]
  \item
    \textbf{\textcolor{darkgreen}{Neural networks}}: \[
    h(\vx)
    = \Theta(\vw^{(L)T} \vx^{(L)})
    = \Theta(\vw^{(L)T} \vv{\Theta}(\mW^{(L-1)} ...))
    \]
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Difference}}:

  \begin{itemize}
  \tightlist
  \item
    RBF has a single layer
  \item
    Neural networks have multiple layers
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Similarities}}:

  \begin{itemize}
  \tightlist
  \item
    Combine features with weights using dot product
  \item
    Extract features from inputs

    \begin{itemize}
    \tightlist
    \item
      RBF features: \(e^{-\gamma \|\vx - \vx_i\|^2}\), always \(> 0\)
    \item
      NN hidden layers: features can be \(> 0\) or \(< 0\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RBF Network vs SVM}
\protect\phantomsection\label{rbf-network-vs-svm}
\begin{itemize}
\tightlist
\item
  The \textbf{\textcolor{red}{model form}} is the same:

  \begin{itemize}
  \tightlist
  \item
    RBF: \[h(\vx) = \sign(\sum_i w_i e^{-\gamma \|\vx - \vx_i\|^2}\]
  \item
    SVM: \[h(\vx) = \sign(\vw^T \vx + b)\]
  \end{itemize}
\item
  The interpretation is completely different (interpolation vs large
  margin)

  \begin{itemize}
  \tightlist
  \item
    RBF: all vectors (or centers of few clusters) contribute to the
    model
  \item
    SVM: only support vectors contribute to the model
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{K-Nearest Neighbor (KNN) Model}
\protect\phantomsection\label{k-nearest-neighbor-knn-model}
\begin{itemize}
\tightlist
\item
  The \textbf{\textcolor{red}{model form}} is like: \[
  h_{\vw}(\vx) = \frac{1}{n} \sum_{\vx_i \text{closest to } \vx} w_i
  \]
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Idea}}:

  \begin{itemize}
  \tightlist
  \item
    Closeness implies a distance (e.g., euclidean metric) or similarity
    (e.g., a kernel)
  \item
    Consider the \(k\) closest points to the evaluation point \(\vx\)
  \item
    Take an average of their response
  \end{itemize}
\end{itemize}
\end{column}

    training point
  \item
    For \(k = N\) \(\implies\) single neighborhood
  \item
    Effective number of parameters: \(\frac{N}{k}\), imagining \(N / k\)
    non-overlapping neighborhoods
  \end{itemize}

\begin{frame}{KNN: Assumptions on the Data}
\protect\phantomsection\label{knn-assumptions-on-the-data}

  \begin{itemize}
  \tightlist
  \item

  \end{itemize}
\end{itemize}

\begin{frame}{K-Means Clustering: Problem Formulation}
\protect\phantomsection\label{k-means-clustering-problem-formulation}
\begin{itemize}
\item
  \(N\) \emph{unlabeled} points \(\{\vx_1, \vx_2, ..., \vx_N\}\)
\item
  Partition points into \(K\) clusters \(S_1, ..., S_K\)

  \begin{itemize}
  \tightlist
  \item
    Each cluster defined by center \(\vmu_k\)
  \item
    Each point \(\vx_i\) assigned to cluster \(c(\vx_i)\)
  \item
    Unknowns are \(c(\vx_1), ..., c(\vx_N), \vmu_1, ..., \vmu_K\)
  \end{itemize}
\item
  Minimize distance between each \(\vx_i\) and assigned center
  \(\vmu_k\) where \(k = c(\vx_i)\) \vspace{-0.5cm} \begin{alignat*}{3}
  J(c_1, ..., c_N, \mu_1, ..., \mu_K) 
  &= \sum_{k=1}^K \sum_{\vx_n \in S_k} \|\vx_n - \vmu_k\| ^ 2
  & \text{(scanning the clusters)}
  \\
  &= \sum_{i=1}^N \| \vx_i - \vmu_{c(\vx_i)} \| ^ 2
  & \text{(scanning the points)} \\
  \end{alignat*} \vspace{-1cm}
\item
  \textbf{\textcolor{red}{K-means clustering is NP-hard}}
  (combinatorial) and thus intractable

  \begin{itemize}
  \tightlist
  \item
    In fact there are \(K^N\) possible assignments
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{K-Means Clustering: Lloyd's Algorithm}
\protect\phantomsection\label{k-means-clustering-lloyds-algorithm}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Start with a random assignment}} of \(N\)
  points to \(K\) clusters

  \begin{itemize}
  \tightlist
  \item
    Better than picking random centroids
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Each iteration}} has 2 steps

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{blue}{Step 1}}: Move centroid

    \begin{itemize}
    \tightlist
    \item
      Move each cluster's centroid to the mean point
    \item
      Iterate over \(K\) clusters
    \item
      \(\vmu_k \leftarrow \frac{1}{|S_k|}\sum_{\vx_n \in S_k}\vx_n\)
    \end{itemize}
  \item
    \textbf{\textcolor{violet}{Step 2}}: Cluster assignment

    \begin{itemize}
    \tightlist
    \item
      Assign each \(\vx_n\) to the closest cluster center
    \item
      Iterate over \(N\) points
    \item
      \(S_k \leftarrow
      \{\vx_n: \| \vx_n - \vmu_k \| \le \|\vx_n - \vmu_l\| \;
      \forall l \ne k \}\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{K-Means Clustering: Convergence}
\protect\phantomsection\label{k-means-clustering-convergence}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{K-means algorithm converges}} since:

  \begin{itemize}
  \tightlist
  \item
    Finite number of possible partitions (and values of objective
    functions)
  \item
    Objective function \(J(\cdot)\) always decreases
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Objective function always decreases}}

  \begin{itemize}
  \tightlist
  \item
    Cost function \(J(\vmu_1, ..., \vmu_K, c_1, ..., c_N)\) depends on:

    \begin{itemize}
    \tightlist
    \item
      Centroids \(c_1, .., c_N\)
    \item
      Point assignments \(\vmu_1, ..., \vmu_K\)
    \end{itemize}
  \item
    K-means minimizes \(J\) by:

    \begin{itemize}
    \tightlist
    \item
      Adjusting centroids (fixed assignments)
    \item
      Adjusting assignments (fixed centroids)
    \end{itemize}
  \item
    Similar to coordinate descent
  \end{itemize}
\item
  Generally converges to a \textbf{\textcolor{blue}{local minimum}}

  \begin{itemize}
  \tightlist
  \item
    Run K-means multiple times with different random initializations
  \item
    Choose best result
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Anomaly Detection: Problem Formulation}
\protect\phantomsection\label{anomaly-detection-problem-formulation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}:

  \begin{itemize}
  \tightlist
  \item
    \(\{\vx_1, ..., \vx_N\}\) examples with features \(\vx \in \bbR^P\)
    for good instances
  \item
    Detect bad/anomalous instances
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Algorithm}}:

  \begin{itemize}
  \tightlist
  \item
    Unknown characteristics of \emph{``bad instances''}
  \item
    Learn common traits of \emph{``good instances''} using unsupervised
    learning

    \begin{itemize}
    \tightlist
    \item
      Find distribution for \(\vx_i\): \(\Pr(\vx \text{ is good})\)
    \end{itemize}
  \item
    Pick features

    \begin{itemize}
    \tightlist
    \item
      Find ``sensitive'' features, e.g., ratio between CPU load and
      network traffic
    \end{itemize}
  \item
    Estimate distribution \(\Pr(\vx \text{ is good})\)
  \item
    Choose threshold \(\varepsilon\)
  \item
    For new instance \(\vx_{new}\), if
    \(\Pr(\vx_{new} \text{ is good}) \le \varepsilon\) flag as anomaly
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Using a Gaussian Model for Anomaly Detection}
\protect\phantomsection\label{using-a-gaussian-model-for-anomaly-detection}
\begin{itemize}

  Aka ``density estimation''
\item

\item
  Ensure that the
  \textbf{\textcolor{red}{features have a Gaussian distribution}}

  \begin{itemize}
  \tightlist
  \item

  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Estimate the parameters}} of the
  Gaussian model \(f_X(\vx)\)
\item
  Given a new example \(\vx_{new}\) \textbf{\textcolor{blue}{compute}}
  \(\Pr(\vx_{new} \text{ is good}) \le \varepsilon\) to flag an anomaly
\end{itemize}

\begin{frame}{Estimate Univariate Gaussian Model}
\protect\phantomsection\label{estimate-univariate-gaussian-model}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  You have \(N\) (scalar) examples \(\vx_1, ..., \vx_N \in \bbR\)
  representing \emph{``good instances''}
\item
  Assume the data is generated by a Gaussian distribution
  \[X \sim \calN(\mu, \sigma)\] which has a PDF: \[
  f_X(x; \mu, \sigma) = \frac{1}{\sqrt{2 \pi} \sigma} \exp(- \frac{(x - \mu)^2}{2 \sigma^2})
  \]
\item
  Estimate mean and sigma with maximum likelihood: \begin{alignat*}{2}
  &\mu = \frac{1}{N}\sum_i x_i \\
  &\sigma^2 = \frac{1}{N-1}\sum_i (x_i - \mu)^2 \\
  \end{alignat*}
\end{itemize}
\end{column}

\begin{column}{0.48\linewidth}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{Estimate Multivariate Independent Gaussian Model}
\protect\phantomsection\label{estimate-multivariate-independent-gaussian-model}
\begin{itemize}
\item
  You have \(N\) examples \(\vx_1, ..., \vx_N \in \bbR^p\) for
  \emph{``good instances''}
\item
  Assume independence of the features, the PDF of a multi-variate
  Gaussian \(X\) is: \[
  f_X(\vx; \vmu, \vsigma)
  = \prod_{i=1}^p f_{X_i}(x_i; \mu_i, \sigma_i)
  \]
\item
  Infer the parameters \(\mu_i\) and \(\sigma_i\) using discrete
  formulas to get the complete model
\end{itemize}
\end{frame}

\begin{frame}{Estimate a Multi-Variate Gaussian Model}
\protect\phantomsection\label{estimate-a-multi-variate-gaussian-model-1}
\begin{itemize}
\item
  The PDF of a multi-variate Gaussian is: \[
  f_X(\vx; \vmu, \mSigma) =
  \frac{1}{(2 \pi)^\frac{n}{2} |\mSigma|^\frac{1}{2}}
  \exp(-\frac{1}{2} (\vx - \vmu)^T \mSigma^{-1} (\vx - \vmu))
  \]
\item
  Estimate: \[
  \vmu \in \bbR^d = \frac{1}{N}\sum_{k=1}^N \vx_k
  \] \[
  \mSigma \in \bbR^{d \times d}
  = \{s_{ij}\}
  = \frac{1}{N - 1} \sum_{k=1}^N (\vx_k - \vmu) (\vx_k - \vmu)^T
  \]
\item
  Model requires more examples to train due to more parameters
\item
  Independence between variables decomposes multivariate Gaussian into
  product of Gaussian distributions
\end{itemize}
\end{frame}

\begin{frame}{Anomaly Detection vs Supervised Learning}
\protect\phantomsection\label{anomaly-detection-vs-supervised-learning}
\begin{itemize}
\item
  Even in unsupervised learning, you need labeled data for model
  evaluation
\item
  Difference with supervised learning:

  \begin{itemize}
  \tightlist
  \item
    Anomaly detection/unsupervised learning: Train only on good examples
  \item
    Supervised learning: Train on both good and bad examples
  \end{itemize}
\item
  Use:

  \begin{itemize}
  \tightlist
  \item
    Anomaly detection/unsupervised learning:

    \begin{itemize}
    \tightlist
    \item
      Learn from good examples due to few anomalous examples
    \item
      Strong prior on the model
    \item
      Future anomalous examples unknown
    \end{itemize}
  \item
    Supervised learning: Less skewed classes in training set
  \item
    Not a clear-cut decision
  \end{itemize}
\end{itemize}
\end{frame}
