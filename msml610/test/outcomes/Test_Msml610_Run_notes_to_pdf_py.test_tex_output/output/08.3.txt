\begin{frame}{Interventions in Causal Networks}
\protect\phantomsection\label{interventions-in-causal-networks-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{The do-operator}}

  \begin{itemize}
  \tightlist
  \item
    Denoted as \(\text{do}(X = x)\)
  \item
    Represents performing an action that \emph{sets} \(X\) to \(x\), not
    \emph{observing} \(X = x\)
  \item
    \(do(X_j = x_j^k)\) removes \(\Pr(x_j | \text{parents}(X_j))\) from
    the product and gives a new joint distribution: \[
    P_{X_j = x_j^k}(x_1, \ldots, x_n) =
    \begin{cases}
    \prod_{i \ne j} \Pr(x_i | \text{parents}(X_i)) & \text{if } X_j = x_j^k \\
    0 & \text{otherwise}
    \end{cases}
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Difference between observation and intervention}}
  \[\Pr(Y | do(X=x) \ne \Pr(Y | X)\]

  \begin{itemize}
  \tightlist
  \item
    Observing \(S = \text{true}\) \emph{provides information} about its
    causes

    \begin{itemize}
    \tightlist
    \item
      E.g., information about weather (and the Markov blanket)
    \end{itemize}
  \item
    Intervening with \(\text{do}(S = \text{true})\) \emph{breaks} those
    causal dependencies

    \begin{itemize}
    \tightlist
    \item
      E.g., it doesn't inform about anything
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Intervention}
\protect\phantomsection\label{intervention}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Estimate causal effect}} of \(X_j\) on \(X_i\)
  with adjustment formula: \[
    \Pr(X_i = x_i | \text{do}(X_j = x_j^k)) =
    \sum_{\text{parents}(X_j)} \Pr(x_i | x_j^k, \text{parents}(X_j)) \Pr(\text{parents}(X_j))
    \]
\item
  \textbf{\textcolor{darkgreen}{Example}}

  \begin{itemize}
  \tightlist
  \item
    In the Sprinkler model \(do(S=\text{true})\), gives the new
    distribution: \[
    \Pr(c, r, w, g | do(S=true)) = \Pr(c)\Pr(r|c)\Pr(w|r, S=true)\Pr(g|w)
    \]

    \begin{itemize}
    \tightlist
    \item
      Only descendants of \(Sprinkler\) (i.e., \(WetGrass\)) change
    \item
      \(Weather\) and \(Rain\) remain unaffected
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Intuition}}

  \begin{itemize}
  \tightlist
  \item
    Do-operator isolates \emph{causal effects} by simulating external
    manipulation
  \item
    Essential for answering ``what if'' questions: \emph{What happens if
    you intervene and change X?}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Counterfactuals}
\protect\phantomsection\label{counterfactuals}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{counterfactual}} describes what would have
  happened in the past under a different scenario

  \begin{itemize}
  \tightlist
  \item
    \emph{``What would the outcome have been, if X had been
    different?''}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Business examples}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``What if we had two suppliers instead of one? Would we have
    fewer delays?''}
  \item
    \emph{``Would customers be more satisfied if we shipped products in
    one week instead of three?''}
  \item
    That's what businesses want, but they can't get it from
    correlation-based models!
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Causal reasoning}}

  \begin{itemize}
  \tightlist
  \item
    Goes beyond correlation and association
  \item
    Requires a causal model to simulate alternate realities
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Actual: \emph{``A student received tutoring and scored 85\%''}
    \item
      Counterfactual: \emph{``What if the student didn't receive
      tutoring?''}
    \item
      Causal model estimates the alternative outcome (e.g., 70\%)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Challenges}}

  \begin{itemize}
  \tightlist
  \item
    Requires strong assumptions and accurate models
  \item
    Difficult to validate directly since counterfactuals are
    unobservable
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Discovery}
\protect\phantomsection\label{causal-discovery}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}

  \begin{itemize}
  \tightlist
  \item
    Causal discovery learns causal network structure from data
  \item
    Identify which variables directly cause others (learn causal
    directions, not just correlations)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Approaches to causal discovery}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Search-based methods}}

    \begin{itemize}
    \tightlist
    \item
      Start with an empty or initial model and iteratively modify it
      (add, reverse, or delete links)
    \item
      Evaluate each candidate network based on fit to data (e.g.,
      likelihood)
    \item
      Use search strategies while ensuring the network remains acyclic
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Constraint-based methods}}

    \begin{itemize}
    \tightlist
    \item
      Infer causal directions from conditional independence tests among
      variables
    \item
      If \(X\) and \(Y\) are independent given \(Z\), this constrains
      possible arrows
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Dealing with complexity}}

  \begin{itemize}
  \tightlist
  \item
    Possible network structures grow super-exponentially with the number
    of variables
  \item
    Complexity penalties to avoid overfitting
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Causality connection}}

  \begin{itemize}
  \tightlist
  \item
    Causal discovery bridges Bayesian learning and causal inference
  \item
    Under certain assumptions, infer causality from observational data
    not only from experiments
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What is a Randomized Controlled Trial?}
\protect\phantomsection\label{what-is-a-randomized-controlled-trial}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{RCTs estimate causal effects}} by comparing
  treatment and control groups

  \begin{itemize}
  \tightlist
  \item
    Randomly assign treatment, not chosen by subjects

    \begin{itemize}
    \tightlist
    \item
      E.g., assign new drug vs placebo by flipping a coin
    \end{itemize}
  \item
    Ensure groups are statistically equivalent except for treatment
  \item
    Isolate treatment effect
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Goal}}: estimate \(\Pr(Y | do(X))\)

  \begin{itemize}
  \tightlist
  \item
    Randomization simulates do-operator by removing incoming arrows to
    \(X\)
  \item
    Eliminate (known and unknown) confounding paths from background
    variables
  \item
    Turn observational data into experimental data

    \begin{itemize}
    \tightlist
    \item
      Use \(\Pr(Y | X)\) to measure \(\Pr(Y | do(X))\)
    \end{itemize}
  \item
    Allow causal inference without knowing the causal graph
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Implement intervention in a principled, unbiased way
  \item
    Foundation for scientific experimentation and evidence-based policy
  \item
    Gold standard of causal inference (when feasible)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Randomized Controlled Trial: Example}
\protect\phantomsection\label{randomized-controlled-trial-example}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Research question}}: \emph{``Does offering an
  after-school tutoring program increase the probability that a student
  passes the end-of-term exam?''}
\item
  \textbf{\textcolor{orange}{Population}}: eligible students in a
  district with proper sample size \(n\)
\item
  \textbf{\textcolor{darkgreen}{Treatment and control}}

  \begin{itemize}
  \tightlist
  \item
    \(X = 1\) (treatment): student is offered/assigned to tutoring
  \item
    \(X = 0\) (control): student is not offered tutoring
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Assignment mechanism (RCT)}}

  \begin{itemize}
  \tightlist
  \item
    Students are randomly assigned to \(X \in \{0, 1\}\)
  \item
    Randomization ensures, in expectation, balance on prior GPA,
    motivation, parental income, etc
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Outcome}}

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(Y \mid do(X)) = \Pr(Y \mid X)\)
  \item
    \(Y_{X=x} = I \{\text{pass exam}\}\) measured at term's end for
    treatment vs control
  \item
    Measure and report \(Y_{X=1} - Y_{X=0}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Randomized Controlled Trial: Limits}
\protect\phantomsection\label{randomized-controlled-trial-limits}
\begin{itemize}
\item
  \textbf{\textcolor{red}{May be unethical}}

  \begin{itemize}
  \tightlist
  \item
    E.g., assigning harmful treatment
  \item
    E.g., you want to verify if asbestos causes cancer
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Can be expensive or impractical}}
\item
  \textbf{\textcolor{darkgreen}{It doesn't always work}}

  \begin{itemize}
  \tightlist
  \item
    Non-compliance: some participants may not follow assigned treatment
  \item
    Attrition: dropout rates may differ between groups
  \item
    May not generalize to broader populations
  \item
    Requires careful implementation and monitoring
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Blind RCT}}: participants don't know which
  group they're in (e.g., placebo)
\item
  \textbf{\textcolor{cyan}{Double-blind RCTs}}: participants and
  investigators/clinicians don't know assignments
\end{itemize}
\end{frame}

\begin{frame}{The Back-Door Adjustment}
\protect\phantomsection\label{the-back-door-adjustment}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Hypotheses}}

  \begin{itemize}
  \tightlist
  \item
    You have a (correct) causal graph
  \item
    You block the back-door paths (needs definition!) that satisfy the
    back-door criterion (needs definitions!)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Thesis}}

  \begin{itemize}
  \tightlist
  \item
    The \emph{adjustment formula} holds
    \[\Pr(Y \mid do(X)) = \sum_z \Pr(Y \mid X, Z=z) \Pr(Z=z)\]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Consequences}}

  \begin{itemize}
  \tightlist
  \item
    It allows you an intervention (level 2 of the causality ladder) only
    using observational data (level 1 of the causality ladder)

    \begin{itemize}
    \tightlist
    \item
      Correlation implies causation
    \end{itemize}
  \item
    This is an alternative to randomized controlled experiments
  \item
    \textbf{\textcolor{violet}{Mind blown!}}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Back-Door Criterion: Overview}
\protect\phantomsection\label{back-door-criterion-overview}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{back-door path}} is any path from \(X\) to
  \(Y\) starting with an arrow into \(X\) and ending pointing into \(Y\)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(X \leftrightarrow ... \leftrightarrow ... \rightarrow Y\)
  \item
    Arrow direction doesn't matter
  \item
    Unblocked paths create spurious associations
  \item
    \emph{Intuition}: Paths make \(X\) and \(Y\) look related even if
    changing \(X\) doesn't change \(Y\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Back-door criterion}}: A set of
  variables \(Z\) satisfies the criterion relative to
  \(X \leftrightarrow ... Z ... \rightarrow Y\) if:

  \begin{itemize}
  \tightlist
  \item
    No variable in \(Z\) is a descendant of \(X\)
  \item
    \(Z\) blocks every path from \(X\) to \(Y\) starting with an arrow
    into \(X\)
  \item
    \emph{Intuition}: Block variables that interfere with the effect
    you're estimating and remove paths where \(X\) and \(Y\) connect
    through common causes
  \end{itemize}
\item
  Condition on variables satisfying the back-door criterion to estimate
  a causal effect without intervention

  \begin{itemize}
  \tightlist
  \item
    This guy won the Nobel prize for CS for this!
    \textbf{\textcolor{blue}{Mind blown!}}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Common Mistakes}
\protect\phantomsection\label{common-mistakes}
\begin{itemize}
\tightlist
\item
  The back-door criterion tells you all and only what you need to
  condition on (i.e., block) to:

  \begin{itemize}
  \tightlist
  \item
    Transform observation in intervention
  \item
    Say that ``correlation IS causation''
  \end{itemize}
\item
  Before this, the solution that researchers used was to ``condition on
  everything''

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{This is incorrect!}}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Common mistakes}}

  \begin{itemize}
  \tightlist
  \item
    Conditioning on a descendant of \(X\) can bias the estimate
  \item
    Controlling for too many variables can open colliders and introduce
    bias
  \item
    Forgetting to block all back-door paths
  \item
    Using variables that lie on the causal path (blocks the mediator of
    the effect)
  \item
    Ignoring unobserved confounders: can make causal effect
    unidentifiable
  \end{itemize}
\item
  \textbf{\textcolor{blue}{1000s of papers and their conclusions are wrong!}}

  \begin{itemize}
  \tightlist
  \item
    In medicine, economics, social science people have used
    observational study incorrectly
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{When Back-Door Adjustment Fails}
\protect\phantomsection\label{when-back-door-adjustment-fails}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Back-door is simple but not universally applicable}}

  \begin{itemize}
  \tightlist
  \item
    No set of observable variables satisfies the back-door criterion

    \begin{itemize}
    \tightlist
    \item
      In order to condition on a variable it needs to be observable
    \item
      E.g., an unobserved or unknown confounders
    \end{itemize}
  \item
    You need to know a good approximation of the true causal graph

    \begin{itemize}
    \tightlist
    \item
      Very easy to omit variables
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Alternatives}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Front-door criterion}: uses mediators
  \item
    \emph{Instrumental variables}: uses external variation
  \item
    \emph{Do-calculus}: symbolic transformations to eliminate \(do()\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Front-Door Adjustment in Causal Inference}
\protect\phantomsection\label{front-door-adjustment-in-causal-inference}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Front-door criterion}} identifies causal
  effects with unobserved confounders

  \begin{itemize}
  \tightlist
  \item
    Applies when a \textbf{\textcolor{darkgreen}{mediator variable}}
    transmits all causal influence from treatment to outcome
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  Assume the causal graph looks like:

  \begin{itemize}
  \tightlist
  \item
    \(X\): treatment or cause
  \item
    \(M\): mediator
  \item
    \(Y\): outcome
  \item
    \(U\): unobserved confounder
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.7.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Hypotheses}}:

  \begin{enumerate}
  \tightlist
  \item
    \emph{All directed paths} from \(X\) to \(Y\) go through \(M\)
  \item
    \emph{No unobserved confounder} affects \(X\) and \(M\)
  \item
    \emph{All backdoor paths} from \(M\) to \(Y\) are blocked by \(X\)
  \end{enumerate}
\item
  \textbf{\textcolor{violet}{Thesis}}: estimate the causal effect
  \(P(Y | do(X))\) despite unobserved \(U\) \begingroup \small
  $$P(Y \mid do(X)) = \sum_m P(M \mid X) \sum_{x'} P(Y \mid M, X') P(X')$$
  \endgroup

  \begin{itemize}
  \tightlist
  \item
    \emph{Intuition}: estimate observed link \(X \to M\) and \(M \to Y\)
  \end{itemize}
\end{itemize}
\end{frame}

  \textbf{\textcolor{darkgreen}{no, not so fast!}}
\item
  \textbf{\textcolor{teal}{Hidden factor}}: \emph{``Parents who care

  \begin{itemize}
  \tightlist
  \item
    Let kids watch more TV to have them eat breakfast and see ads
  \item
    Buy more cereal anyway
  \end{itemize}
\end{itemize}

\begin{itemize}
\tightlist
\item
  Hidden factor \(U\) confounds ``Watch ads'' and ``Buy cereals''

  \begin{itemize}
  \tightlist
  \item

  \item
    Observing \(X\) and \(Y\) without controlling for \(U\) leads to

  \item

\begin{frame}{Cereal and Ads: Solutions}
\protect\phantomsection\label{cereal-and-ads-solutions}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Strategy 1: Back-door adjustment}}

  \begin{itemize}
  \tightlist
  \item
    If you \emph{know} and can \emph{measure} \(U\) ``how much parents
    care about breakfast'', include \(U\) as a control variable in
    analysis
  \item
    \emph{Intuition}:

    \begin{itemize}
    \tightlist
    \item
      Compare families with \emph{the same} breakfast attitudes (\(U\)
      fixed)
    \item
      See if ads (\(X\)) still change cereal buying (\(Y\))
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Strategy 2: Use randomization}}

  \begin{itemize}
  \tightlist
  \item
    Randomized experiments break link between \(X\) and \(U\)

    \begin{itemize}
    \tightlist
    \item
      Randomly show ads to some families and not others
    \item
      Randomization ignores parental breakfast attitude; differences in
      buying come from ads
    \end{itemize}
  \item
    This is why controlled experiments are gold standard for causal
    inference
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Strategy 3: Front-door Adjustment}}
\end{itemize}
\end{frame}

  \tightlist

  \begin{enumerate}

    \end{itemize}
  \end{enumerate}

\begin{frame}{Do-Calculus}
\protect\phantomsection\label{do-calculus}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Do-calculus}} is a formal system for reasoning
  about causal effects in graphical models (Judea Pearl, 2000)
\item
  \textbf{\textcolor{orange}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    You care about \textbf{\textcolor{darkgreen}{causal effects}} like:
    \[
    \Pr(Y | do(X = x))
    \]

    \begin{itemize}
    \tightlist
    \item
      I.e., distribution of \(Y\) if you intervene and set \(X\) to
      \(x\), breaking causal links into \(X\)
    \end{itemize}
  \item
    You have \textbf{\textcolor{teal}{observational data}} like: \[
    \Pr(Y | X = x)
    \]
  \item
    In general \(\Pr(Y | X) \ne \Pr(Y | do(X = x))\) due to
    ``correlation is not causation'' (confounding, \ldots)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Solution}}: do-calculus provides algebraic
  rules to transform intervention expressions (do-operator, i.e.,,
  \(do(X = x)\)) into expressions computable from observational data,
  given certain conditions
\end{itemize}
\end{frame}

\begin{frame}{The Rules of Do-Calculus}
\protect\phantomsection\label{the-rules-of-do-calculus}
\begin{itemize}
\item
  Do-calculus provides
  \textbf{\textcolor{red}{three transformation rules}} for manipulating
  expressions involving \(do()\):

  \begin{enumerate}
  \item
    \textbf{\textcolor{darkgreen}{Insertion/Deletion of Observations:}}
    If \(Y \perp Z \mid X, W\) in \(G_{\overline{X}}\) (where incoming
    edges to X are removed), then:
    \[\Pr(Y \mid do(X), Z, W) = \Pr(Y \mid do(X), W)\]
  \item
    \textbf{\textcolor{blue}{Action/Observation Exchange:}} If
    \(Y \perp Z \mid X, W\) in \(G_{\overline{X}, \underline{Z}}\)
    (incoming edges to X removed, outgoing from Z removed), then:
    \[\Pr(Y \mid do(X), do(Z), W) = \Pr(Y \mid do(X), Z, W)\]
  \item
    \textbf{\textcolor{violet}{Insertion/Deletion of Actions:}} If
    \(Y \perp Z \mid X, W\) in \(G_{\overline{X}, \overline{Z(W)}}\)
    (incoming edges to X and to Z excluding those from W removed), then:
    \[\Pr(Y \mid do(X), do(Z), W) = \Pr(Y \mid do(X), W)\]
  \end{enumerate}
\item
  These rules allow the systematic reduction of expressions involving
  \(do()\) into observational terms, if the causal graph permits
\end{itemize}
\end{frame}

\begin{frame}{Back/Front-door Adjustments and Do-calculus}
\protect\phantomsection\label{backfront-door-adjustments-and-do-calculus}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{back-door}} and
  \textbf{\textcolor{orange}{front-door}} criteria are
  \textbf{\textcolor{darkgreen}{specific applications}} of do-calculus

  \begin{itemize}
  \tightlist
  \item
    They are simpler, graphical conditions that allow
    \(P(Y \mid do(X))\) to be expressed using observational
    probabilities
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Back-door adjustment}}: If a set of variables
  \(Z\) blocks all back-door paths from \(X\) to \(Y\) (paths that go
  into \(X\)), then: \[
  \Pr(Y | do(X)) = \sum_z \Pr(Y | X, Z) \Pr(Z)
  \]
\item
  \textbf{\textcolor{cyan}{Front-door adjustment}}: If there exists a
  variable \(Z\) such that:

  \begin{enumerate}
  \tightlist
  \item
    Z is affected by X
  \item
    Z affects Y
  \item
    All back-door paths from X to Z are blocked
  \item
    All back-door paths from Z to Y are blocked by X then: \[
    \Pr(Y | do(X)) = \sum_z \Pr(Z | X) \sum_{x'} \Pr(Y | Z, X') \Pr(X')
    \]
  \end{enumerate}
\end{itemize}
\end{frame}
