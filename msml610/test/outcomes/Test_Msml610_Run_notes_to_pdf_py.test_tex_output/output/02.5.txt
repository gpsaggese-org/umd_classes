\begin{frame}{How to Make Progress in ML Research}
\protect\phantomsection\label{how-to-make-progress-in-ml-research}
\begin{itemize}
\item
  In ML there are
  \textbf{\textcolor{red}{many possible directions for research}}, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Different data preprocessing methods
  \item
    Different features
  \item
    Different models
  \item
    Different training algorithms
  \item
    Different evaluation techniques
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{What to do?}}
\item
  \textbf{\textcolor{blue}{Right approach}}

  \begin{itemize}
  \tightlist
  \item
    Evaluate models systematically using a single number

    \begin{itemize}
    \tightlist
    \item
      Implement metrics (e.g., accuracy, F1 score) for insight
    \item
      Use cross-validation for model validation
    \end{itemize}
  \item
    Statistical tests to ensure differences are not random

    \begin{itemize}
    \tightlist
    \item
      Utilize hypothesis testing for genuine improvements
    \item
      Conduct A/B testing in real-world
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How to Measure Classifier's Performance?}
\protect\phantomsection\label{how-to-measure-classifiers-performance}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Success/hit/win rate (or error/miss rate)}}

  \begin{itemize}
  \tightlist
  \item
    Measures correct predictions proportion
  \item
    Important for overall accuracy
  \item
    E.g., 80 correct out of 100 = 80\% success rate
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Log probability/cross-entropy error}}

  \begin{itemize}
  \tightlist
  \item
    Evaluates classification with probabilities 0 to 1
  \item
    Lower cross-entropy loss = better performance
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Precision/recall/F-score}}

  \begin{itemize}
  \tightlist
  \item
    Useful for imbalanced data
  \item
    Precision: correctly predicted positives / total predicted positives

    \begin{itemize}
    \tightlist
    \item
      E.g., 0.75 precision = 75\% true positives
    \end{itemize}
  \item
    Recall: correctly predicted positives / actual positives

    \begin{itemize}
    \tightlist
    \item
      E.g., 0.60 recall = 60\% actual positives identified
    \end{itemize}
  \item
    F-score: weighted harmonic mean of precision and recall
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Utility function}}

  \begin{itemize}
  \tightlist
  \item
    Customizes evaluation to prioritize errors and success

    \begin{itemize}
    \tightlist
    \item
      E.g., true/false positives/negatives
    \end{itemize}
  \item
    E.g., in medical diagnosis, prioritize minimizing false negatives
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Training vs Test Set}
\protect\phantomsection\label{training-vs-test-set}
\begin{itemize}
\tightlist
\item
  Performance on train set \(E_{in}\) is an
  \textbf{\textcolor{red}{optimistic estimate}} of \(E_{out}\)

  \begin{itemize}
  \tightlist
  \item
    A model can have:

    \begin{itemize}
    \tightlist
    \item
      0\% error rate on training data (e.g., memorizing responses for
      training set)
    \item
      50\% error rate on test set (e.g., answering randomly)
    \end{itemize}
  \end{itemize}
\item
  To \textbf{\textcolor{blue}{properly evaluate model performance}}:

  \begin{itemize}
  \tightlist
  \item
    Use test set not involved in training
  \item
    Training and test sets should be representative samples
  \item
    E.g., credit risk problem

    \begin{itemize}
    \tightlist
    \item
      Cannot use data from a Florida bank branch to assess a model built
      with New York data
    \item
      Populations have different characteristics
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Lots of Data Scenario vs Scarce Data Scenario}
\protect\phantomsection\label{lots-of-data-scenario-vs-scarce-data-scenario}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Lots of data scenario}}

  \begin{itemize}
  \tightlist
  \item
    Ideal to have lots of data (ideally infinite)
  \item
    Learn on lots of data

    \begin{itemize}
    \tightlist
    \item
      \(\to\) Fit all degrees of freedom of a complex model
    \end{itemize}
  \item
    Predict on lots of data

    \begin{itemize}
    \tightlist
    \item
      \(\to\) Assess precise out-of-sample performance
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Scarce data scenario}}

  \begin{itemize}
  \tightlist
  \item
    Often data is scarce

    \begin{itemize}
    \tightlist
    \item
      E.g., facial recognition datasets with limited annotated data
    \end{itemize}
  \item
    Cannot use all data as a training set
  \item
    Hold out data to estimate performance metrics and bounds

    \begin{itemize}
    \tightlist
    \item
      Split data 70-30 or 80-20 in train and test sets
    \item

    \end{itemize}
  \item
    Other approaches:

    \begin{itemize}
    \tightlist
    \item

    \item
      Use transfer learning with pre-trained models on related tasks
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Splitting Data Into Training, Validation, Test Sets}
\protect\phantomsection\label{splitting-data-into-training-validation-test-sets}
\begin{itemize}
\tightlist
\item
  Training, validation, and test sets \textbf{\textcolor{red}{must be}}:

  \begin{itemize}
  \tightlist
  \item
    Distinct
  \item
    Representative of the problem

    \begin{itemize}
    \tightlist
    \item
      E.g., each class in all sets represented according to original
      data
    \end{itemize}
  \item
    Sized based on data and problem needs
  \end{itemize}
\item
  Ensure sets have the \textbf{\textcolor{blue}{same distribution}}:

  \begin{itemize}
  \tightlist
  \item
    Stratified sampling

    \begin{itemize}
    \tightlist
    \item
      E.g., each class label proportionally represented in each set
    \end{itemize}
  \item
    Shuffle and sample

    \begin{itemize}
    \tightlist
    \item
      Achieves randomization, maintains distribution
    \end{itemize}
  \item
    Sample and check statistics of variables

    \begin{itemize}
    \tightlist
    \item
      E.g., mean, std dev, PDF
    \item
      Compare statistics to ensure each set mirrors broader dataset
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Rule of Thumbs for Data Set Splits}
\protect\phantomsection\label{rule-of-thumbs-for-data-set-splits}
\begin{itemize}
\tightlist
\item
  If \(n\) is \textbf{\textcolor{red}{large}} \(\to\) use a 60-20-20
  split

  \begin{itemize}
  \tightlist
  \item
    Training: 60\%
  \item
    Validation: 20\%
  \item
    Test: 20\%
  \end{itemize}
\item
  If \(n\) is \textbf{\textcolor{darkgreen}{medium}} \(\to\) use a 60-40
  split

  \begin{itemize}
  \tightlist
  \item
    Training: 60\%
  \item
    Test: 40\%
  \item
    Not possible to learn hyperparameters, so no validation set
  \end{itemize}
\item
  If \(n\) is \textbf{\textcolor{blue}{small}} \(\to\) use
  cross-validation and report ``small data size''

  \begin{itemize}
  \tightlist
  \item
    Use K-fold cross-validation
  \item
    Be cautious of the increased chance of high accuracy by chance
  \item
    Is machine learning even suitable for this problem?
  \end{itemize}
\end{itemize}
\end{frame}

\protect\phantomsection\label{can-you-use-test-set-for-training}

  Once the model is selected and validated, reuse all data (including

  \begin{itemize}
  \tightlist
  \item
    Ensures the model benefits from all information
  \end{itemize}
\item
  More data is generally better, though returns diminish after a certain

  \begin{itemize}
  \tightlist
  \item

  \item

  \item
    Use learning curves
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{In-Sample vs Out-Of-Sample Error Expressions}
\protect\phantomsection\label{in-sample-vs-out-of-sample-error-expressions}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal of ML}}: find a function \(h\) that
  approximates the unknown function \(f\), \(h \approx f\) over the
  space of inputs \(\vx \in \calX\) (``script X'')
\item
  Usually the \textbf{\textcolor{darkgreen}{error is point-wise}}: \[
  e(h(\vx_i), f(\vx_i))
  \]

  \begin{itemize}
  \tightlist
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Squared error: \(e(\vx) = (h(\vx) - f(\vx))^2\)
    \item
      0-1 binary error: \(e(\vx) = I[h(\vx) == f(\vx)]\)
    \item
      Log probability: \(e(\vx) = - \log( \Pr(h(\vx) == f(\vx)) )\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{In-sample error}} computed using all points
  in the training set:

  \[
  E_{in}{(h)} = \frac{1}{N} \sum_{i=1}^N e(h(\vx_i), f(\vx_i))
  \]
\item
  \textbf{\textcolor{violet}{Out-of-sample error}} computed on the
  entire space of inputs \(\calX\):

  \[
  E_{out}(h) = \EE_{\vx \in \calX}[e(h(\vx), f(\vx))]
  \]
\end{itemize}
\end{frame}

\begin{frame}{Mean Squared Error (MSE)}
\protect\phantomsection\label{mean-squared-error-mse}
\begin{itemize}
\tightlist
\item
  MSE is the
  \textbf{\textcolor{red}{average squared difference of error}}: \[
  \text{MSE}
  \defeq \frac{1}{N} \sum_{i=1}^N (h(\vx_i) - f(\vx_i))^2
  \]

  \begin{itemize}
  \tightlist
  \item
    Measures estimator quality, quantifying difference between estimated
    and actual values
  \item
    E.g., in house price prediction, determines how close predicted
    prices are to actual prices
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pros:}}

  \begin{itemize}
  \tightlist
  \item
    Related to Gaussian error
  \item
    Optimization friendly
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons:}}

  \begin{itemize}
  \tightlist
  \item
    Doesn't share unit of measure with output

    \begin{itemize}
    \tightlist
    \item
      Distorts error interpretation
    \end{itemize}
  \item
    Sensitive to outliers

    \begin{itemize}
    \tightlist
    \item
      Single large error can disproportionately affect MSE
    \item
      Use median absolute deviation (MAD), median of squared error for
      robustness against outliers
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Root Mean Squared Error (RMSE)}
\protect\phantomsection\label{root-mean-squared-error-rmse}
\begin{itemize}
\item
  RMSE is the
  \textbf{\textcolor{red}{standard deviation of Mean Squared Error (MSE)}}:
  \[
  \text{RMSE} \defeq \sqrt{\text{MSE}}
  = \sqrt{\frac{1}{N} \sum_{i=1}^N (h(\vx_i) - f(\vx_i))^2}
  \]
\item
  \textbf{\textcolor{darkgreen}{Pros:}}

  \begin{itemize}
  \tightlist
  \item
    Same units as output, allowing intuition of magnitude compared to
    mean
  \item
    Facilitates comparison between data sets or models since metric is
    normalized to output's scale
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons:}}

  \begin{itemize}
  \tightlist
  \item
    Sensitive to outliers, which can excessively affect metric
  \item
    May not be suitable for outliers or skewed distributions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Median-Based Metrics}
\protect\phantomsection\label{median-based-metrics}
\begin{itemize}
\item
  Use metrics based on median (i.e., the 0.5 quantile of absolute
  error):
\item
  \textbf{\textcolor{red}{Median absolute deviation}}:

  \[
  \text{MAD} \defeq \text{median}_i(|h(\vx_i) - f(\vx_i)|)
  \]
\item
  \textbf{\textcolor{darkgreen}{Median squared error}}:

  \[
  \defeq \text{median}_i(|h(\vx_i) - f(\vx_i)|^2)
  \]
\item
  \textbf{\textcolor{blue}{Pros:}}

  \begin{itemize}
  \tightlist
  \item
    Robust to outliers
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Cons:}}

  \begin{itemize}
  \tightlist
  \item
    Difficult to evaluate
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How to Choose an Error Measure?}
\protect\phantomsection\label{how-to-choose-an-error-measure}
\begin{itemize}
\tightlist
\item
  Error measure depends on \textbf{\textcolor{red}{application}} and
  \textbf{\textcolor{darkgreen}{"customer"}} should define acceptable
  error level for their use case, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Medical applications have low error tolerance
  \item
    Recommendation systems have higher tolerance
  \end{itemize}
\item
  Otherwise, pick error measure that is

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{blue}{Plausible}}:

    \begin{itemize}
    \tightlist
    \item
      E.g., squared error for Gaussian noise
    \end{itemize}
  \item
    \textbf{\textcolor{violet}{Friendly}}:

    \begin{itemize}
    \tightlist
    \item
      E.g., measures allowing closed-form solutions simplify
      calculations
    \item
      Convex optimization-friendly measures ensure algorithms find
      global minimum easily
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Error Metrics for Skewed Classes}
\protect\phantomsection\label{error-metrics-for-skewed-classes}
\begin{itemize}
\tightlist
\item
  When \textbf{\textcolor{red}{classes are skewed}} (i.e., one class is
  very rare)

  \begin{itemize}
  \tightlist
  \item
    Accuracy can be misleading
  \item
    Use metrics like confusion matrix, precision, and recall
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example:}}

  \begin{itemize}
  \tightlist
  \item
    Train a classifier to distinguish tumors as:

    \begin{itemize}
    \tightlist
    \item
      \(y = 1\): malignant
    \item
      \(y = 0\): benign
    \end{itemize}
  \item
    Classifier's error rate is 1\% (i.e., guess correctly 99\% of the
    time)

    \begin{itemize}
    \tightlist
    \item
      1\% error rate seems good!
    \end{itemize}
  \item
    But only 0.5\% of patients have cancer

    \begin{itemize}
    \tightlist
    \item
      A trivial classifier that always outputs \(y = 0\) has a 0.5\%
      error rate!
    \item
      Now a 1\% error rate does not look good anymore
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Confusion Matrix}
\protect\phantomsection\label{confusion-matrix}
\begin{itemize}
\tightlist
\item
  Assume

  \begin{itemize}
  \tightlist
  \item
    Actual/predicted class \(\in \{0, 1\}\)
  \item
    Typically \(y = 1\) encodes the rare class to predict
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\tightlist
\item
  You have 4 possible cases:

  \begin{itemize}
  \tightlist
  \item
    \(act = 1\), \(pred = 1\) \(\to\) true positive (TP)
  \item
    \(act = 0\), \(pred = 0\) \(\to\) true negative (TN)
  \item
    \(act = 1\), \(pred = 0\) \(\to\) false negative (FN)

    \begin{itemize}
    \tightlist
    \item
      Model outputs \(pred = 0\), but it is wrong
    \end{itemize}
  \item
    \(act = 0\), \(pred = 1\) \(\to\) false positive (FP)

    \begin{itemize}
    \tightlist
    \item
      Model outputs \(pred = 1\), but it is wrong
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Precision vs Recall: Definition}
\protect\phantomsection\label{precision-vs-recall-definition}
\begin{itemize}
\item
  Assume that \(y = 1\) encodes the rare event
\item
  \textbf{\textcolor{red}{Precision}} measures how often there is a true
  positive \emph{given that pred = 1} \begingroup \small
  $$
  \vspace{-0.2cm}
  \text{precision}
  \defeq \Pr(\text{TP} | \text{pred == 1})
  = \frac{|\text{pred == 1} \land \text{act == 1}|}{|\text{pred == 1}|}
  = \frac{\text{TP}}{\text{TP} + \text{FP}}
  $$
  \endgroup
\item
  \textbf{\textcolor{blue}{Recall}} measures how often there is a true
  positive \emph{given that act = 1} \begingroup \small
  $$
  \vspace{-0.2cm}
  \text{recall}
  \defeq \Pr(\text{TP} | \text{act == 1})
  = \frac{\text{TP}}{|\text{act == 1}|}
  = \frac{\text{TP}}{\text{TP} + \text{FN}}
  $$
  \endgroup
\item
  Both are conditional probability measuring the fraction of TP under
  different circumstances:

  \begin{itemize}
  \tightlist
  \item
    (Pre)cision: pred = 1
  \item
    Rec(a)ll: act = 1
  \end{itemize}
\item
  Precision/recall are widely used in information retrieval

  \begin{itemize}
  \tightlist
  \item
    E.g., a search engine:

    \begin{itemize}
    \tightlist
    \item
      Returns 30 pages; only 20 are relevant \(\to\) precision =
      \(\frac{20}{30} = \frac{2}{3}\)
    \item
      Fails to return other 40 relevant pages \(\to\) recall =
      \(\frac{20}{(40 + 20)} = \frac{20}{60} = \frac{1}{3}\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Precision / Recall as Quality / Quantity}
\protect\phantomsection\label{precision-recall-as-quality-quantity}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{"Increasing precision"}} means when you
  predict 1, you are more likely to be right

  \begin{itemize}
  \tightlist
  \item
    A higher precision indicates fewer false positives

    \begin{itemize}
    \tightlist
    \item
      E.g., in a spam email detection system, ``precision is 90\%''
      means 90\% of the emails marked as spam are actually spam
    \end{itemize}
  \item
    Measures ``quality'' of prediction
  \end{itemize}
\item
  \textbf{\textcolor{blue}{"Increasing recall"}} means you predict more
  instances when the outcome is 1

  \begin{itemize}
  \tightlist
  \item
    A higher recall means fewer false negatives

    \begin{itemize}
    \tightlist
    \item
      E.g., in a spam email detection system, ``recall is 80\%''
      indicates 80\% of all actual spam emails were correctly identified
      as spam
    \end{itemize}
  \item
    Measures ``quantity'' of prediction (coverage)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Precision / Recall for Trivial Classifiers}
\protect\phantomsection\label{precision-recall-for-trivial-classifiers}
\begin{itemize}
\item
  A
  \textbf{\textcolor{red}{classifier that outputs always most common class}}
  has:

  \begingroup \small

  \begin{alignat*}{3}
  \text{precision}
  & \defeq \Pr(\text{TP} | \text{pred == 1})
  = 0
  & \text{(since TP = 0)}
  \\
  \text{recall}
  & \defeq \Pr(\text{TP} | \text{act == 1})
  = 0
  & \text{(since TP = 0)}
  \\
  \end{alignat*} \endgroup
\item
  A \textbf{\textcolor{blue}{classifier that outputs always rare class}}
  has: \begingroup \small
  \begin{alignat*}{3}
  \text{recall}
  & = 1
  & \text{(since FN = 0)}
  \\
  \text{precision}
  & \defeq \Pr(\text{TP} | \text{pred == 1})
  & \text{(by definition)}
  \\
  &= \frac{\text{TP}}{\text{TP + FP}} 
  & \text{($TP + FP = n$ because}
  \\
  &= \frac{\#(y = 1)}{n}
  & \text{classifier always emits 1)}
  \\
  &= \Pr(\text{pos}) \approx 0
  & \text{(the positive class is very rare)}
  \\
  \end{alignat*}
  \endgroup
\item
  A trivial classifier has precision or recall close to 0
\end{itemize}
\end{frame}

\begin{frame}{Trading Off Precision and Recall}
\protect\phantomsection\label{trading-off-precision-and-recall}
\begin{itemize}
\item
  In theory, you want to increase
  \textbf{\textcolor{red}{both precision and recall}}
\item
  In practice, you can modify the threshold of a probabilistic
  classifier to
  \textbf{\textcolor{blue}{trade off precision and recall}}
\item
  E.g., use logistic regression to predict cancer:

  \begin{itemize}
  \tightlist
  \item
    With a threshold \(\theta = 0.5\), the classifier has:

    \begin{itemize}
    \tightlist
    \item
      Precision = \(\frac{\text{TP}}{|\text{pred == 1}|}\)
    \item
      Recall = \(\frac{\text{TP}}{|\text{act == 1}|}\)
    \end{itemize}
  \item
    Increase threshold \(\theta\)

    \begin{itemize}
    \tightlist
    \item
      Output 1 only if more confident
    \item
      I.e., increase precision
    \end{itemize}
  \item
    Decrease threshold \(\theta\)

    \begin{itemize}
    \tightlist
    \item
      Output 1 more often
    \item
      Decrease the chances of missing a possible case of cancer
    \item
      I.e., increase recall
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Precision-Recall: Pros / Cons}
\protect\phantomsection\label{precision-recall-pros-cons}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros:}}

  \begin{itemize}
  \tightlist
  \item
    Give insight on the behavior of a classifier (e.g., confusion
    matrix)
  \item
    Avoid mistaking a trivial classifier for a good classifier
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons:}}

  \begin{itemize}
  \tightlist
  \item
    You have two different numbers, thus it is difficult to compare
    classifiers to each other
  \item
    Solutions: F-score, AUC
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Precision-Recall Curves}
\protect\phantomsection\label{precision-recall-curves}
\begin{itemize}
\item
  Aka ROC curves
\item
  Show precision vs recall trade-off for a classifier

  \begin{itemize}
  \tightlist
  \item
    Plot the curve on a precision-recall plane (\(y =\) precision,
    \(1 - x =\) recall)
  \item
    E.g., changing threshold of logistic regression
  \end{itemize}
\item
  Precision-recall plot can have
  \textbf{\textcolor{red}{different shapes}}, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Diagonal (pure luck)
  \item
    Convex up (better than luck)
  \item
    Convex down (worse than luck)
  \end{itemize}
\item
  A curve \textbf{\textcolor{blue}{higher}} than another means a better
  classifier, since for the same recall you get higher precision

  \begin{itemize}
  \tightlist
  \item
    Best classifier (precision = recall = 1) is in the top-right corner
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Area Under the Curve}
\protect\phantomsection\label{area-under-the-curve}
\begin{itemize}
\tightlist
\item
  AUC is the
  \textbf{\textcolor{red}{area under the precision-recall curve}}

  \begin{itemize}
  \tightlist
  \item
    Provides a robust metric by integrating over all thresholds
  \item
    Higher AUC indicates better class differentiation
  \item
    AUC = 0.5 suggests no discriminative power, like random guessing
  \item
    AUC closer to 1.0 indicates high performance
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros:}}

  \begin{itemize}
  \tightlist
  \item
    Single number summarizing classifier behavior, useful for comparing
    models
  \item
    Does not require selecting a threshold for performance calculation
  \item
    Handles imbalanced datasets effectively
  \end{itemize}
\item
  E.g., consider a classifier for medical diagnosis

  \begin{itemize}
  \tightlist
  \item
    AUC helps understand model's ability to distinguish between patients
    with and without a disease
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{F-Score}
\protect\phantomsection\label{f-score}
\begin{itemize}
\item
  The F-score is defined as the
  \textbf{\textcolor{red}{harmonic mean of precision and recall}}:

  \[
  \text{F-score}
  \defeq \frac{2}{\frac{1}{P} + \frac{1}{R}}
  = 2 \frac{P \cdot R}{P + R}
  \]
\item
  \textbf{\textcolor{blue}{Interpretation:}}

  \begin{itemize}
  \tightlist
  \item
    Trivial classifiers

    \begin{itemize}
    \tightlist
    \item
      \(P = 0\) or \(R = 0\) \(\implies\) F-score = 0
    \end{itemize}
  \item
    Perfect classifiers

    \begin{itemize}
    \tightlist
    \item
      \(P = R = 1\) \(\implies\) F-score = 1
    \end{itemize}
  \item
    For F-score to be large, both \(P\) and \(R\) must be high
  \end{itemize}
\item
  Why not just averaging \(P, R\)?

  \begin{itemize}
  \tightlist
  \item
    A classifier that always outputs 1 has:

    \begin{itemize}
    \tightlist
    \item
      \(R = 1\) and \(P = 0\)
    \item
      \(\frac{P + R}{2} = \frac{1}{2}\)
    \item
      We prefer a low value (ideally 0)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Problem of Model Selection}
\protect\phantomsection\label{the-problem-of-model-selection}
\begin{itemize}
\tightlist
\item
  Model selection:

  \begin{itemize}
  \tightlist
  \item
    Chooses the best model from candidates based on performance
  \item
    Is needed when multiple hypotheses explain data
  \end{itemize}
\item
  Across models certain parameters are fixed
\item
  Others need selection, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Set of features

    \begin{itemize}
    \tightlist
    \item
      E.g., select a subset from 100 variables
    \end{itemize}
  \item
    Learning algorithms

    \begin{itemize}
    \tightlist
    \item
      E.g., decide how to train a neural network
    \end{itemize}
  \item
    Model types

    \begin{itemize}
    \tightlist
    \item
      E.g., linear regression vs.~SVM
    \end{itemize}
  \item
    Model complexity

    \begin{itemize}
    \tightlist
    \item
      E.g., polynomials of degree \(d < 10\)
    \end{itemize}
  \item
    Values of regularization parameter

    \begin{itemize}
    \tightlist
    \item
      E.g., try values like 0.01, 0.1, 1.0
    \end{itemize}
  \end{itemize}
\item
  Evaluate metrics (e.g., model accuracy, precision, recall) with
  cross-validation
\end{itemize}
\end{frame}

\begin{frame}{Model Selection Process}
\protect\phantomsection\label{model-selection-process}
\begin{enumerate}
\item
  Split data into \(D_{train}, D_{val}, D_{test}\)

  \begin{itemize}
  \tightlist
  \item
    Commonly: 60\% training, 20\% validation, 20\% test
  \item
    Like splitting 80\% training between two learning phases
  \end{itemize}
\item
  Given \(N\) hypotheses, learn on \(D_{train}\) to get
  \(g_1, ..., g_N\)
\item
  Evaluate hypotheses on \(D_{val}\) estimating errors
  \(E_{val}^{(1)}, ..., E_{val}^{(N)}\)
\item
  Pick model \(g_m\) with minimum \(E_{val}^{(m)}\)
\item
  Use test set \(D_{test}\) to estimate fair performance of model
  \(g_m\), i.e., \(E_{val} \approx E_{out}\)
\item
  Retrain model with entire \(D = D_{train} \cup D_{val} \cup D_{test}\)
  to get final \(g_m^{*}\)
\end{enumerate}
\end{frame}

\begin{frame}{Model Selection as Learning}
\protect\phantomsection\label{model-selection-as-learning}
\begin{itemize}
\tightlist
\item
  \emph{``Picking the model with smallest \(E_{val}\)''} is a
  \textbf{\textcolor{red}{form of learning}}:

  \begin{itemize}
  \tightlist
  \item
    Hypothesis set: \(\{g_1, ... , g_N\}\)
  \item
    Training set: \(D_{val}\)
  \item
    Pick the best model \(g_m\)
  \end{itemize}
\item
  After model selection, you need to
  \textbf{\textcolor{blue}{assess the true performance}} on \(D_{test}\)

  \begin{itemize}
  \tightlist
  \item
    Experimentally \[
    E_{val}(g_m) < E_{out}(g_m)
    \]

    \begin{itemize}
    \tightlist
    \item
      I.e., \(E_{val}(g_m)\) is a (optimistically) biased estimate of
      \(E_{out}(g_m)\)
    \end{itemize}
  \item
    Theoretically:

    \begin{itemize}
    \tightlist
    \item
      The penalty for model complexity with a finite set of hypotheses
      is \[
      E_{out}(g_m) \le E_{val}(g_m) + O(\sqrt{\log(N / K)})
      \]
    \item
      Use VC dimension for an infinite number of hypotheses (e.g.,
      choice of \(\lambda\) for regularization)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ensemble Learning: Intuition}
\protect\phantomsection\label{ensemble-learning-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Idea}}

  \begin{itemize}
  \tightlist
  \item
    A group of weak learners can form a strong learner
  \item
    Combine multiple models to improve prediction accuracy
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Combine outputs of models}} \(X_i\) to
  build a model \(X^*\) better than any \(X_i\)

  \begin{itemize}
  \tightlist
  \item
    Utilize diversity in model predictions to improve accuracy
  \item
    Each model contributes its unique perspective, reducing overfitting
  \item
    E.g., like a panel of voting experts
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: in computer vision detecting a
  face is difficult task (circa 2010)

  \begin{itemize}
  \tightlist
  \item
    Look for different features:

    \begin{itemize}
    \tightlist
    \item
      Are there eyes?
    \item
      Is there a nose?
    \item
      Are eyes and nose in the correct position?
    \end{itemize}
  \item
    Each feature is weak by itself, but together they become reliable
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ensemble Learning: Different Techniques}
\protect\phantomsection\label{ensemble-learning-different-techniques}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Bagging}} (bootstrap + aggregation)

  \begin{itemize}
  \tightlist
  \item
    Reduces variance by averaging predictions from different models
  \item
    E.g., decision trees + bagging = random forest

    \begin{itemize}
    \tightlist
    \item
      Creates multiple versions of a decision tree
    \item
      Each tree is trained on a random sample of data
    \item
      Averages predictions to improve accuracy
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Boosting}}

  \begin{itemize}
  \tightlist
  \item
    Reduces bias by focusing on errors made by previous models
  \item
    Sequentially adds models, each correcting its predecessor
  \item
    E.g., \textcolor{blue}{\texttt{adaBoost}} increases weights of
    incorrectly classified data points to learn the next model
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Stacking}}

  \begin{itemize}
  \tightlist
  \item
    Uses a meta-model to combine separate models using weights
  \item
    E.g., a stacking ensemble

    \begin{itemize}
    \tightlist
    \item
      Uses logistic regression as a meta-model
    \item
      Combines predictions of other models (e.g., decision trees,
      support vector machines, neural networks)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ensemble Learning: Relation with Statistics}
\protect\phantomsection\label{ensemble-learning-relation-with-statistics}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Bagging}}

  \begin{itemize}
  \tightlist
  \item
    Improves performance by adding randomized variants (mimicking
    multiple training sets)
  \item
    Reduce variance without affecting bias
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Boosting}}

  \begin{itemize}
  \tightlist
  \item
    Use another model to learn the residuals, i.e., difference between
    predicted and true values
  \item
    Related to ``forward stagewise additive models''
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Stacking}}

  \begin{itemize}
  \tightlist
  \item
    If we have 3 independent classifiers, each with
    \(\Pr(\text{correct}) = 0.7\) \begin{alignat*}{2}
    \Pr(\text{majority correct})
    &= \Pr(\text{at least 2 classifiers correct}) \\
    &= {3 \choose 2} 0.7^2 0.3 + 0.7^3 \\
    &= 3 \times 0.7^2 \times 0.3 + 0.7^3 \\
    &\approx 0.78 > 0.7
    \end{alignat*}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Ensemble Learning: Pros and Cons}
\protect\phantomsection\label{ensemble-learning-pros-and-cons}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Hypothesis set \(\calH\) is increased by combining hypotheses from
    different models
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    More computationally intensive to train and evaluate
  \item
    Loss of interpretability
  \item
    Risk of overfitting (model complexity is increased)
  \item
    Ensemble learning contradicts Occam's razor, which advocates
    simplicity
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{When Ensemble Learning Works}
\protect\phantomsection\label{when-ensemble-learning-works}
\begin{itemize}
\tightlist
\item
  Combining multiple models with ensemble learning works when models:

  \begin{itemize}
  \tightlist
  \item
    Are very different from each other
  \item
    Treat a reasonable percentage of the data correctly

    \begin{itemize}
    \tightlist
    \item
      E.g., you cannot do much if all classifiers have 50\% accuracy
    \end{itemize}
  \item
    Complement each other: they are specialists in a part of the domain
    where the others don't perform well
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How to Combine Outputs in Ensemble Learning}
\protect\phantomsection\label{how-to-combine-outputs-in-ensemble-learning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Regression}}

  \begin{itemize}
  \tightlist
  \item
    Weighted average of prediction
  \item
    E.g., by accuracy of each model or by a prior
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Classification}}

  \begin{itemize}
  \tightlist
  \item
    Weighted vote of predicted classes
  \item
    It needs an odd number of models to break ties
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Probabilistic classification}}

  \begin{itemize}
  \tightlist
  \item
    Weighted average of class probabilities
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Learn a meta-learner}} (i.e., stacking) to
  combine multiple models
\end{itemize}
\end{frame}

\subsubsection{Bagging}\label{bagging}

\begin{frame}{Bagging}
\protect\phantomsection\label{bagging-1}
\begin{itemize}
\item
  Bagging stands for ``Bootstrap AGGregation''
\item
  \textbf{\textcolor{red}{Learning procedure}}

  \begin{itemize}
  \tightlist
  \item
    Several training datasets are extracted randomly by sampling with
    replacement from the original dataset (i.e., bootstrap)
  \item
    Learn multiple models, one for each training set
  \item
    Combine outputs using various methods
  \item
    Result is a better model than a single model
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Why bagging works?}}

  \begin{itemize}
  \tightlist
  \item
    From the bias-variance decomposition view, combining multiple
    models:

    \begin{itemize}
    \tightlist
    \item
      Reduces the variance component
    \item
      Without compromising the bias (bagged models are typically
      unbiased)
    \end{itemize}
  \item
    Bagging mimics extracting more training sets (though not
    independent) from the unknown distribution
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bagging and Instability in Learning Algorithms}
\protect\phantomsection\label{bagging-and-instability-in-learning-algorithms}
\begin{itemize}
\tightlist
\item
  Bagging works best with highly different models, especially non-linear
  models

  \begin{itemize}
  \tightlist
  \item
    Introduce randomization in the learning algorithm intentionally
  \end{itemize}
\item
  \textbf{\textcolor{red}{Decision Trees}}

  \begin{itemize}
  \tightlist
  \item
    Disable pruning
  \item
    Break ties randomly when selecting the best attribute to split
  \item
    E.g., bagging trees results in random forests
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Multilayer Perceptrons}}

  \begin{itemize}
  \tightlist
  \item
    Use different initial weights in backpropagation to reach different
    local minima
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Nearest Neighbor Classifier}}

  \begin{itemize}
  \tightlist
  \item
    Use a random subset of features
  \item
    Resampling the training set has limited impact, as it is equivalent
    to changing example weights
  \end{itemize}
\end{itemize}
\end{frame}

\subsubsection{Boosting}\label{boosting}

\begin{frame}{Boosting}
\protect\phantomsection\label{boosting-1}
\begin{itemize}
\tightlist
\item
  Boosting builds models that complement each other

  \begin{itemize}
  \tightlist
  \item
    Typically use homogeneous models, i.e., parametrized models from
    \(\calH\)
  \end{itemize}
\item
  Strong classifiers can be built from weak classifiers

  \begin{itemize}
  \tightlist
  \item
    E.g., decision stumps = decision trees with one level
  \end{itemize}
\item
  \textbf{\textcolor{red}{Why boosting works?}}

  \begin{itemize}
  \tightlist
  \item
    Boosting implements forward stagewise additive modeling
  \item
    Use another model to learn residuals (difference between predicted
    and true values)
  \end{itemize}
\item
  Boosting does not work for linear regression:

  \begin{itemize}
  \tightlist
  \item
    Combination of linear models is still a linear model
  \item
    OLS finds optimal weights in one step
  \item
    Combining linear regressions from different attributes is equivalent
    to a single multiple linear regression
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Adaboost.M1}
\protect\phantomsection\label{adaboost.m1}
\begin{itemize}
\tightlist
\item
  Widely used for classification
\item
  Assume examples can be weighted in the cost function used to learn

  \begin{itemize}
  \tightlist
  \item
    Otherwise use resampling
  \end{itemize}
\item
  \textbf{\textcolor{red}{Learning procedure}}

  \begin{itemize}
  \tightlist
  \item
    Start with equal weights for all examples
  \item
    Iterate:

    \begin{itemize}
    \tightlist
    \item
      Learn a classifier based on current weights for examples
    \item
      Weight the answer of each model by overall score (e.g., accuracy)
      or probability
    \item
      Evaluate the ensemble
    \item
      Adjust weights for examples classified correctly/incorrectly
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\subsubsection{Stacking}\label{stacking}

\begin{frame}{Stacking}
\protect\phantomsection\label{stacking-1}
\begin{itemize}
\item
  Stacking is about learning how to combine models (not necessarily of
  the same type)
\item
  The problem is that with voting / averaging we don't know which model
  to trust
\item
  Instead of voting or weighting we can use a meta-learner (level 1) to
  learn how to pick / mix models (level 0)
\item
  \textbf{\textcolor{red}{Learning procedure}}

  \begin{itemize}
  \tightlist
  \item
    Learn ``level 0'' models
  \item
    Learn ``level 1'' model using hold-out data from learning of level 0
    models (like in model selection)

    \begin{itemize}
    \tightlist
    \item
      Build training data with predicted values from level 0 models
    \item
      Then learn level 1
    \item
      Use a simple model for level 1 (e.g., linear models or trees) to
      avoid overfitting
    \item
      Use probabilities from level 0, so level 1 can assess the
      confidence of each model
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}
