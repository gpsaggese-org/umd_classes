\begin{frame}{EDA vs Inference}
\protect\phantomsection\label{eda-vs-inference}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Exploratory data analysis}}

  \begin{itemize}
  \tightlist
  \item
    Summarize, interpret, check data
  \item
    Visually inspect the data
  \item
    Compute descriptive statistics
  \item
    Communicate results
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Inferential statistics / inference}}

  \begin{itemize}
  \tightlist
  \item
    Draw insights from a limited set of data
  \item
    Make predictions for future unobserved data points
  \item
    Understand a phenomenon
  \item
    Choose among competing explanations for the same observations
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Good vs Bad Way to Do Statistics}
\protect\phantomsection\label{good-vs-bad-way-to-do-statistics}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Bad}}
  \includegraphics[width=0.125in,height=\textheight,keepaspectratio,alt={Smile Emoji}]{assets/emoji/angry_devil.png}

  \begin{itemize}
  \tightlist
  \item
    Learn a collection of ``statistical recipes''

    \begin{itemize}
    \tightlist
    \item
      Make assumption / approximate to make math workable
    \end{itemize}
  \item
    Given data and problem

    \begin{itemize}
    \tightlist
    \item
      Pick one recipe
    \item
      Try until you get a ``low'' p-value
    \end{itemize}
  \item
    For machine learning

    \begin{itemize}
    \tightlist
    \item
      Iterate until you get a ``good'' fit on out-of-sample data
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Good}}
  \includegraphics[width=0.125in,height=\textheight,keepaspectratio,alt={Smile Emoji}]{assets/emoji/emoji_smile.png}

  \begin{itemize}
  \tightlist
  \item
    General approach to statistical inference (Bayesian statistics)

    \begin{itemize}
    \tightlist
    \item
      Remove limitations from closed analytical form
    \end{itemize}
  \item
    Probabilistic approach unifies (seemingly) disparate methods

    \begin{itemize}
    \tightlist
    \item
      E.g., statistical methods and machine learning
    \item
      E.g., \textcolor{blue}{\texttt{statsmodels}} linear regression vs
      \textcolor{blue}{\texttt{sklearn}} decision tree
    \item
      Deep unity of different recipes
    \end{itemize}
  \item
    Modern tools (e.g., PyMC3) solve previously unsolvable models
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Data}
\protect\phantomsection\label{data}
\begin{itemize}
\item
  Data \textbf{\textcolor{red}{comes from}}:

  \begin{itemize}
  \tightlist
  \item
    Experiments
  \item
    Simulations
  \item
    Surveys
  \item
    Field observations
  \end{itemize}
\item
  Data is stochastic due to \textbf{\textcolor{orange}{uncertainty}}

  \begin{itemize}
  \tightlist
  \item
    Ontological: system is intrinsically stochastic
  \item
    Technical: measurement precision is limited or noisy
  \item
    Epistemic: conceptual limitations in understanding
  \end{itemize}
\item
  Collecting data is \textbf{\textcolor{darkgreen}{costly}}

  \begin{itemize}
  \tightlist
  \item
    Consider questions before collecting data
  \item
    Experiment design is a branch of statistics for data collection
  \end{itemize}
\item
  Data is \textbf{\textcolor{teal}{rarely clean and tidy}}
\item
  Data needs to be \textbf{\textcolor{cyan}{interpreted}} through mental
  and formal models
\end{itemize}
\end{frame}

\begin{frame}{Models}
\protect\phantomsection\label{models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Models}} are simplified descriptions of a
  given system/process

  \begin{itemize}
  \tightlist

  \item
    VC dimension made it mathematical precise

    \begin{itemize}
    \tightlist
    \item
      \emph{``You need at least 10 data points per effective degree of
      freedom of the hypothesis set''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Goals}}

  \begin{itemize}
  \tightlist
  \item
    Capture the most relevant aspects of the system
  \item
    Ignore minor details
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayes' Theorem: Recap}
\protect\phantomsection\label{bayes-theorem-recap}
\begin{itemize}
\tightlist
\item
  \textbf{\black{Bayes' theorem}} posits that for model parameters
  \(\theta\) and data \(X\) \[
  \textcolor{red}{\Pr(\theta | X)}
  = \frac{\textcolor{teal}{\Pr(X | \theta)} \cdot \textcolor{blue}{\Pr(\theta)}}{\textcolor{violet}{\Pr(X)}}
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\red{$\Pr(\theta | X)$}}

    \begin{itemize}
    \tightlist
    \item
      \textbf{\black{Posterior}}: probability for parameters \(\theta\)
      after seeing data \(X\)
    \end{itemize}
  \item
    \textbf{\teal{$\Pr(X | \theta)$}}

    \begin{itemize}
    \tightlist
    \item
      \textbf{\black{Likelihood}} (aka ``statistical model''):
      plausibility of data \(X\) given parameters \(\theta\)
    \end{itemize}
  \item
    \textbf{\blue{$\Pr(\theta)$}}

    \begin{itemize}
    \tightlist
    \item
      \textbf{\black{Prior}}: knowledge about parameter \(\theta\)
      before any data
    \end{itemize}
  \item
    \textbf{\violet{$\Pr(X)$}}

    \begin{itemize}
    \tightlist
    \item
      \textbf{\black{Evidence}} (``marginal likelihood''): probability
      of observing data \(X\)
    \item
      ``Marginal'' as it averages over all possible parameter values
    \end{itemize}
  \end{itemize}
\item
  In other words: \[
  \textcolor{red}{\text{Posterior}}
  = \frac{\textcolor{teal}{\text{Likelihood}} \cdot \textcolor{blue}{\text{Prior}}}{\textcolor{violet}{\text{Evidence}}}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Models}
\protect\phantomsection\label{bayesian-models}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Probability}} measures uncertainty about
  parameters
\item
  \textbf{\textcolor{darkgreen}{Bayes' theorem}} updates probabilities
  with new data, reducing uncertainty (hopefully) \[
  \Pr(hyp | data) = \frac{\Pr(data | hyp) \Pr(hyp)}{\Pr(data)}
  \]
\item
  \textbf{\textcolor{blue}{Bayesian modeling workflow}}

  \begin{enumerate}
  \tightlist
  \item
    Design a model using probabilities based on data and assumptions

    \begin{itemize}
    \tightlist
    \item
      Assumptions on data generation
    \item
      Model can be a crude approximation
    \end{itemize}
  \item
    Apply Bayes' theorem to ``condition'' the model on data
  \item
    Validate model against:

    \begin{itemize}
    \tightlist
    \item
      Data
    \item
      Subject expertise
    \item
      Related models
    \end{itemize}
  \end{enumerate}

  \begin{itemize}
  \tightlist
  \item
    Steps may involve backtracking:

    \begin{itemize}
    \tightlist
    \item
      Correct coding errors
    \item
      Improve model
    \item
      Gather more or different data
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Coin Example: Problem}
\protect\phantomsection\label{coin-example-problem}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}:

  \begin{itemize}
  \tightlist
  \item
    Toss a coin \(N\) times
  \item
    Record the number of heads \(Y\) and tails \(N - Y\)
  \item
    Question: \emph{``How biased is the coin?''}
  \end{itemize}
\item
  There is \textbf{\textcolor{darkgreen}{true uncertainty}}

  \begin{itemize}
  \tightlist
  \item
    An underlying parameter exists, but it is unknown
  \item
    \(\theta\) represents the coin bias

    \begin{itemize}
    \tightlist
    \item
      \(0\): always tails
    \item
      \(1\): always heads
    \item
      \(0.5\): half tails, half heads
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Model assumptions}}:

  \begin{itemize}
  \tightlist
  \item
    Independent Identically Distributed (IID)

    \begin{itemize}
    \tightlist
    \item
      Independence: coin tosses don't affect each other
    \item
      Identically distributed: coin's bias is constant
    \end{itemize}
  \item
    Likelihood \(Y | \theta\) as a binomial distribution

    \begin{itemize}
    \tightlist
    \item
      Probability of \(Y\) heads out of \(N\) tosses, given \(\theta\)
    \end{itemize}
  \item
    Prior \(\theta\) as a beta distribution

    \begin{itemize}
    \tightlist
    \item
      Adopts several shapes
    \item
      Beta is the conjugate prior of the binomial distribution
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Binomial Distribution}
\protect\phantomsection\label{binomial-distribution}
\begin{columns}[T]
\begin{column}{0.4\linewidth}
Probability of \(k\) heads out of \(n\) tosses given bias \(p\)

\begin{align*}
  & X \sim Binomial(n, p) \\
  & \Pr(k) = \frac{n!}{k! (n - k)!} p^k (1 - p)^{n-k} \\
\end{align*}
\end{column}

\begin{column}{0.55\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L07.1.Binomial_distribution.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Beta Distribution}
\protect\phantomsection\label{beta-distribution}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\item
  Continuous PDF in {[}0, 1{]}
\item
  Adopts several shapes

  \begin{itemize}
  \tightlist
  \item
    Uniform, increasing, decreasing, Gaussian-like, U-like
  \item
    \(\alpha\): ``success'' parameter
  \item
    \(\beta\): ``failure'' parameter
  \item
    \(\alpha > \beta\): Skews toward 1, higher probability of success
  \item
    \(\alpha = \beta\): Symmetric, centered around 0.5
  \end{itemize}
\item
  Models probability or proportion

  \begin{itemize}
  \tightlist
  \item
    E.g., probability of success in a Bernoulli trial \(\theta\)
  \end{itemize}
\item
  Beta is the conjugate prior of the binomial distribution
\end{itemize}
\end{column}

\begin{column}{0.5\linewidth}
\begin{align*}
  & X \sim Beta(\alpha, \beta) \\
  & \Pr(\theta)
    = \frac{\Gamma(\alpha + \beta)}{\Gamma(\alpha) \Gamma(\beta)}
    \theta^{\alpha - 1} (1 - \theta)^{\beta - 1} \\
\end{align*}

\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L07.1.Beta_distribution.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Conjugate Prior of a Likelihood}
\protect\phantomsection\label{conjugate-prior-of-a-likelihood}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Conjugate prior}} is a prior that, when
  combined with a likelihood, returns a posterior with the same
  functional form as the prior

  \begin{itemize}
  \tightlist
  \item
    E.g.,
  \end{itemize}

  {\def\LTcaptype{none} % do not increment counter
  \begin{longtable}[]{@{}lll@{}}
  \toprule\noalign{}
  \textbf{Prior} & \textbf{Likelihood} & \textbf{Posterior} \\
  \midrule\noalign{}
  \endhead
  Beta & Binomial & Beta \\
  Normal & Normal & Normal \\
  \bottomrule\noalign{}
  \end{longtable}
  }
\item
  \textbf{\textcolor{blue}{Properties}}

  \begin{itemize}
  \tightlist
  \item
    Prior and posterior have the same distribution
  \item
    Posterior has a closed analytical form

    \begin{itemize}
    \tightlist
    \item
      Update parameters from the prior using data in multiple iterations
    \end{itemize}
  \item
    Ensures tractability of the posterior
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Coin Example: Analytical Solution}
\protect\phantomsection\label{coin-example-analytical-solution}
\begin{itemize}
\item
  The \textbf{\red{posterior}} is proportional to
  \textbf{\green{likelihood}} \(\times\) \textbf{\blue{prior}}
  \[\textcolor{red}{\Pr(\theta | y)} \propto \textcolor{darkgreen}{\Pr(y | \theta)} \textcolor{blue}{\Pr(\theta)}\]
\item
  Substituting \textbf{\textcolor{darkgreen}{\text{likelihood}}} with a
  Binomial and \textbf{\textcolor{blue}{\text{prior}}} with a Beta
  \begin{align*}
  & \Pr(\theta \mid Y)
  \\
  & = \underbrace{\frac{N!}{y!(N - y)!} \theta^y (1 - \theta)^{N - y}}_{\text{likelihood}}
  \underbrace{\frac{\Gamma(\alpha + \beta)}{\Gamma(\alpha) \Gamma(\beta)}
   \theta^{\alpha - 1} (1 - \theta)^{\beta - 1}}_{\text{prior}}
  \\
  & \propto
  \underbrace{\theta^y (1 - \theta)^{N - y}}_{\text{likelihood}}
  \underbrace{\theta^{\alpha - 1} (1 - \theta)^{\beta - 1}}_{\text{prior}}
  \\
  & = \theta^{y + \alpha - 1} (1 - \theta)^{ N - y + \beta - 1}
  \\
  & = \text{Beta} \left( \alpha_{\text{prior}} + y, \beta_{\text{prior}} + N - y \right)
  \\
  \end{align*}
\end{itemize}

\vspace{-0.7cm}

\begin{itemize}
\tightlist
\item
  This is how
  \textbf{\textcolor{black}{\text{the posterior is updated}}} given the
  data
\end{itemize}
\end{frame}

\begin{frame}{Frequentist Approach vs Priors}
\protect\phantomsection\label{frequentist-approach-vs-priors}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Detractors of Bayesian approach}} complain
  that:

  \begin{itemize}
  \tightlist
  \item
    \emph{``One should let the data speak''}
  \item
    The prior doesn't let the data speak for itself
  \end{itemize}
\item
  \includegraphics[width=0.14583in,height=\textheight,keepaspectratio]{assets/emoji/balance_scale.png}
  \textbf{\textcolor{blue}{Counterpoints}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``Data doesn't speak, but murmurs''}

    \begin{itemize}
    \tightlist
    \item
      Data doesn't have meaning per-se
    \item
      Make sense of data only in context of models (e.g., mental models,
      mathematical models)
    \item
      A prior is a mathematical model
    \end{itemize}
  \item
    Every statistical model has a prior, even if not explicit

    \begin{itemize}
    \tightlist
    \item
      Frequentist statistics still makes assumptions (i.e., has a
      prior), but are hidden
    \item
      E.g., maximum likelihood estimate (MLE) in frequentist approach
      corresponds to a uniform prior and mode of the posterior
    \item
      E.g, MLE is a point-estimate, not a distribution of plausible
      values
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\item
  \textbf{\textcolor{red}{Assumptions are clear and explicit}}

\item
  \textbf{\textcolor{orange}{Prior}}

\item

\item

  \textbf{\textcolor{cyan}{speeds up inference}}

\begin{frame}{How to Choose Priors}
\protect\phantomsection\label{how-to-choose-priors}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Weakly-informative priors}} (aka ``flat'',
  ``vague'', ``diffuse priors'')

  \begin{itemize}
  \tightlist
  \item
    Provide minimal information

    \begin{itemize}
    \tightlist
    \item
      Coefficient of linear regression centered around 0:
      \(\beta \sim Normal(0, 10)\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Regularizing priors}}

  \begin{itemize}
  \tightlist
  \item
    Known information about the parameter

    \begin{itemize}
    \tightlist
    \item
      Parameter is positive: \(\sigma \sim HalfCauchy(0, 5)\)
    \item
      Parameter close to zero, above/below a number, or in a range
    \item
      \(\beta \sim Laplace(0, 1)\) (lasso prior) encourages sparsity
    \item
      \(\beta \sim Normal(0, 1)\) discourages extreme values
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Informative priors}}

  \begin{itemize}
  \tightlist
  \item
    Strong priors from previous knowledge (expert opinion, studies)

    \begin{itemize}
    \tightlist
    \item
      From experimental data: \(\beta_1 \sim Normal(2.5, 0.5^2)\)
    \item
      From previous data, about 5\% of cases positive:
      \(p \sim Beta(2, 38)\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Prior elicitation}}

  \begin{itemize}
  \tightlist
  \item
    Compute least informative distribution given constraints

    \begin{itemize}
    \tightlist
    \item
      Estimate distribution using maximum entropy to satisfy constraints
    \item
      E.g., beta distribution with 90\% of mass between 0.1 and 0.7
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Confidence Intervals vs Credible Intervals}
\protect\phantomsection\label{confidence-intervals-vs-credible-intervals}
\begin{itemize}
\tightlist
\item
  People confuse:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Frequentist confidence intervals}}
  \item
    \textbf{\textcolor{orange}{Bayesian credible intervals}}
  \end{itemize}
\item
  In the frequentist framework, there is a true (unknown) parameter
  value

  \begin{itemize}
  \tightlist
  \item
    A \textbf{\textcolor{darkgreen}{confidence interval}} may or may not
    contain the true parameter value
  \item
    Interpretation of a 95\% confidence interval

    \begin{itemize}
    \tightlist
    \item
      \includegraphics[width=0.125in,height=\textheight,keepaspectratio]{assets/emoji/wrong.png}
      No: \emph{``There is a 95\% probability that the true value is in
      this interval''}
    \item
      \includegraphics[width=0.125in,height=\textheight,keepaspectratio]{assets/emoji/check_mark.png}
      Yes: \emph{``If repeated many times, 95\% of intervals would
      contain the true value''}
    \end{itemize}
  \end{itemize}
\item
  In the Bayesian framework, parameters are random variables

  \begin{itemize}
  \tightlist
  \item
    Interpretation of a 95\%
    \textbf{\textcolor{teal}{Bayesian credible interval}}

    \begin{itemize}
    \tightlist
    \item
      \emph{``There is a 95\% probability that the true parameter lies
      within this interval, given the observed data''}
    \item
      Bayesian \textbf{\textcolor{cyan}{credible interval}} is intuitive
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Confidence Intervals vs Credible Intervals (ELI5)}
\protect\phantomsection\label{confidence-intervals-vs-credible-intervals-eli5}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Confidence Interval (Frequentist)}}

  \begin{itemize}
  \tightlist
  \item
    Imagine fishing in a lake without seeing the fish
  \item
    You throw your net
  \item
    95\% confidence interval: \emph{``If I threw this net 100 times,
    about 95 nets would catch the fish.''}
  \item
    Important: Once the net is thrown, it either caught the fish or not.
    The 95\% makes sense across many attempts
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Credible Interval (Bayesian)}}

  \begin{itemize}
  \tightlist
  \item
    Imagine a magical map showing where fish \emph{probably} are, based
    on past observations
  \item
    95\% credible interval: \emph{``Given my map, there's a 95\% chance
    the fish is inside this part of the lake.''}
  \item
    The fish's location is uncertain, and probability describes your
    belief
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Probabilistic Programming Languages}
\protect\phantomsection\label{probabilistic-programming-languages}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Steps}}:

  \begin{enumerate}
  \tightlist
  \item
    Specify models using code
  \item
    Numerical models solve inference problems without need of user to
    understand how

    \begin{itemize}
    \tightlist
    \item
      Universal inference engines
    \item
      \textcolor{blue}{\texttt{PyMC3}}: flexible Python library for
      probabilistic programming
    \item
      \textcolor{blue}{\texttt{Theano}}: library to define, optimize,
      evaluate mathematical expressions using tensors
    \item
      \textcolor{blue}{\texttt{ArviZ}}: library to interpret
      probabilistic model results
    \end{itemize}
  \end{enumerate}
\item
  \textbf{\textcolor{darkgreen}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    Compute results without analytical closed form
  \item
    Treat model solving as a black box
  \item
    Focus on model design, evaluation, interpretation
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Probabilistic programming languages}}

  \begin{itemize}
  \tightlist
  \item
    Similar impact as Fortran on scientific computing
  \item
    Build algorithms but ignore computational details
  \end{itemize}
\end{itemize}
\end{frame}

  \begin{itemize}
  \tightlist
  \item

  \end{itemize}
\item
  \textbf{\textcolor{red}{Workflow}}

  \begin{itemize}
  \tightlist
  \item
    Model the prior \(\theta\) and the likelihood \(Y | \theta\)
    \begin{equation*}
      \begin{cases}
      \theta \sim \text{Beta}(\alpha = 1, \beta = 1) \\
      Y \sim \text{Binomial}(n = 1, p = \theta) \\
      \end{cases}
    \end{equation*}
  \item
    Observe samples of the variable \(Y\)
  \item
    Run inference
  \item
    Generate samples of the posterior
  \item
    Summarize posterior

    \begin{itemize}
    \tightlist
    \item
      E.g., Highest-Posterior Density (HPD)
    \end{itemize}
  \item
    \ldots{}
  \end{itemize}
\end{itemize}
\end{frame}
