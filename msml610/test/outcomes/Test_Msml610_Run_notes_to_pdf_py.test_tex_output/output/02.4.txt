\subsubsection{Gradient Descent}\label{gradient-descent}

\begin{frame}{The Problem of Minimizing a Function}
\protect\phantomsection\label{the-problem-of-minimizing-a-function}
\begin{columns}[T]
\begin{column}{0.75\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}: minimize scalar function \(J(\vw)\) of
  \(n\)-variables \(\vw\)

  \begin{itemize}
  \tightlist
  \item
    E.g., in-sample error \(E_{in}(\vw)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solutions}}:

  \begin{enumerate}
  \tightlist
  \item
    Analytical solution

    \begin{itemize}
    \tightlist
    \item
      Impose the gradient of \(J(\vw)\) to equal 0
    \item
      Find a closed-form solution for \(\vw^*\)
    \end{itemize}
  \item
    Numerical solution:

    \begin{itemize}
    \tightlist
    \item
      Use an iterative method to update \(\vw\) to reach the minimum
      value of \(J(\vw)\)
    \item
      E.g., gradient descent
    \item
      It works even if there is an analytical solution
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{column}

\begin{column}{0.3\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L02.4.Gradient_descent_2.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Gradient Descent: Intuition}
\protect\phantomsection\label{gradient-descent-intuition}
\begin{columns}[T]
\begin{column}{0.75\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}:

  \begin{itemize}
  \tightlist
  \item
    You are on a hilly surface and we want to walk down to the bottom of
    the hill
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    At each point:

    \begin{itemize}
    \tightlist
    \item
      You look around
    \item
      You move a step in the direction where the surface is steepest
    \end{itemize}
  \item
    You keep doing until we reach the bottom
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Gradient descent}}

  \begin{itemize}
  \tightlist
  \item
    Is a general technique for minimizing twice-differentiable functions
  \item
    In general, converge to a local minimum
  \item
    If \(J(\vw)\) is convex, converge to the global minimum

    \begin{itemize}
    \tightlist
    \item
      E.g., logistic regression and linear models
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.3\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L02.4.Gradient_descent_1.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Gradient descent with fixed learning rate (1/3)}
\protect\phantomsection\label{gradient-descent-with-fixed-learning-rate-13}
\begin{itemize}
\item
  Consider the contour plot of a function
\item
  \textbf{\textcolor{red}{Start}} from a point \(\vw(0)\) (random, the
  origin, \ldots)
\item
  \textbf{\textcolor{darkgreen}{At each step}}

  \begin{itemize}
  \item
    Move a fixed amount \(\eta\) in weight space (fixed learning rate):
    \[
    \vw(t + 1) = \vw(t) + \eta \hat{\vvv}
    \] where \(\hat{\vvv}\) is a unit vector
  \item
    Pick \(\hat{\vvv}\) to move to a value of \(E_{in}(\vw)\) as
    negative as possible
  \end{itemize}
\item
  The \textbf{\textcolor{blue}{change for $E\_{in}$}} is:
  \begingroup \footnotesize
  \begin{align*}
  \Delta E_{in}
  & = E_{in}(\vw(t + 1)) - E_{in}(\vw(t))
  & \\
  & = E_{in}(\vw(t) + \eta \hat{\vvv}) - E_{in}(\vw(t))
  & \text{ (replacing the expression of $\vw(t + 1)$)}
  \\
  & \approx \eta \nabla E_{in}(\vw(t))^T \hat{\vvv} + O(\eta ^ 2)
  & \text{ (using Taylor expansion)}
  \\
  \end{align*}
  \endgroup

  \begin{itemize}
  \tightlist
  \item
    Gradient descent keeps only \(O(\eta)\) the term and ignores the
    rest
  \item
    Conjugate gradient considers up to \(O(\eta^2)\) and ignores higher
    infinitesimals
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Gradient Descent with Fixed Learning Rate (2/3)}
\protect\phantomsection\label{gradient-descent-with-fixed-learning-rate-23}
\begin{itemize}
\item
  You have: \[
  \Delta E_{in}
  \approx \eta \nabla E_{in}(\vw(t))^T \hat{\vvv}
  \]
\item
  The \textbf{\textcolor{red}{minimal value of the scalar product}}
  \(\Delta E_{in}\):

  \begin{itemize}
  \tightlist
  \item
    Happens when
    \(\hat{\vvv} = - \frac{\nabla E_{in}(\vw(t))}{\|\nabla E_{in}(\vw(t))\|}\)
  \item
    Is \(- \eta \|\nabla E_{in}(\vw(t))\|\)
  \end{itemize}
\item
  The \textbf{\textcolor{darkgreen}{change in weights}} \(\Delta \vw\)
  is: \begingroup \footnotesize
  \begin{align*}
  \Delta \vw
  & = \vw(t + 1) - \vw(t) \\
  & = \eta \hat{\vvv} \\
  & = - \eta \frac{\nabla E_{in}(\vw(t))}{\|\nabla E_{in}(\vw(t))\|}
  \end{align*}
  \endgroup
\item
  Called \textbf{\textcolor{blue}{"gradient descent"}} since it descends
  along the gradient of the function to optimize
\end{itemize}
\end{frame}

\begin{frame}{Gradient Descent with Fixed Learning Rate (3/3)}
\protect\phantomsection\label{gradient-descent-with-fixed-learning-rate-33}
\begin{itemize}
\tightlist
\item
  Each \(j\) component of the weight \(\vw\) is updated with the partial
  derivative with respect to that coordinate: \begin{align*}
  \vw(t + 1)
  &= \vw(t) - \eta \frac{\nabla E_{in}(\vw(t))}{\|\nabla E_{in}(\vw(t))\|} \\
  w_j(t + 1)
  &= w_j(t) - \eta
  \frac{1}{\|\nabla E_{in}(\vw(t))\|}
  \frac{\partial E_{in}(\vw)}{\partial w_j}
  \end{align*}

  \begin{itemize}
  \tightlist
  \item
    The update of all components should be
    \textbf{\textcolor{red}{simultaneous}}, i.e., computed at once
  \end{itemize}
\item
  A step of the optimization when we update the solution (weights) is
  called \textbf{\textcolor{blue}{epoch}}
\end{itemize}
\end{frame}

\begin{frame}{Gradient Descent: Stopping Criteria}
\protect\phantomsection\label{gradient-descent-stopping-criteria}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{In theory}}, stop when
  \(\Delta E_{in} = \vv{0}\)

  \begin{itemize}
  \tightlist
  \item
    Numerically, this might not occur
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{In practice}}, stop when:

  \begin{itemize}
  \tightlist
  \item
    Variation of \(E_{in}\) is smaller than threshold
    \(\Delta E_{in} < \theta\)
  \item
    Reached a certain number of iterations
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Monitoring gradient descent}}

  \begin{itemize}
  \tightlist
  \item
    In theory, only compute derivatives of function \(J(\vw)\) to
    optimize
  \item
    In practice, monitor progress by recomputing cost function
    \(J(\vw)\) periodically to ensure it decreases
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Gradient Descent with Fixed Learning Rate}
\protect\phantomsection\label{gradient-descent-with-fixed-learning-rate}
\begin{itemize}
\tightlist
\item
  Consider a 1D convex function

  \begin{itemize}
  \tightlist
  \item
    If \textbf{\textcolor{red}{$\eta$ is too small}}:

    \begin{itemize}
    \tightlist
    \item
      Linear approximation of \(E_{in}\) is effective
    \item
      Many steps needed to converge to minimum
    \end{itemize}
  \item
    If \textbf{\textcolor{darkgreen}{$\eta$ is too large}}:

    \begin{itemize}
    \tightlist
    \item
      Linear approximation fails (higher terms affect values)
    \item
      It ``bounces around''
    \end{itemize}
  \end{itemize}
\end{itemize}

\includegraphics[width=0.9\linewidth,height=\textheight,keepaspectratio]{msml610/lectures_source/figures/L02.4.Gradient_descent_3.png}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Idea}}: vary learning rate \(\eta\) during
  gradient descent

  \begin{itemize}
  \tightlist
  \item
    Smaller learning rates may find a better minimum
  \item
    Reduce \(\eta\) as a function of iterations
  \item
    \textbf{\textcolor{violet}{Cons}}: introduces an additional
    parameter to tune
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Gradient Descent with Variable Learning Rate}
\protect\phantomsection\label{gradient-descent-with-variable-learning-rate}
\begin{itemize}
\item
  In \textbf{\textcolor{red}{gradient descent with fixed learning rate}}
  (constant change in weight space), use: \[
  \Delta \vw = - \eta \frac{\nabla J}{\|\nabla J\|}
  \]
\item
  \textbf{\textcolor{darkgreen}{To converge quickly:}}

  \begin{itemize}
  \tightlist
  \item
    Move fast (large change) in weight space (large \(\eta\)) when
    surface is steep (large gradient)
  \item
    Move slow (small change) in weight space (small \(\eta\)) near
    minimum to avoid bouncing (small gradient)
  \item
    Ideally, \(\eta\) should increase with slope:
    \(\eta \propto \|\nabla J\|\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Gradient descent with variable learning rate}}:
  \[
  \Delta \vw = - \eta \nabla J
  \]
\end{itemize}
\end{frame}

\begin{frame}{Feature Scaling in Gradient Descent}
\protect\phantomsection\label{feature-scaling-in-gradient-descent}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: Different gradient components have
  different errors due to numerical approximation, causing bouncing

  \begin{itemize}
  \tightlist
  \item
    Unscaled features lead to slow, unstable convergence
  \item
    E.g., feature ranging from 1 to 1000 vs.~0.01 to 1 causes
    inefficient updates
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}: Converge faster by scaling
  features to the same range

  \begin{itemize}
  \tightlist
  \item
    Min-max scaling
  \item
    Standardization (transforms feature values to mean 0, standard
    deviation 1)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Issues with Batch Gradient Descent}
\protect\phantomsection\label{issues-with-batch-gradient-descent}
\begin{itemize}
\item
  Consider squared error with \(n\) samples: \[
  E_{in}(\vw)
  \triangleq \frac{1}{n} \sum_{i=1}^n (h_{\vw}(\vx_i) - y_i) ^ 2
  \]
\item
  \textbf{\textcolor{red}{Batch Gradient Descent (BGD)}} updates each
  weight component: \[
  \vw(t + 1) = \vw(t) - \eta \frac{\nabla E_{in}}{\|\nabla E_{in}\|}
  \]

  \begin{itemize}
  \item
    Coordinate update for squared error: \[
    w_j(t + 1) = w_j(t) -
    \eta \frac{2}{n}
    \sum_{i=1}^n (h_{\vw}(\vx_i) - y_i)
    \frac{\partial h_{\vw}(\vx_i)}{\partial w_j}
    \]
  \item
    \textbf{\textcolor{blue}{Cons}}: gradient descent with many training
    examples (e.g., \(N = 10^6\))

    \begin{itemize}
    \tightlist
    \item
      Is computationally expensive, requiring gradient evaluation from
      all examples for one update
    \item
      Requires storing all data in memory
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\subsubsection{Stochastic Gradient
Descent}\label{stochastic-gradient-descent}

\begin{frame}{Stochastic Gradient Descent}
\protect\phantomsection\label{stochastic-gradient-descent-1}
\begin{itemize}
\item
  Idea of \textbf{\textcolor{red}{Stochastic Gradient Descent (SGD)}}:
  update the weights only for one training example picked at random
\item
  \textbf{\textcolor{darkgreen}{Algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Pick one \((\vx_n, y_n)\) at a time from the available examples
  \item
    Compute \(\nabla e(h(\vx_n), y_n)\) to update the weights:
    \(\Delta \vw = -\eta \nabla e\)
  \item
    Update the weight considering only one random example: \[
    w_j(t + 1) = w_j(t) -
    \eta \frac{2}{n}
    (h_{\vw}(\vx_t) - y_t)
    \frac{\partial h_{\vw}(\vx_t)}{\partial w_j}
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}: in Stochastic Gradient Descent

  \begin{itemize}
  \tightlist
  \item
    Path in weight space is random
  \item
    Oscillates around local minimum
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Why SGD works}}

  \begin{itemize}
  \tightlist
  \item
    \(\nabla e\) is a function of a random var \(\vx_n\)
  \item
    The average direction of SGD is the same direction as batch version
    \[
    \EE[\nabla e]
    = \frac{1}{N} \sum \nabla e(h(\vx_n), y_n)
    = \nabla \frac{1}{N} \sum e(h(\vx_n), y_n)
    = \nabla E_{in}
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Mini-Batch Gradient Descent}
\protect\phantomsection\label{mini-batch-gradient-descent}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Mini-Batch Gradient Descent}} brings together
  characteristics of both Batch and Stochastic Gradient Descent

  \begin{itemize}
  \tightlist
  \item
    Use \(b\) examples to make an update to the current weight
  \item
    \(b\) represents the batch size
  \item
    A common choice is \(b = 32\) or \(b = 64\)
  \end{itemize}
\item
  Mini-batch GD offers a balance between SGD noisiness and full-batch
  approaches, using small, random data samples for updates
\end{itemize}
\end{frame}

\begin{frame}{Coordinate Descend}
\protect\phantomsection\label{coordinate-descend}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Idea}}: minimize \(J(x_0, ..., x_n)\) by
  optimizing along one direction \(x_i\) at a time

  \begin{itemize}
  \tightlist
  \item
    Instead of computing all derivatives
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Pick a random starting point \(\vw(0)\)
  \item
    Pick a random order for the coordinates \(\{ x_i \}\)
  \item
    Find the minimum along the current coordinate (1D optimization
    problem)
  \item
    Move to the next coordinate \(x_{i+1}\)
  \item
    The sequence of \(\vw(t)\) is decreasing
  \item
    A minimum is found if there is no improvement after one cycle of
    scanning all coordinates
  \item
    The minimum is local
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Gradient Descent vs Pseudo-Inverse}
\protect\phantomsection\label{gradient-descent-vs-pseudo-inverse}
\begin{itemize}
\item
  For linear models, you can use pseudo-inverse or gradient descent to
  find optimal \(\vw^*\)
\item
  \textbf{\textcolor{red}{Gradient descent}}

  \begin{itemize}
  \tightlist
  \item
    Choose learning rate \(\eta\)
  \item
    Requires many iterations
  \item
    Monitor stopping criteria, oscillations
  \item
    Effective for many features \(P\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pseudo-inverse}}

  \begin{itemize}
  \tightlist
  \item
    No parameter selection
  \item
    Converges in one iteration
  \item
    Computes a \(P \times P\) matrix \((\mX^T \mX)^{-1}\)

    \begin{itemize}
    \tightlist
    \item
      Complexity of matrix inversion \(O(P^3)\)
    \item
      For \(P \approx 10,000\), prefer gradient descent
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}
