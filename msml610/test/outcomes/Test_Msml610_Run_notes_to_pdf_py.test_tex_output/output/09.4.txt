\begin{frame}{Tracking Objects}
\protect\phantomsection\label{tracking-objects}
\begin{itemize}
\item
  Many problems can be formulated as
  \textbf{\textcolor{red}{tracking objects}}
\item
  \textbf{\textcolor{darkgreen}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    Navigation of aircraft, drones, autonomous cars
  \item
    Robotics: arm kinematics to predict the position of joints
  \item
    Sensor fusion: merge multiple sensor readings
  \item
    Finance: predict economic variables
  \item
    Computer vision: track moving objects across video
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Kalman filter}}

  \begin{itemize}
  \tightlist
  \item
    Used for state estimation in dynamic systems with noisy, uncertain
    measurements
  \item
    Track state over time using model to make predictions and
    observations
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Some Guiding Principles}
\protect\phantomsection\label{some-guiding-principles}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{The world is noisy}}

  \begin{itemize}
  \tightlist
  \item
    E.g., a car might swerve around a pothole or brake for a pedestrian
  \item
    E.g., wind or ice might change the car's path
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Sensors are noisy}}

  \begin{itemize}
  \tightlist
  \item
    E.g., a kitchen scale gives different readings for the same object
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Knowledge is uncertain}}

  \begin{itemize}
  \tightlist
  \item
    E.g., you alter beliefs on how things work based on evidence and its
    strength
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Combine past information and system knowledge}}
  to estimate future information

  \begin{itemize}
  \tightlist
  \item
    E.g., if a car moves at speed \(X\) at time \(t\), the speed at time
    \(t+1\) is \(\approx X\)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Data (even if noisy) > guess}}

  \begin{itemize}
  \tightlist
  \item
    Never discard information, no matter how poor it is
  \item
    E.g., two sensors, even if one is less accurate, are better than one
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Example of Estimating Body Weight}
\protect\phantomsection\label{example-of-estimating-body-weight}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: estimate your weight over time while
  going to the gym
\item
  Two possible approaches

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Predict your weight}}

    \begin{itemize}
    \tightlist
    \item
      Track calorie intake
    \item
      Track energy expense
    \item
      Compute expected weight gain
    \item
      \emph{Cons}: Difficult to track food intake and exercise
      accurately
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Measure your weight}}

    \begin{itemize}
    \tightlist
    \item
      Use a scale
    \item
      \emph{Cons}: Scale is noisy, water weight fluctuates, different
      clothes
    \end{itemize}
  \end{enumerate}
\end{itemize}

\begin{frame}{Example of Estimating Body Weight}
\protect\phantomsection\label{example-of-estimating-body-weight-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{What's the real weight?}}

  \begin{itemize}
  \tightlist
  \item
    You need to blend prediction and measurement \[
    \text{estimate = 0.6} \times \text{prediction} + (1 - 0.6) \times \text{measurement}
    \]
  \item
    Because of coefficient 0.6 you believe the prediction is more likely
    correct than the measurement
  \item
    The prediction is always between the prediction and the measurement
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Algorithm}}

  \begin{enumerate}
  \tightlist
  \item
    Start with an initial guess

    \begin{itemize}
    \tightlist
    \item
      Assume it's correct for now
    \end{itemize}
  \item
    Predict the next weight based on the model
  \item
    Measure the weight
  \item
    Estimate the next weight by merging values:
  \item
    Go back to first step
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{g-h Filter}
\protect\phantomsection\label{g-h-filter}
\begin{itemize}
\tightlist
\item
  The previous algorithm is called \textbf{\textcolor{red}{g-h filter}}

  \begin{itemize}
  \tightlist
  \item
    \(g\): scaling used to blend predicted state and measurement
  \item
    \(h\): scaling used to update the parameter of the system model
    based on the measurements
  \end{itemize}
\item
  g-h filters have different values of \(g\) and \(h\) to achieve
  different properties

  \begin{itemize}
  \tightlist
  \item
    E.g., pick \(g\) to minimize the transient error when the derivative
    of the signal has a step (i.e., a discontinuity of the slope)
  \end{itemize}
\item
  Many filters (including Kalman filter) are just generalizations of a
  g-h filter
\end{itemize}
\end{frame}

\begin{frame}{g-h Filter Algorithm: Pseudo-Code}
\protect\phantomsection\label{g-h-filter-algorithm-pseudo-code}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{enumerate}
\tightlist
\item
  \textbf{\textcolor{red}{Initialization}}

  \begin{itemize}
  \tightlist
  \item
    Initialize state of the filter
  \item
    Initialize belief in the state
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Predict}}

  \begin{itemize}
  \tightlist
  \item
    Use system model to predict state at next time step
  \item
    Adjust belief to account for uncertainty in prediction
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Update}}

  \begin{itemize}
  \tightlist
  \item
    Get measurement and associated belief about its accuracy
  \item
    Use as estimate of the next state a point between estimated state
    and measurement
  \end{itemize}
\end{enumerate}
\end{column}

\begin{frame}{Interpretation of \(g\)}
\protect\phantomsection\label{interpretation-of-g}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  If \(g = 0\):

  \begin{itemize}
  \tightlist
  \item
    Follow system model, ignore measurements
  \end{itemize}
\item
  If \(g\) is small:

  \begin{itemize}
  \tightlist
  \item
    Follow model more than measurements
  \end{itemize}
\item
  If \(g\) increases:

  \begin{itemize}
  \tightlist
  \item
    Follow measurements more, ignore prediction
  \item
    Useful with accurate measurements, inaccurate model
  \end{itemize}
\item
  If \(g = 1\):

  \begin{itemize}
  \tightlist
  \item
    Follow only measurements, ignore model
  \item
    Too large \(\to\) follow measurements, reject no noise
  \end{itemize}
\end{itemize}
\end{column}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{Interpretation of \(h\)}
\protect\phantomsection\label{interpretation-of-h}
\begin{columns}[T]
\begin{column}{0.4\linewidth}
\begin{itemize}
\tightlist
\item
  Estimate model parameters from data:

  \begin{itemize}
  \tightlist
  \item
    Change of weight
  \item
    Rate of change of measurements
  \item
    Car speed on different terrains
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.55\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L09.4.varying_h1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  If \(h = 0\):

  \begin{itemize}
  \tightlist
  \item
    Filter follows previous rate of change
  \item
    Adapts slowly to signal changes
  \end{itemize}
\item
  If \(h = 1\):

  \begin{itemize}
  \tightlist
  \item
    Filter reacts rapidly to significant signal variations
  \end{itemize}
\item
  If \(h\) is:

  \begin{itemize}
  \tightlist
  \item
    Smaller \(\to\) Smoother, slower rate adaptation
  \item
    Larger \(\to\) Faster rate adjustment
  \end{itemize}
\item
  \textbf{\textcolor{red}{Note}}: Incorrect initial state is similar to
  a changing state
\end{itemize}
\end{frame}

  \textbf{\textcolor{orange}{Filter}}: algorithm to estimate system

\item
  \textbf{\textcolor{darkgreen}{State of the system}}

  \begin{itemize}
  \tightlist
  \item
    Part of the state might be hidden
  \end{itemize}

\begin{itemize}
