\begin{frame}{Is Machine Learning Possible?}
\protect\phantomsection\label{is-machine-learning-possible}
\begin{itemize}
\tightlist
\item
  The function can assume
  \textbf{\textcolor{red}{any value outside data}}

  \begin{itemize}
  \tightlist
  \item
    E.g., with summer temperature data, the function could assume a
    different value for winter
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{How to learn an unknown function?}}

  \begin{itemize}
  \tightlist
  \item
    Estimating at unseen points seems impossible in general
  \item
    Requires assumptions or models about behavior
  \end{itemize}
\item
  Difference between:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{blue}{Possible}}

    \begin{itemize}
    \tightlist
    \item
      No knowledge of the unknown function
    \item
      E.g., could be linear, quadratic, or sine wave outside known data
    \end{itemize}
  \item
    \textbf{\textcolor{violet}{Probable}}

    \begin{itemize}
    \tightlist
    \item
      Some knowledge of the unknown function from domain knowledge or
      historical data patterns
    \item
      E.g., if historical weather data forms a sinusoidal pattern,
      unknown points likely follow that pattern
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

  \begin{itemize}
  \tightlist
  \item
    We want to estimate \(\Pr(\text{pick a \red{red} marble}) = \mu\)
    where the value of \(\mu\) is unknown
  \item
    We pick \(N\) marbles independently with replacement
  \item
    The fraction of \red{red} marbles is \(\nu\)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L05.1.Bin_with_marbles.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  Does \(\nu\) say anything about \(\mu\)?

  \begin{itemize}
  \tightlist
  \item
    \textbf{``No''}

    \begin{itemize}
    \tightlist
    \item
      In strict terms, we don't know anything about the marbles we
      didn't pick
    \item
      The sample can be mostly \green{green}, while the bin is mostly
      \red{red}
    \item
      This is \emph{possible}, but \emph{not probable}
    \end{itemize}
  \item
    \textbf{``Yes''}

    \begin{itemize}
    \tightlist
    \item
      Under certain conditions, the sample frequency is close to the
      real frequency
    \end{itemize}
  \end{itemize}
\item
  \textbf{Possible vs probable}

  \begin{itemize}
  \tightlist
  \item
    It is \textbf{possible} that we don't know anything about the
    marbles in the bin
  \item
    It is \textbf{probable} that we know something
  \item
    Hoeffding inequality makes this intuition formal
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Hoeffding Inequality}
\protect\phantomsection\label{hoeffding-inequality}
\begin{itemize}
\item
  Consider a Bernoulli random variable \(X\) with probability of success
  \(\mu\)
\item
  Estimate the mean \(\mu\) using \(N\) samples with
  \(\nu = \frac{1}{N} \sum_i X_i\)
\item
  The \textbf{\textcolor{red}{probably approximately correct}} (PAC)
  statement holds: \[
  \Pr(|\nu - \mu| > \varepsilon) \le \frac{2}{e^{2 \varepsilon^2 N}}
  \]
\item
  \textbf{\textcolor{blue}{Remarks:}}

  \begin{itemize}
  \tightlist
  \item
    Valid for all \(N\) and \(\varepsilon\), not an asymptotic result
  \item
    Holds only if you sample \(\nu\) and \(\mu\) at random and in the
    same way
  \item
    If \(N\) increases, it is exponentially small that \(\nu\) will
    deviate from \(\mu\) by more than \(\varepsilon\)
  \item
    The bound does not depend on \(\mu\)
  \item
    Trade-off between \(N\), \(\varepsilon\), and the bound:

    \begin{itemize}
    \tightlist
    \item
      Smaller \(\varepsilon\) requires larger \(N\) for the same
      probability bound
    \item
      Since \(\nu \in [\mu - \varepsilon, \mu + \varepsilon]\), you want
      small \(\varepsilon\) with a large probability
    \end{itemize}
  \item
    It is a statement about \(\nu\) and not \(\mu\) although you use it
    to state something about \(\nu\) (like for a confidence interval)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Supervised Learning: Bin Analogy (2/2)}
\protect\phantomsection\label{supervised-learning-bin-analogy-22}
\begin{itemize}
\tightlist
\item
  Let's connect the bin analogy, Hoeffding inequality, and feasibility
  of machine learning

  \begin{itemize}
  \tightlist
  \item
    You know \(f(\vx)\) at points \(\vx \in \calX\)
  \item
    You choose an hypothesis \(h: \calX \rightarrow \calY = \{0, 1\}\)
  \item
    Each point \(\vx \in \calX\) is a marble
  \item
    You color \red{red} if the hypothesis is correct
    \(h(\vx) = f(\vx)\), \green{green} otherwise
  \item
    The in-sample error \(E_{in}(h)\) corresponds to \(\nu\)
  \item
    The marbles of unknown color corresponds to \(E_{out}(h) = \mu\)
  \item
    \(\vx_1, ..., \vx_n\) are picked randomly and independently from a
    distribution over \(\calX\) which is the same as for \(E_{out}\)
  \end{itemize}
\item
  Hoeffding inequality holds and bounds the error going from in-sample
  to out-of-sample \[
  \Pr(|E_{in} - E_{out}| > \varepsilon) \le c
  \]

  \begin{itemize}
  \tightlist
  \item
    Generalization over unknown points (i.e., marbles) is possible
  \item
    \textbf{Machine learning is possible!}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Validation vs Learning: Bin Analogy}
\protect\phantomsection\label{validation-vs-learning-bin-analogy}
\begin{itemize}
\tightlist
\item
  You have learned that for a given \(h\), in-sample performance
  \(E_{in}(h) = \nu\) needs to be close to out-of-sample performance
  \(E_{out}(h) = \mu\)

  \begin{itemize}
  \tightlist
  \item
    This is the \textbf{\textcolor{red}{validation setup}}, after you
    have already learned a model
  \end{itemize}
\item
  In a \textbf{\textcolor{darkgreen}{learning setup}} you have \(h\) to
  choose from \(M\) hypotheses

  \begin{itemize}
  \tightlist
  \item
    You need a bound on the out-of-sample performance of the chosen
    hypothesis \(h \in \calH\), regardless of which hypothesis you
    choose
  \item
    You need a Hoeffding counterpart for the case of choosing from
    multiple hypotheses \begingroup \small
    \begin{alignat*}{2}
    & \forall g \in \calH = \{h_1, ... , h_M\} \; \Pr(|E_{in}(g) - E_{out}(g)| > \varepsilon)
    &
    \\
    & \hspace{1cm} \le \Pr(\bigcup_{i=1}^M (|E_{in}(h_i) - E_{out}(h_i) | > \varepsilon))
    &
    \\
    & \hspace{1cm} \le \sum_{i=1}^M \Pr(|E_{in}(h_i) - E_{out}(h_i)| > \varepsilon)
    & \text{  (by the union bound)}
    \\
    & \hspace{1cm} \le 2 M \exp(-2 \varepsilon^2 N)
    & \text{  (by Hoeffding)}
    \\
    \end{alignat*}
    \endgroup
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Problem}}: the bound is weak
\end{itemize}
\end{frame}

\begin{frame}{Validation vs Learning: Coin Analogy}
\protect\phantomsection\label{validation-vs-learning-coin-analogy}
\begin{itemize}
\item
  In a \textbf{\textcolor{red}{validation set-up}}, you have a coin and
  want to determine if it is fair
\item
  Assume the coin is unbiased: \(\mu = 0.5\)
\item
  Toss the coin 10 times
\item
  How likely is that you get 10 heads (i.e., the coin looks biased
  \(\nu = 0\))? \[
  \Pr(\text{coin shows } \nu = 0) = 1 / 2^{10} = 1 / 1024 \approx 0.1\%
  \]
\item
  \textbf{\textcolor{blue}{Conclusion}}: the probability that the
  out-of-sample performance (\(\nu=0.0\)) is completely different from
  the in-sample perf (\(\mu=0.5\)) is very low
\end{itemize}
\end{frame}

\begin{frame}{Validation vs Learning: Coin Analogy}
\protect\phantomsection\label{validation-vs-learning-coin-analogy-1}
\begin{itemize}
\item
  In a \textbf{\textcolor{red}{learning set-up}}, you have many coins
  and you need to choose one and determine if it's fair
\item
  If you have 1000 fair coins, how likely is it that at least one
  appears totally biased using 10 experiments?

  \begin{itemize}
  \tightlist
  \item
    I.e., out-of-sample performance is completely different from
    in-sample performance \[
    \begin{aligned}
    \Pr(\text{at least one coin has } \nu = 0) &
    = 1 - \Pr(\text{all coins have } \nu \neq 0)\\
    &= 1 - (\Pr(\text{a coin has } \nu \neq 0)) ^{10}\\
    &= 1 - (1 - \Pr(\text{a coin has } \nu = 0)) ^{10}\\
    &= 1 - (1 - 1 / 2 ^ {10}) ^ {1000}\\
    &\approx 0.63\%
    \end{aligned}
    \]
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Conclusion}}: It is probable, more than 50\%
\end{itemize}
\end{frame}

\begin{frame}{Validation vs Learning: Hoeffding Inequality}
\protect\phantomsection\label{validation-vs-learning-hoeffding-inequality}
\begin{itemize}
\tightlist
\item
  In \textbf{\textcolor{red}{validation / testing}}

  \begin{itemize}
  \tightlist
  \item
    Use Hoeffding to assess how well our \(g\) (the \emph{chosen
    hypothesis}) approximates \(f\) (the \emph{unknown hypothesis}): \[
    \Pr(|E_{in} - E_{out}| > \varepsilon) \le 2 \exp(-2 \varepsilon^2 N)
    \] where: \begingroup \small
    \begin{alignat*}{2}
    & E_{in}(g) = \frac{1}{N} \sum_i e(g(\vx_i), f(\vx_i)) \\
    & E_{out}(g) = \EE_{\vx}[e(g(\vx), f(\vx))] \\
    \end{alignat*}
    \endgroup
  \item
    Since the hypothesis \(g\) is final and fixed, Hoeffding inequality
    guarantees that you can learn since it gives a bound for \(E_{out}\)
    to track \(E_{in}\)
  \end{itemize}
\item
  In \textbf{\textcolor{darkgreen}{learning}}

  \begin{itemize}
  \tightlist
  \item
    Need to account that our hypothesis is the best of \(M\) hypotheses,
    so: \[
    \Pr(|E_{in} - E_{out}| > \varepsilon) \le 2 M \exp(-2 \varepsilon^2 N)
    \]
  \item
    The bound for \(E_{out}\) from Hoeffding is weak
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Questions}}:

  \begin{itemize}
  \tightlist
  \item
    Is the bound weak because it needs to be?
  \item
    Is it possible to replace it with a stricter bound?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Intuition Why Bound for Hoeffding Is Weak}
\protect\phantomsection\label{intuition-why-bound-for-hoeffding-is-weak}
\begin{itemize}
\item
  The Hoeffding inequality and the union bound applied to training set
  \[
  \Pr(|E_{in} - E_{out}| > \varepsilon) \le 2 M \exp(-2 \varepsilon^2 N)
  \] is \textbf{\textcolor{red}{artificially}} too loose
\item
  \(M\) was coming from the bad event: \begin{alignat*}{2}
  \calB_i
  &= \textit{"hypothesis $h_i$ does not generalize out-of-sample"} \\
  &= "|E_{in}(h_i) - E_{out}(h_i)| > \varepsilon"
  \end{alignat*}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\item
  Since \(g \in \{h_1, h_2, \cdots, h_M\}\) then \(\Pr(\calB)
  \le \Pr(\bigcup_i \calB_i)
  \le \sum_i \Pr(\calB_i)\)
\item
  The union bound assumes the events are disjoint, leading to a
  conservative estimate if events overlap
\item
  \textbf{\textcolor{blue}{In reality}}, bad events are extremely
  overlapping because bad hypotheses are extremely similar
\end{itemize}
\end{column}

\end{frame}

\begin{frame}[fragile]{Dichotomy: Definition}
\protect\phantomsection\label{dichotomy-definition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: classify \(N\) (fixed) points
  \(\vx_1, ..., \vx_N\) with an hypothesis set \(\calH\) of multi-class
  classifiers
\item
  Consider an assignment \(D\) of the points to certain class
  \(\vd_1, ..., \vd_N\)
\item
  \(D\) is a \textbf{\textcolor{darkgreen}{dichotomy}} for hypothesis
  set \(\calH\) \(\iff\) there exists \(h \in \calH\) that gets the
  desired classification \(D\)
\end{itemize}

\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \item
    4 points in a plane \(A, B, C, D\)
  \item
    Binary classification
  \item
    \(\calH\) = \{ bidimensional perceptrons \}
  \item
    Moving the separating hyperplane, you get different classifications
    for the points (i.e., dichotomies)

\begin{verbatim}
      D1    D2   D3   D4   D...
A     o     x      ...
B     x     x
C     o     o
D     x     o      ...
\end{verbatim}
  \item
    There are at most \(2^N\) dichotomies
  \item
    Certain classifications are not possible (e.g., XOR assignment)
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Dichotomies vs Hypotheses}
\protect\phantomsection\label{dichotomies-vs-hypotheses}
\begin{itemize}
\item
  An \textbf{\textcolor{red}{hypothesis}} classifies each point of
  \(\calX\): \(\calX \rightarrow \{-1, +1\}\)
\item
  A \textbf{\textcolor{darkgreen}{dichotomy}} classifies each point of a
  fixed set: \(\{\vx_1, ..., \vx_N\} \rightarrow \{-1, +1\}\)

  \begin{itemize}
  \tightlist
  \item
    Dichotomies are ``mini-hypotheses'', i.e., hypotheses restricted to
    given points
  \item
    A dichotomy depends on:

    \begin{itemize}
    \tightlist
    \item
      The number of points \(N\)
    \item
      Hypothesis set \(\calH\) (i.e., the possible models)
    \item
      Where the points are placed
    \item
      How the points are assigned
    \end{itemize}
  \end{itemize}
\item
  The \textbf{\textcolor{blue}{number of different dichotomies}} is
  indicated by \(|\calH(\vx_1, ..., \vx_N)|\)

  \begin{itemize}
  \tightlist
  \item
    The number of dichotomies is always finite, since
    \(|\calH(\vx_1, ..., \vx_N)| \le N^K\)
  \item
    The number of hypotheses is usually infinite, i.e.,
    \(|\calH| = \infty\)
  \end{itemize}
\item
  The ``complexity'' of \(\calH\) is related to the number of hypothesis
\item
  From the training set point of view what matters are dichotomies and
  not hypotheses

  \begin{itemize}
  \tightlist
  \item
    Many (infinite) hypotheses can correspond to the same dichotomy
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Growth Function}
\protect\phantomsection\label{growth-function}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{growth function}} counts the maximum
  number of possible dichotomies on \(N\) points for a hypothesis set
  \(\calH\):

  \[
  m_{\calH}(N)
  = \max_{\vx_1, \cdots, \vx_N \in \calX} |\calH(\vx_1, \cdots, \vx_N)|
  \]
\item
  \textbf{\textcolor{darkgreen}{Why growth function?}}

  \begin{itemize}
  \tightlist
  \item
    The dichotomies depend on point distribution and assignment
  \item
    The growth function considers the maximum by placing points in the
    most ``favorable way'' for the hypothesis set
  \end{itemize}
\item
  To compute \(m_{\calH}(N)\) by \textbf{\textcolor{blue}{brute force}}:

  \begin{itemize}
  \tightlist
  \item
    Consider all possible placements of \(N\) points
    \(\vx_1, ..., \vx_N\)
  \item
    Consider all possible assignments of the points to the classes
  \item
    Consider all possible hypotheses \(h \in \calH\)
  \item
    Compute the corresponding dichotomy for \(h\) on
    \(\vx_1, ..., \vx_N\)
  \item
    Count the number of different dichotomies
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What Can Vary in a Dichotomy}
\protect\phantomsection\label{what-can-vary-in-a-dichotomy}
\begin{itemize}
\item
  Given:

  \begin{itemize}
  \tightlist
  \item
    An hypothesis set \(\calH\) (e.g., bidimensional perceptrons)
  \item
    \(N\) (fixed) points \(\vx_1, ..., \vx_N\)
  \item
    An assignment \(D\) of the points to certain class
    \(\vd_1, ..., \vd_N\)
  \end{itemize}
\item
  \(D\) is a \textbf{\textcolor{red}{dichotomy}} for hypothesis set
  \(\calH\) \(\iff\) there exists \(h \in \calH\) that gets the desired
  classification \(D\)
\item
  There are various quantities in the definition of dichotomy

  \begin{itemize}
  \tightlist
  \item
    The hypothesis set \(\calH\)

    \begin{itemize}
    \tightlist
    \item
      It is fixed
    \end{itemize}
  \item
    The number of dimensions of the input space

    \begin{itemize}
    \tightlist
    \item
      It is fixed through the hypothesis set \(\calH\)
    \end{itemize}
  \item
    The number of points \(N\)

    \begin{itemize}
    \tightlist
    \item
      Input to the growth function \(m_{\calH}(N)\)
    \end{itemize}
  \item
    How the points are assigned to the classes \(\vd_1, ..., \vd_N\)

    \begin{itemize}
    \tightlist
    \item
      It is a free parameter, removed by how each hypothesis in
      \(\calH\) ``splits'' the space
    \end{itemize}
  \item
    Where the points are positioned \(\vx_1, ..., \vx_N\)

    \begin{itemize}
    \tightlist
    \item
      It is a free parameter, removed by the growth function through
      \(\max\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Growth Function Is Increasing}
\protect\phantomsection\label{growth-function-is-increasing}
\begin{itemize}
\item
  \(m_{\calH}(N)\) increases (although not monotonically) with \(N\)
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    The number of dichotomies on \(N=3\) points \(m_{\calH}(3)\) is
    smaller or equal than the number of dichotomies on \(N=4\) points
  \item
    In fact we can ignore a new point and get the same classification
  \end{itemize}
\item
  \(m_{\calH}(N)\) increases with the complexity of \(\calH\)
\item
  \(m_{\calH}(N)\) increases with the number of dimensions in the input
  space (i.e., feature space)
\end{itemize}
\end{frame}

\begin{frame}{Break Point of an Hypothesis Set}
\protect\phantomsection\label{break-point-of-an-hypothesis-set}
\begin{itemize}
\item
  Given an hypothesis set \(\calH\)
\item
  A hypothesis set \(\calH\)
  \textbf{\textcolor{red}{shatters $N$ points}} \(\iff\)
  \(m_{\calH}(N) = 2^N\)

  \begin{itemize}
  \tightlist
  \item
    There is a position of \(N\) points and a class assignment that you
    can classify using \(h \in \calH\)
  \item
    It does not mean all sets of \(N\) points can be classified in any
    way
  \end{itemize}
\item
  \(k\) is a \textbf{\textcolor{blue}{break point}} for \(\calH\)
  \(\iff m_{\calH}(k) < 2^k\)

  \begin{itemize}
  \tightlist
  \item
    I.e., no data set of size \(k\) can be shattered by \(\calH\)
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      For 2D perceptron: a break point is 4
    \item
      For positive rays: a break point is 2
    \item
      For positive intervals: a break point is 3
    \item
      For convex set on a plane: there is no break point
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Break Point for an Hypothesis Set and Learning}
\protect\phantomsection\label{break-point-for-an-hypothesis-set-and-learning}
\begin{itemize}
\item
  If there is a break point for a hypothesis set \(\calH\), it can be
  shown that:

  \begin{itemize}
  \tightlist
  \item
    \(m_{\calH}(N)\) is polynomial in \(N\)
  \item
    Instead of Hoeffding's inequality for learning \[
    \Pr(|E_{in}(g) - E_{out}(g)| > \varepsilon) \le 2 M e^{-2 \varepsilon^2 N}
    \] you can use the Vapnik-Chervonenkis inequality: \[
    \Pr(\text{bad generalization}) \le
    4 m_\calH(2N) e^{-\frac{1}{8} \varepsilon^2 N}
    \]
  \item
    Since \(m_{\calH}(N)\) is polynomial in \(N\), it will be dominated
    by the negative exponential, given enough examples
  \item
    You can have a generalization bound: machine learning works!
  \end{itemize}
\item
  A hypothesis set can be characterized from the learning point of view
  by the \textbf{\textcolor{red}{existence and value of a break point}}
\end{itemize}
\end{frame}

\begin{frame}{VC Dimension of an Hypothesis Set}
\protect\phantomsection\label{vc-dimension-of-an-hypothesis-set}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{VC dimension of a hypothesis set}}
  \(\calH\), denoted as \(d_{VC}(\calH)\), is defined as the largest
  value of \(N\) for which \(m_{\calH}(N) = 2^N\)

  \begin{itemize}
  \tightlist
  \item
    I.e., the VC dimension is the most points \(\calH\) can shatter
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Properties}} of the VC dimension: if
  \(d_{VC}(\calH) = N\) then

  \begin{itemize}
  \item
    Exists a constellation of \(N\) points that can be shattered by
    \(\calH\)

    \begin{itemize}
    \tightlist
    \item
      Not all sets of \(N\) points can be shattered
    \item
      If \(N\) points were placed randomly, they could not be
      necessarily shattered
    \end{itemize}
  \item
    \(\calH\) can \emph{shatter} \(N\) points for any
    \(N \le d_{VC}(\calH)\)
  \item
    The \emph{smallest break point} is \(d_{VC} - 1\)
  \item
    The \emph{growth function} in terms of the VC dimension is
    \(m_{\calH} \le \sum_{i=0}^{d_{VC}} {N \choose i}\)
  \item
    The VC dimension is the \emph{order of the polynomial bounding}
    \(m_{\calH}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{VC Dimension: Interpretation}
\protect\phantomsection\label{vc-dimension-interpretation}
\begin{itemize}
\item
  The VC dimension \textbf{\textcolor{red}{measures the complexity}} of
  a hypothesis set in terms of
  \textbf{\textcolor{blue}{effective parameters}}
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    A perceptron in a \(d\)-dimensional space has \(d_{VC} = d + 1\)
  \item
    In fact \(d_{VC}\) is the number of perceptron parameters!
  \item
    E.g., for a 2D perceptron (\(d = 2\)), the break point is 2, so
    \(d_{VC} = 3\)
  \end{itemize}
\item
  The VC dimension considers the model as a black box in order to
  estimate effective parameters

  \begin{itemize}
  \tightlist
  \item
    How many points \(N\) a model can shatter, not the number of
    parameters
  \end{itemize}
\item
  Not all parameters contribute to degrees of freedom

  \begin{itemize}
  \tightlist
  \item
    E.g., combining \(N\) 1D perceptrons gives \(2N\) parameters, but
    the effective degrees of freedom remain 2
  \end{itemize}
\item
  A complex hypothesis \(\calH\):

  \begin{itemize}
  \tightlist
  \item
    Has more parameters (higher VC dimension \(d_{VC}\))
  \item
    Requires more examples for training
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{VC Generalization Bounds}
\protect\phantomsection\label{vc-generalization-bounds}
\begin{itemize}
\item
  How many data points are needed to obtain
  \(\Pr(|E_{in} - E_{out}| > \varepsilon) \le \delta\)?
\item
  The VC inequality states \[
  \Pr(\text{bad generalization}) \le
    4 m_\calH(2N) e^{-\frac{1}{8} \varepsilon^2 N}
  \]
\end{itemize}

\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \(N^d e^{-N}\) abstracts the upper bound term

  \begin{itemize}
  \tightlist
  \item
    Plot \(N^d e^{-N}\) vs.~\(N\): Power dominates for small \(N\),
    exponential for large \(N\) and brings it to 0
  \item
    Vary \(d\) (VC dimension) function peaks for larger \(N\), then
    approaches the region of interest \(< 1\)
  \end{itemize}
\end{itemize}

\begin{frame}{VC Generalization Bounds}
\protect\phantomsection\label{vc-generalization-bounds-1}
\begin{itemize}
\item
  The VC inequality \[
  \Pr(|E_{in} - E_{out}| > \varepsilon) \le 
    4 m_\calH(2N) e^{-\frac{1}{8} \varepsilon^2 N}
  \] can be used in several ways to relate \(\varepsilon\), \(\delta\),
  and \(N\), e.g.,
\item
  Examples

  \begin{itemize}
  \tightlist
  \item
    ``Given \(\varepsilon\) = 1\% error, how many examples \(N\) are
    needed to get \(\delta = 0.05\)?''
  \item
    ``Given \(N\) examples, what's the probability of an error larger
    than \(\varepsilon\)?''
  \end{itemize}
\item
  You can equate \(\delta\) to
  \(4 m_{\calH}(2N) e^{\frac{1}{8}\varepsilon^2 N}\) and solve for
  \(\varepsilon\), getting \[
  \Omega(N, \calH, \delta)
  = \sqrt{\frac{8}{N} \ln \frac{4 m_{\calH}(2N)}{\delta}}
  \]
\item
  Then you can say \(|E_{out} - E_{in}| \le \Omega(N, \calH, \delta)\)
  with probability \(\ge 1 - \delta\)

  \begin{itemize}
  \tightlist
  \item
    The generalization bounds are then:
    \(\Pr(E_{out} \le E_{in} + \Omega) \ge 1 - \delta\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How to Void the VC Analysis Guarantee}
\protect\phantomsection\label{how-to-void-the-vc-analysis-guarantee}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  Consider the case where data is genuinely non-linear

  \begin{itemize}
  \tightlist
  \item
    E.g., ``o'' points in the center and ``x'' in the corners
  \end{itemize}
\item
  Transform to high-dimensional \(\calZ\) with: \[
  \Phi: \vx = (x_0, ... , x_d) \rightarrow \vz = (z_0, ... , z_{\tilde{d}})
  \]
\item
  \(d_{VC} \le \tilde{d} + 1\); smaller \(\tilde{d}\) improves
  generalization

  \begin{itemize}
  \tightlist
  \item
    Use \(\vz = (1, x_1, x_2, x_1 x_2, x_1^2, x_2^2)\)
  \item
    Why not \(\vz = (1, x_1^2, x_2^2)\)?
  \item
    Why not \(\vz = (1, x_1^2 + x_2^2)\)?
  \item
    Why not \(\vz = (x_1^2 + x_2^2 - 0.6)\)?
  \end{itemize}
\end{itemize}
