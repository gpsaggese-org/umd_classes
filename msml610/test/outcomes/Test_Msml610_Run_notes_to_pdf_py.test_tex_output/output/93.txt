\begin{frame}{Field: definition}
\protect\phantomsection\label{field-definition}
A field \(\bbF = (X, +, *)\) is a set \(X\) with two binary operations +
and *, satisfying the following 6 axioms:

\begin{enumerate}
\tightlist
\item
  Closed with respect to + and *: \[
  \begin{aligned}
  & a, b \in X \implies a + b \in X \\
  & a, b \in X \implies a * b \in X
  \end{aligned}
  \]
\item
  Commutativity of + and *: \[
  \begin{aligned}
  & a + b = b + a \\
  & a * b = b * a
  \end{aligned}
  \]
\item
  Associativity of + and *: \[
  \begin{aligned}
  & a + (b + c) = (a + b) + c = a + b + c \\
  & a * (b * c) = (a * b) * c = a * b * c
  \end{aligned}
  \]
\item
  Distributivity of multiplication over addition:
  \[a * (b + c) = a * b + a * c\]
\item
  Existence of + and * identity elements, \(0\) and \(1\): \[
  \begin{aligned}
  & a + 0 = a \\
  & a * 1 = a
  \end{aligned}
  \]
\item
  Existence of + and * inverses (defining subtraction and division): \[
  \begin{aligned}
  & a + (-a) = 0 \\
  & a * \frac{1}{a} = 1 \text{ for } a \ne 0
  \end{aligned}
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Field: examples}
\protect\phantomsection\label{field-examples}
\begin{itemize}
\tightlist
\item
  Examples

  \begin{itemize}
  \tightlist
  \item
    The set of \(\bbR, \bbC\), GF(2)
  \item
    The set of rational numbers \(\mathbb{Q}\), i.e., numbers that can
    be written as fraction \(\frac{a}{b}\) with \(a, b \in \bbZ\) and
    \(b \ne 0\)
  \end{itemize}
\item
  Non-examples

  \begin{itemize}
  \tightlist
  \item
    The set of positive integers \(\mathbb{N} = 1, 2, 3, \dots\) is not
    a field
  \item
    The set of integers \(\mathbb{Z}\) = \ldots, -2, -1, 0, 1, 2,
    \ldots\$ is not a field
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Vector space: definition}
\protect\phantomsection\label{vector-space-definition}
\begin{itemize}
\tightlist
\item
  A ``vector space \(\calV\) over a field \(\bbF\)'' is a triple
  \((\calV, \bbF, +, \cdot)\) where:

  \begin{itemize}
  \tightlist
  \item
    \(\calV\) is a set of vectors
  \item
    \(\bbF\) is a field of scalars
  \item
    \(+\) is a sum operation between vectors
  \item
    \(\cdot\) is a scalar multiplication
  \end{itemize}
\item
  A vector space needs to satisfy the following 2 properties:

  \begin{enumerate}
  \tightlist
  \item
    Closed with respect to to scalar multiplication: if
    \(\vx \in \calV\), then \(\alpha \cdot \vx \in \calV\)
  \item
    Closed with respect to to vector addition: if
    \(\vx, \vy \in \calV\), then \(\vx + \vy \in \calV\)
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Linear combination of vectors}
\protect\phantomsection\label{linear-combination-of-vectors}
\begin{itemize}
\item
  The vector
  \[\alpha_1 \vvv_1 + \alpha_2 \vvv_2 + ... + \alpha_n \vvv_n\] is a
  linear combination of vectors \(\vvv_1, ..., \vvv_n\) with
  coefficients \(\alpha_1, ..., \alpha_n\)
\item
  A linear combination can be written in matrix form:
  \[\mV \cdot \valpha \text{ or } \valpha^T \cdot \mV^T\] where
  \(\mV = (\vvv_1 | ... | \vvv_n)\)
\end{itemize}
\end{frame}

\begin{frame}{Span of vectors}
\protect\phantomsection\label{span-of-vectors}
\begin{itemize}
\tightlist
\item
  The span of \(n\) \(m\)-dimensional vectors is the set of all linear
  combinations of the \(n\) vectors: \[
  \Span(\vvv_1, ..., \vvv_n)
  = \{ \mV \cdot \valpha \text{ with } \valpha \in \bbF^n \}
  = \{ \vvv \in \bbF^m: \vvv = \sum_{i=1}^n \alpha_i \vvv_i \}
  \]
\item
  E.g., the span of vectors is a vector space
\end{itemize}
\end{frame}

\begin{frame}{Null space of a matrix}
\protect\phantomsection\label{null-space-of-a-matrix}
\begin{itemize}
\tightlist
\item
  Null space of the columns of a matrix \(\mA\) is defined as the set:
  \[\Null(\mA) = \{ \vvv: \mA \cdot \vvv = \vv{0} \}\]

  \begin{itemize}
  \tightlist
  \item
    In words, all the vectors that are coefficients of linear
    combinations of columns of \(\mA\) yielding the zero vector
  \end{itemize}
\item
  Null space is a vector space
\end{itemize}
\end{frame}

\begin{frame}{Homogeneous linear system associated with null space}
\protect\phantomsection\label{homogeneous-linear-system-associated-with-null-space}
\begin{itemize}
\tightlist
\item
  From the definition of matrix-vector multiplication the vector
  \(\vvv\) is in \(\Null(\mA)\) \(\iff\) \(\vvv\) is a solution of the
  homogeneous linear system involving the columns of \(\mA\): \[
  \begin{aligned}
  &\va^T_1 \cdot \vvv = 0\\
  &\va^T_2 \cdot \vvv = 0\\
  &...\\
  &\va^T_m \cdot \vvv = 0\\
  \end{aligned}
  \]
\item
  Note that the notation is a bit confusing since we mean the transpose
  of the columns \(\va_i\) of \(\mA\) and not the rows of \(\mA\)
\end{itemize}
\end{frame}

\begin{frame}{Dot product on a vector space: definition}
\protect\phantomsection\label{dot-product-on-a-vector-space-definition}
\begin{itemize}
\tightlist
\item
  Given a field of scalars \(\bbF\) and a vector space \(\calV\) over
  \(\bbF\), an inner product is a mapping:
  \[\langle \cdot , \cdot \rangle: \calV \times \calV \rightarrow \bbF\]
  that satisfies the 3 axioms:

  \begin{enumerate}
  \tightlist
  \item
    Conjugate symmetry (almost commutativity):
    \[\langle \vx, \vy \rangle = \overline{\langle \vy, \vx \rangle}\]
  \item
    Linearity in the first argument:
    \[\langle a \vx, \vy \rangle = a \langle \vx, \vy \rangle\] \[
    \langle \vx + \vy, \vz \rangle = \langle \vx, \vz\rangle +
    \langle \vy, \vz \rangle
    \]
  \item
    Positive definitiveness: \[
    \begin{aligned}
    & \langle \vx, \vx \rangle \ge 0 \\
    & \langle \vx, \vx \rangle = 0 \implies \vx = \vv{0} \\
    \end{aligned}
    \]
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Vector inner product}
\protect\phantomsection\label{vector-inner-product}
\begin{itemize}
\tightlist
\item
  Aka ``dot product'', ``scalar product''
\item
  Given \(\vx, \vy \in \bbF^n\) (i.e., same number of components and
  also same ``label'' for each element), the inner product of \(\vx\)
  and \(\vy\) is defined as: \[
  \langle \vx, \vy \rangle
  = \vx^T \cdot \vy
  = \sum_{i=1}^n x_i y_i \in \bbF
  \]
\item
  \(\vx^T \cdot \vy\) is read ``x dotted y'' or ``x transposed y''
\end{itemize}
\end{frame}

\begin{frame}{Affine space: definition}
\protect\phantomsection\label{affine-space-definition}
\begin{itemize}
\tightlist
\item
  If \(\vc\) is a vector and \(\calV\) is a vector space then
  \[\calA = \vc + \calV = \{ \vc + \vvv: \vvv \in \calV\}\] is called an
  affine space
\item
  An affine space is a vector space translated by a point represented by
  a vector

  \begin{itemize}
  \tightlist
  \item
    E.g., a plane or a line that do not contain the origin
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Affine space: example of plane passing through 3 points}
\protect\phantomsection\label{affine-space-example-of-plane-passing-through-3-points}
\begin{itemize}
\tightlist
\item
  Given 3 not collinear vectors: \(\vu_1\), \(\vu_2\), and \(\vu_3\),
  the plane containing the endpoints of the 3 vectors can be represented
  as \(\calA = \vu_1 + \calV\) where
  \(\calV = \Span(\vu_2 - \vu_1, \vu_3 - \vu_1)\)
\item
  \textbf{\textcolor{red}{Note}}: the span of the 3 vectors has
  dimension 3, but an affine space with dimension 2
\end{itemize}
\end{frame}

\begin{frame}{Affine combination: definition}
\protect\phantomsection\label{affine-combination-definition}
\begin{itemize}
\item
  An affine combination is a linear combination of vectors where the sum
  of the (positive or negative) coefficients is 1: \[
  \alpha_1 \vu_1 + \ldots + \alpha_n \vu_n
  \text{ where } \sum_i \alpha_i = 1
  \]
\item
  In matrix form:
  \(\mU \cdot \valpha \text{ where } \vv{1}^T \valpha = 1\)
\end{itemize}
\end{frame}

\begin{frame}{Affine hull of vectors}
\protect\phantomsection\label{affine-hull-of-vectors}
\begin{itemize}
\tightlist
\item
  Given vectors \(\vu_1, ..., \vu_n\), the set of all affine
  combinations is the affine hull: \[
  \calA
  = \{ \vvv = \sum_i^n \alpha_i \vu_i : \sum_i \alpha_i = 1 \}
  = \{ \vvv = \mU \valpha : \vv{1}^T \valpha = 1 \}
  \]
\item
  The affine hull includes each point because if \(\valpha\) has a
  single 1 in position \(i\) and all others 0, we get \(\vu_i\)
\end{itemize}
\end{frame}

\begin{frame}{Affine hull of vectors is an affine space}
\protect\phantomsection\label{affine-hull-of-vectors-is-an-affine-space}
\begin{itemize}
\tightlist
\item
  We can write the affine hull of \(\vu_1, ..., \vu_n\) as an affine
  space: \[\vu_i + \Span(\vu_1 - \vu_i, ..., \vu_n - \vu_i)\]

  \begin{itemize}
  \tightlist
  \item
    Thus an affine space is an affine hull and vice versa
  \end{itemize}
\item
  This is the dual of ``the span of vectors is a vector space''
\end{itemize}
\end{frame}

\begin{frame}{The solution set of non-homogeneous linear system is empty
or affine space}
\protect\phantomsection\label{the-solution-set-of-non-homogeneous-linear-system-is-empty-or-affine-space}
\begin{itemize}
\item
  Consider the solution of a system of non-homogeneous linear equations
  \[
  \{ \vx : \va_1^T \vx = \beta_1, ..., \va_m^T \vx = \beta_m \}
  \text{ or in matrix form }
  \mA \cdot \vx = \vbeta
  \]
\item
  The solution set is either empty or an affine space
\item
  There are two cases: either the non-homogeneous system has solution,
  or not
\item
  Consider the case where the system of equations has no solutions
  (e.g., it is contradictory, e.g., \(x = 1, x = 2\)), then the solution
  set is empty
\item
  Consider the case where there is a solution
\item
  Each linear system \(\mA \vx = \vbeta\) has an associated homogeneous
  linear system \(\mA \vx = \vv{0}\)
\item
  If \(\vu_1\) is a solution of the non-homogeneous system (i.e.,
  \(\mA \vu_1 = \vbeta\)), then any other solution \(\vu_2\) is a
  solution (i.e., \(\mA \vu_2 = \vbeta\)) \(\iff\) \(\vu_2 - \vu_1\) is
  in the vector space which is the solution of the homogeneous linear
  system (i.e., \(\mA (\vu_1 - \vu_2) = \vv{0}\))
\item
  So if the system has at least one solution, we can write the set of
  solutions as \(\vu + \calV\) where \(\calV\) is the vector space
  defined by the homogeneous linear system
\item
  A corollary is that a system of linear equations has only one solution
  if (but not iff) the homogeneous linear system has only the zero
  vector as solution
\end{itemize}
\end{frame}

\begin{frame}{Vector space vs affine space: summary}
\protect\phantomsection\label{vector-space-vs-affine-space-summary}
\begin{itemize}
\tightlist
\item
  Linear combination vs affine combination
\item
  Span of vectors vs affine hull of vectors

  \begin{itemize}
  \tightlist
  \item
    Span (affine hull) is the set of all linear (affine) combinations
  \end{itemize}
\item
  Vector space vs affine space
\end{itemize}
\end{frame}

\begin{frame}{Matrix}
\protect\phantomsection\label{matrix}
\begin{itemize}
\item
  A matrix \(\mA \in \bbF^{m \times n}\) is a two dimensional array with
  dimensions \(m \times n\) of elements from a field \(\bbF\)
\item
  Matrix notation

  \begin{itemize}
  \tightlist
  \item
    \(\mA \in \bbR^{m \times n}\) has \(m\) rows and \(n\) columns
  \item
    \(A_{ij}\) is the element on \(i\)-th row and \(j\)-th column \[
    \mA =
    \begin{pmatrix}
    A_{1, 1} & ...      & A_{1, n} \\
    ...      & A_{i,j}  & ... \\
    A_{m, 1} & ...      & A_{m, n} \\
    \end{pmatrix}
    \]
  \item
    The convention is first rows and then columns (i.e., y-x instead of
    the more usual x-y) for both elements and dimensions of a matrix
  \end{itemize}
\item
  Notation for rows and columns in a matrix

  \begin{itemize}
  \tightlist
  \item
    \(j\)-th column is \(\va_j\) or \(\va_{:,j}\) (using
    \textcolor{blue}{\texttt{numpy}} notation)
  \item
    \(i\)-th row is \(\va^T_i\) or \(\va_{i,:}\)
  \item
    Fixing a coordinate (e.g., row) one gets the orthogonal indices
    (e.g., column)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear functions over vector spaces: definition}
\protect\phantomsection\label{linear-functions-over-vector-spaces-definition}
\begin{itemize}
\tightlist
\item
  Consider two vector spaces \(\calV\) and \(\calW\) over the same field
  \(\bbF\)
\item
  A linear function \(f: \calV \rightarrow \calW\) satisfies two
  properties:

  \begin{enumerate}
  \tightlist
  \item
    \(f(\alpha \vvv) = \alpha f(\vvv)\)
  \item
    \(f(\vu + \vvv) = f(\vu) + f(\vvv)\)
  \end{enumerate}
\item
  Linear functions ``push linear combination through'': \[
  f(\alpha_1 \vvv_1 + ... + \alpha_n \vvv_n)
  = \alpha_1 f(\vvv_1) + ... + \alpha_n f(\vvv_n)
  \]

  \begin{itemize}
  \tightlist
  \item
    Equivalent to the 2 properties of linear functions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Matrix and linear function}
\protect\phantomsection\label{matrix-and-linear-function}
\begin{itemize}
\tightlist
\item
  From matrix to linear function

  \begin{itemize}
  \tightlist
  \item
    Given a matrix \(\mA \in \bbF^{n \times m}\) we can define the
    function: \[f(\vx) = \mA \cdot \vx\]
  \item
    The function \(f()\) maps \(m\)-vectors into \(n\)-vectors

    \begin{itemize}
    \tightlist
    \item
      The domain is \(\bbF^m\) for the matrix-vector product to be
      defined
    \item
      The co-domain is \(\bbF^n\)
    \end{itemize}
  \item
    \(f(\vx)\) is a linear function because of the properties of
    matrix-vector product
  \end{itemize}
\item
  From linear function to matrix

  \begin{itemize}
  \item
    Consider a linear function \(f: \bbF^m \rightarrow \bbF^n\)
  \item
    We want to find a matrix \(\mA\) such that
    \(f(\vx) = \mA \cdot \vx\)
  \item
    \textbf{\emph{Solution}}

    \begin{itemize}
    \tightlist
    \item
      We know that \(\mA \in \bbF^{n \times m}\) from matrix-vector
      product definition
    \item
      If we compute \(\mA \cdot \vv{e_i}\), where
      \(e_i = (0, ..., 0, 1, ...0)\) is the \(i\)-th standard generator,
      we obtain \(\va_i\) (i.e., the \(i\)-th column of \(\mA\))
    \item
      Thus \(\mA\) is the matrix with columns equal to the standard
      generators transformed by \(f()\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear functions: examples and non-examples}
\protect\phantomsection\label{linear-functions-examples-and-non-examples}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Identity function}} is linear

  \begin{itemize}
  \tightlist
  \item
    Corresponds to the identity matrix
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Rotation}} is linear transformations

  \begin{itemize}
  \tightlist
  \item
    Corresponds to an orthonormal matrix
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Scaling}} each coordinate independently is
  linear transformation

  \begin{itemize}
  \tightlist
  \item
    Corresponds to a diagonal matrix
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Translation}} is \emph{not} a linear
  function

  \begin{itemize}
  \tightlist
  \item
    Since it does not satisfy either of the two linearity properties

    \begin{enumerate}
    \tightlist
    \item
      \(f(\alpha \vvv) = \alpha f(\vvv)\)
    \item
      \(f(\vu + \vvv) = f(\vu) + f(\vvv)\)
    \end{enumerate}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Kernel of a Linear Function}
\protect\phantomsection\label{kernel-of-a-linear-function}
\begin{itemize}
\item
  Any linear function \(f: \calV \rightarrow \calW\) maps the zero
  vector of \(\calV\) to the zero vector of \(\calW\)
\item
  The \textbf{\textcolor{red}{kernel}} of a linear function \(f\) is the
  set of vectors that are transformed by \(f\) into the \(\vv{0}\)
  vector \[\Ker(f) = \{ \vvv: f(\vvv) = \vv{0} \}\]
\item
  If linear function \(f\) is expressed in matrix form
  \(f(\vx) = \mA \cdot \vx\) then its kernel is the null space of the
  columns of the associated matrix \(\mA\) \[\Ker(f) = \Null(\mA)\]
\end{itemize}
\end{frame}

\begin{frame}{Domain, Image, Co-domain of a Function}
\protect\phantomsection\label{domain-image-co-domain-of-a-function}
\begin{itemize}
\item
  Consider a (linear or not) function \(f: \calV \rightarrow \calW\)
\item
  The \textbf{\textcolor{red}{domain}} of \(f\) \(\calV\) is the set of
  all values where the function is defined
\item
  The \textbf{\textcolor{darkgreen}{image of domain}} \(f(\calV)\) is
  the set of all values that the function can assume
\item
  The \textbf{\textcolor{blue}{co-domain}} \(\calW\) is the set where
  the function assumes its value

  \begin{itemize}
  \tightlist
  \item
    E.g., \(\bbR^2\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{One-to-one Function}
\protect\phantomsection\label{one-to-one-function}
\begin{itemize}
\item
  Consider a function \(f: \calV \rightarrow \calW\)
\item
  A function is \textbf{\textcolor{red}{one-to-one}} (or injective) iff
  any two different elements \(v_1 \ne v_2 \in \calV\) have different
  images \(f(v_1) \ne f(v_2)\)
\item
  Equivalently:

  \begin{itemize}
  \tightlist
  \item
    Using contrapositive: if two elements \(v_1\) and \(v_2\) have the
    same image \(f(v_1) = f(v_2)\), then they are equal \(v_1 = v_2\)
  \item
    In terms of set cardinality: \(|f(\calV)| = |\calV|\), i.e., the
    image of the domain has the same number of elements as the domain
  \end{itemize}
\item
  A \emph{linear} function is one-to-one iff its kernel is the trivial
  vector space, \(\Ker(f) = \{ \vv{0} \}\)

  \begin{itemize}
  \tightlist
  \item
    Equivalently the associated matrix has \(\Null(\mA) = \{ \vv{0} \}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Onto function}
\protect\phantomsection\label{onto-function}
\begin{itemize}
\item
  Consider a function \(f: \calV \rightarrow \calW\)
\item
  A function is \textbf{\textcolor{red}{onto}} (or surjective) iff for
  any element of its co-domain \(w \in \calW\), there exists an element
  of the domain \(v \in \calV\) that is transformed into it, i.e.,
  \(f(v) = w\)
\item
  Equivalently in terms of set cardinality:

  \begin{itemize}
  \tightlist
  \item
    \(f(\calV) = \calW\), i.e., the image of the domain is equal to the
    co-domain
  \end{itemize}
\item
  Any function can be made surjective by restricting \(\calW\) to
  \(f(\calV)\)
\end{itemize}
\end{frame}

\begin{frame}{Invertible function}
\protect\phantomsection\label{invertible-function}
\begin{itemize}
\item
  A function \(f: \calV \rightarrow \calW\) is invertible iff it is both
  one-to-one (injective) and onto (surjective), i.e.,
  \[\forall w \in W \quad \exists! v \in V: f(v) = w\]
\item
  Equivalently in terms of set cardinality:

  \begin{itemize}
  \tightlist
  \item
    \(|\calV| = |\calW|\), i.e., the co-domain and the domain have the
    same number of elements
  \end{itemize}
\item
  Consider an invertible function \(f: \calV \rightarrow \calW\), the
  inverse of \(f\) is:

  \begin{itemize}
  \tightlist
  \item
    \(f^{-1}: \calW \rightarrow \calV\)
  \item
    \(f \circ f^{-1}\) is the identity function
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear function composition in matrix terms}
\protect\phantomsection\label{linear-function-composition-in-matrix-terms}
\begin{itemize}
\item
  There is a correspondence between linear functions and matrices
\item
  Consider two matrices \(\mA\) and \(\mB\) and the two associated
  functions:

  \begin{itemize}
  \tightlist
  \item
    \(f(\vy) = \mA \cdot \vy\)
  \item
    \(g(\vx) = \mB \cdot \vx\)
  \end{itemize}
\item
  The composed function is defined as:
  \[h(\vx) = (f \circ g)(\vx) = f(g(\vx))\]
\item
  It can be shown that the associated matrix to the composed function is
  \[h(\vx) = \mA \cdot \mB \cdot \vx\]
\end{itemize}
\end{frame}

\begin{frame}{Matrix inverse}
\protect\phantomsection\label{matrix-inverse}
\begin{itemize}
\item
  Using the definition of inverse functions, two square matrices \(\mA\)
  and \(\mB\) are inverses iff \(\mA \cdot \mB = \mI\)

  \begin{itemize}
  \tightlist
  \item
    We indicate the (unique) inverse of \(\mA\) with \(\mA^{-1}\)
  \end{itemize}
\item
  Given an invertible (square) matrix \(\mA\), then the associated
  function \(f(\vx) = \mA \cdot \vx\) is an invertible function, i.e.,
  \(f()\) is one-to-one and onto
\item
  The matrix-vector equation \(\mA \cdot \vx = \vb\) has one and only
  one solution \(\vx\) for any \(\vb\), i.e., \(\mA^{-1} \vb\)
\item
  If \(\mA\) and \(\mB\) are invertible and can be multiplied, then
  \(\mA \cdot \mB\) is invertible
\end{itemize}
\end{frame}

\begin{frame}{Coordinate representation of a vector}
\protect\phantomsection\label{coordinate-representation-of-a-vector}
\begin{itemize}
\item
  Consider:

  \begin{itemize}
  \tightlist
  \item
    \(\calV\) a vector space

    \begin{itemize}
    \tightlist
    \item
      \(\calV\) is not necessarily like \(\bbF^n\), i.e., the vector
      space does not need to be a ``numeric'' vector representation
    \end{itemize}
  \item
    Vectors \(\va_1, ..., \va_n \in \calV\)
  \item
    \(\vvv \in \Span(\va_1, ..., \va_n) \in \calV\)
  \end{itemize}
\item
  You can represent \(\vvv\) in terms of \(\va_1, ..., \va_n\) using the
  coordinates \(\vu = (\alpha_1, ..., \alpha_n)\) such that:
  \[\vvv = \sum_{i=1}^n \alpha_i \va_i\] or in terms of the matrix
  \(\mA\) that has \(\va_i\) as columns: \(\vvv = \mA \cdot \vu\)
\item
  A generator set allows to represent any vector in terms of coordinates
  (although not unique)
\item
  To find the coordinates \(\vu\) of a vector \(\vvv\) with respect to a
  generator \(\mA\), solve the matrix-vector equation for \(\vu\):
  \[\mA \cdot \vu = \vvv\]
\end{itemize}
\end{frame}

\begin{frame}{Linear dependence between vectors: definition}
\protect\phantomsection\label{linear-dependence-between-vectors-definition}
\begin{itemize}
\item
  A set \(S\) of \(n\) vectors \(\vvv_1, ..., \vvv_n\) are
  \textbf{\textcolor{red}{linearly dependent}} \emph{iff} \[
  \exists (\alpha_1, ..., \alpha_n) \ne \vv{0}_n:
  \alpha_1 \vvv_1 + ... + \alpha_n \vvv_n = \vv{0}_k
  \]
\item
  In words, the zero vector can be written as a \emph{non-trivial}
  linear combination of the vectors
\item
  Equivalently: at least one vector of the set \(S\) can be expressed as
  linear combination of the remaining using coefficients not all zero
\item
  \textbf{\textcolor{blue}{Remarks}}

  \begin{itemize}
  \tightlist
  \item
    One of the vectors \(\vvv_i\) can be the zero vector
  \item
    The notion of linear dependence / independence applies to \emph{set
    of vectors} and not to a single vector

    \begin{itemize}
    \tightlist
    \item
      The zero vector \(\vv{0}\) is not ``linearly dependent'', rather
      the set containing only the zero vector \(\{ \vv{0} \}\) is
      linearly dependent
    \end{itemize}
  \item
    A \emph{trivial} linear combination of the vectors
    \(\vx_1, \vx_2, ..., \vx_n\) is a combination of vectors with
    coefficients all equal to zero

    \begin{itemize}
    \tightlist
    \item
      It is always equal to the zero vector \(\vv{0}\)
    \end{itemize}
  \end{itemize}
\item
  E.g., \(S = \{(1, 0, 0), (0, 2, 0), (2, 4, 0)\}\) are linearly
  dependent since
  \[2 \cdot (1, 0, 0) + 2 \cdot (0, 2, 0) -1 \cdot (2, 4, 0) = (0, 0, 0)\]
\end{itemize}
\end{frame}

\begin{frame}{Linear independence: definition}
\protect\phantomsection\label{linear-independence-definition}
\begin{itemize}
\tightlist
\item
  Linear independence is the negation of the definition of linear
  dependence:
  \[\neg (\exists \valpha \ne \vv{0}: \sum \alpha_i \vvv_i = \vv{0})\]

  \begin{itemize}
  \tightlist
  \item
    By negating the existential quantifier:
    \(\neg (\exists x: P(x)) \iff \forall x: \neg P(x)\)
    \[\forall \valpha \ne \vv{0} \quad \sum \alpha_i \vvv_i \ne \vv{0}\]
  \end{itemize}
\item
  Equivalently by contrapositive
  \((P \implies Q) \iff (\neg Q \implies \neg P)\)
  \[\sum \alpha_i \vvv_i = \vv{0} \implies \valpha = \vv{0}\]

  \begin{itemize}
  \tightlist
  \item
    In words, the only combination of the vectors that gives the zero
    vector is the trivial linear combination
  \end{itemize}
\item
  E.g., \((1, 0, 0), (0, 2, 0), (0, 0, 3)\) are linearly independent
  since: \begin{alignat*}{2}
  & \alpha_1 \cdot (1, 0, 0) + \alpha_2 \cdot (0, 2, 0) +
  \alpha_3 \cdot (0, 0, 3) \\
  & = (\alpha_1, 2 \alpha_2, 3 \alpha_3) \\
  & = (0, 0, 0) \\
  & \iff \alpha_1 = \alpha_2 = \alpha_3 = 0
  \end{alignat*}
\end{itemize}
\end{frame}

\begin{frame}{Property of linear dependence / independence}
\protect\phantomsection\label{property-of-linear-dependence-independence}
\begin{itemize}
\item
  Any set of vectors that contains the \(\vv{0}\) is linearly dependent

  \begin{itemize}
  \tightlist
  \item
    In fact we can make a non-trivial combination of the zero setting
    the coefficient of the zero vector non zero and all remaining zero
  \item
    Thus the set \(\{ \vv{0} \}\) is not linearly independent
  \end{itemize}
\item
  A subset of linearly independent vectors is still independent
\item
  A superset of linearly dependent vectors is still dependent

  \begin{itemize}
  \tightlist
  \item
    This is the contrapositive of the previous proposition
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear dependence and null space}
\protect\phantomsection\label{linear-dependence-and-null-space}
\begin{itemize}
\item
  By definition linear dependence of vectors \(\vvv_1, ..., \vvv_n\)
  means that there is a non-trivial combination of those vectors that is
  equal to zero vector
\item
  If \(\mV\) is the matrix with \(\vvv_i\) as columns, linear dependence
  is written: \[\exists \vx \ne \vv{0}: \mV \cdot \vx = \vv{0}\]
\item
  In matrix form \[\Null(\mV) \ne \{ \vv{0} \}\] i.e., the null space of
  columns of \(\mV\) is not trivial
\item
  Linear independence is equivalent to \(\Null(\mV) = \{ \vv{0} \}\)
  i.e., the null space of the columns of \(\mV\) is the trivial space
\end{itemize}
\end{frame}

\begin{frame}{Linear one-to-one function and linear independence}
\protect\phantomsection\label{linear-one-to-one-function-and-linear-independence}
\begin{itemize}
\item
  A linear one-to-one function \(f\) preserve linear independence
\item
  It can be proven by showing that a linear combination of transformed
  independent vectors is \(\vv{0}\) only if it is a trivial combination:
  \begin{alignat*}{1}
  & \vx_i \text{ independents} \\
  & \sum \alpha_i f(\vx_i) = \vv{0} \implies \valpha = \vv{0} \\
  \end{alignat*}
\end{itemize}
\end{frame}

\begin{frame}{Superfluous-vector lemma}
\protect\phantomsection\label{superfluous-vector-lemma}
\begin{itemize}
\item
  Given a set of vectors \(S\)
\item
  We already know that vectors are linearly dependent \emph{iff} a
  vector \(\vvv \in S\) is a non-trivial linear combination of the
  remaining vectors
\item
  This is equivalent to:
  \[\exists \vvv \in S : \Span(S) = \Span(S - \{\vvv\})\]

  \begin{itemize}
  \tightlist
  \item
    In words, one can always remove at least a vector dependent from a
    generator set without changing its span
  \end{itemize}
\item
  \textbf{\textcolor{red}{Intuition}}: it establishes a relationship
  between linear dependence and span
\item
  \textbf{\textcolor{blue}{Corollary}}:

  \begin{itemize}
  \tightlist
  \item
    You can always \emph{add} or \emph{remove} linearly dependent
    vectors from a set, without changing its span
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Basis of a vector space: definition}
\protect\phantomsection\label{basis-of-a-vector-space-definition}
\begin{itemize}
\item
  A set of vectors \(B\) is a \textbf{\textcolor{red}{basis}} of the
  vector space \(\calV\) iff by definition:

  \begin{itemize}
  \tightlist
  \item
    \(B\) is a set of generators for \(\calV\), i.e.,
    \(\Span(B) = \calV\)
  \item
    Vectors in \(B\) are linearly independent
  \end{itemize}
\item
  In words, a basis is a set of linearly independent generators
\item
  Often you care about the order of the vectors

  \begin{itemize}
  \tightlist
  \item
    You should use \(B\) as a tuple or a matrix, and not a set of
    vectors
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Unique-representation lemma}
\protect\phantomsection\label{unique-representation-lemma}
\begin{itemize}
\item
  Consider:

  \begin{itemize}
  \tightlist
  \item
    \(\calV\) is a vector space over a scalar field \(\bbF\)
  \item
    \(B = (\vb_1, ..., \vb_n)\) be a basis for \(\calV\)
  \end{itemize}
\item
  For any vector \(\vvv \in \calV\) there is exactly one representation
  of \(\vvv\) in terms of the basis vectors
\item
  \textbf{\textcolor{red}{Intuition}}: a vector space can always be
  represented in terms of coordinate vectors \(\bbF^n\) using a basis
\item
  There is a function \(f: \bbF^n \rightarrow \calV\) that:

  \begin{itemize}
  \tightlist
  \item
    transforms the coordinates \(\vx\) with respect to a basis \(\mB\)
    into a vector \(\vvv = \mB \cdot \vx \in \calV\)
  \item
    is one-to-one (because of linear independence of basis vectors and
    linearity of function); and
  \item
    is onto (because a generator)
  \end{itemize}
\item
  There is also a function \(g: \calV \rightarrow \bbF^n\) that

  \begin{itemize}
  \tightlist
  \item
    Computes the (unique) coordinates of \(\vvv \in \calV\) with respect
    to the basis \(B\)
  \item
    \(f\) and \(g\) are inverse of each other
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Change of basis transformation}
\protect\phantomsection\label{change-of-basis-transformation}
\begin{itemize}
\tightlist
\item
  Consider two bases for the same vector space \(\calV\)

  \begin{itemize}
  \tightlist
  \item
    \(\va_1, ..., \va_n\)
  \item
    \(\vc_1, ..., \vc_k\)
  \item
    We assume \(n \ne k\) (we don't know yet that all bases of a vector
    space must have the same cardinality)
  \end{itemize}
\item
  We know that:

  \begin{itemize}
  \tightlist
  \item
    The matrix \(\mA\) with \(\va_i\) as columns transforms the
    coordinates with respect to to \(\va_i\) into a vector in \(\calV\)
  \item
    The matrix \(\mC^{-1}\) transforms a vector in \(\calV\) into its
    coordinates with respect to to \(\vc_i\)
  \end{itemize}
\item
  We can compose the 2 transformations to get a transformation from the
  coordinate space \(\va\) into the coordinate space \(\vc\)
  \[\vc = \mC^{-1} \cdot \mA \cdot \va\]

  \begin{itemize}
  \tightlist
  \item
    This is called ``a change of basis transformation''
  \item
    The entire transformation is also invertible, i.e., we can go from
    coordinates of \(\vc\) to \(\va\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Simplified exchange lemma}
\protect\phantomsection\label{simplified-exchange-lemma}
\begin{itemize}
\tightlist
\item
  Consider:

  \begin{itemize}
  \tightlist
  \item
    A set \(S\) of vectors, not necessarily linearly independent
  \item
    \(\vz \ne \vv{0} \in \Span(S)\)
  \end{itemize}
\item
  There is a vector \(\vw \in S\) such that:
  \[\Span(S \cup \{\vz\} - \{\vw\}) = \Span(S)\]

  \begin{itemize}
  \tightlist
  \item
    In words, one can exchange a vector from \(S\) with a vector from
    its span without changing the span
  \end{itemize}
\item
  Example

  \begin{itemize}
  \tightlist
  \item
    \(S = \{\vvv_1 = (1, 0, 0), \vvv_2 = (0, 1, 0), \vvv_3 = (1, 2, 0)\}\)
    and \(\vz = (-1, -1, 0)\)
  \item
    Note that \(\vz = -1 \cdot \vvv_1 - 1 \cdot \vvv_2\)
  \item
    WLOG we can express \(\vvv_1\) in terms of \(\vz, \vvv_2\)
  \item
    Thus we can exchange \(\vz\) with \(\vvv_1\) without changing the
    span of \(S\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Exchange lemma}
\protect\phantomsection\label{exchange-lemma}
\begin{itemize}
\tightlist
\item
  Consider

  \begin{itemize}
  \tightlist
  \item
    A set \(S\) of vectors
  \item
    A subset of \emph{linearly independent} \(A \subseteq S\) (aka
    ``protected subset'' of \(S\))
  \item
    Pick \(\vz \in \Span(S)\) such that \(A \cup \{\vz\}\) is linearly
    independent (i.e., \(\vz \not \in \Span(A)\))
  \end{itemize}
\item
  There is a vector \(\vw \in S - A\) such that:
  \[\Span(S \cup \{\vz\} - \{\vw\}) = \Span(S)\]
\item
  In words, we can exchange \(\vz\) with a vector of the set, without
  changing the span and the protected subset
\end{itemize}
\end{frame}

\begin{frame}{Morphing lemma}
\protect\phantomsection\label{morphing-lemma}
\begin{itemize}
\item
  Consider

  \begin{itemize}
  \tightlist
  \item
    A vector space \(\calV\)
  \item
    \(G\) be a set of generators for \(\calV\)
  \item
    \(B\) be a basis for \(\calV\)
  \end{itemize}
\item
  \(|B| \le |G|\), i.e., the cardinality of a basis is always smaller
  than the cardinality of a generator
\item
  It is called morphing lemma since you can morph \(G\) into \(B\)
  without changing the span
\end{itemize}
\end{frame}

\begin{frame}{A basis as smallest generator set}
\protect\phantomsection\label{a-basis-as-smallest-generator-set}
\begin{itemize}
\item
  Using the morphing lemma, a basis for \(\calV\) is the smallest
  generating set for \(\calV\)
\item
  Subset-basis lemma: any \emph{finite} generator set \(S\) includes a
  basis \(B\) for \(\calV = \Span(S)\)
\item
  All bases for \(\calV\) have the same size
\end{itemize}
\end{frame}

\begin{frame}{Dimension of vector space}
\protect\phantomsection\label{dimension-of-vector-space}
\begin{itemize}
\tightlist
\item
  The dimension of a vector space \(\calV\), written \(\Dim(\calV)\), is
  the size of a (any) basis for \(\calV\)
\item
  By definition of basis, \(\Dim(\calV)\) is the (exact) number of
  linearly independent vectors generating a vector space
\end{itemize}
\end{frame}

\begin{frame}{Rank of a set S of vectors / matrix}
\protect\phantomsection\label{rank-of-a-set-s-of-vectors-matrix}
\begin{itemize}
\item
  Rank is the dimension of the space spanned by a set:
  \[\Rank(S) \defeq \Dim(\Span(S))\]
\item
  By definition of basis, the rank of a set of vectors \(S\) is the
  number of linearly independent vectors included in \(S\)
\item
  MEM: \(\Rank\) and \(\Dim\) measure the same thing, but for vectors
  and vector spaces
\end{itemize}
\end{frame}

\begin{frame}{Rank and dimension: example}
\protect\phantomsection\label{rank-and-dimension-example}

  dependent, thus \(\Rank(S)\) is \(< 3\)
\item
  Actually \(\Rank(S) = 2\) since the first 2 vectors are linearly

\begin{frame}{Nullity of a set of vectors / matrix}
\protect\phantomsection\label{nullity-of-a-set-of-vectors-matrix}
\begin{itemize}
\tightlist
\item

\begin{frame}{Dimension of trivial vector space}
\protect\phantomsection\label{dimension-of-trivial-vector-space}
\begin{itemize}
\item
  Consider the vector space \(\calV = \Span(\{ \vv{0} \})\)
\item
  One could think that the dimension is 1 since it is spanned by 1
  vector
\item
  This is not possible since the set with the zero vector is not

\item
  In reality \(\calV\) is spanned by the empty set, thus

\begin{frame}{Subspace dimension lemma}
\protect\phantomsection\label{subspace-dimension-lemma}
\begin{itemize}
\item
  \(\calU\) is a vector subspace of \(\calW\)
\item
  \textbf{\emph{Thesis}} \[\Dim(\calU) \le \Dim(\calW)\] and
  \[\Dim(\calU) = \Dim(\calW) \iff \calU = \calW\]
\item
  \textbf{\emph{Proof}}
\item
  One can use a modified version of the exchange lemma to have basis of
  \(\calW\) include a basis of \(\calU\)
\end{itemize}
\end{frame}

\begin{frame}{Subspace dimension lemma: example}

\item

\item
  Since the vectors are independent then \(\Dim(V) = 2\) and

\begin{frame}{Matrix rank theorem: lemma}
\protect\phantomsection\label{matrix-rank-theorem-lemma}
\begin{itemize}
\tightlist

  For any matrix \(A\) \[\Rank(\Row(\mA)) \le \Rank(\Col(\mA))\]
\end{itemize}
\end{frame}

\begin{frame}{Matrix rank theorem: lemma proof}
\protect\phantomsection\label{matrix-rank-theorem-lemma-proof}
\begin{itemize}
\item
  Assume that \(\mA\) is \(m \times n\)
\item
  Consider the column space \(\mA = (\va_1 | \va_2 | ... | \va_n)\)
  which is generated by \(n\) vectors with \(m\) components
\item
  This space has a dimension \(k = \Rank(\Col(\mA)) \le n\) and let's
  call the basis \(\mB\)
\item
  So each
  \(\va_i = \alpha_1 \vb_1 + ... + \alpha_k \vb_k = \mB \cdot \valpha\)
  where the \(\vb_j\) has \(m\) components
\item
  If we write all vectors \(\va_i = \mA \valpha_i\) and use the
  matrix-matrix product in terms of matrix-vector product we get
  \(\mA = \mB \mC\) (\(m \times n = (m \times k) \cdot (k \times n)\))
\item
  If we transpose we have:
  \[\mA^T = \mC^T \cdot \mB^T = (n \times k) \cdot (k \times m)\] so we
  can express the rows of \(\mA^T\) (i.e., the columns of \(\mA\)) as a
  linear combination of the \(k\) rows of \(\mB^T\) (vector-matrix
  product)
\item
  So the \(k\) rows of \(\mB^T\) are a generator for rows of \(\mA\)
\item
  They contain a basis for \(\Row(\mA)\) because of basis generator
  lemma: \(\Rank(\Row(\mA)) \le k = \Rank(\Col(\mA))\)
\end{itemize}
\end{frame}

\begin{frame}{Matrix rank theorem}
\protect\phantomsection\label{matrix-rank-theorem}
\begin{itemize}
\tightlist
\item
  For every matrix \(\mA\), row rank equals column rank:
  \[\Rank(\Row(\mA)) = \Rank(\Col(\mA))\]
\end{itemize}
\end{frame}

\begin{frame}{Matrix rank theorem: proof}
\protect\phantomsection\label{matrix-rank-theorem-proof}
\begin{itemize}
\item
  Penultimate step: to prove \(a = b\) we can prove that \(a \le b\) and
  the converse \(b \le a\)
\item
  We can always exchange rows and columns of a matrix by transposing
\item
  So we get from the lemma:
  \[\Rank(\Row(\mA^T)) \le \Rank(\Col(\mA^T))\] but
  \(\Row(\mA^T) = \Col(\mA)\) so we have
  \[\Rank(\Col(\mA)) \le \Rank(\Row(\mA))\] and we can use the
  penultimate step to reach the thesis
\end{itemize}
\end{frame}

\begin{frame}{Direct sum of vector spaces: definition}
\protect\phantomsection\label{direct-sum-of-vector-spaces-definition}
\begin{itemize}
\tightlist
\item
  Given two vector spaces \(\calU\) and \(\calV\), subsets of the same
  vector space \(\calW\) \[\calU, \calV \subseteq \calW\]
\item
  Assume \(\calU \cap \calV = \{ \vv{0} \}\)
\item
  We define direct sum of \(\calU\) and \(\calV\): \[
  \calU \oplus \calV
  = \{ \vu + \vvv: \vu \in \calU, \vvv \in \calV \}
  \]
\item
  MEM: It's like a linear combination of vectors from the two
  sub-spaces, the coefficients \(\alpha, \beta\) are provided by the
  underlying linearity of vector space
\end{itemize}
\end{frame}

\begin{frame}{Direct sum: geometric interpretation}
\protect\phantomsection\label{direct-sum-geometric-interpretation}
\begin{itemize}
\tightlist
\item
  \(\calU, \calV \subseteq \bbR^n\) are spanned by two independent
  vectors (i.e., two non-parallel lines)
\item
  The direct sum \(\calU \oplus \calV\) is the plane containing both
  lines
\end{itemize}
\end{frame}

\begin{frame}{Direct sum: properties}
\protect\phantomsection\label{direct-sum-properties}
\begin{itemize}
\tightlist
\item
  \(\calU \oplus \calV\) is a vector space (by def closed with respect
  to + and *)
\item
  The union of two generators (resp. bases) for \(\calU\) and for
  \(\calV\) is a generator (resp. basis) for \(\calU \oplus \calV\)
\item
  Each vector has a unique representation as sum of a vector from
  \(\calU\) and \(\calV\) (by unique representation lemma)
\item
  The dimension of \(\calU \oplus \calV\) is the sum of the dimensions
\end{itemize}
\end{frame}

\begin{frame}{Direct sum: example}
\protect\phantomsection\label{direct-sum-example}
\begin{itemize}
\tightlist
\item
  \(\calU = \Span(1000, 0100)\) and \(\calV = \Span(0010)\) over GF(2)
\item
  The vectors have no intersection besides \(\vv{0}\) since their bases
  have no intersection
\item
  \(\calU \oplus \calV
    = \{0000, 1000, 0100, 0010, 1100, 1010, 0110, 1110\}
    = \Span(1000, 0100, 0010)\)
\end{itemize}
\end{frame}

\begin{frame}{Complementary subspace}
\protect\phantomsection\label{complementary-subspace}
\begin{itemize}
\tightlist
\item
  Two spaces \(\calU, \calV \subseteq \calW\) are complementary
  subspaces of \(\calW\) \(\iff\) their direct sum is
  \(\calU \oplus \calV = \calW\)
\item
  Thus, any vector in \(\calW\) can be written as the (unique) sum of a
  vector of \(\calU\) and a vector of \(\calV\)
\end{itemize}
\end{frame}

\begin{frame}{Existence of complementary subspaces}
\protect\phantomsection\label{existence-of-complementary-subspaces}
\begin{itemize}
\tightlist
\item
  Given a space \(\calW\) and its subspace \(\calU \subseteq \calW\),
  exists one (unique) subspace \(\calV\) such that \(\calU, \calV\) are
  complementary subspaces
\item
  It can be proved using basis of \(\calU\) and \(\calV\)
\end{itemize}
\end{frame}
