\begin{frame}{The Illusion of Understanding}
\protect\phantomsection\label{the-illusion-of-understanding}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Prediction vs. Explanation}}

  \begin{itemize}
  \tightlist
  \item
    Models find patterns in historical data
  \item
    Can't explain mechanisms or answer ``what if'' scenarios
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{High accuracy $\neq$ Understanding causation}}

  \begin{itemize}
  \tightlist
  \item
    A model with 95\% churn prediction accuracy won't tell you what
    \emph{causes} churn
  \item
    Accurate sales forecasts can't predict effects of \emph{pricing
    changes}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{The Real Problem}}

  \begin{itemize}
  \tightlist
  \item
    Without causation, you react to past events
  \item
    You can't proactively \emph{change outcomes}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Association, Correlation, and Causation}
\protect\phantomsection\label{association-correlation-and-causation}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Association}}

  \begin{itemize}
  \tightlist
  \item
    Variables co-occur or are statistically dependent
  \item
    E.g., \emph{``People who buy diapers also buy beer''}
  \item
    No info on direction or mechanism
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.2.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Correlation}}

  \begin{itemize}
  \tightlist
  \item
    Linear relationship between variables
  \item
    E.g., \emph{``Ice cream sales and drowning deaths correlate''} (both
    seasonal)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Causation}}

  \begin{itemize}
  \tightlist
  \item
    Changing one variable \emph{directly} changes another
  \item
    E.g., \emph{``Asbestos causes cancer''}
  \item
    Requires causal model or experiment
  \end{itemize}
\item
  \textbf{\textcolor{violet}{The illusion}}: high correlation
  \emph{feels like} understanding

  \begin{itemize}
  \tightlist
  \item
    Real understanding means knowing what happens when you
    \emph{intervene}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What ML Systems Can Tell You}
\protect\phantomsection\label{what-ml-systems-can-tell-you}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pattern recognition}}

  \begin{itemize}
  \tightlist
  \item
    Find correlations in data
  \item
    Identify similar items (e.g., through clustering, classification)
  \item
    Forecast based on past trends
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Predictive accuracy on held-out test data}}

  \begin{itemize}
  \tightlist
  \item
    Accurate prediction if conditions remain stable

    \begin{itemize}
    \tightlist
    \item
      Only works when training and future data match
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Feature importance}}

  \begin{itemize}
  \tightlist
  \item
    Find variables with strongest statistical associations
  \item
    Doesn't distinguish correlation from causation
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What ML Systems Cannot Tell You}
\protect\phantomsection\label{what-ml-systems-cannot-tell-you}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Causation from observational data alone}}

  \begin{itemize}
  \tightlist
  \item
    Strong correlations \(\leftarrow\) confounding, reverse causation,
    or coincidence
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Effects of interventions}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``If we change X, what happens to Y?''}

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``If we lower prices by 10\%, will revenue
      increase?''}
    \end{itemize}
  \item
    Observational data can't answer without causal assumptions
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Counterfactuals}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``What would have happened with a different decision?''}

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``Would customer churn if we offered them a
      discount?''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Fairness}}

  \begin{itemize}
  \tightlist
  \item
    A model can be \emph{statistically} unbiased but \emph{causally}
    biased
  \item
    \emph{Statistically unbiased}: \(\EE[Prediction] = \EE[True Value]\)
  \item
    \emph{Causally biased}: model uses variables that are proxies for
    protected attributes, even if those variables aren't explicitly
    protected

    \begin{itemize}
    \tightlist
    \item
      E.g., using variables correlated to race, gender in decisions
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Optimal decisions}}

  \begin{itemize}
  \tightlist
  \item
    ML optimizes for accuracy, not business outcomes

    \begin{itemize}
    \tightlist
    \item
      E.g., a 90\% accurate model might lead to worse decisions than an
      85\% accurate one, depending on the consequences of different
      errors
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Problem 1: Correlation is Not Causation!}
\protect\phantomsection\label{problem-1-correlation-is-not-causation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Correlation}}: statistical relationships
  between variables

  \begin{itemize}
  \tightlist
  \item
    Pros:

    \begin{itemize}
    \tightlist
    \item
      Find patterns in past data to predict future
    \end{itemize}
  \item
    Cons:

    \begin{itemize}
    \tightlist
    \item
      Doesn't explain cause
    \item
      Variables move together by coincidence or hidden factors
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Causation}} explains how changing one
  variable influences another

  \begin{itemize}
  \tightlist
  \item
    Cannot be concluded from correlation alone
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Data doesn't understand causes and effects}}

  \begin{itemize}
  \tightlist
  \item
    Only humans identify variables and relationships from context
  \item
    Without causation, you can't make intelligent decisions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Correlation is Not Causation: Examples}
\protect\phantomsection\label{correlation-is-not-causation-examples}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Intuitively}}

  \begin{itemize}
  \tightlist
  \item
    You know that association is not causation
  \item
    Humans confuse association with causation, e.g.,

    \begin{itemize}
    \tightlist
    \item
      \emph{``You eat X, get stomachaches''} \(\to\) infer \emph{``X is
      bad for you''}
    \item
      \emph{``Buy stock before price spikes''} \(\to\) infer \emph{``you
      can time markets''}
    \end{itemize}
  \end{itemize}
\item
  Sometimes both directions are possible, e.g.,

  \begin{itemize}
  \tightlist
  \item
    \emph{``Top-notch consulting improves businesses''} or
  \item
    \emph{``Do successful businesses hire top consultants?''}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Problem 4: How ML Systems Make Decisions Today}

\tightlist
\item
  \textbf{\textcolor{red}{Typical workflow}}

  \begin{itemize}
  \tightlist
  \item
    Collect data \(\to\) Train model \(\to\) Make predictions
  \item
    Deploy: \emph{``If model predicts X, take action Y''}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Critical limitation}}

  \begin{itemize}
  \tightlist
  \item

    \item

\begin{frame}{Data Science for Decisions: Examples}
\protect\phantomsection\label{data-science-for-decisions-examples}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Recommendation systems}} (e.g., Netflix,
  Amazon, TikTok)

  \begin{itemize}
  \tightlist
  \item
    Model predicts which items users will click based on historical
    clicks
  \item
    System recommends top-N items based on predicted clicks
  \item
    \emph{Problem}: Can't run counterfactual experiments offline

    \begin{itemize}
    \tightlist
    \item
      Recommending novel items has no historical data to learn from
    \item
      Optimizing purely for clicks may harm long-term engagement (users
      get bored, echo chambers, filter bubbles)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Hiring pipelines}}

  \begin{itemize}
  \tightlist
  \item
    Model scores candidates
  \item
    HR uses scores to rank applicants
  \item
    \emph{Problem}: Training data encodes historical hiring bias
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Fraud detection}}

  \begin{itemize}
  \tightlist
  \item
    Model flags suspicious transactions
  \item
    System blocks transactions automatically
  \item
    \emph{Problem}: High false positive rate without knowing how to
    improve
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Problem 5: Feedback Loops}
\protect\phantomsection\label{problem-5-feedback-loops}
\begin{itemize}
\item
  Prediction systems can change the world through
  \textbf{\textcolor{red}{feedback loops}}
\item
  \textbf{\textcolor{darkgreen}{Recommendation systems}}

  \begin{itemize}
  \tightlist
  \item
    Model recommends sensational content \(\to\) users click more
  \item
    Future training data has more clicks on sensational content
  \item
    Model recommends even more sensational content
  \item
    \emph{Problem}: Radicalization, echo chambers, mental health issues
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Criminal justice relapse prediction}}

  \begin{itemize}
  \tightlist
  \item
    Minorities policed more \(\to\) arrested more \(\to\) data more
    biased
  \item
    Model trained on arrest data (biased against minorities)
  \item
    Judges use models for bail/sentencing
  \item
    \emph{Problem}: Model becomes increasingly biased over time
  \end{itemize}
\item
  \textbf{\textcolor{violet}{College admissions}}

  \begin{itemize}
  \tightlist
  \item
    Model trained on past acceptance decisions (which encode historical
    racism, sexism)
  \item
    Model used to screen applications
  \item
    Filter for applicants is biased
  \item
    \emph{Problem}: Historical biases are perpetuated
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Problem 6: Distribution Shift}
\protect\phantomsection\label{problem-6-distribution-shift}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{The world is always changing}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Concept drift}}

    \begin{itemize}
    \tightlist
    \item
      I.e., \(\Pr(Y | X)\), the link between inputs and predictions
      changes over time, the ``rules''
    \item
      E.g., customer preferences change over times \(\to\) model trained
      on 2020 data becomes less accurate by 2023
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Covariate shift}}

    \begin{itemize}
    \tightlist
    \item
      I.e., \(\Pr(X)\), input distribution changes but relationship
      stays the same
    \item
      E.g., more young customers join platform \(\to\) model trained on
      older customer base performs worse
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Label shift}}

    \begin{itemize}
    \tightlist
    \item
      I.e., \(\Pr(Y)\), outcome distribution changes
    \item
      E.g., fraud rate increases due to new fraud tactics \(\to\) model
      trained on old fraud rate is less effective
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Why causal models help}}

  \begin{itemize}
  \tightlist
  \item
    Encode mechanisms, not just patterns

    \begin{itemize}
    \tightlist
    \item
      Mechanisms more stable across distribution shifts
    \end{itemize}
  \item
    Understand ``why'' \(\to\) predict in new contexts
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Optimization vs.~Inference vs.~Decision Theory}
\protect\phantomsection\label{optimization-vs.-inference-vs.-decision-theory}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Three paradigms that ML conflates}}
\item
  \textbf{\textcolor{darkgreen}{Optimization}}: find best action given
  \emph{known} objective

  \begin{itemize}
  \tightlist
  \item
    Assumes world behavior is known

    \begin{itemize}
    \tightlist
    \item
      E.g., minimize cost, maximize throughput, tune parameters
    \end{itemize}
  \item
    \emph{Tools}: gradient descent, linear programming, evolutionary
    algorithms
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Statistical inference}}: estimate parameters
  from data

  \begin{itemize}
  \tightlist
  \item
    Learns from observations given model structure

    \begin{itemize}
    \tightlist
    \item
      E.g., estimate treatment effects, compute confidence intervals
    \end{itemize}
  \item
    \emph{Tools}: regression, Bayesian estimation, hypothesis testing
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Decision theory}}: choose actions under
  \emph{uncertainty}

  \begin{itemize}
  \tightlist
  \item
    E.g., \emph{``Launch product with uncertain demand?''}

    \begin{itemize}
    \tightlist
    \item
      Requires reasoning about interventions and outcomes
    \end{itemize}
  \item
    \emph{Tools}: utility functions, expected value of information,
    causal models
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Cost of Ignoring Causality (1/2)}
\protect\phantomsection\label{the-cost-of-ignoring-causality-12}
\begin{itemize}
\item
  Wrong decisions, wasted resources occur when organizations ignore
  causality
\item
  \textbf{\textcolor{red}{Simpson's Paradox}}

  \begin{itemize}
  \tightlist
  \item
    Aggregate data shows opposite trend from subgroups
  \item
    E.g., Berkeley admissions: overall bias shown, but no bias within
    departments
  \item
    \emph{Causal fix}: condition on the right variables using DAG
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Confounding bias}}

  \begin{itemize}
  \tightlist
  \item
    Hidden variable drives both treatment and outcome
  \item
    E.g., ice cream sales and drowning deaths both rise in summer
  \item
    \emph{Causal fix}: identify and control for confounders
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Collider bias}}

  \begin{itemize}
  \tightlist
  \item
    Conditioning on common effect creates false associations
  \item
    E.g., in hospitalized patients unrelated diseases appear correlated
    (since they both cause hospitalization)
  \item
    \emph{Causal fix}: don't condition on colliders
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Cost of Ignoring Causality (2/2)}
\protect\phantomsection\label{the-cost-of-ignoring-causality-22}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Feedback loop amplification}}

  \begin{itemize}
  \tightlist
  \item
    Biased predictions reinforce bias in future training data
  \item
    E.g., more police to minority neighborhoods \(\to\) more arrests
    \(\to\) model learns neighborhoods are more dangerous \(\to\) send
    more police
  \item
    \emph{Causal fix}: model feedback explicitly
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Intervention backfire}}

  \begin{itemize}
  \tightlist
  \item
    Acting on correlation leads to wrong actions
  \item
    E.g., hospitals with more staff have higher mortality rates

    \begin{itemize}
    \tightlist
    \item
      Confounded by patient severity: sicker patients need more staff
    \end{itemize}
  \item
    A naive intervention (\emph{``reduce staff to lower mortality''})
    would be catastrophic
  \item
    \emph{Causal fix}: distinguish correlation from causal effect before
    acting
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal AI}
\protect\phantomsection\label{causal-ai}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Understands the why}}

  \begin{itemize}
  \tightlist
  \item
    Determines cause-and-effect between variables
  \item
    E.g., \emph{``Did marketing campaign increase sales?''}
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Identify interventions}}

  \begin{itemize}
  \tightlist
  \item
    Identifies variables and interventions to change outcomes
  \item
    E.g., \emph{``Which lifestyle changes reduce blood pressure?''}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Predict counterfactuals}}

  \begin{itemize}
  \tightlist
  \item
    Hypothesize outcomes under different circumstances
  \item
    E.g., student grades if they attended a different school
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Avoid bias}}

  \begin{itemize}
  \tightlist
  \item
    Ensure fairness beyond training data biases by accounting for
    confounding variables
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Improve decisions}}

  \begin{itemize}
  \tightlist
  \item
    Understand relationships for better choices
  \item
    E.g., improve supply chain by understanding impact of decisions on
    logistics
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Rung 1: Association}
\protect\phantomsection\label{rung-1-association}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Question}}: \emph{``How would seeing \(X\)
  change your belief in \(Y\)?''}
\item
  \textbf{\textcolor{darkgreen}{Math}}: \(\Pr(Y|X)\)
\item
  \textbf{\textcolor{blue}{Activity}}:

  \begin{itemize}
  \tightlist
  \item
    Passive observation to determine if \(X, Y\) relate
  \item
    Traditional AI/ML based on this
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``Trees have green leaves in spring''}
  \item
    \emph{``What does a symptom tell you about disease?''}
  \item
    \emph{``What does a survey tell you about elections?''}
  \end{itemize}
\end{itemize}

\begin{frame}{Rung 2: Intervention}
\protect\phantomsection\label{rung-2-intervention}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Question}}: \emph{``What happens to \(Y\) if
  you do \(X\)?''}
\item
  \textbf{\textcolor{darkgreen}{Math}}: \(\Pr(Y | do(X), Z)\)
\item
  \textbf{\textcolor{blue}{Activity}}:

  \begin{itemize}
  \tightlist
  \item
    Understand impact of action \(X\) on \(Y\) under conditions \(Z\)
  \item
    Requires causal model, not just observations
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``Spring makes leaves turn green''}
  \item
    \emph{``Did headache disappear from pain reliever or food?''}
  \item
    \emph{``If I take aspirin, will headache cure?''}
  \item
    \emph{``If we ban sodas, what happens?''}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Rung 3: Counterfactuals}
\protect\phantomsection\label{rung-3-counterfactuals}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Question}}: \emph{``Was \(X\) that caused
  \(Y\)?''}
\item
  \textbf{\textcolor{darkgreen}{Symbol}}: \(\Pr(Y_X | x', y')\)
\item
  \textbf{\textcolor{blue}{Activity}}:

  \begin{itemize}
  \tightlist
  \item
    Imagine what would happen if facts were different
  \item
    Highest form of reasoning---requires understanding cause-and-effect
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``What if child received adult drug dose?''}
  \item
    \emph{``What would jury conclude?''}
  \item
    \emph{``Why did marketing campaign fail?''}
  \end{itemize}
\end{itemize}

\begin{frame}{Correlation vs Causation Model}

\begin{itemize}

\begin{frame}{Why Probabilistic Machine Learning Matters}
\protect\phantomsection\label{why-probabilistic-machine-learning-matters}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Question}}: When is knowing \emph{how
  confident} a model is more important than the prediction itself?

  \begin{itemize}
  \tightlist
  \item
    Medical diagnosis: A classifier that says ``cancer'' needs to know
    how sure it is
  \item
    Autonomous driving: When should the car defer to human control?
  \item
    Financial risk: Which decisions warrant caution despite high
    confidence?
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Key advantage}}: Probabilistic models
  output full distributions, not point estimates

  \begin{itemize}
  \tightlist
  \item
    Standard ML: ``The diagnosis is pneumonia''
  \item
    Probabilistic ML: ``87\% pneumonia, 13\% flu, plus credible
    intervals for each''
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Why it matters in practice}}

  \begin{itemize}
  \tightlist
  \item
    Uncertainty quantification drives better decisions under risk
  \item
    Limited or noisy data is handled naturally, not forced into
    deterministic assumptions
  \item
    Priors encode domain expertise formally, crucial when data is
    expensive or scarce
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Intuition}}: Probability embraces what we
  \emph{don't} know, treating it as information rather than failure
\end{itemize}
\end{frame}

\begin{frame}{Probabilistic ML: Robustness Through Uncertainty}
\protect\phantomsection\label{probabilistic-ml-robustness-through-uncertainty}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: Deterministic models fail silently

  \begin{itemize}
  \tightlist
  \item
    Overfit to small datasets (commit to one ``best'' explanation)
  \item
    Produce overconfident predictions on out-of-distribution data
  \item
    Struggle with missing values (requires imputation or row deletion)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Solution}}: Probabilistic frameworks solve
  these naturally

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Uncertainty propagation}}: Bayesian
    model averaging marginalizes over plausible explanations, building
    in Occam's razor

    \begin{itemize}
    \tightlist
    \item
      Naturally penalizes unnecessary complexity
    \item
      Confidence grows only when data strongly supports a model
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Principled handling of incomplete data}}:
    Integrate over unknown values instead of guessing or dropping rows
  \item
    \textbf{\textcolor{cyan}{Composability}}: Chain probabilistic models
    (graphical models, hierarchical Bayes) and reason about how
    uncertainty flows through pipelines

    \begin{itemize}
    \tightlist
    \item
      Deterministic models bolted together hide how errors accumulate
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key insight}}: Probability isn't just about
  predictions---it's about decision-making under uncertainty

  \begin{itemize}
  \tightlist
  \item
    When to act, when to wait for more data, when to defer to humans
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Why Explainability Matters}
\protect\phantomsection\label{why-explainability-matters}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: Modern AI systems are powerful but
  opaque

  \begin{itemize}
  \tightlist
  \item
    Deep neural networks with millions of parameters are hard to
    interpret
  \item
    Business logic becomes implicit rather than explicit
  \item
    Regulators and stakeholders demand accountability
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Question}}: When do we need
  explanations?

  \begin{itemize}
  \tightlist
  \item
    High-stakes decisions: healthcare diagnosis, loan approval, criminal
    justice
  \item
    Safety-critical systems: autonomous vehicles, aircraft control
  \item
    Building trust: users and decision-makers need to verify model
    behavior
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: Medical diagnosis

  \begin{itemize}
  \tightlist
  \item
    Model predicts patient has 85\% likelihood of disease X
  \item
    Question: Which symptoms drove this prediction?
  \item
    Without explanation: doctor cannot assess clinical validity
  \item
    With explanation: doctor can override or accept the prediction
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Key idea}}: Explainability enables
  accountability and builds trust
\end{itemize}
\end{frame}

\begin{frame}{Approaches to Explainability}
\protect\phantomsection\label{approaches-to-explainability}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: Explainability is the degree to
  which we can understand why a model made a particular prediction

  \begin{itemize}
  \tightlist
  \item
    Global explanation: \emph{``How does the model work overall?''}
  \item
    Local explanation: \emph{``Why did the model predict this specific
    outcome?''}
  \item
    Feature importance: \emph{``Which input features most influence the
    prediction?''}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Main approaches}}:

  \begin{enumerate}
  \tightlist
  \item
    Intrinsic methods: Design inherently interpretable models

    \begin{itemize}
    \tightlist
    \item
      Decision trees, linear regression, rule-based systems
    \item
      Transparent structure allows direct inspection
    \end{itemize}
  \item
    Post-hoc methods: Explain predictions after training

    \begin{itemize}
    \tightlist
    \item
      Feature attribution (LIME, SHAP, integrated gradients)
    \item
      Attention mechanisms in neural networks
    \item
      Counterfactual explanations: \emph{``What would need to change for
      a different prediction?''}
    \end{itemize}
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Example}}: SHAP (SHapley Additive
  exPlanations)

  \begin{itemize}
  \tightlist
  \item
    Computes each feature's contribution to moving the prediction from
    baseline to actual
  \item
    Assigns a value to each feature showing how much it pushed the
    decision
  \item
    Enables ranking features by importance and visualizing their impact
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Remark}}: Trade-off between model
  performance and explainability

  \begin{itemize}
  \tightlist
  \item
    Simpler models are easier to explain but may have lower accuracy
  \item
    Complex models achieve better performance but require post-hoc
    explanation techniques
  \item
    Choice depends on the application's requirements for accuracy vs
    interpretability
  \end{itemize}
\end{itemize}
\end{frame}
