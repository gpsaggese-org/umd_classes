\begin{frame}{Why Time Series Causality Is Different}
\protect\phantomsection\label{why-time-series-causality-is-different}
\begin{itemize}
\tightlist
\item
  Time series provide structure to exploit

  \begin{itemize}
  \tightlist
  \item
    The \emph{arrow of time} (causes precede effects) eliminates
    backward arrows
  \item
    Information from cohorts of related data
  \end{itemize}
\item
  \textbf{\textcolor{red}{Univariate Time series causal inference}}
  compares a unit to \emph{itself} over time

  \begin{itemize}
  \tightlist
  \item
    E.g., compare a city's crime rate \emph{before} vs \emph{after} a
    new policy
  \item
    Unit is its own control, but observations are \emph{not} independent
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Cross-sectional causal inference}}
  compares units at a single point in time

  \begin{itemize}
  \tightlist
  \item
    E.g., compare outcomes for treated vs untreated patients in one
    cohort
  \item
    Assumes units are (conditionally) exchangeable
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Challenges}}

  \begin{itemize}
  \tightlist
  \item
    Observations are correlated across time (autocorrelation)
  \item
    Treatments and confounders may drift or trend
  \item
    Often impossible to run experiments
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Did the Federal Reserve rate cut in 2020 cause the stock market
    rebound?
  \item
    Cannot rerun 2020 with a different policy \(\to\) must use temporal
    structure to reason counterfactually
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Temporal Causal Structures: Basic Setup}
\protect\phantomsection\label{temporal-causal-structures-basic-setup}
\begin{itemize}
\tightlist
\item
  Observe a sequence of variables over time:

  \begin{itemize}
  \tightlist
  \item
    Treatment: \(T_t\) (continuous, binary)
  \item
    Outcome: \(Y_t\)
  \item
    Covariates: \(\vX_t\)
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\item
  A simple \textbf{\textcolor{red}{temporal causal graph}} (first-order)
\item
  \textbf{\textcolor{blue}{Key principle}}: \emph{effects cannot precede
  causes}

  \begin{itemize}
  \tightlist
  \item
    \(T_t\) can affect \(Y_{t}, Y_{t+1}, ...\) but never \(Y_{t-1}\)
  \item
    Eliminates half of the arrows a-priori
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Temporal Causal Structures: Key Quantities}
\protect\phantomsection\label{temporal-causal-structures-key-quantities}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: The causal effect of a
  \emph{sustained intervention} starting at time \(t^*\): \[
  \tau = \EE[Y_t | do(T_{t^*:t} = 1)]
       - \EE[Y_t | do(T_{t^*:t} = 0)]
  \]
\item
  \textbf{\textcolor{darkgreen}{Intuition}}: average effect when we
  \emph{force} the treatment from \(t^*\) onward
\item
  \textbf{\textcolor{blue}{Definition}} \emph{Dynamic treatment} is when
  treatment at time \(t\) depends on history
  \((\vX_{1:t}, Y_{1:t-1}, T_{1:t-1})\)

  \begin{itemize}
  \tightlist
  \item
    Common in medicine: dosing depends on patient response so far
  \end{itemize}
\item
  Two common special cases:

  \begin{enumerate}
  \tightlist
  \item
    \_Contemporaneous effect: \(T_t \to Y_t\)
  \item
    \emph{Lagged effect}: \(T_t \to Y_{t+k}\) for some lag \(k > 0\)
  \end{enumerate}
\item
  \textbf{\textcolor{violet}{Example}}: monetary policy

  \begin{itemize}
  \tightlist
  \item
    Rate cut \emph{this quarter} affects GDP \emph{over several
    quarters} (lagged effect)
  \item
    Fed adjusts rates based on past GDP \(\to\) feedback loop
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Autocorrelation}
\protect\phantomsection\label{autocorrelation}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Autocorrelation}}: \(Y_t\) is correlated with
  \(Y_{t-1}, Y_{t-2}, ...\)

  \begin{itemize}
  \tightlist
  \item
    Standard errors computed assuming i.i.d. data are \emph{too small}
  \item
    Significance tests reject the null far more than their nominal rate
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Intuition}}: autocorrelated data carries
  less \emph{independent} information than the number of observations
  suggests
\item
  \textbf{\textcolor{blue}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    If quantities are autocorrelated, naive OLS standard errors mislead
    us
  \item
    Fix: Newey-West or HAC (heteroskedasticity- and
    autocorrelation-consistent) standard errors
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}: regressing daily stock returns
  on a dummy for ``news day''
\end{itemize}
\end{frame}

\begin{frame}{Non-Stationarity and Trends}
\protect\phantomsection\label{non-stationarity-and-trends}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: a \emph{stationary} process has
  constant mean, variance, and autocovariance over time

  \begin{itemize}
  \tightlist
  \item
    Many real-world series are \emph{not} stationary (e.g., GDP, prices,
    populations grow)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Spurious regression}}

  \begin{itemize}
  \tightlist
  \item
    Two series trending upward over time look correlated even if
    unrelated
  \item
    E.g., number of pirates vs global temperature (both trend over
    decades) but correlation \(\neq\) causation
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solutions}}:

  \begin{itemize}
  \tightlist
  \item
    Difference the series: work with \(\Delta Y_t = Y_t - Y_{t-1}\)
  \item
    Detrend: remove a fitted trend before analysis
  \item
    Use cointegration methods if two series share a common trend
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Feedback Loops and Simultaneity}
\protect\phantomsection\label{feedback-loops-and-simultaneity}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Fact}}: in time series, cause and effect often
  influence each other creating \emph{feedback loops}
\item
  \textbf{\textcolor{darkgreen}{Problem}}: this creates
  \emph{simultaneity bias} in regression

  \begin{itemize}
  \tightlist
  \item
    \(Y_t\) and \(T_t\) are jointly determined
  \item
    OLS estimates are inconsistent
  \item
    Cannot interpret coefficients as causal effects
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \tightlist
  \item
    Monetary policy \(\to\) inflation \(\to\) monetary policy
  \item
    Advertising \(\to\) sales \(\to\) advertising budget
  \end{itemize}
\end{itemize}

\begin{frame}{Structural Vector Autoregressions (VARs)}
\protect\phantomsection\label{structural-vector-autoregressions-vars}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: Model variables together,
  allowing simultaneous influence
\item
  \textbf{\textcolor{darkgreen}{Key idea}}

  \begin{itemize}
  \tightlist
  \item
    Treat policy \(T_t\) and outcome \(Y_t\) symmetrically
  \item
    Estimate response to shocks (i.e., unexpected, exogenous changes)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Mathematical formulation}}: VAR(p) model with
  \(k\) variables \[
  \vY_t = \vc + A_1 \vY_{t-1} + A_2 \vY_{t-2} + ... + A_p \vY_{t-p} + \vv{\varepsilon}_t
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \(\vY_t\) = vector of \(k\) time series
  \item
    \(A_i\) = coefficient matrices (dependence on lagged values)
  \item
    \(\vv{\varepsilon}_t\) = structural shocks (uncorrelated across
    equations)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Identification challenge}}: How do you
  determine causality direction?

  \begin{itemize}
  \tightlist
  \item
    Impose \emph{structural assumptions}

    \begin{itemize}
    \tightlist
    \item
      E.g., monetary policy reacts to inflation \emph{within} quarter;
      inflation reacts \emph{next} quarter
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Instrumental Variables (IV)}
\protect\phantomsection\label{instrumental-variables-iv}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Key idea}}:

  \begin{itemize}
  \tightlist
  \item
    Find a variable \(Z\) that affects treatment \(T\) but not outcome
    \(Y\) directly
  \item
    Use \(Z\) to predict \(T\), then measure how predicted \(\hat{T}\)
    correlates with \(Y\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Advantage}}: breaks the feedback loop by
  providing exogenous variation in treatment
\item
  \textbf{\textcolor{blue}{Problem}}: must find a credible instrument

  \begin{itemize}
  \tightlist
  \item
    E.g., studying effect of schooling on earnings, use distance to
    nearest college (affects schooling access, not earnings directly)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Limitation}}: only estimates the effect on
  ``compliers'': units whose treatment changes with the instrument
\end{itemize}
\end{frame}

\begin{frame}{Natural Experiments}
\protect\phantomsection\label{natural-experiments}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Key idea}}: exploit an unexpected, external
  event that randomly assigns treatment to some units but not others

  \begin{itemize}
  \tightlist
  \item
    E.g., lottery, policy rollout, shock, discontinuity in time
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Advantage}}: randomization eliminates
  confounding without requiring an instrument
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \tightlist
  \item
    Vietnam War draft lottery randomly assigns military service
  \item
    Compare drafted vs non-drafted on lifetime earnings
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Limitation}}

  \begin{itemize}
  \tightlist
  \item
    Only valid when the shock is truly exogenous and affects treatment
    assignment
  \item
    Cannot replicate in observational data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Unobserved Confounders That Vary Over Time}
\protect\phantomsection\label{unobserved-confounders-that-vary-over-time}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Time-varying confounders make identification harder
  \item
    \(U_t\) may affect \emph{both} \(T_t\) and \(Y_t\) and change slowly
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}: evaluating a diet on weight
  loss

  \begin{itemize}
  \tightlist
  \item
    Season (winter vs summer), stress, and motivation all change over
    time
  \item
    Hard to measure but drive both adherence and outcome
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    If a confounder is \emph{time-invariant} simple before-after
    comparisons difference it out

    \begin{itemize}
    \tightlist
    \item
      E.g., a person's genetics
    \item
      This is the core idea behind DiD and fixed-effects models
    \end{itemize}
  \item
    If the confounder is \emph{time-varying} it contaminates the
    comparison even after differencing

    \begin{itemize}
    \tightlist
    \item
      E.g., mood, weather
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Other Common Challenges}
\protect\phantomsection\label{other-common-challenges}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Anticipation effects}}: units change behavior
  \emph{before} treatment if they expect it

  \begin{itemize}
  \tightlist
  \item
    E.g., consumers delay purchases before a known tax cut
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Spillovers / SUTVA violations}}:
  treatment on one unit affects another

  \begin{itemize}
  \tightlist
  \item
    E.g., advertising in one region spills to neighboring regions
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Structural breaks}}: the data-generating
  process changes mid-sample

  \begin{itemize}
  \tightlist
  \item
    E.g., 2008 financial crisis, COVID-19 shock
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Missing data and irregular sampling}}:
  standard time series tools assume equispaced observations

  \begin{itemize}
  \tightlist
  \item
    E.g., sensor dropouts, holidays, weekends
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{When Temporal Structure Helps}
\protect\phantomsection\label{when-temporal-structure-helps}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Time orders causes before effects}}: rules out
  many causal directions a priori

  \begin{itemize}
  \tightlist
  \item
    Future cannot cause past \(\to\) prunes the graph
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pre-treatment history as a control}}:
  unit is its own baseline

  \begin{itemize}
  \tightlist
  \item
    Time-invariant confounders are automatically controlled for
  \item
    E.g., genes, geography, institutional features
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Natural experiments are easier to spot}}:
  policy changes, shocks, and discontinuities give quasi-experimental
  variation

  \begin{itemize}
  \item
    Interrupted Time Series (ITS), DiD, and synthetic control all
    exploit this
  \item
    \textbf{\textcolor{violet}{Example}}: studying the effect of a
    sugary-drink ban

    \begin{itemize}
    \tightlist
    \item
      Compare the same city before and after
    \item
      Genes, climate, culture are held fixed
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{When Temporal Structure Misleads}
\protect\phantomsection\label{when-temporal-structure-misleads}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Trends fake causation}} (spurious regression)

  \begin{itemize}
  \tightlist
  \item
    Two series both trending upward look causally linked even if
    unrelated
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Seasonality hides effects}}: a holiday bump
  can mask a policy effect
\item
  \textbf{\textcolor{darkgreen}{Lagged effects confuse attribution}}:
  today's outcome may reflect yesterday's treatment

  \begin{itemize}
  \tightlist
  \item
    Wrong lag specification \(\to\) wrong effect estimate
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Feedback inflates correlations}}: if \(Y_t\)
  drives \(T_{t+1}\), regressing \(Y_t\) on \(T_t\) also picks up the
  reverse effect
\item
  \textbf{\textcolor{cyan}{Regime changes invalidate models}}: a model
  fit before COVID may not hold after
\item
  \textbf{\textcolor{blue}{Key takeaway}}: temporal data is informative,
  but blindly applying cross-sectional tools to it creates more problems
  than it solves
\end{itemize}
\end{frame}

\begin{frame}{Granger Causality: Intuition}
\protect\phantomsection\label{granger-causality-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Intuition}}: Granger Causality (Nobel 2003)
  \emph{``Does knowing the past of \(X\) help me predict \(Y\) beyond
  what \(Y\)'s own past already tells me?''}

  \begin{itemize}
  \tightlist
  \item
    Causes must come before effects, and effects should carry
    information about their causes
  \end{itemize}
\item
  ``Granger-causing'' (a bit of a misnomer)

  \begin{itemize}
  \tightlist
  \item
    Is \emph{not} causation in Pearl's do-calculus sense
  \item
    Is a \emph{predictive} notion based on temporal precedence
  \item
    Answers \emph{``does X help predict Y?''} not \emph{``does changing
    X change Y?''}
  \item
    \(X \to Y\) and \(Y \to X\) can both hold (common in economic
    feedback systems)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Ice cream sales} does not Granger-cause \emph{drowning}
  \item
    \emph{Temperature} Granger-causes both \emph{ice cream sales} and
    \emph{drowning}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Granger Causality: Formal Definition}
\protect\phantomsection\label{granger-causality-formal-definition}
\begin{itemize}
\item
  Consider two stationary time series \(X_t\) and \(Y_t\)
\item
  Compare two predictors of \(Y_t\):

  \begin{enumerate}
  \tightlist
  \item
    \emph{Restricted model}: uses only past \(Y\) \[
    Y_t = \alpha + \sum_{i=1}^p \phi_i Y_{t-i} + \varepsilon_t
    \]
  \item
    \emph{Unrestricted model}: adds past \(X\) \[
    Y_t = \alpha + \sum_{i=1}^p \phi_i Y_{t-i}
                 + \sum_{j=1}^p \beta_j X_{t-j} + \eta_t
    \]
  \end{enumerate}
\item
  \textbf{\textcolor{red}{Definition}}: \(X\) \emph{Granger-causes}
  \(Y\) iff the unrestricted model has \emph{strictly smaller} forecast
  error than the restricted one

  \begin{itemize}
  \tightlist
  \item
    Equivalently: F-test on
    \(H_0: \beta_1 = \beta_2 = ... = \beta_p = 0\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Granger Causality: Key Assumptions}
\protect\phantomsection\label{granger-causality-key-assumptions}
\begin{enumerate}
\tightlist
\item
  \textbf{\textcolor{red}{Stationarity}}: both series have
  time-invariant statistical properties

  \begin{itemize}
  \tightlist
  \item
    Non-stationary series can produce spurious Granger causality
  \item
    Pre-test with ADF / KPSS; difference if needed
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Correct lag order $p$}}: too few lags miss
  the effect; too many overfit

  \begin{itemize}
  \tightlist
  \item
    Choose via AIC, BIC, or cross-validation
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Linearity}} (in the basic formulation)

  \begin{itemize}
  \tightlist
  \item
    Non-linear Granger causality extensions exist (e.g., transfer
    entropy)
  \end{itemize}
\item
  \textbf{\textcolor{teal}{No omitted common causes}} (the big one)

  \begin{itemize}
  \tightlist
  \item
    If \(Z\) drives both \(X\) and \(Y\) with different lags, \(X\)
    appears to Granger-cause \(Y\)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Correct sampling frequency}}

  \begin{itemize}
  \tightlist
  \item
    If the true effect operates faster than the sampling interval,
    Granger causality may fail to detect or reverse direction
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Granger Causality: Limitations}
\protect\phantomsection\label{granger-causality-limitations}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Granger causality $\neq$ causation}}

  \begin{itemize}
  \tightlist
  \item
    It tests \textbf{\textcolor{orange}{predictive}} relationships, not
    \textbf{\textcolor{darkgreen}{interventional}} ones
  \item
    A Granger-causal relationship can be entirely due to a common driver
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Example of spurious Granger causality}}:

  \begin{itemize}
  \tightlist
  \item
    Barometer reading \(X_t\) Granger-causes storms \(Y_{t+1}\)
  \item
    But raising the barometer manually does \emph{not} cause a storm
  \item
    Atmospheric pressure is the actual common cause
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Aggregation problems}}:

  \begin{itemize}
  \tightlist
  \item
    Monthly data may miss daily Granger effects
  \item
    Sampling slower than the process \(\to\) spurious bidirectional
    causality
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cannot handle contemporaneous effects}} in
  the basic form

  \begin{itemize}
  \tightlist
  \item
    If \(X_t\) affects \(Y_t\) instantly, lag-based tests miss it
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Requires long time series}} for reliable
  inference

  \begin{itemize}
  \tightlist
  \item
    Typically \(T \gg p\) (e.g., 100+ observations for 4 lags)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Granger Causality: Practical Example 1}
\protect\phantomsection\label{granger-causality-practical-example-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Question}}: does advertising spend
  Granger-cause sales?

  \begin{itemize}
  \tightlist
  \item
    \(X_t\): weekly ad spend
  \item
    \(Y_t\): weekly sales
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Caveat}}: this does \emph{not} mean ``if
  we spend more on ads, sales will rise''

  \begin{itemize}
  \tightlist
  \item
    Both could be driven by seasonal demand \(\to\) need an experiment
    or a causal design to get an interventional answer
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Question}}: do Twitter sentiment scores
  Granger-cause stock returns?

  \begin{itemize}
  \tightlist
  \item
    A widely-studied question; results are mixed
  \item
    Twitter sentiment \emph{does} Granger-cause short-horizon returns
    for some stocks and horizons, but effect sizes are small
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Interrupted Time Series: Motivation}
\protect\phantomsection\label{interrupted-time-series-motivation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Scenario}}: suppose a policy is implemented at
  a single known date \(t^*\) across all units

  \begin{itemize}
  \tightlist
  \item
    E.g., new speed limit enforced from January 1st
  \item
    E.g., a website redesign launched on a specific day
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    No control group is available: the whole population is treated
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    Use the \emph{pre-intervention} trajectory of the series to forecast
    what would have happened \emph{without} the intervention
  \item
    Compare the forecast (\emph{counterfactual}) to what actually
    happened
  \item
    The gap is the estimated causal effect
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Seatbelt law: \textasciitilde15-20\% reduction in fatalities
  \item
    Netflix password ban:

    \begin{itemize}
    \tightlist
    \item
      Expected: subscriber growth from monetizing shared accounts
    \item
      Actual: mixed results (growth in certain regions, churn in others)
    \end{itemize}
  \item
    iOS privacy changes:

    \begin{itemize}
    \tightlist
    \item
      Apps must ask permission to track
    \item
      60-70\% opt-in to tracking \(\to\) 15-25\% opt-in
    \item
      ad-targeting effectiveness tanked, Meta reported billions in
      revenue losses
    \end{itemize}
  \end{itemize}
\end{itemize}

\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.3.png}}
\end{frame}

\begin{frame}{ITS: Segmented Regression Formulation}
\protect\phantomsection\label{its-segmented-regression-formulation}
\begin{itemize}
\item
  \emph{Segmented regression} model: \[
  Y_t = \beta_0 + \beta_1 t + \beta_2 D_t + \beta_3 (t - t^*) D_t + \varepsilon_t
  \] where \(D_t = I(t \ge t^*)\)
\item
  \textbf{\textcolor{red}{Interpretation}}

  \begin{itemize}
  \tightlist
  \item
    \(\beta_0\): pre-intervention intercept
  \item
    \(\beta_1\): pre-intervention slope (trend)
  \item
    \(\beta_2\): \emph{level change} at the intervention (immediate
    jump)
  \item
    \(\beta_3\): \emph{slope change} after the intervention (trend
    shift)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Questions}}:

  \begin{enumerate}
  \tightlist
  \item
    Is there a jump at \(t^*\)?

    \begin{itemize}
    \tightlist
    \item
      Test \(\beta_2 = 0\)
    \end{itemize}
  \item
    Does the trajectory change?

    \begin{itemize}
    \tightlist
    \item
      Test \(\beta_3 = 0\)
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{ITS: Details}
\protect\phantomsection\label{its-details}
\begin{itemize}
\tightlist
\item
  \emph{Autocorrelation} is almost always present

  \begin{itemize}
  \tightlist
  \item
    Fit with autocorrelation robust methods (Newey-West, ARIMA)
  \end{itemize}
\item
  \emph{Robustness checks}:

  \begin{itemize}
  \tightlist
  \item
    Vary the window size (use 1, 2, 5 years pre-period)
  \item
    Try placebo intervention dates: do you find fake effects?
  \item
    Check for seasonality, holidays, co-occurring events
  \end{itemize}
\item
  \emph{Visual diagnostic}:

  \begin{itemize}
  \tightlist
  \item
    Always plot the series with the fitted model and the intervention
    date marked
  \item
    Does ITS result survives?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{ITS: Strengths and Weaknesses}
\protect\phantomsection\label{its-strengths-and-weaknesses}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Works when \emph{no control group exists} (universal policy changes)
  \item
    Uses the unit as its own control \(\to\) time-invariant confounders
    cancel
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Relies on \emph{extrapolation} of the pre-period trend
  \item
    Vulnerable to \emph{co-occurring events} (\emph{``was it the policy
    or the recession?''})
  \item
    Autocorrelation inflates errors if ignored
  \item
    Needs a long, stable pre-period (rare in practice)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Rule of thumb}}: ITS is most credible when:

  \begin{enumerate}
  \tightlist
  \item
    The intervention is sharply timed
  \item
    No other major events occur near \(t^*\)
  \item
    The pre-period is long and stable
  \item
    The outcome has low noise relative to the effect size
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{DiD: Motivation and Setup}
\protect\phantomsection\label{did-motivation-and-setup}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Motivating Example}}

  \begin{itemize}
  \tightlist
  \item
    Suppose a policy is rolled out in \emph{some} units but not others

    \begin{itemize}
    \tightlist
    \item
      E.g., New Jersey raises its minimum wage in 1992; Pennsylvania
      does not
    \item
      Two groups: treated (NJ) and control (PA); two periods: before and
      after
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Definition}}: Difference-in-Differences
  (DiD) estimates a causal effect by comparing the change in outcomes
  over time between a group that received treatment and a group that
  didn't

  \begin{itemize}
  \tightlist
  \item
    Assume both groups would have followed the same trend in the absence
    of treatment
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Procedure}}

  \begin{itemize}
  \tightlist
  \item
    Subtracts out time-invariant differences between groups
  \item
    Subtracts out common time trends across groups
  \item
    What's left is attributed to the policy
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}: Card \& Krueger (1994) studied
  the NJ vs PA minimum wage

  \begin{itemize}
  \tightlist
  \item
    Standard economic theory predicted that raising the minimum wage
    would reduce employment (since labor becomes more costly)
  \item
    DiD estimate: essentially no employment effect (a surprising result)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DiD: The Core Idea}
\protect\phantomsection\label{did-the-core-idea}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.4.png}}

\begin{itemize}
\tightlist
\item
  \emph{DiD estimator} (with 2 groups, 2 periods): \[
  \hat{\tau}_{DiD}
    = (\bar{Y}^{T}_{\text{post}} - \bar{Y}^{T}_{\text{pre}})
      - (\bar{Y}^{C}_{\text{post}} - \bar{Y}^{C}_{\text{pre}})
  \]
\end{itemize}
\end{frame}

\begin{frame}{DiD: Regression Formulation}
\protect\phantomsection\label{did-regression-formulation}
\begin{itemize}
\tightlist
\item
  Equivalent regression form (with unit \(i\) and time \(t\)): \[
  Y_{it} = \alpha + \beta \cdot \text{Treat}_i
         + \gamma \cdot \text{Post}_t
         + \tau \cdot (\text{Treat}_i \cdot \text{Post}_t) + \varepsilon_{it}
  \]

  \begin{itemize}
  \tightlist
  \item
    \(\beta\): time-invariant difference between groups
  \item
    \(\gamma\): common time trend
  \item
    \(\tau\): \emph{the DiD causal effect} (coefficient on the
    interaction)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Parallel Trends Assumption}
\protect\phantomsection\label{parallel-trends-assumption}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Assumption}}: in the absence of treatment, the
  treated and control units would have followed the \emph{same trend}
  over time \[
  \EE[Y^{(0)}_{it} - Y^{(0)}_{i,t-1} | \text{Treat}_i = 1]
  = \EE[Y^{(0)}_{it} - Y^{(0)}_{i,t-1} | \text{Treat}_i = 0]
  \] where \(Y^{(0)}\) denotes the untreated potential outcome
\item
  \textbf{\textcolor{orange}{Intuition}}: levels can differ, but
  \emph{changes} should match

  \begin{itemize}
  \tightlist
  \item
    E.g., NJ may always have higher wages than PA, but both move
    together absent the policy
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Question}}: how do we test it?

  \begin{itemize}
  \tightlist
  \item
    We cannot test it: it is a \emph{counterfactual} statement
  \item
    But we can check the \emph{pre-treatment} trends as a proxy
  \item
    If they are parallel \emph{before} treatment, we gain confidence
    (but no proof)
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Visual check}}: plot the two group means over
  time; do they move in lockstep before treatment?
\item
  \textbf{\textcolor{cyan}{Red flags}}:

  \begin{itemize}
  \tightlist
  \item
    Significant pre-trends with opposite sign of the effect
  \item
    Divergence starting \emph{before} the policy (anticipation or
    selection)
  \item
    Using too short a pre-period \(\to\) can't check parallel trends
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DiD: Robustness Checks}
\protect\phantomsection\label{did-robustness-checks}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Placebo treatments in time}}: pretend the
  treatment happened earlier

  \begin{itemize}
  \tightlist
  \item
    Should get no effect if the design is sound
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Placebo outcomes}}: apply DiD to unrelated
  outcomes that \emph{shouldn't} change

  \begin{itemize}
  \tightlist
  \item
    E.g., if studying minimum wage on employment, check it has no effect
    on commodity prices
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Alternative control groups}}: use
  different comparison units

  \begin{itemize}
  \tightlist
  \item
    Results should be stable across reasonable choices
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Sensitivity to functional form}}: levels
  vs.~logs, linear vs.~trend adjustment
\item
  \textbf{\textcolor{cyan}{Falsification tests}}: subset the data in
  ways that should \emph{not} yield an effect

  \begin{itemize}
  \tightlist
  \item
    If they do, your design picks up something other than the treatment
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{DiD: Summary of Strengths and Limits}
\protect\phantomsection\label{did-summary-of-strengths-and-limits}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Removes time-invariant confounders with unit fixed effects
  \item
    Removes common shocks with time fixed effects
  \item
    Uses both treated and control units, making it more credible than
    ITS alone
  \item
    Flexible: extends to multiple periods, staggered adoption,
    covariates
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Parallel trends assumption is \emph{untestable}: can only check
    pre-trends
  \item
    Time-varying confounders still bias the estimate
  \item
    Staggered adoption with heterogeneous effects breaks naive TWFE
  \item
    Standard errors require clustering --- easy to get wrong
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Rule of thumb}}: DiD is credible when:

  \begin{enumerate}
  \tightlist
  \item
    Pre-trends are visually parallel
  \item
    No co-occurring shocks hit treated and controls differently
  \item
    Staggered timing is handled with modern estimators
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Synthetic Control: Motivation}
\protect\phantomsection\label{synthetic-control-motivation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: DiD works when a credible control
  unit exists

  \begin{itemize}
  \tightlist
  \item
    But what if \emph{no single control unit} matches the treated one?
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}: studying the effect of German
  reunification (1990) on West Germany's per-capita GDP

  \begin{itemize}
  \tightlist
  \item
    No other single country matches West Germany's pre-1990 size, growth
    path, and economic structure
  \item
    Synthetic Control instead builds a weighted combination of other
    OECD countries that closely tracks West Germany's pre-reunification
    GDP, then uses that as the counterfactual
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key idea}}

  \begin{itemize}
  \tightlist
  \item
    Construct a \emph{weighted average} of control units that best
    matches the treated unit's pre-treatment characteristics
  \item
    This weighted average is the synthetic control
  \item
    Compare treated unit to the synthetic control in the post-period
  \item
    The gap is the estimated causal effect
  \end{itemize}
\end{itemize}
\end{frame}

  units eligible for inclusion in the weighted combination

  \begin{itemize}
  \tightlist
  \item
    Should exclude units treated by the same or similar interventions
  \end{itemize}

\begin{frame}{Synthetic Control: Formal Setup}
\protect\phantomsection\label{synthetic-control-formal-setup}
\begin{itemize}
\item
  There are \(J+1\) units:

  \begin{itemize}
  \tightlist
  \item
    Unit 1 is treated at time \(t^*\)
  \item
    Units \(2, ..., J+1\) are untreated (donors)
  \end{itemize}
\item
  \(\vX_1\) be a vector of pre-treatment predictors for the treated unit
\item
  \(\vX_0\) the matrix of the same predictors for donors (size
  \(k \times J\))
\item
  Pick \emph{weights} \(\vw = (w_2, ..., w_{J+1})\) satisfy: \[
  w_j \ge 0, \quad \sum_{j=2}^{J+1} w_j = 1
  \]

  \begin{itemize}
  \tightlist
  \item
    Non-negativity and unit-sum \(\to\) weights are a \emph{convex
    combination} of donors
  \item
    Ensures the synthetic control is in the \emph{convex hull} of the
    donor pool \(\to\) no extrapolation
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Synthetic Control: Optimization}
\protect\phantomsection\label{synthetic-control-optimization}
\begin{itemize}
\item
  Choose \(\vw^*\) to minimize pre-treatment mismatch: \[
  \vw^* = \arg\min_{\vw} (\vX_1 - \vX_0 \vw)^\top \vV (\vX_1 - \vX_0 \vw)
  \] subject to \(w_j \ge 0, \sum w_j = 1\)
\item
  \(\vV\) is a \(k \times k\) diagonal matrix of \emph{predictor
  importance weights}

  \begin{itemize}
  \tightlist
  \item
    Chosen to minimize mean-squared prediction error on pre-treatment
    outcomes
  \item
    Nested optimization (inner: find \(\vw\); outer: find \(\vV\))
  \end{itemize}
\item
  The \emph{estimated causal effect} at time \(t \ge t^*\): \[
  \hat{\tau}_t = Y_{1t} - \sum_{j=2}^{J+1} w^*_j Y_{jt}
  \]

  \begin{itemize}
  \tightlist
  \item
    Gap between treated unit's outcome and the weighted donor outcome
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Synthetic Control: Inference via Placebos}
\protect\phantomsection\label{synthetic-control-inference-via-placebos}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Problem}}: there is only \emph{one} treated
  unit, so standard inference doesn't apply
\item
  \textbf{\textcolor{darkgreen}{Solution}}: \emph{placebo tests} by
  permutation

  \begin{enumerate}
  \tightlist
  \item
    Pretend each donor was the treated unit
  \item
    Compute the synthetic control and causal-effect trajectory for each
  \item
    Compare the \emph{actual} treated unit's effect against this ``null
    distribution''
  \end{enumerate}
\item
  If the treated unit's effect is \emph{larger in magnitude} than most
  placebo effects \(\to\) evidence of a real effect

  \begin{itemize}
  \tightlist
  \item
    Compute an empirical \(p\)-value from the placebos
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Variant}}: filter placebos to keep only those
  with good pre-treatment fit

  \begin{itemize}
  \tightlist
  \item
    A donor with bad pre-fit produces noise, not a legitimate null
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Synthetic Control: When It Works Well}
\protect\phantomsection\label{synthetic-control-when-it-works-well}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Long pre-treatment period}}: many time points
  to match on

  \begin{itemize}
  \tightlist
  \item
    Rule of thumb: at least 10+ pre-treatment observations
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Rich donor pool}}: many untreated units
  with diverse characteristics

  \begin{itemize}
  \tightlist
  \item
    Enables good pre-treatment fit via convex combinations
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Small number of treated units}} (usually one)

  \begin{itemize}
  \tightlist
  \item
    Designed for aggregated units: countries, states, cities, firms
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Outcome with moderate noise}}: low enough
  to detect the signal
\end{itemize}
\end{frame}

\begin{frame}{Synthetic Control: When It Fails}
\protect\phantomsection\label{synthetic-control-when-it-fails}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Poor pre-treatment fit}}: no convex
  combination matches the treated unit

  \begin{itemize}
  \tightlist
  \item
    The treated unit is an \emph{outlier} in the donor space
  \item
    Reported effects are unreliable
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Treatment contaminates donors}}
  (spillovers)

  \begin{itemize}
  \tightlist
  \item
    E.g., neighboring states also affected by a policy violate the
    design
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Short pre-period or noisy outcome}}
  \(\to\) fit is unreliable or overfits noise
\item
  \textbf{\textcolor{teal}{Confounding unobserved events}}: a
  co-occurring shock to the treated unit near \(t^*\) masquerades as the
  treatment effect
\item
  \textbf{\textcolor{cyan}{Diagnostic}}: always plot the pre-treatment
  fit

  \begin{itemize}
  \tightlist
  \item
    A poor pre-fit \(\to\) don't trust the post-period gap
  \end{itemize}
\end{itemize}
\end{frame}

\begin{itemize}
\tightlist
\item
  Use \textbf{\textcolor{red}{Granger causality}} when:

  \begin{itemize}
  \tightlist
  \item
    You want a fast predictive screen on two long time series
  \item
    Intervention / experiment is not available
  \end{itemize}
\item
  Use \textbf{\textcolor{darkgreen}{Interrupted Time Series (ITS)}}
  when:

  \begin{itemize}
  \tightlist
  \item
    A single intervention has a sharp date and no comparison units exist
  \item
    Pre-period is long and stable
  \end{itemize}
\item
  Use \textbf{\textcolor{blue}{Difference-in-Differences (DiD)}} when:

  \begin{itemize}
  \tightlist
  \item
    Some units are treated and others are not
  \item
    Parallel pre-trends are plausible
  \item
    You have panel data with units and time
  \end{itemize}
\item
  Use \textbf{\textcolor{violet}{Synthetic Control}} when:

  \begin{itemize}
  \tightlist
  \item
    One (or few) treated units with a rich donor pool t- No single donor
    is a credible match, but a weighted combination is
  \end{itemize}
\end{itemize}
\end{frame}
