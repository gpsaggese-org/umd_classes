\begin{frame}{Data Cleaning}
\protect\phantomsection\label{data-cleaning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Purpose of data cleaning}}

  \begin{itemize}
  \tightlist
  \item
    Ensure data quality for accurate model training
  \item
    Detect and correct errors or inconsistencies in the dataset
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Typical steps in data cleaning}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Remove duplicates}: Identical records are eliminated
  \item
    \emph{Correct data entry errors}: Fix misspellings or misformatted
    entries
  \item
    \emph{Standardize data}:

    \begin{itemize}
    \tightlist
    \item
      Convert dates formatted as both
      \textcolor{blue}{\texttt{MM/DD/YYYY}} and
      \textcolor{blue}{\texttt{DD-MM-YYYY}} into consistent format
    \item
      String normalization (e.g., lowercase conversion)
    \item
      Type conversion (e.g., strings to integers)
    \item
      Dealing with unexpected characters or encodings
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Relevance to ML models}}

  \begin{itemize}
  \tightlist
  \item
    Poor data quality leads to biased or incorrect predictions
  \item
    Clean data reduces variance and improves generalization
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Handling Outliers and Missing Data}
\protect\phantomsection\label{handling-outliers-and-missing-data}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Outliers}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Definition}: Data points significantly different from others
  \item
    \emph{Causes}: Measurement errors, variability in the data
  \item
    \emph{Detection}: Box plots, Z-scores, or interquartile method
  \item
    \emph{Treatment}: Removal, capping, or transformation (e.g., log
    scale)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Missing data}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Detection}: Count of null values or incomplete entries
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Remediation}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Deletion}: Remove rows or columns with too many missing values
  \item
    \emph{Imputation}:

    \begin{itemize}
    \tightlist
    \item
      Mean/median/mode substitution
    \item
      K-nearest neighbors (KNN)
    \item
      Regression or model-based approaches
    \item
      E.g., for a missing temperature reading, impute using the mean of
      the day's surrounding values
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Normalization and Standardization}
\protect\phantomsection\label{normalization-and-standardization}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}: Adjust feature scales for better
  convergence and learning

  \begin{itemize}
  \tightlist
  \item
    \emph{Normalize}: Rescale to \([0, 1]\)
  \item
    \emph{Standardize}: Zero mean, unit variance
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Why it helps}}

  \begin{itemize}
  \tightlist
  \item
    Equal feature contribution in distance-based models
  \item
    Faster convergence in gradient-based algorithms
  \item
    Enables regularization
  \item
    Easier feature interpretation
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Common methods}}

  \begin{itemize}
  \tightlist
  \item
    Min-Max normalization:
    \(x' = \frac{x - x_{min}}{x_{max} - x_{min}}\)
  \item
    Z-score standardization: \(x' = \frac{x - \mu}{\sigma}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Encoding Categorical Data}
\protect\phantomsection\label{encoding-categorical-data}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: Convert non-numeric categories into
  numeric representations
\item
  \textbf{\textcolor{darkgreen}{Label encoding}}

  \begin{itemize}
  \tightlist
  \item
    Assigns an integer to each category

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{red}},
      \textcolor{blue}{\texttt{green}}, \textcolor{blue}{\texttt{blue}}
      \(\to\) \textcolor{blue}{\texttt{1}},
      \textcolor{blue}{\texttt{2}}, \textcolor{blue}{\texttt{3}}
    \end{itemize}
  \item
    Can mislead models if order is not meaningful
  \end{itemize}
\item
  \textbf{\textcolor{blue}{One-hot encoding}}

  \begin{itemize}
  \tightlist
  \item
    Creates binary vector per category

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{red}},
      \textcolor{blue}{\texttt{green}}, \textcolor{blue}{\texttt{blue}}
      \(\to\) \textcolor{blue}{\texttt{[1,0,0]}},
      \textcolor{blue}{\texttt{[0,1,0]}},
      \textcolor{blue}{\texttt{[0,0,1]}}
    \end{itemize}
  \item
    Avoids ordinal assumption
  \item
    Increases dimensionality
  \end{itemize}
\end{itemize}
\end{frame}

  \end{itemize}
\item
  \textbf{\textcolor{blue}{Why it helps}}

  \begin{itemize}

  \item
    Encodes domain knowledge
  \item
    Improves model expressiveness and performance
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Dimensionality Reduction}
\protect\phantomsection\label{dimensionality-reduction}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: Reduce number of features while
  preserving key information
\item
  \textbf{\textcolor{darkgreen}{Why it helps}}

  \begin{itemize}
  \tightlist
  \item
    Reduce overfitting
  \item
    Reduce data redundancy
  \item
    Improve model speed
  \item
    Allow visualization
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Common techniques}}

  \begin{itemize}
  \tightlist
  \item
    PCA: linear combinations that maximize variance
  \item
    LDA: projects data to maximize class separability
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}

  \begin{itemize}
  \tightlist
  \item
    Reduce 1024x640 image pixels to 10 principal components
  \item
    Quantize color images into gray scale
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Discretization}
\protect\phantomsection\label{discretization}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: Convert continuous values into
  categorical bins
\item
  \textbf{\textcolor{darkgreen}{Why it helps}}

  \begin{itemize}
  \tightlist
  \item
    Simplifies models or enables categorical algorithms
  \item
    Helps detect threshold effects in data
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Techniques}}

  \begin{itemize}
  \tightlist
  \item
    Equal-width binning
  \item
    Quantile binning
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}

  \begin{itemize}
  \tightlist
  \item
    Discretize age

    \begin{itemize}
    \tightlist
    \item
      \textcolor{blue}{\texttt{Child}}: \([0, 13)\)
    \item
      \textcolor{blue}{\texttt{Teen}}: \([13, 20)\)
    \item
      \textcolor{blue}{\texttt{Adult}}: \([20, 65)\)
    \item
      \textcolor{blue}{\texttt{Senior}}: \([65, \infty)\)
    \end{itemize}
  \item
    Age 32 \(\to\) \textcolor{blue}{\texttt{Adult}}
  \end{itemize}
\end{itemize}
\end{frame}
