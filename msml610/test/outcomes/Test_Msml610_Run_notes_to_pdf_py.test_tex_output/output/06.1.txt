\begin{frame}{Logic-Based AI Under Uncertainty: Problem}
\protect\phantomsection\label{logic-based-ai-under-uncertainty-problem}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: Logic-based AI systems

  \begin{itemize}
  \tightlist
  \item
    Based on \emph{propositional logic}
  \item
    Represent \emph{actions} using \emph{rules} like:

    \begin{itemize}
    \tightlist
    \item
      \emph{``If preconditions P hold, then action A causes effect E''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{``If I turn the car key, the engine starts''}
  \item
    \emph{But}: the battery might be dead, there's no fuel, the starter
    is broken, \ldots{}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Problem}}: Real-world agents face
  \emph{uncertainty} from:

  \begin{enumerate}
  \tightlist
  \item
    Partial observability

    \begin{itemize}
    \tightlist
    \item
      Agent can't see the full state of the world
    \end{itemize}
  \item
    Non-determinism

    \begin{itemize}
    \tightlist
    \item
      Actions don't always have predictable outcomes
    \end{itemize}
  \item
    Adversarial conditions

    \begin{itemize}
    \tightlist
    \item
      Other agents may interfere
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Logic-Based Under Uncertainty: Naive Solution}
\protect\phantomsection\label{logic-based-under-uncertainty-naive-solution}
\begin{itemize}
\item
  \textbf{\textcolor{red}{(Naive) Solution}}: A possible solution to
  logic-based AI under uncertainty is to use:

  \begin{enumerate}
  \tightlist
  \item
    \emph{Belief state}, i.e., set of all possible current world states

    \begin{itemize}
    \tightlist
    \item
      Cons: difficult to enumerate and track
    \end{itemize}
  \item
    \emph{Causal and exhaustive augmentation} for rules

    \begin{itemize}
    \tightlist
    \item
      Cons: Must consider all possible preconditions for acting, even
      unlikely ones
    \end{itemize}
  \item
    \emph{Contingent plans} to handle every possible sensor report and
    belief

    \begin{itemize}
    \tightlist
    \item
      Cons: Plans become large and complex
    \item
      No guaranteed plan may exist, yet action is required
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Naive Solution: Example of Starting a Car}
\protect\phantomsection\label{naive-solution-example-of-starting-a-car}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Goal}}: build a system to start a car by
  turning the car key
\item
  \textbf{\textcolor{darkgreen}{Remark}}: Obvious preconditions

  \begin{itemize}
  \tightlist
  \item
    The car has fuel
  \item
    The car battery is charged
  \item
    The car key is the right one
  \item
    \ldots{}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Remark}}: Less obvious preconditions

  \begin{itemize}
  \tightlist
  \item
    The electrical system is intact (no blown fuses)
  \item
    The fuel pump is working
  \item
    The fuel lines are not clogged
  \item
    The engine oil level is sufficient
  \item
    \ldots{}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Problem}}: Every precondition requires
  something else in order to do to achieve the goal!
\end{itemize}
\end{frame}

\begin{frame}{Causal and Exhaustive Augmentation}
\protect\phantomsection\label{causal-and-exhaustive-augmentation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: Causal and exhaustive
  augmentation of logic-rules

  \begin{itemize}
  \tightlist
  \item
    To use propositional logic under uncertainty, need to augment the
    left-side of rules \(X \implies Y\) to make it:

    \begin{enumerate}
    \tightlist
    \item
      \emph{Causal}: identify true causal-effect relationships
    \item
      \emph{Exhaustive}: identify all possible conditions \(X\) leading
      to the outcome \(Y\)
    \end{enumerate}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Logical qualification problem}}

  \begin{itemize}
  \tightlist
  \item
    Need to enumerate all the preconditions necessary for an action to
    succeed
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Limitations}}: Difficulties

  \begin{itemize}
  \tightlist
  \item
    Every action can have more hidden conditions needed in order to
    succeed

    \begin{itemize}
    \tightlist
    \item
      E.g., start the engine \(\implies\) turn car key \(\implies\)
      starter motor is working \(\implies\) battery is charged
      \(\implies\) \ldots{}
    \end{itemize}
  \item
    Preconditions can change depending on the situation

    \begin{itemize}
    \tightlist
    \item
      E.g., \ldots{} \(\implies\) battery is charged \(\implies\) it's
      not too cold \(\implies\) \ldots{}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal and Exhaustive Augmentation: Limitations}
\protect\phantomsection\label{causal-and-exhaustive-augmentation-limitations}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Limitations}} of logical qualification

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{orange}{Laziness}}

    \begin{itemize}
    \tightlist
    \item
      Too much effort is required to create all possible rules
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Theoretical ignorance}}

    \begin{itemize}
    \tightlist
    \item
      Science doesn't always have a complete theory of a domain
    \item
      E.g., medical science doesn't know all the ``rules''
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Practical ignorance}}

    \begin{itemize}
    \tightlist
    \item
      Even if you knew all the rules, you might not have all the
      necessary information
    \item
      E.g., not all required medical tests can be performed for a
      particular patient
    \end{itemize}
  \end{enumerate}
\item
  \textbf{\textcolor{cyan}{AI winter (mid-1980s, 1990s)}} due to expert
  systems failure

  \begin{itemize}
  \tightlist
  \item
    The real world is complex and open-ended
  \item
    Logical rules can't capture all necessary and sufficient conditions
  \end{itemize}
\end{itemize}

\begin{frame}{The Paradox of Probability and Knowledge}
\protect\phantomsection\label{the-paradox-of-probability-and-knowledge}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Paradox}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``There is no uncertainty in the actual world!''}
  \item
    E.g., the grass is wet, but either it has rained or not
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Knowledge is subjective}}

  \begin{itemize}
  \tightlist
  \item
    Probabilities relate to a knowledge state, not to the real world
  \item
    Updating knowledge changes probability statements
  \end{itemize}
\end{itemize}

\begin{frame}{The Paradox of Probability and Knowledge: Example}
\protect\phantomsection\label{the-paradox-of-probability-and-knowledge-example}
\begin{itemize}
\tightlist
\item
  Initially, you observe \emph{``wet grass''}

  \begin{itemize}
  \tightlist
  \item
    From past data you know that \(\Pr(Rain | WetGrass) = 0.8\)
  \item
    Current belief: \emph{``80\% chance it rained if grass is wet''}
  \item
    This is the Bayesian prior
  \end{itemize}
\item
  You further observe that \emph{``sprinkler was on''}

  \begin{itemize}
  \tightlist
  \item
    Wet grass could be due to the sprinkler, not rain
  \item
    Belief changes: \(\Pr(Rain | WetGrass \land Sprinkler) = 0.4\)
  \item
    This is a Bayesian update
  \end{itemize}
\item
  You learn from the weather report that there was no rain

  \begin{itemize}
  \tightlist
  \item
    Belief changes: \emph{``It didn't rain''}
  \item
    Evidence overrides prior:
    \(\Pr(Rain | WetGrass \land WeatherReport) = 0\)
  \end{itemize}
\item
  All the statements were true even if contradictory

  \begin{itemize}
  \tightlist
  \item
    You need non-monotonic logic
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Full Joint Probability Distribution}
\protect\phantomsection\label{full-joint-probability-distribution}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}

  \begin{itemize}
  \tightlist
  \item
    Consider a set of random variables \(X_1, X_2, \dots, X_n\)
  \item
    The \emph{full joint probability distribution} assigns a probability
    to every possible world: \[
    \Pr(X_1 = x_1, X_2 = x_2, \dots, X_n = x_n)
    \] where a \textbf{\textcolor{orange}{possible world}} is a
    particular assignment of values to all variables
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Fact}}: full joint probability can
  answer any probabilistic query about the domain
\item
  \textbf{\textcolor{teal}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Size grows exponentially \(k^n\) with the number of variables \(n\)
    and number of values \(k\)
  \item
    Impractical for real-world problems with many variables
  \item
    Manually specifying each entry is tedious or impossible
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Remark}}: (conditional and absolute)
  independence simplifies modeling

  \begin{itemize}
  \tightlist
  \item
    In the real world, many variables are not fully dependent on all
    others
  \item
    Reduces the number of variables needed in the model
  \item
    Makes compact and structured representations possible

    \begin{itemize}
    \tightlist
    \item
      E.g., factorized probabilistic models, Bayesian networks
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Independence of Random Variables: Definition}
\protect\phantomsection\label{independence-of-random-variables-definition}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: two random variables \(X\) and
  \(Y\) are \textbf{\textcolor{darkgreen}{independent}} iff: \[
  \Pr(X, Y) = \Pr(X) \cdot \Pr(Y)
  \]
\item
  \textbf{\textcolor{blue}{Equivalently}} knowing \(Y\) tells nothing
  about \(X\) \[
  \Pr(X | Y) = \Pr(X)
  \]

  \begin{itemize}
  \tightlist
  \item
    E.g., the events ``coin flip'' and ``weather'' are independent \[
      \Pr(Coin=Heads | Weather=Rainy) = \Pr(Coin=Heads)
      \]
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Consequence}}: Independence of random
  variables

  \begin{itemize}
  \tightlist
  \item
    Reduces the number of parameters needed to model a system, e.g., \[
    \Pr(X_1 | X_2, X_3) = \Pr(X_1)
    \]
  \item
    Allows factorization of joint distribution, e.g., \[
    \Pr(X_1, X_2, X_3) = \Pr(X_1) \cdot \Pr(X_2) \cdot \Pr(X_3)
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Conditional Independence: Definition}
\protect\phantomsection\label{conditional-independence-definition}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: two random variables \(X\) and
  \(Y\) are \emph{conditionally independent} given a random variable
  \(Z\) iff knowing \(Z\) makes \(X\) and \(Y\) independent: \[
  X \perp Y | Z \iff \Pr(X, Y | Z) = \Pr(X | Z) \cdot \Pr(Y | Z)
  \]
\item
  \textbf{\textcolor{darkgreen}{Example}}

  \begin{itemize}
  \tightlist
  \item
    \(X\) = \emph{``it is raining today''}
  \item
    \(Y\) = \emph{``a person is carrying an umbrella''}
  \item
    \(Z\) = \emph{``the weather forecast''}
  \item
    \emph{Without knowing \(Z\)}: there is a relationship between rain
    \(X\) and umbrellas \(Y\)
  \item
    \emph{Given / knowing \(Z\)}: once the forecast is known \(Z\),
    knowing whether someone carries an umbrella \(Y\) gives no
    information about rain \(X\)
  \item
    Thus, \(X\) and \(Y\) are conditionally independent given \(Z\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Conditional independence more common than absolute independence
  \item
    Simplify probabilistic models by factorizing the joint conditional
    distribution into product of individual conditional distributions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Conditional Independence: Example}
\protect\phantomsection\label{conditional-independence-example}
\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\item
  Two events can become independent once we know a third event
\item
  Consider:

  \begin{itemize}
  \tightlist
  \item
    \(Fire\) = \emph{``there is a fire in the house''}
  \item
    \(Toast\) = \emph{``someone burned toast''}
  \item
    \(Alarm\) = \emph{``the fire alarm rings''}
  \item
    \(Call\) = \emph{``a friend calls to check if everything is ok''}
  \end{itemize}
\item
  \textbf{\textcolor{red}{Dependencies}}

  \begin{itemize}
  \tightlist
  \item
    \(Alarm\) depends on \(Fire\) or \(Toast\)
  \item
    \(Call\) depends on whether \(Alarm\) rings
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.3\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.2.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Conditional independence}}

  \begin{itemize}
  \tightlist
  \item
    \(Call\) is conditionally independent of \(Fire\) and \(Toast\)

  \item

\begin{frame}{CI: Garden World Example}

\begin{frame}{Bayesian Networks: Definition}
\protect\phantomsection\label{bayesian-networks-definition}
\begin{itemize}
\tightlist
\item
  Aka:

  \begin{itemize}
  \tightlist
  \item
    ``Bayes nets'',
  \item
    ``Belief networks''
  \item
    ``Probabilistic networks''
  \item
    ``Graphical models'' (somehow a broader class of statistical models)
  \item
    ``Causal networks'' (arrows have special meaning)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: A \emph{Bayesian network} is a
  Directed Acyclic Graph (DAG)

  \begin{enumerate}
  \tightlist
  \item
    \emph{Nodes} \(X_i\) correspond to random variables (discrete or
    continuous)
  \item
    \emph{Edges} \(X \to Y\) connect nodes and represent dependencies
    among variables

    \begin{itemize}
    \tightlist
    \item
      E.g., \(X = Parent(Y)\), \(Y = Child(X)\), ancestors, spouses,
      \ldots{}
    \end{itemize}
  \item
    Each node \(X_i\) is associated with a \emph{conditional probability
    table} (CPT or CPD): \[
    \Pr(X_i | Parents(X_i))
    \] quantifying the effect of the parents on the node

    \begin{itemize}
    \tightlist
    \item
      If a node has no parents, it has a prior probability
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Network: Intuition}
\protect\phantomsection\label{bayesian-network-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Remark}}: Bayesian networks are the analogous
  for uncertain knowledge to propositional logic for definite knowledge

  \begin{itemize}
  \tightlist
  \item
    \emph{Propositional logic} = rigid rules, i.e.,
    \textcolor{blue}{\texttt{True}} or \textcolor{blue}{\texttt{False}}
  \item
    \emph{Bayesian networks} = flexible, degrees of belief
  \item
    Replace \(X \implies Y\) with \(\Pr(Y | X)\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}: garden world example

  \begin{itemize}
  \tightlist
  \item
    \(R\) = ``It is raining''
  \item
    \(W\) = ``The grass is wet''
  \item
    Propositional logic

    \begin{itemize}
    \tightlist
    \item
      If \(R \rightarrow W\) and \(R\) is true, then \(W\) must be true
    \end{itemize}
  \item
    Bayesian network

    \begin{itemize}
    \tightlist
    \item
      \(\Pr(R = True) = 0.2\)
    \item
      \(\Pr(W | R) = 0.9\)
    \item
      \(\Pr(W | \neg R) = 0.1\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: medical diagnosis

  \begin{itemize}
  \tightlist
  \item
    Propositional logic

    \begin{itemize}
    \tightlist
    \item
      \emph{``Patient has disease \(D\)'' \(\implies\) ``Patient has
      symptom \(S\)''}
    \end{itemize}
  \item
    Bayesian network

    \begin{itemize}
    \tightlist
    \item
      \emph{``Probability of \(S\) given \(D\) is high, but not
      certain''}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Network and Full-joint distribution}
\protect\phantomsection\label{bayesian-network-and-full-joint-distribution}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Theorem}}: it can be shown that
  \emph{topology} and \emph{conditional probabilities} are sufficient to
  specify any full joint distribution, very concisely (often)
\item
  The topology of the network (nodes and edges) specifies conditional
  independence relationships

  \begin{itemize}
  \tightlist
  \item
    E.g., \(X \to Y\) means \emph{``\(X\) has a direct influence on
    \(Y\)''}, i.e., \emph{``\(X\) relates to \(Y\)''} (not necessarily
    \emph{\(X\) causes \(Y\)})
  \end{itemize}
\item
  Nodes are:

  \begin{itemize}
  \tightlist
  \item
    Directly influenced by their parents
  \item
    Indirectly influenced by all their ancestors
  \end{itemize}
\item
  \textbf{\textcolor{blue}{How to build a Bayesian Network}}:

  \begin{itemize}
  \tightlist
  \item
    Domain experts can decide what relationships exist among domain
    variables, determining the topology
  \item
    Conditional probabilities can be specified or estimated
  \end{itemize}
\end{itemize}
\end{frame}

  \end{itemize}

\begin{itemize}
\tightlist
\item
  \(Rain\) and \(Sprinkler\) are
  \textbf{\textcolor{blue}{conditionally independent}} given \(Weather\)
  \[Rain \perp Sprinkler | Weather\]

  \begin{itemize}
  \tightlist
  \item
    Once \(Weather\) is fixed, knowing whether it rained tells you
    nothing about the \(Sprinkler\)
  \item
    The correlation is ``explained'' by \(Weather\) (confounder)
  \end{itemize}

  \end{itemize}

  \begin{itemize}

\begin{frame}{Conditional Probability Table}
\protect\phantomsection\label{conditional-probability-table}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: \emph{Conditional Probability
  Table} (CPT) encodes the probability of one node \(X_i\) given its
  parents \(Parents(X_i)\) \[
  \Pr(X_i | Parents(X_i))
  \]
\item
  Each row of the CPT contains the conditional probability of the node
  under a conditioning case (i.e., a possible combination of the values
  for the parent nodes)
\item
  \textbf{\textcolor{blue}{Note}}:

  \begin{itemize}
  \tightlist
  \item
    Natural for discrete variables, but can be extended to continuous
    variables
  \item
    A conditional probability table summarizes an infinite set of
    circumstances in the table
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Conditional Probability Table: Example}
\protect\phantomsection\label{conditional-probability-table-example}
\begin{itemize}
\tightlist
\item
  Consider the events:

  \begin{itemize}
  \tightlist
  \item
    \(T\) = ``Traffic Jam''
  \item
    \(S\) = ``Snow''
  \item
    \(R\) = ``Rush Hour''
  \end{itemize}
\item
  The CPT models
  \(\Pr(\text{Traffic Jam} | \text{Snow}, \text{Rush Hour}) = \Pr(T | S, R)\)
\end{itemize}

\begingroup \scriptsize

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[]{@{}lllll@{}}
\toprule\noalign{}
T & P(T\textbar S,R) & P(T\textbar S,-R) & P(T\textbar-S,R) &
P(T\textbar-S,-R) \\
\midrule\noalign{}
\endhead
True & 0.80 & 0.10 & 0.50 & 0.05 \\
False & 0.20 & 0.90 & 0.50 & 0.95 \\
\bottomrule\noalign{}
\end{longtable}
}

\endgroup

\begin{itemize}
\tightlist
\item
  As for any conditional probability

  \begin{itemize}
  \tightlist
  \item
    The sum of probabilities on the left side of the conditioning is 1
  \item
    Not on the right side
  \end{itemize}
\item
  \textbf{\textcolor{red}{Interpretation}}

  \begin{itemize}
  \tightlist
  \item
    Traffic is highly likely when there is both snow and rush hour
  \item
    Traffic is unlikely when neither condition holds
  \end{itemize}
\end{itemize}
\end{frame}
