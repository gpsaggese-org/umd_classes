\begin{frame}{Naive Bayes Model}
\protect\phantomsection\label{naive-bayes-model}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}: predict classes \(H_1, ..., H_n\)
  using evidence \(\vE\):

  \begin{itemize}
  \tightlist
  \item
    Use Bayes' rule of conditional probability to decide output class
    \(H_1, ... H_n\) given evidence \(\vE\): \[
    \Pr(H_j|\vv{E}) = \frac{\Pr(\vv{E}|H_j) \Pr(H_j)}{\Pr(\vv{E})}
    \] where \(\vv{E}\) is the vector of features
  \item
    Training data estimates joint probability \(\Pr(H_i, \vE)\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Assumptions:}}

  \begin{enumerate}
  \tightlist
  \item
    Features are independent
  \item
    Features are equally important (or at least all relevant)
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Naive Bayes model:}}

  \begin{itemize}
  \tightlist
  \item
    Called ``naive'' due to the simplifying assumption of independence,
    even if not true
  \item
    Works surprisingly well
  \end{itemize}
\end{itemize}
\end{frame}

\item
  Use Bayes' rule to decide class \(H_j\): \[
  \Pr(H_j|\vv{E}) = \frac{\Pr(\vv{E}|H_j) \Pr(H_j)}{\Pr(\vv{E})}
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \(H_j\): event to predict

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{play = yes}}
    \end{itemize}
  \item
    \(\vv{E}\): event with feature values

    \begin{itemize}
    \tightlist
    \item
      E.g.,
      \textcolor{blue}{\texttt{outlook=sunny, temperature=high, humidity=high, windy=yes}}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Naive Bayes: Weather Prediction Example}
\protect\phantomsection\label{naive-bayes-weather-prediction-example-2}
\begin{itemize}
\tightlist
\item
  The \textbf{\textcolor{red}{model}} is: \begingroup \scriptsize
  $$
  \Pr(H_j|\vv{E}) = \frac{\Pr(\vv{E}|H_j) \Pr(H_j)}{\Pr(\vv{E})}
  $$
  \endgroup where:

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(H_j)\): \textbf{\textcolor{orange}{prior probability}}
    (probability of the outcome before evidence \(\vv{E}\))

    \begin{itemize}
    \tightlist
    \item
      E.g., \(\Pr(\text{play = yes})\)
    \item
      Computed from training set as: \begingroup \scriptsize
      $\Pr(H_j) = \sum_{k=1}^N \Pr(H_j \land \vv{E}_k)$
      \endgroup
    \end{itemize}
  \item
    \(\Pr(\vv{E})\):
    \textbf{\textcolor{darkgreen}{probability of the evidence}}

    \begin{itemize}
    \tightlist
    \item
      Computed from training set
    \item
      Not needed as it is common to the probability of each class
    \end{itemize}
  \item
    \(\Pr(\vv{E}|H_j)\):
    \textbf{\textcolor{teal}{conditional probability}}

    \begin{itemize}
    \tightlist
    \item
      Computed as independent probabilities (the ``naive'' assumption):
      \begingroup \scriptsize
      \begin{alignat*}{2}
      \Pr(\vv{E}|H_j)
      &= \Pr(E_1=e_1, E_2=e_2, ..., E_n=e_n | H_j) \\
      &\approx \Pr(E_1=e_1|H_j) \cdot ... \cdot \Pr(E_n=e_n|H_j) \\
      \end{alignat*}
      \endgroup
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Interpretation of Bayes theorem}}

  \begin{itemize}
  \tightlist
  \item
    The prior is modulated through the conditional probability and the
    probability of the evidence
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Naive Bayes: Why Independence Assumption}
\protect\phantomsection\label{naive-bayes-why-independence-assumption}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Independence assumption}} factors joint
  probability into marginal probabilities: \begin{alignat*}{2}
  \Pr(\vv{E}|H_j)
  &= \Pr(E_1=e_1, E_2=e_2, ..., E_n=e_n | H_j) \\
  &\approx \Pr(E_1=e_1|H_j) \cdot ... \cdot \Pr(E_n=e_n|H_j) \\
  \end{alignat*}
\item
  \textbf{\textcolor{darkgreen}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    Simplifies probability computation
  \item
    Aids generalization due to few samples needed
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}:

  \begin{itemize}
  \tightlist
  \item
    Assumption may not be true
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Estimating Probabilities: MLE}
\protect\phantomsection\label{estimating-probabilities-mle}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Maximum Likelihood Estimate}} (MLE) estimates
  event probability by counting occurrences among all possible events:
  \[
  \Pr(\vv{E} = \vv{e}') = \frac{\# I(\vv{E} = \vv{e}')}{K}
  \]
\item
  For Naive Bayes, we need to estimate probability of each feature: \[
  \Pr(E_i = e') = \frac{\# I(E_i = e')}{\sum_{j=1}^V \# I(E_i = e_k)}
  \]
\end{itemize}

\vspace{-0.5cm}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Value \(e'\) of feature \(E_i\) not in training set with output
    class \(H_j\)
  \item
    Estimated probability \(\Pr(E_i=e'|H_j) = 0\)
  \item
    Plugging \(\Pr(E_i=e'|H_j) = 0\) into \[
    \Pr(H_j|\vv{E}) \approx \Pr(E_1=e_1|H_j) \cdot ... \cdot \Pr(E_n=e_n|H_j)
    \]
  \item
    Yields \(\Pr(H_j|\vv{E}) = 0\) \(\implies\) impossible to decide
    output
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Estimating Probabilities: Laplace Estimator}
\protect\phantomsection\label{estimating-probabilities-laplace-estimator}
\begin{itemize}
\item
  Use Laplace estimator for events not in training set instead of MLE
\item
  \textbf{\textcolor{red}{Maximum likelihood estimate}} (MLE) \[
  \Pr(E_i = e') = \frac{\# I(E_i = e')}{\sum_{j=1}^V \# I(E_i = e_k)}
  \]
\item
  \textbf{\textcolor{blue}{Laplace estimator}}

  \begin{itemize}
  \tightlist
  \item
    Adds 1 to each count and \(V\) (number of feature values) to
    denominator: \[
    \Pr(E_i = e')
    = \frac{1 + \# I(E_i = e')}{\sum_{j=1}^V (1 + \# I(E_i = e'))}
    = \frac{1 + \# I(E_i = e')}{V + \sum_{j=1}^V \# I(E_i = e')}
    \]
  \item
    Blends prior of equiprobable feature values with MLE estimates
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Splitting at One Node: Problem Formulation}
\protect\phantomsection\label{splitting-at-one-node-problem-formulation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Consider the $m$-th node}} of a decision tree

  \begin{itemize}
  \tightlist
  \item
    Given \(p_i = (\vx_i, y_i)\) where \(i = 1, ..., N_m\), with
    training vectors \(\vx_i\) and labels \(y_i\)
  \item
    Candidate splits are \(\theta = (j, t_m)\) with feature \(j\) and
    threshold \(t_m\)
  \item
    Each split partitions data into subsets: \begin{alignat*}{2}
    Q_{L}(j, t_m) & = \{ p_i = (\vx_i, y_i): x_j \le t_m \} \\
    Q_{R}(j, t_m) & = \{ p_i \notin Q_{L} \} \\
    \end{alignat*}
  \item
    Impurity of a split is defined as: \[
    H(j, t_m) = \frac{n_{L}}{N_m} H(Q_{L}) + \frac{n_{R}}{N_m} H(Q_{R})
    \]
  \item
    Find split \((j, t_m)^*\) minimizing \(H(j, t_m)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Recurse}} on \(Q_{L}(j, t_m)^*\) and
  \(Q_{R}(j, t_m)^*\)
\end{itemize}
\end{frame}

\begin{frame}{Measures of Node Impurity for Classification}

\begin{frame}{Probability of Classification in a Node}
\protect\phantomsection\label{probability-of-classification-in-a-node}
\begin{itemize}
\item
  Assume \(m\)-th node has \(N_m\) objects \(x_i\) in \(K\) classes
\item
  Compute class probability in \(m\)-th node as: \[
  \hat{f}_{m, k}
  \defeq \Pr(\text{pick object of class $k$ in $m$-th node })
  = \frac{1}{N_m} \sum_{x_i \in \text{ node }} I(c(x_i) = k)
  \] where \(y_i = c(x_i)\) is the correct class for element \(x_i\)
\item
  E.g., if there are \(N_m = 10\) objects belonging to \(K=3\) classes

  \begin{itemize}
  \tightlist
  \item
    3 red, 6 blue, 1 green
  \item
    The probability of classification \(\hat{f}_{m, k}\) are:

    \begin{itemize}
    \tightlist
    \item
      Red = 3/10
    \item
      Blue = 6/10
    \item
      Green = 1/10
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Misclassification Error: Definition}
\protect\phantomsection\label{misclassification-error-definition}
\begin{itemize}
\item
  You have several class probabilities \(p_i\) and need a single
  probability

  \begin{itemize}
  \tightlist
  \item
    Consider worst case: most common class \(k'\) in node
  \end{itemize}
\item
  \textbf{\textcolor{red}{Misclassification error}} \[
  H_M(p) \defeq 1 - \max_{k}{p_k}
  \]
\item
  \textbf{\textcolor{darkgreen}{Binary classifier}}

  \begin{itemize}
  \tightlist
  \item
    Best case (perfect purity)

    \begin{itemize}
    \tightlist
    \item
      Only one class in node
    \item
      \(H_M(p) = 0\)
    \end{itemize}
  \item
    Worst case (perfect impurity)

    \begin{itemize}
    \tightlist
    \item
      50-50 split between classes in node
    \item
      \(H_M(p) = 0.5\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Multi-class classifier}} with \(K\) classes

  \begin{itemize}
  \tightlist
  \item
    Misclassification error has upper bound: \(1 / K\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Gini Impurity Index: Definition}
\protect\phantomsection\label{gini-impurity-index-definition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Gini index}} \(H_G(p)\) is probability of
  picking an element randomly and classifying it incorrectly

  \begin{itemize}
  \tightlist
  \item
    By using the law of total probability: \begingroup \small
    \begin{alignat*}{2}
    H_G(p)
    & = \Pr(\text{pick elem of $k$-th class}) \cdot
    \Pr(\text{misclassification | elem of $k$ class}) \\
    & = \sum_{k=1}^K p_k \cdot (1 - p_k)
    \end{alignat*}
    \endgroup
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Binary classifier}}

  \begin{itemize}
  \tightlist
  \item
    \(H_G(p)\) is between 0 (perfect purity) and 0.5 (perfect impurity)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Multi-class classifier}} with \(K\) classes

  \begin{itemize}
  \tightlist
  \item
    \(H_G(p)\) has upper bound:
    \(1 - K \frac{1}{K}^2 = 1 - \frac{1}{K}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Information Gain: Definition}
\protect\phantomsection\label{information-gain-definition}
\begin{itemize}
\item
  Aka cross-entropy (remember entropy is \(- p \log p\))
\item
  \textbf{\textcolor{red}{Information gain}} \[
  H_{IG} = - \sum_{k=1}^K p_k \cdot \log_2(p_k)
  \]
\item
  \textbf{\textcolor{darkgreen}{Binary classifier}}

  \begin{itemize}
  \tightlist
  \item
    \(H_{IG}\) varies between 0 (perfect purity) and 1 (perfect
    impurity)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Multi-class classifier}} with \(K\) classes

  \begin{itemize}
  \tightlist
  \item
    \(H_{IG}\) has upper bound: \(\log{K}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Measures of Impurity: Examples}
\protect\phantomsection\label{measures-of-impurity-examples}
\begin{itemize}
\item
  Consider the case of 16 elements in a node , belonging to 2 classes
\item
  If all elements are of the same class:

  \begin{itemize}
  \tightlist
  \item
    Misclassification error \(= 0\)
  \item
    Gini index \(= 1 - (1 - 0) = 0\)
  \item
    Information gain \(= - (1 \cdot \log_2(1) - 0 \cdot \log_2(0)) = 0\)
  \end{itemize}
\item
  If one element is of one class:

  \begin{itemize}
  \tightlist
  \item
    Misclassification error =
    \(1 - \max(\frac{1}{16}, \frac{1}{15}) = \frac{1}{16}\)
  \item
    Gini index \(= 1 - ((\frac{1}{16})^2 + (\frac{1}{15})^2) = 0.12\)
  \item
    Information gain \(= -(\frac{1}{16} \log_2(\frac{1}{16}) +
    \frac{15}{16} \log_2(\frac{15}{16})) = 0.34\)
  \end{itemize}
\item
  If elements are split in the two classes equally:

  \begin{itemize}
  \tightlist
  \item
    Misclassification error \(= \frac{8}{16} = 0.5\)
  \item
    Gini index \(= 1 - (\frac{8}{16}^2 + \frac{8}{16}^2) = 0.5\)
  \item
    Information gain \(= -( \frac{8}{16} \log_2(\frac{8}{16}) +
    \frac{8}{16} \log_2(\frac{8}{16})) = 1\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Measures of Impurity for Regression}
\protect\phantomsection\label{measures-of-impurity-for-regression}
\begin{itemize}
\item
  For continuous variables with \(N_m\) observations at a node, minimize
  mean squared error:

  \[
  \begin{aligned}
  & c_m = \frac{1}{N_m} \sum_{i \in N_m} y_i & \text{ (average class)}\\
  & H = \frac{1}{N_m} \sum_{i \in N_m} (y_i - c_m)^2 & \text{ (variance)}\\
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{From Decision Trees to Random Forests}
\protect\phantomsection\label{from-decision-trees-to-random-forests}
\begin{itemize}
\item
  Decision trees are \textbf{\textcolor{red}{high capacity models}}:

  \begin{itemize}
  \tightlist
  \item
    Low bias
  \item
    High variance
  \end{itemize}
\item
  Idea: apply ensemble methods to trees \(\rightarrow\) ``random
  forests''
\item
  \textbf{\textcolor{darkgreen}{Bagging}}

  \begin{itemize}
  \tightlist
  \item
    Reduces variance in ``unstable'' non-linear models

    \begin{itemize}
    \tightlist
    \item
      Learning trees is unstable
    \end{itemize}
  \item
    Best with complex models (e.g., fully grown trees)

    \begin{itemize}
    \tightlist
    \item
      Boosting works best with weak models (e.g., shallow decision
      trees, aka tree stumps)
    \end{itemize}
  \item
    Works for regression and classification
  \item
    Customizable for trees

    \begin{itemize}
    \tightlist
    \item
      Different types of randomization in trees
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bias-variance trade-off}} in random forests

  \begin{itemize}
  \tightlist
  \item
    Forest bias could increase compared to a single non-random tree
  \item
    Forest variance reduced by averaging, usually compensating for bias
    increase
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear Regression Model}
\protect\phantomsection\label{linear-regression-model}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Data set}}

  \begin{itemize}
  \tightlist
  \item
    \((\vx_1, y_1), ... , (\vx_N, y_N)\)
  \item
    \(N\) examples
  \item
    \(P\) features, \(\vx_i \in \bbR^P\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Linear regression model}} \[
  h(\vx) = \sum_{i=1}^{P} w_i x_i = \vw^T \vx \in \bbR
  \]
\item
  Add \textbf{\textcolor{blue}{bias term}} \(w_0\) to model with
  \(x_0 = 1\) to data \[
  h(\vx) = w_0 + \sum_{i=1}^{P} w_i x_i = \sum_{i=0}^{P} w_i x_i = \vw^T \vx
  \]
\end{itemize}
\end{frame}

\begin{frame}{Linear Regression: In-sample Error}
\protect\phantomsection\label{linear-regression-in-sample-error}
\begin{itemize}
\item
  For regression, use
  \textbf{\textcolor{red}{squared error for in-sample error}}: \[
  E_{in}(h) = \frac{1}{N} \sum_{i=1}^{N} (h(\vx_i) - f(\vx_i)) ^ 2
  \]
\item
  Squared error for \textbf{\textcolor{darkgreen}{linear regression}} \[
  E_{in}(h)
  = E_{in}(\vw)
  = \frac{1}{N} \sum_{i=1}^{N} (\vw^T \vx_i - y_i) ^ 2
  \]
\item
  Squared error \textbf{\textcolor{blue}{in vector form}} \[
  E_{in}(h)
  = \frac{1}{N} \| \mX \vw - \vy \| ^ 2
  = \frac{1}{N} (\mX \vw - \vy)^T (\mX \vw - \vy)
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \(\mX\) is the matrix with examples \(\vx_i^T\) as rows (``design
    matrix'')
  \item
    \(\mX\) is a tall matrix with few parameters (\(P\)) and many
    examples (\(N\))
  \item
    \(\vy\) is the column vector with all outputs (target vector)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear Regression: Find Optimal Model}
\protect\phantomsection\label{linear-regression-find-optimal-model}
\begin{itemize}
\item
  You want to minimize
  \(E_{in}(\vw) = (\mX \vw - \vy)^T (\mX \vw - \vy)\) with respect to
  \(\vw\) \begingroup \small
  \begin{alignat*}{1}
  & \nabla E_{in}(\vw^*) = \vv{0} \\
  & \frac{2}{N}\mX^T(\mX \vw^* - \vy) = \vv{0} \\
  & \mX^T \mX \vw^* = \mX^T \vy \\
  \end{alignat*}
  \endgroup
\item
  If the square matrix \(\mX^T \mX\) is invertible: \[
  \vw^* = (\mX^T \mX)^{-1} \mX^T \vy = \mX^\dagger \vy
  \]
\item
  The matrix \(\mX^\dagger \defeq (\mX^T \mX)^{-1} \mX^T\) is called
  \textbf{\textcolor{red}{pseudo-inverse}}

  \begin{itemize}
  \tightlist
  \item
    It generalizes the inverse for non-square matrices, in fact:

    \begin{itemize}
    \tightlist
    \item
      \(\mX^\dagger \mX = \mI\)
    \item
      If \(\mX\) is square and invertible, then
      \(\mX^\dagger = \mX^{-1}\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Complexity of One-Step Learning}
\protect\phantomsection\label{complexity-of-one-step-learning}
\begin{itemize}
\tightlist
\item
  Learning with the pseudo-inverse is
  \textbf{\textcolor{red}{one-step learning}}

  \begin{itemize}
  \tightlist
  \item
    Contrasts with iterative methods, e.g., gradient descent
  \end{itemize}
\item
  Inverting a square matrix of size \(P\) is related to the number of
  parameters, not examples \(N\)

  \begin{itemize}
  \tightlist
  \item
    Complexity of one-step learning is \(O(P^3)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Linear Models Are Linear in What?}
\protect\phantomsection\label{linear-models-are-linear-in-what}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{model is linear}} when the signal
  \(s = \sum_{i=0}^P w_i x_i = \vw^T \vx\) is linear
  \textbf{\textcolor{darkgreen}{with variables}}

  \begin{itemize}
  \tightlist
  \item
    Unknown variables: weights \(w_i\)
  \item
    Inputs \(x_i\) are fixed
  \end{itemize}
\item
  Applying a \textbf{\textcolor{blue}{non-linear transform to inputs}}
  \(z_i = \Phi(x_i)\) keeps the model linear, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Positive/negative part (e.g., \(z_i = RELU(x_i), RELU(-x_i)\))
  \item
    Waterfall (conditioning model to different feature ranges)
  \item
    Thresholding (e.g., \(z_i = \min(x_i, T)\))
  \item
    Indicator variables (e.g., \(z_i = I(x_i > 0)\))
  \item
    Winsorizing (replace outliers with a large constant value)
  \end{itemize}
\item
  Applying a
  \textbf{\textcolor{violet}{non-linear transform to weights}}
  \(z_i = \Phi(w_i)\) makes the model non-linear
\end{itemize}
\end{frame}

\begin{frame}{Non-Linear Transformations in Linear Models}
\protect\phantomsection\label{non-linear-transformations-in-linear-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Transform variables}}

  \begin{itemize}
  \tightlist
  \item
    Use \(\Phi: \calX \rightarrow \calZ\)
  \item
    Transform each point \(\vx_n \in \calX = \bbR^d\) into a point in
    feature space \(\vz_n = \Phi(\vx_n) \in \calZ = \bbR^{\tilde{d}}\)
    with \(d \ne \tilde{d}\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Learn}}

  \begin{itemize}
  \tightlist
  \item
    Learn linear model in \(\calZ\), obtaining \(\vv{\tilde{w}}\) for
    separating hyperplane
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Predict}}

  \begin{itemize}
  \tightlist
  \item
    Evaluate model on new point in \(\calZ\): \[
    y = \sign(\vv{\tilde{w}}^T \Phi(\vx))
    \text{or} \quad y = \vv{\tilde{w}}^T \Phi(\vx)
    \]
  \end{itemize}
\item
  Compute \textbf{\textcolor{violet}{decision boundary}}

  \begin{itemize}
  \tightlist
  \item
    In \(\calX\) if \(\Phi\) is invertible; or
  \item
    By classifying any \(\vx \in \calX\) in \(\calZ\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Perceptron Learning Algorithm (PLA)}
\protect\phantomsection\label{perceptron-learning-algorithm-pla}
\begin{itemize}
\item
  First machine learning algorithm discovered
\item
  \textbf{\textcolor{red}{Algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Training set \(\calD = \{(\vx_1, y_1), ..., (\vx_n, y_n) \}\)
  \item
    Initialize weights \(\vw\)

    \begin{itemize}
    \tightlist
    \item
      Random values
    \item
      Use linear regression for classification as seed
    \end{itemize}
  \item
    Pick a misclassified point \(\sign(\vw^T \vx_i) \ne y_i\) from
    training set \(\calD\)
  \item
    Update weights: \(\vw(t + 1) = \vw(t) + y_i \vx_i\)

    \begin{itemize}
    \tightlist
    \item
      Like stochastic gradient descent
    \end{itemize}
  \item
    Iterate until no misclassified points
  \end{itemize}
\item
  Algorithm \textbf{\textcolor{darkgreen}{converges}} (slowly) for
  linearly separable data
\item
  \textbf{\textcolor{blue}{Pocket version of PLA}}

  \begin{itemize}
  \tightlist
  \item
    Idea

    \begin{itemize}
    \tightlist
    \item
      Continuously update solution
    \item
      Keep best solution ``in the back pocket''
    \end{itemize}
  \item
    Have a solution if stopping after max iterations
  \end{itemize}
\end{itemize}
\end{frame}
