\begin{frame}{What is probability?}
\protect\phantomsection\label{what-is-probability}
\begin{itemize}
\tightlist
\item
  Probability is the mathematical language for quantifying uncertainty

  \begin{itemize}
  \tightlist
  \item
    It provides a framework for understanding and modeling the
    likelihood of different outcomes
  \item
    E.g., the probability of flipping a coin and it landing on heads is
    0.5
  \end{itemize}
\item
  Probability lets you imagine possible universes and quantify their
  probability

  \begin{itemize}
  \tightlist
  \item
    Each universe represents a different possible outcome / scenario
  \item
    Assign probabilities to these universes based on available
    information or assumptions
  \item
    E.g., in a simple dice game

    \begin{itemize}
    \tightlist
    \item
      Assuming a fair die
    \item
      There are 6 universes where you roll a 1, 2, \ldots, 6 on a die
    \item
      Each universe has a probability of 1/6
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sample outcome and sample space}
\protect\phantomsection\label{sample-outcome-and-sample-space}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Sample space}} \(\Omega\) is the set of all
  possible outcomes of a random experiment, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Toss a coin once: \(\Omega = \{H, T\}\)
  \item
    Toss a die: \(\Omega = \{1, 2, 3, 4, 5, 6\}\)
  \item
    Toss a coin twice: \(\Omega = \{HH, HT, TH, TT\}\)
  \item
    Measure bulb lifetime in hours: real number in \(0 \le 4000\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Sample outcome}} is the realization of an
  experiment

  \begin{itemize}
  \tightlist
  \item
    E.g., toss a coin twice

    \begin{itemize}
    \tightlist
    \item
      Sample outcomes are \(\omega \in \Omega = \{HH, HT, TH, TT\}\)
    \end{itemize}
  \end{itemize}
\item
  The same experiment can be represented in different sample spaces
  \(\Omega\)

  \begin{itemize}
  \tightlist
  \item
    E.g., toss a coin twice, possible sample spaces are:

    \begin{itemize}
    \tightlist
    \item
      \(\Omega = \{HH, HT, TH, TT\}\)
    \item
      \(\Omega = \{\text{Results from first and second toss are the same or not}\}\)
    \item
      \(\Omega = \{\text{The first toss is H}\}\)
    \item
      \(\Omega = \{\text{There is only one H, or not}\}\)
    \item
      \(\Omega = \{\text{Number of heads}\} = \{0, 1, 2\}\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Event}
\protect\phantomsection\label{event}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Event}} is a subset of the sample space
  \(\Omega\)

  \begin{itemize}
  \tightlist
  \item
    Combine outcomes \(\omega \in \Omega\) of an experiment that
    interest us
  \end{itemize}
\item
  \textbf{\textcolor{blue}{The event $A$ happened}} when the outcome
  \(\omega\) belongs to \(A \subseteq \Omega\), e.g.,

  \begin{itemize}
  \tightlist
  \item
    Toss a coin twice
  \item
    Sample space: \(\Omega = \{TT, TH, HT, HH\}\)
  \item
    Event \emph{``the first toss is heads''} is \(A = \{HH, HT\}\)
  \end{itemize}
\item
  Interesting events

  \begin{itemize}
  \tightlist
  \item
    Impossible event: \(\emptyset\)
  \item
    Certain event: \(\Omega\)
  \item
    Complement of event \(A\): \(- A = \Omega - A\)
  \item
    Outcomes in \(A\) but not in \(B\): \(A - B = A \cap (\Omega - B)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Event space \(\calF\)}
\protect\phantomsection\label{event-space-calf}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Event space}} \(\calF\) is the set of all
  possible events in \(\Omega\), i.e., set of subsets of sample space
  \(\Omega\)\\
  \[\calF \defeq \calP(\Omega) = \{ A: A \subseteq \Omega \}\]
\item
  \textbf{\textcolor{blue}{Sample space vs event space}}

  \begin{itemize}
  \tightlist
  \item
    Sample space \(\Omega\) contains all possible outcomes of a random
    experiment
  \item
    Events \(\calF\) are subsets of the possible outcomes of the
    experiment
  \item
    ``Sample space vs event space'' provides flexibility to formulate
    the problem for better resolution/understanding

    \begin{enumerate}
    \tightlist
    \item
      Define sample space to include outcomes that already encode
      interesting events
    \item
      Define outcomes at maximum granularity (E.g., sample space is
      \(\{HHHH, HHHT, ..., TTTT\}\)) and then combine outcomes in events
    \end{enumerate}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Properties of event space}
\protect\phantomsection\label{properties-of-event-space}
\begin{itemize}
\tightlist
\item
  An event space \(\calF\) must have these properties to define a
  probability function:

  \begin{enumerate}
  \tightlist
  \item
    Impossible event is included: \(\emptyset \in \calF\)
  \item
    Closed under complement:\\
    \(A \in \calF \implies \Omega - A \in \calF\)
  \item
    Closed under \emph{finite} union:\\
    \(A_1, ..., A_n \in \calF \implies \bigcup_i A_i \in \calF\)
  \end{enumerate}
\item
  \textbf{\textcolor{red}{Note}}: Using \(\calF = \calP(\Omega)\)
  ensures these properties
\end{itemize}
\end{frame}

\begin{frame}{Two mutually exclusive events}
\protect\phantomsection\label{two-mutually-exclusive-events}
\begin{itemize}
\item
  Aka disjoint, distinct
\item
  Two events \(A\), \(B\) are mutually exclusive \(\iff\)
  \(A \cap B = \emptyset\)

  \begin{itemize}
  \tightlist
  \item
    The events cannot happen at the same time
  \item
    No outcome can belong to both
  \end{itemize}
\item
  E.g., roll a die

  \begin{itemize}
  \tightlist
  \item
    Sample space \(\Omega = \{1, 2, 3, 4, 5, 6\}\)
  \item
    Events \(A = \{1, 2\}\) and \(B = \{3, 4\}\) are mutually exclusive
  \item
    Events ``odd number'' and ``even number'' are mutually exclusive
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Mutually vs pairwise exclusive}
\protect\phantomsection\label{mutually-vs-pairwise-exclusive}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  Two exclusive events can't happen at the same time
\item
  It's not obvious what it means that \emph{several} events are
  \emph{exclusive}
\item
  \textbf{\textcolor{red}{Pairwise exclusive events:}}

  \begin{itemize}
  \tightlist
  \item
    Events \(A_1, A_2, \ldots, A_n\) are pairwise exclusive \(\iff\)
    \(A_i \cap A_j = \emptyset \; \forall i \ne j\)
  \item
    Any pair of events has no intersection
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Mutually exclusive events:}}

  \begin{itemize}
  \tightlist
  \item
    Events \(A_1, A_2, \ldots, A_n\) are mutually exclusive \(\iff\)
    \(\bigcap_i A_i = \emptyset\)
  \item
    All events have no intersection
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Mutually exclusive as Venn diagrams}
\protect\phantomsection\label{mutually-exclusive-as-venn-diagrams}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  Pairwise exclusive is a stronger property than mutually exclusive:

  \begin{itemize}
  \tightlist
  \item
    Any pair of events has no intersection (pairwise exclusive) implies
    all events have no intersection (mutually exclusive)

    \begin{itemize}
    \tightlist
    \item
      The reverse is not true
    \end{itemize}
  \item
    E.g., 3 events \(A, B, C\) can have no common element, but \(A\) and
    \(B\) can have a non-empty intersection
  \end{itemize}
\item
  Pairwise exclusive sets mean any event does not overlap with any other
  event: they are separated
\item
  Mutual, but not pairwise, exclusivity means there is a chain of sets
  without a single intersection
\end{itemize}

\begin{frame}{Partition of \(\Omega\)}
\protect\phantomsection\label{partition-of-omega}
\begin{itemize}
\tightlist
\item
  Partition is a sequence of sets \(A_1, A_2, ...\) such that:

  \begin{itemize}
  \tightlist
  \item
    Union is \(\Omega\): \(\cup A_i = \Omega\)
  \item
    Pairwise exclusive: \(A_i \cap A_j = \emptyset\)
  \end{itemize}
\item
  Monotone increasing \(\iff A_1 \subseteq A_2 \subseteq ...\)

  \begin{itemize}
  \tightlist
  \item
    We define \(A_n \rightarrow A\) as:
    \[\lim_{n \to \infty} A_n = \cup A_i = A\]
  \end{itemize}
\item
  Monotone decreasing \(\iff A_1 \supseteq A_2 \supseteq ...\)

  \begin{itemize}
  \tightlist
  \item
    We define \(A_n \rightarrow A\) as:
    \[\lim_{n \to \infty} A_n = \cap A_i = A\]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Probability measure}
\protect\phantomsection\label{probability-measure}
\begin{itemize}
\tightlist
\item
  A probability measure is a function:
  \[\Pr: \text{ event space } \calF \rightarrow \bbR\] satisfying the 3
  axioms:

  \begin{itemize}
  \tightlist
  \item
    Probability is non-negative: \(\Pr(A) \ge 0\)
  \item
    Probability of \emph{the certain event} is 1: \(\Pr(\Omega) = 1\)
  \item
    Probability of a \emph{finite} union of disjoint events is the sum
    of their probabilities: if \(A_1, ..., A_n\) are pairwise disjoint
    events, then \(\Pr(\cup A_i) = \sum_{i} \Pr(A_i)\)
  \end{itemize}
\item
  Probability is defined:

  \begin{itemize}
  \tightlist
  \item
    On the event space \(\calF = \calP(\Omega)\)
  \item
    Not on the sample space \(\Omega\) (i.e., set of all realizations of
    an experiment)
  \end{itemize}
\item
  This ensures the 3 properties of the event space \(\calF\) hold
\item
  A probability measure associates a probability with each ``possible
  world''
\end{itemize}
\end{frame}

\begin{frame}{Set operations on events and probability measure}
\protect\phantomsection\label{set-operations-on-events-and-probability-measure}
\begin{itemize}
\tightlist
\item
  For any events \(A, B \in \calF\)

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(\emptyset) = 0\)
  \item
    \(0 \le \Pr(A) \le 1\)
  \item
    \(\Pr(-A) = \Pr(\Omega - A) = 1 - \Pr(A)\)
  \item
    \(A \subseteq B \implies \Pr(A) \leq \Pr(B)\)
  \item
    \(A \subseteq B \implies \Pr(B - A) = \Pr(B) - \Pr(A)\)
  \item
    \(\Pr(A \cap B) \le \min(\Pr(A), \Pr(B))\)
  \item
    \(\Pr(A) = \Pr(A \cap B) + \Pr(A \cap -B)\)
  \item
    \(\Pr(A \cup B) = \Pr(A) + \Pr(B) - \Pr(A \cap B)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Union upper / lower bound}
\protect\phantomsection\label{union-upper-lower-bound}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Union upper bound}}
  \[\Pr(A \cup B) \le \Pr(A) + \Pr(B)\]

  \begin{itemize}
  \tightlist
  \item
    Consequence of:

    \begin{itemize}
    \tightlist
    \item
      \(\Pr(A \cup B) = \Pr(A) + \Pr(B) - \Pr(A \cap B)\)
    \item
      \(\Pr(\cdot) \ge 0\)
    \end{itemize}
  \item
    Useful to upper bound probability when \(\Pr(A \cap B)\) is unknown
    or hard to compute
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Union lower bound}}
  \[\Pr(A \cup B) \ge \max(\Pr(A), \Pr(B))\]

  \begin{itemize}
  \tightlist
  \item
    From the relationship between probability of union and intersection
    of \(A\) and \(B\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Intersection bound}} \[
  \Pr(A \cap B) \le \min(\Pr(A), \Pr(B))
  \]

  \begin{itemize}
  \tightlist
  \item
    From the relationship between probability of union and intersection
    of \(A\) and \(B\),
    \(\Pr(A \cap B) = \Pr(A) + \Pr(B) - \Pr(A \cup B) \le \Pr(A) + \Pr(B)\)
  \item
    This is the dual of \(\Pr(A \cup B) \ge \max(\Pr(A), \Pr(B))\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Two independent events}
\protect\phantomsection\label{two-independent-events}
\begin{itemize}
\tightlist
\item
  Two events \(A\) and \(B\) are independent \(\iff\)
  \[\Pr(A \cap B) = \Pr(A) \cdot \Pr(B)\]

  \begin{itemize}
  \tightlist
  \item
    In words, the probability of the intersection of independent events
    is the product of the probabilities of the events
  \item
    Aka ``multiplication rule''
  \end{itemize}
\item
  Complement of independent events are independent:
  \[\Pr(\neg (A \cup B)) = \Pr(\neg A) \Pr(\neg B)\]
\end{itemize}
\end{frame}

\begin{frame}{Exclusive vs independent events}
\protect\phantomsection\label{exclusive-vs-independent-events}
\begin{itemize}
\item
  Mutually exclusive (aka disjoint, distinct):
  \[\Pr(A \cup B) = \Pr(A) + \Pr(B) \text{   (addition rule)}\]
\item
  Independent:
  \[\Pr(A \cap B) = \Pr(A) \cdot \Pr(B) \text{   (multiplication rule)}\]
\item
  Two mutually exclusive \(A\) and \(B\) with both non-null probability
  cannot be independent

  \begin{itemize}
  \tightlist
  \item
    In fact if \(\Pr(A) > 0\) and \(\Pr(B) > 0\), then
    \(\Pr(A \cap B) = \Pr(\emptyset) = 0 \ne \Pr(A) \cdot \Pr(B)\)
  \item
    In words, typical exclusive events are not independent
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Set of mutually / pairwise independent events}
\protect\phantomsection\label{set-of-mutually-pairwise-independent-events}
\begin{itemize}
\item
  A finite set of events \(\{A_i: i \in I\}\) is
  \textbf{\textcolor{red}{mutually independent}} \(iff\)
  \[\Pr(\cap_k A_k) = \prod_k \Pr(A_k)\]

  \begin{itemize}
  \tightlist
  \item
    The probability of every subset of events can be factored into the
    product of the probabilities
  \item
    Mutual independence \(\iff\) each event is independent from any
    intersection of a subset of the remaining events
  \end{itemize}
\item
  A finite set of events \(\{A_i: i \in I\}\) is
  \textbf{\textcolor{blue}{pairwise independent}} \(iff\)
  \[\Pr(A_i \cap A_j) = \Pr(A_i) \Pr(A_j) \; \forall i, j \in I, i \ne j\]
  E.g., a pair of events are independent
\item
  For more than 2 events, mutually independent events \(\implies\)
  pairwise independent events, but the converse is not true

  \begin{itemize}
  \tightlist
  \item
    Note: this is the opposite relationship with respect to mutually vs
    pairwise exclusivity since pairwise exclusive is stronger than
    mutually exclusive
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Probabilistic Principle of Inclusion-Exclusion}
\protect\phantomsection\label{probabilistic-principle-of-inclusion-exclusion}
\begin{itemize}
\item
  Aka PPIE \[\Pr(A \cup B) = \Pr(A) + \Pr(B) - \Pr(A \cap B)\]
\item
  \textbf{\textcolor{red}{E.g.,}}

  \begin{itemize}
  \tightlist
  \item
    Probability of throwing two dice and getting at least one 6
  \item
    Interpret ``union'' as ``at least one'' 6 and use the complement:
    \[\Pr(\text{one 6}) = \Pr(\text{at least one 6}) = 1 - \Pr(\text{no 6})\]
  \item
    Or use PPIE and independence: \begin{align*} &\Pr(A=6 \cup B=6) \\
    &\hspace{1cm} = \Pr(A=6) + \Pr(B=6) - \Pr(A=6 \cap B=6) \\ &\hspace{1cm} =
    \Pr(6) + \Pr(6) - \Pr(A=6) \cdot \Pr(B=6) \\ \end{align*}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{PPIE with N events}
\protect\phantomsection\label{ppie-with-n-events}
\begin{itemize}
\item
  The probability of the union of \(n\) events \(\Pr(\cup A_i)\) is the
  sum / subtraction of the probability of intersection of all subsets of
  events \[
  \Pr(\cup A_i) =
  \sum_{i=1}^k (-1)^{k+1} (\sum_{1 < i_1 \le ... < i_k \le n}
  \Pr(A_{i_1} \cap ... \cap A_{i_k}))
  \]
\item
  PPIE for 3 events: \begin{align*}
  &\Pr(A \cup B \cup C) = \\
  &\hspace{1cm} \Pr(A) + \Pr(B) + \Pr(C) -\\
  &\hspace{1cm} (\Pr(A \cap B) + \Pr(A \cap C) + \Pr(B \cap C)) + \\
  &\hspace{1cm} \Pr(A \cap B \cap C) \\
  \end{align*}
\end{itemize}
\end{frame}

\begin{frame}{Conditional probability}
\protect\phantomsection\label{conditional-probability}
\begin{itemize}
\item
  Given an event \(B\) with \(\Pr(B) > 0\), the conditional probability
  of \(A\) given \(B\) is:
  \[\Pr(A|B) \defeq \frac{\Pr(A \cap B)}{\Pr(B)}\]
\item
  Conditional probability \(\Pr(\cdot |B)\) is a probability:

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(B) > 0\) since conditioning to an event that cannot happen is
    undefined, like \(\frac{0}{0}\)
  \item
    It can be proved that it verifies the 3 axioms of a probability
    measure

    \begin{itemize}
    \tightlist
    \item
      E.g., \(\Pr(A \cap B) \le \Pr(B)\), so \(\Pr(A | B) \le 1\)
    \end{itemize}
  \item
    The rules of probability apply to events left of the bar, but not
    right of the bar, \(\Pr(X|\cdot)\)

    \begin{itemize}
    \tightlist
    \item
      E.g., \(\Pr(X|A \cup B) \ne \Pr(X|A) + \Pr(X|B)\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Conditional probability: intuition}
\protect\phantomsection\label{conditional-probability-intuition}
\begin{itemize}
\item
  \(\Pr(A|B)\) is the probability of \(A\) when \(B\) has happened

  \begin{itemize}
  \tightlist
  \item
    I.e., the fraction of times that \(A\) happens when \(B\) has
    already happened
  \item
    It changes the sample space \(\Omega\) to reflect a world where
    \(B\) has happened, so we normalize by \(\Pr(B)\)
  \end{itemize}
\item
  E.g., if the probability of it raining today \(A\) is 10\%, given that
  it's cloudy \(B\), and clouds appear in 50\% of the days, then
  \(\Pr(A|B) = \frac{\Pr(A \cap B)}{\Pr(B)}\)
\item
  Sometimes it is easier to compute a probability in a world where \(B\)
  has happened

  \begin{itemize}
  \tightlist
  \item
    E.g., if you know a card drawn is a king, the probability it is a
    heart is 1/4
  \end{itemize}
\item
  Conditional probability refines predictions with new evidence
\end{itemize}
\end{frame}

\begin{frame}{Conditional probability: example}
\protect\phantomsection\label{conditional-probability-example}
What is the probability of getting 1 from a die, given that the die
yielded an odd number?

\textbf{\textcolor{red}{Solution}}

\begin{enumerate}
\tightlist
\item
  Computing the probability directly in the new world where the event
  ``die is odd'', i.e., in a different sample space \(\Omega\)

  \begin{itemize}
  \tightlist
  \item
    We get that there are 3 possible outcomes, equally probable, so
    \(\frac{1}{3}\)
  \end{itemize}
\item
  Using the definition of conditional probability, without changing the
  sample space \[
  \Pr(X = 1 | X \text{ is odd})
  \defeq \frac{\Pr(X = 1 \land X \text{ is odd})}{\Pr(X \text{ is odd})}
  = \frac{1}{6} / \frac{1}{2} = \frac{1}{3}
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Conditional probability: example}
\protect\phantomsection\label{conditional-probability-example-1}
\begin{itemize}
\tightlist
\item
  The probability that ``it is Friday and that a student is absent'' is
  0.03
\item
  Today is Friday: what is the probability that the student is absent?
\end{itemize}

\textbf{\textcolor{red}{Solution}}

\begin{itemize}
\tightlist
\item
  The answer is not \(\Pr(A \cap B)\) since that is the probability of
  both events happening at the same time, while we know that one event
  has already happened
\item
  \(A\) = ``Friday'' and \(B\) = ``student is absent''
\item
  We want to know \(\Pr(B|A) = \frac{\Pr(B \cap A)}{\Pr(A)}\) and we
  know that \(\Pr(A) = 1 / 7\) and \(\Pr(A \cap B) = 0.03\)
\end{itemize}
\end{frame}

\begin{frame}{Probability of the intersection of two non-independent
events}
\protect\phantomsection\label{probability-of-the-intersection-of-two-non-independent-events}
\begin{itemize}
\item
  It holds:
  \[\Pr(A \cap B) = \Pr(A|B) \cdot \Pr(B) = \Pr(B|A) \cdot \Pr(A)\]
\item
  This is useful to compute the probability of the intersection event
  when \(A\) and \(B\) are not independent
\item
  If they are independent the probability is factored in the product
  \[\Pr(A \cap B) = \Pr(A) \cdot \Pr(B)\]
\end{itemize}
\end{frame}

\begin{frame}{Conditional probability: example marbles}
\protect\phantomsection\label{conditional-probability-example-marbles}
\begin{itemize}
\tightlist
\item
  In a bag there are 2 blue marbles and 3 red marbles
\item
  What is the chance of drawing 2 marbles and both are blue?
\end{itemize}

\textbf{\textcolor{red}{Solution}}

\begin{enumerate}
\item
  Using conditional probability: \[
  \Pr(\text{1 B and 1 B})
  = \Pr(\text{1 B} | \text{bag}) \cdot \Pr(\text{1 B} | \text{bag - 1 B})
  = 2 / 5 \times 1 / 4 = 2 / 20 = 1 / 10
  \]
\item
  By counting:

  \begin{itemize}
  \tightlist
  \item
    There are \(5! / (3! \times 2!) = 10\) possible ways of picking 2
    marbles from a set of 2 blue and 3 red marbles (Mississippi formula)
  \item
    We are interested in only one
  \end{itemize}
\item
  Distinguish the marbles as all different (i.e., B1, B2, R1, R2, R3)

  \begin{itemize}
  \tightlist
  \item
    There are 20 possibles ways of picking the marbles (\(5 \times 4\))
    and we are interested in two permutations
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Prosecutor's fallacy}
\protect\phantomsection\label{prosecutors-fallacy}
\begin{itemize}
\tightlist
\item
  The prosecutor's fallacy represents that: \[\Pr(A|B) \ne \Pr(B|A)\]
\end{itemize}

\textbf{\textcolor{red}{Problem}}

\begin{itemize}
\tightlist
\item
  There is a medical test for a disease \(D\) which has outcomes \(+\)
  and \(-\)
\item
  The test is fairly accurate and has \(\Pr(+|D)\) (true positive rate)
  = \(\Pr(-|\overline{D})\) (true negative rate) = 90\%
\item
  You take the test and get a positive: what's the probability that you
  have the disease?
\end{itemize}

\textbf{\textcolor{blue}{Solution}}

\begin{itemize}
\tightlist
\item
  You want to know \(\Pr(D|+)\) and not \(\Pr(+|D)\) which is 90\%
\item
  In fact \(\Pr(D|+)\) depends (according to Bayes' theorem) on
  \(\Pr(+|D) = 90\%\), but also on \(\Pr(D)\) (how likely is the
  disease) and \(\Pr(+)\) (how often the machine report a positive)
\item
  E.g., if the disease is vanishingly rare \(\Pr(D) \to 0\)
\end{itemize}
\end{frame}

\begin{frame}{Prosecutor's fallacy: example}
\protect\phantomsection\label{prosecutors-fallacy-example}
\begin{itemize}
\item
  The probability that a person is Argentinian being the Pope, is not
  the probability that a person is the Pope being Argentinian
\item
  Numerically
\end{itemize}

\begingroup
\small

\[\Pr(\text{x is Pope} | \text{x is from Argentina})
  = \frac{\Pr(\text{x is Pope} \land \text{x is from Argentina})}
    {\Pr(\text{x is from Argentina})}
  = \frac{1}{47,000,000}\]
\[\Pr(\text{x is from Argentina} | \text{x is Pope})
  = \frac{\Pr(\text{x is from Argentina} \land \text{x is Pope})}
    {\Pr(\text{x is Pope})}
  = \frac{1}{1} = 1\] \endgroup
\end{frame}

\begin{frame}{Independent events and conditional probability}
\protect\phantomsection\label{independent-events-and-conditional-probability}
\begin{itemize}
\item
  \(A\) and \(B\) are independent \(\iff \Pr(A|B) = \Pr(A)\)
\item
  In words, knowing that \(B\) happened does not change the probability
  of \(A\): that's why the events are said to be ``independent''
\item
  The (less intuitive) definition
  \(\Pr(A \cap B) = \Pr(A) \cdot \Pr(B)\) is equivalent to this property
\item
  \(A\) and \(B\) are independent \(iff\)
  \[\Pr(A|B) = \Pr(A|\neg B) \land \Pr(B|A) = \Pr(B|\neg A)\]
\item
  \(A\) and \(B\) are dependent \(iff\)
  \[\Pr(A|B) \ne \Pr(A|\neg B) \lor \Pr(B|A) \ne \Pr(B|\neg A)\]
\item
  Dependent events can be contemporaneous or not, e.g.,

  \begin{itemize}
  \tightlist
  \item
    2 horses winning the same race
  \item
    2 marbles picked from the same bag one after another
  \end{itemize}
\item
  When we say ``at least one of coin flip is \ldots{}'' introduces a
  dependency between events
\end{itemize}
\end{frame}

\begin{frame}{Odds: definition}
\protect\phantomsection\label{odds-definition}
\begin{itemize}
\item
  Given an event \(A\) with a success probability \(p\) (i.e., modelled
  as a Bernoulli), the odds of \(A\) are defined as \[\frac{p}{1 - p}\]
\item
  Odds are the ratio between payoff when winning and losing
  \[\text{odds} = \frac{\text{lose payoff}}{\text{win payoff}}\] that
  makes the game fair
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    A game is fair when the expected earnings from playing it are equal
    to 0
  \item
    If odds are 1/3 one should be paid 3 times more when winning than
    when losing
  \item
    When odds are \textless{} 1 then ``the odds are against you''
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of odds}
\protect\phantomsection\label{interpretation-of-odds}
\begin{itemize}
\tightlist
\item
  Consider a game where:

  \begin{itemize}
  \tightlist
  \item
    You flip a coin with probability \(p\) of head
  \item
    If it comes up heads you win \(X > 0\), otherwise you lose \(Y > 0\)
  \end{itemize}
\item
  What is the value of \(X\) and \(Y\) for the game to be fair?
\end{itemize}

\textbf{\textcolor{red}{Solution}}

\begin{itemize}
\item
  The game is fair when expected earnings are 0: \[
  \EE[\text{earnings}] = p X - (1 - p) Y = 0
  \implies
  \frac{Y}{X} = \frac{p}{1 - p} = \text{odds}
  \]
\item
  Thus \[Y = \frac{p}{1 - p} X = \text{odds} \cdot X\]
\item
  The odds indicate how many times one should be paid in case of losing
  in a game with probability \(p\)

  \begin{itemize}
  \tightlist
  \item
    If \(p = 0.5\), then odds \(= 1\), so one should be paid the same as
    winning since the game is fair
  \item
    If \(p > 0.5\), since odds \(= \frac{1}{1 / p - 1}\), then
    \(1/p < 2\), \((1/p - 1) < 1\), and odds \(> 1\);
    \(Y = \text{odds} \cdot X > X\), i.e., you need to be paid more if
    you lose than if you win
  \item
    If \(p < 0.5\), then odds \(< 1\) and
    \(Y = \text{odds} \cdot X < X\), i.e., for a long shot you need to
    be paid more if you win, since it is unlikely to happen
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Law of total probability}
\protect\phantomsection\label{law-of-total-probability}
\begin{itemize}
\item
  Let \(A_1, ..., A_k\) be a partition of the sample space \(\Omega\)
  with \(\Pr(A_i) > 0\)
\item
  For any event \(B\): \[\Pr(B) = \sum_{i=1}^{k} \Pr(B|A_i) \Pr(A_i)\]
\item
  It expresses the probability of an event using conditional
  probabilities of events partitioning the event space \(\Omega\)

  \begin{itemize}
  \tightlist
  \item
    Computing conditional probabilities can be easier as our perspective
    changes when conditioning on an event
  \end{itemize}
\end{itemize}

\textbf{\textcolor{red}{Proof}}

\begin{itemize}
\tightlist
\item
  Define \(C_i = B \cap A_i\)
\item
  All \(C_i\) are disjoint and their union is \(B\) (i.e., \(C_i\) are a
  partition of \(B\))
\item
  By the axiom about the probability of the union of disjoint events and
  by the definition of conditional probability: \[
  \Pr(B)
  = \Pr(\cup_i (B \cap A_i))
  = \sum_i \Pr(B \cap A_i)
  = \sum_i \Pr(B | A_i) \cdot \Pr(A_i)
  \]
\end{itemize}
\end{frame}

\begin{frame}{Law of total probability for two events}
\protect\phantomsection\label{law-of-total-probability-for-two-events}
\begin{itemize}
\tightlist
\item
  For any event \(B\) with \(\Pr(B) > 0\) \[
  \begin{aligned}
  \Pr(A)
  &= \Pr(A | B) \Pr(B) + \Pr(A | \overline{B}) \Pr(\overline{B})\\
  &= \Pr(A | B) \Pr(B) + \Pr(A | \overline{B}) (1 - \Pr(B))
  \end{aligned}
  \]
\item
  So one needs 3 quantities to compute \(\Pr(A)\)

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(A|B)\)
  \item
    \(\Pr(A|\overline{B})\)
  \item
    \(\Pr(B)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayes' theorem for 2 events}
\protect\phantomsection\label{bayes-theorem-for-2-events}
\begin{itemize}
\item
  If \(\Pr(A) > 0, \Pr(B) > 0\), then \[
  \Pr(A|B)
  = \frac{\Pr(B|A) \cdot \Pr(A)}{\Pr(B)}
  \]
\item
  MEM: one multiplies by \(\Pr(A) / \Pr(B)\) as in the original formula
\item
  Bayes' theorem for 2 events allows to invert the conditioning of
  events
\item
  Using the Law of total probability, we can express everything in terms
  of the same conditional probabilities: \[
  \Pr(A|B)
  = \frac{\Pr(B|A) \cdot \Pr(A)}{\Pr(B|A) \cdot \Pr(A) +
  \Pr(B|\overline{A}) \cdot \Pr(\overline{A})}
  \]
\end{itemize}
\end{frame}

\begin{frame}{General form of Bayes' theorem}
\protect\phantomsection\label{general-form-of-bayes-theorem}
\begin{itemize}
\item
  Assume (same hypothesis of law of total probability)

  \begin{itemize}
  \tightlist
  \item
    \(A_1, ..., A_k\) be a partition of sample space \(\Omega\)
  \item
    \(\Pr(A_i) > 0\)
  \item
    \(\Pr(B) > 0\)
  \end{itemize}
\item
  Then: \[\Pr(A_i|B) = \frac{\Pr(B|A_i) \cdot \Pr(A_i)}{\Pr(B)}\]
\item
  Bayes' theorem:

  \begin{itemize}
  \tightlist
  \item
    Computes the probability of different events \(A_i\) partitioning
    \(\Omega\)
  \item
    After a given event \(B\) has happened
  \item
    In terms of the inverted conditioned probabilities \(B | A_i\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of Bayes' theorem as update of beliefs}
\protect\phantomsection\label{interpretation-of-bayes-theorem-as-update-of-beliefs}
\begin{itemize}
\tightlist
\item
  Bayes' theorem states: \[
  \red{\Pr(A_i|B)} = \frac{\green{\Pr(B|A_i)} \cdot \blue{\Pr(A_i)}}{\violet{\Pr(B)}}
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \textcolor{red}{$\Pr(A_i|B)$} = posterior probability of
    \textcolor{blue}{$A_i$}
  \item
    \textcolor{darkgreen}{$\Pr(B|A_i)$} = conditional (inverted)
    probability
  \item
    \textcolor{blue}{$\Pr(A_i)$} = prior probability of
    \textcolor{blue}{$A_i$}
  \item
    \textcolor{violet}{$\Pr(B)$} = probability of
    \textcolor{violet}{\text{$B$}}
  \end{itemize}
\item
  Bayes' theorem expresses the posterior probability of each event
  \(A_i\) using:

  \begin{itemize}
  \tightlist
  \item
    The conditional probabilities of \(B | A_i\) (known or estimated)
  \item
    The prior probability of \(A_i\), i.e., the probability before event
    \(B\)
  \item
    The probability of the ``updating event'' \(B\)
  \end{itemize}
\item
  In other words, the events of interest are:

  \begin{itemize}
  \tightlist
  \item
    \(A_i\), for which we have prior probabilities
  \item
    Then an event \(B\) occurs
  \item
    Using Bayes' theorem, we update our belief about \(A_i\) after \(B\)
    occurs
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayes' theorem + Law of total probability}
\protect\phantomsection\label{bayes-theorem-law-of-total-probability}
\begin{itemize}
\tightlist
\item
  Under the same hypothesis of Bayes' theorem: \[
  \Pr(A_i|B) =
  \frac{\Pr(B|A_i) \cdot \Pr(A_i)}{\sum_{j=1}^k \Pr(B|A_j) \cdot \Pr(A_j)}
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(A_i|B)\) = posterior probability of \(A_i\)
  \item
    \(\Pr(B|A_i)\) = conditional (inverted) probability
  \item
    \(\Pr(A_i)\) = prior probability of \(A_i\)
  \end{itemize}
\item
  All conditional probabilities on the RHS are of the same type and
  inverted
\end{itemize}
\end{frame}

\begin{frame}{Bayes' theorem vs Law of total probability}
\protect\phantomsection\label{bayes-theorem-vs-law-of-total-probability}
\begin{itemize}
\tightlist
\item
  Both Bayes' theorem and Law of total probability use:

  \begin{itemize}
  \tightlist
  \item
    An event
  \item
    A partitioning of the sample space
  \end{itemize}
\item
  Law of total probability

  \begin{itemize}
  \tightlist
  \item
    The probability of the given event \(A\) is expressed in terms of
    the probabilities of the partitioning events \(B_i\)
  \end{itemize}
\item
  Bayes' theorem

  \begin{itemize}
  \tightlist
  \item
    The probabilities of the partitioning events \(A_i\) are updated
    given an event \(B\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Making decisions using Bayes' theorem}
\protect\phantomsection\label{making-decisions-using-bayes-theorem}
\begin{itemize}
\item
  Bayes' theorem is used to make decisions (e.g., choose among outcomes)
  using an event and prior information
\item
  Important points:

  \begin{itemize}
  \tightlist
  \item
    Bayes' theorem calculates the probability of an event based on prior
    knowledge of conditions related to the event
  \item
    It applies to various fields, such as finance, healthcare, and
    machine learning, where decision-making under uncertainty is
    required
  \item
    The formula for Bayes' theorem is: \[
    \Pr(A|B) = \frac{\Pr(B|A) \cdot \Pr(A)}{\Pr(B)}
    \]
  \end{itemize}
\item
  Examples:

  \begin{itemize}
  \tightlist
  \item
    In a medical diagnosis, Bayes' theorem can be used to determine the
    probability of a disease given a positive test result, considering
    the overall prevalence of the disease and the accuracy of the test
  \item
    In spam filtering, Bayes' theorem helps decide the probability that
    an incoming email is spam based on features like word frequency
  \item
    In weather forecasting, it can update the probability of rain based
    on new data
  \item
    In financial markets, it can assist in estimating the probability of
    stock movements given prior trends and new economic data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayes' theorem: detect spam email}
\protect\phantomsection\label{bayes-theorem-detect-spam-email}
\begin{itemize}
\tightlist
\item
  We want to partition emails in 3 categories:

  \begin{itemize}
  \tightlist
  \item
    \(A_1\)=\textcolor{blue}{\texttt{spam}},
    \(A_2\)=\textcolor{blue}{\texttt{low priority}},
    \(A_3\)=\textcolor{blue}{\texttt{high priority}}
  \end{itemize}
\item
  Receive an email with the word \textcolor{blue}{\texttt{free}}: what
  is the probability that it is spam?
\end{itemize}

\textbf{\textcolor{red}{Solution}}

\begin{itemize}
\item
  Prior knowledge (e.g., from a large corpus of emails):

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(A_1)=.7\), \(\Pr(A_2)=.2\), \(\Pr(A_3)=.1\)
  \item
    Sum is 1 since the events are a partition of the sample space
  \end{itemize}
\item
  \(B\) is the event ``email contains the word free''
\item
  From previous experience (e.g., a large corpus of emails) we know:

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(B|A_1)=.9\), \(\Pr(B|A_2)=.01\), \(\Pr(B|A_3)=.01\)
  \item
    Sum is not 1 since \(\Pr(B|.)\) is not a probability function
  \end{itemize}
\item
  Using Bayes' theorem, compute the probability of the event
  \[A_1|B = \text{"email is spam, given that it contains free"}\] \[
  \Pr(A_1|B)
  = \frac{\Pr(B|A_1) \cdot \Pr(A_1)}
  {\Pr(B|A_1) \cdot \Pr(A_1) + \Pr(B|A_2) \cdot \Pr(A_2) + \Pr(B|A_3)\cdot\Pr(A_3)}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Bayes' theorem: rain example}
\protect\phantomsection\label{bayes-theorem-rain-example}
\begin{itemize}
\tightlist
\item
  Consider the events:

  \begin{itemize}
  \tightlist
  \item
    \(W\) = ``weatherman predicts rain''
  \item
    \(R\) = ``it rains''
  \end{itemize}
\item
  A weatherman:

  \begin{itemize}
  \tightlist
  \item
    Predicts rain correctly 90\% of the time when it rains
  \item
    Predicts rain incorrectly 10\% of the time when it does not rain
  \end{itemize}
\item
  Given it rains 5 days a year, what's the probability it rains tomorrow
  given the weatherman predicts rain?
\end{itemize}

\textbf{\textcolor{red}{Solution}}

\begin{itemize}
\item
  Assume

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(W | R) = 90\%\)
  \item
    \(\Pr(W | \overline{R}) = 10\%\)
  \item
    \(\Pr(R) = 5 / 365\)
  \item
    \(\Pr(W | R) = .9\)
  \item
    \(\Pr(W | \overline{R}) = .1\)
  \end{itemize}
\item
  Using Bayes: \[
  \Pr(R|W)
  = \frac{\Pr(W|R) \Pr(R)}{\Pr(W|R) \Pr(R) +
  \Pr(W|\overline{R}) \Pr(\overline{R})}
  = 0.9 * 5 / (0.9 * 5 + 0.1 * 360) = 0.11111
  \]
\item
  Note the prosecutor fallacy in action

  \begin{itemize}
  \tightlist
  \item
    One might think \(\Pr(R | W) = 0.9\), but this cannot be accurate
    since it rains infrequently
  \item
    If \(\Pr(R) = \Pr(W) = 0.5\), then \(\Pr(R|W)\) would equal
    \(\Pr(W|R)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Frequentist interpretation}
\protect\phantomsection\label{frequentist-interpretation}
\begin{itemize}
\item
  Probabilities can be interpreted as the outcome of long-run
  experiments
\item
  This is a way to empirically estimate probabilities, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Q: What is the probability of a car tire exploding when filled 50\%
    beyond the manufacturer's recommendation?
  \item
    A: Fill 100 tires and see how many explode
  \end{itemize}
\item
  There is a problem for one-time events, e.g.,

  \begin{itemize}
  \tightlist
  \item
    Q: What is the probability of life on Mars?

    \begin{itemize}
    \tightlist
    \item
      The true probability is 0 or 1, depending on life being on Mars or
      not
    \item
      You can use scientific knowledge to estimate it
    \item
      If we go to Mars and find life, then the probability is 1
    \item
      Proving the probability is 0 is more difficult: you need to check
      everywhere on Mars
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Random variable (RV)}
\protect\phantomsection\label{random-variable-rv}
\begin{itemize}
\item
  A RV \(X\) is a function defined from the sample space to real
  numbers: \[X: \Omega \rightarrow \bbR\]
\item
  A RV is concerned with outcomes of an experiment (i.e., sample space),
  and not with events
\item
  Events are subsets of the sample space \(A \subseteq \Omega\) that are
  transformed by \(X\) into subsets of \(A \subseteq \bbR\), and vice
  versa
\item
  A random variable can be:

  \begin{itemize}
  \tightlist
  \item
    Discrete: takes finite \emph{or} countably infinite values
  \item
    Continuous: takes uncountably infinite values
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Random variables link sample space and data}
\protect\phantomsection\label{random-variables-link-sample-space-and-data}
\begin{itemize}
\tightlist
\item
  Sample space \(\Omega\) is a set, but to process data we need numbers

  \begin{itemize}
  \tightlist
  \item
    Random variable is the link between \(\Omega\) and numbers
  \end{itemize}
\item
  A relation for a RV \(X \in A_0 \subseteq \bbR\) (e.g., \(X = 1\) or
  \(X \ge 1\)) corresponds to an event \(X^{-1}(A_0) \subseteq \Omega\)

  \begin{itemize}
  \tightlist
  \item
    For simplicity, we refer to \(X \in A_0\) as ``event \(A_0\)'' since
    the relation induces an event
  \item
    Once the RV is introduced, it is like the sample space disappears
    and the outcome of an experiment is just a number
  \end{itemize}
\item
  A RV can have a simple conceptual description of an experiment, which
  can help understanding and reason about the problem, e.g.,

  \begin{itemize}
  \tightlist
  \item
    ``\(X\) = number of heads when tossing 5 fair coins''
  \item
    \(X \ne 0\) is the event ``there are no heads tossing 5 fair coins''
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RVs are not defined in a unique way}
\protect\phantomsection\label{rvs-are-not-defined-in-a-unique-way}
\begin{itemize}
\item
  Different RVs \(X\) can be associated with the same sample space
  \(\Omega\)
\item
  RVs introduce degrees of freedom in describing and solving a problem,
  just like a sample space
\item
  E.g., for 2 coin tosses:

  \begin{itemize}
  \tightlist
  \item
    X as ``binary representation'' = \{HH: 0, HT: 1, TH: 2, TT: 3\}
  \item
    X as ``number of tails'' = \{HH: 0, HT: 1, TH: 1, TT: 2\}
  \item
    X as ``number of heads'' = \{HH: 2, HT: 1, TH: 1, TT: 0\}
  \item
    X as ``two coin tosses are equal'' = \{HH: 1, HT: 0, TH: 0, TT: 1\}
  \end{itemize}
\item
  Different outcomes of the experiment can be associated with the same
  number

  \begin{itemize}
  \tightlist
  \item
    E.g., we might want to associate distinct numbers to interesting
    events
  \item
    E.g., for 2 coin tosses: \{HH: 0, HT: 1, TH: 1, TT: 2\}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Cumulative Distribution Function of a RV}
\protect\phantomsection\label{cumulative-distribution-function-of-a-rv}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{Cumulative Distribution Function (CDF)}}
  of a \emph{continuous} or \emph{discrete} RV \(X\) is defined as:
  \[F_X(x) \defeq \Pr(X \le x) \text{ for } x \in \bbR\]
\item
  Why it is useful:

  \begin{itemize}
  \tightlist
  \item
    CDF combines RV \(X\) (to infer events) together with \(\Pr(\cdot)\)
    into a function \(\bbR \rightarrow [0, 1]\)
  \item
    CDF is a way to infer ``standard'' events using the total ordering
    in \(\bbR\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Properties of CDF}
\protect\phantomsection\label{properties-of-cdf}
\begin{enumerate}
\tightlist
\item
  Limits: \[\lim_{x \rightarrow -\infty} F(x) = 0\]
  \[\lim_{x \rightarrow +\infty} F(x) = 1\]
\item
  Not decreasing: \[x_1 < x_2 \implies F(x_1) \le F(x_2)\]
\item
  Continuous from the right:
  \[\lim_{\varepsilon \rightarrow 0^+} F(x + \varepsilon) = F(x) \; \forall x\]
\end{enumerate}
\end{frame}

\begin{frame}{CDF of discrete RV in terms of probability}
\protect\phantomsection\label{cdf-of-discrete-rv-in-terms-of-probability}
\begin{itemize}
\item
  The CDF of a discrete RV evaluated in one point \(x\) is the sum of
  probability of all outcomes \(u \le x\):
  \[F_X(x) \defeq \Pr(X \le x) = \sum_{u \le x} \Pr(X = u)\]
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    Experiment: Toss a fair coin twice
  \item
    RV \(X\) = ``count the number of tails''
  \item
    \(X\) is \(\{HH: 0, HT: 1, TH: 1, TT: 2\}\)
  \item
    \(F_X(x)\) is \(\{0: 1/4, 1: 3/4, 2: 1\}\)
  \end{itemize}
\item
  Plot of \(F_X(x)\) for a discrete RV:

  \begin{itemize}
  \tightlist
  \item
    Is a staircase function
  \item
    The jump at \(x_i\) is equal to \(\Pr(x_i)\)
  \item
    The step has the same value on the right
  \item
    Is monotonically increasing function
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Probability Mass Function for a discrete RV}
\protect\phantomsection\label{probability-mass-function-for-a-discrete-rv}
\begin{itemize}
\item
  The \textbf{\textcolor{red}{Probability Mass Function (PMF)}} of a
  discrete RV \(X\) is a function \(f_X(x)\) such that: \[
  f_X(x) =
  \begin{cases}
  \Pr(X = x_i) & x_i \in \{x_1, ..., x_n\}\\
  0 & \text{otherwise}
  \end{cases}
  \]
\item
  A finite PMF can always be represented with a table

  \begin{itemize}
  \tightlist
  \item
    E.g., for 2 coin tosses \{HH: 0, HT: 1, TH: 1, TT: 2\}, and PMF is
    \{0: 1/4, 1: 1/2, 2: 1/4\}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Properties of PMF}
\protect\phantomsection\label{properties-of-pmf}
\begin{enumerate}
\tightlist
\item
  Integral is 1: \(\sum_{x_i} f_X(x) = 1\)
\item
  Always non-negative: \(f_X(x) \ge 0\)
\end{enumerate}

\begin{itemize}
\item
  These properties descend from the properties of CDF
\item
  CDF of discrete RV in terms of PMF
  \[F_X(x) = \Pr(X \le x) = \sum_{u \le x} f_X(u)\] where \(f_X(u)\) is
  the PMF of \(X\)
\end{itemize}
\end{frame}

\begin{frame}{PMF: example of coin flip}
\protect\phantomsection\label{pmf-example-of-coin-flip}
\begin{itemize}
\tightlist
\item
  \(X\) represents the outcome of a coin flip (Bernoulli)
\item
  \(X = 0\) represents tails and \(X = 1\) represents heads, with a
  given probability \(p\): \[
  f_X(x) =
  \begin{cases}
  p     & $X = 1$ \\
  1 - p & $X = 0$ \\
  \end{cases}
  \]
\item
  The PMF of \(X\) can be written as one-line:
  \[f_X(x) = p^x (1 - p)^{(1 - x)}\] with \(x = 0, 1\)
\end{itemize}
\end{frame}

\begin{frame}{Discrete RV in terms of continuous RV}
\protect\phantomsection\label{discrete-rv-in-terms-of-continuous-rv}
\begin{itemize}
\tightlist
\item
  A discrete RV \(f_X(x)\) can be expressed a continuous RV \(f^*_X(x)\)
  with Dirac delta impulses in its PDF

  \begin{itemize}
  \tightlist
  \item
    Otherwise the probability of a single event would be 0 as in a
    continuous RV \[f^*_X(x) = \sum_{i=1}^n f_X(x_i) \delta(x - x_i)\]
  \end{itemize}
\item
  With this definition all formulas for continuous RV apply to a
  discrete RV

  \begin{itemize}
  \tightlist
  \item
    E.g., CDF is just the integral of the PDF and it has jumps in the
    deltas
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Integrals in terms of PDF and CDF}
\protect\phantomsection\label{integrals-in-terms-of-pdf-and-cdf}
\begin{itemize}
\tightlist
\item
  Any integral involving a PDF in the form (e.g., same form of theorem
  of mean): \[\int g(x) f_X(x) dx\] can be rewritten in terms of CDF:
  \[\int g(x) dF_X\]
\item
  This is because of relationship between CDF and PDF in terms of
  derivative \[dF_X = \frac{df_X}{dx}\]
\end{itemize}
\end{frame}

\begin{frame}{Empirical CDF of a (discrete or continuous) RV}
\protect\phantomsection\label{empirical-cdf-of-a-discrete-or-continuous-rv}
\begin{itemize}
\tightlist
\item
  Consider

  \begin{itemize}
  \tightlist
  \item
    \(X\) (discrete or continuous) with a certain CDF \(F(x)\)
  \item
    Take IID samples \(X_1, ..., X_n\) from \(X\)
  \end{itemize}
\item
  The empirical CDF is defined as:
  \[\hat{F}_n(x) = \frac{1}{n} \sum_{i=1}^n I(X_i \le x)\]
\item
  Note that \(X_i\) can be

  \begin{itemize}
  \tightlist
  \item
    A RV, then \(\hat{F}_n(x)\) is the empirical CDF RV
  \item
    A realization of a RV \(x_i\), then \(\hat{F}_n(x)\) is a sample
    realization of the empirical CDF RV
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Empirical PMF}
\protect\phantomsection\label{empirical-pmf}
\begin{itemize}
\tightlist
\item
  The empirical PMF is defined as:
  \[\hat{f}_n(x) \defeq \frac{d \hat{F_n}(x)}{dx}\]
\item
  Using the definition of empirical CDF:
  \[\hat{F}_n(x) = \frac{1}{n} \sum_{i=1}^n I(X_i \le x)\]
\item
  Then \[
  \hat{f}_n(x)
  = \frac{d}{dx} \frac{1}{n} \sum_i I(X_i \le x)
  = \frac{1}{n} \sum_i \frac{d}{dx} I(X_i \le x)
  = \frac{1}{n} \sum_i \delta(x - x_i)
  \]
\end{itemize}
\end{frame}

\begin{frame}{Empirical PMF of a (discrete or continuous) RV}
\protect\phantomsection\label{empirical-pmf-of-a-discrete-or-continuous-rv}
\begin{itemize}
\tightlist
\item
  We can represent the PMF using \(n\) samples, summing on the \(m\)
  support points \(x_i\):
  \[f^*_X(x) = \frac{1}{n} \sum_{i=1}^m f_X(x_i) \delta(x - x_i)\] or in
  terms of the \(n\) samples \(x_j\) where multiple samples are
  accounted one at a time:
  \[f^*_X(x) = \frac{1}{n} \sum_{j=1}^n \delta(x - x_j)\]
\end{itemize}
\end{frame}

\begin{frame}{Integral of empirical CDF / PMF}
\protect\phantomsection\label{integral-of-empirical-cdf-pmf}
\begin{itemize}
\tightlist
\item
  Given a relationship like
  \[y(x) = \int g(x) d \hat{F}_n = \int g(x) \hat{f}_n(x) dx\] using the
  expression for the empirical PMF:
  \[\hat{f}_n(x) = \frac{1}{n} \sum_{i=1}^n \delta(x - x_i)\] we get: \[
  y(x)
  = \int g(x) \frac{1}{n} \sum_i \delta(x - x_i) dx
  = \frac{1}{n} \sum_i \int g(x) \delta(x - x_i) dx
  = \frac{1}{n} \sum_i g(x_i)
  \]
\end{itemize}
\end{frame}

\begin{frame}{Probability Density Function for continuous RV var}
\protect\phantomsection\label{probability-density-function-for-continuous-rv-var}
\begin{itemize}
\tightlist
\item
  The \textbf{\textcolor{red}{Probability Density Function (PDF)}} is
  defined as \[f_X(x) = \frac{dF_X(u)}{du} \biggr\rvert_{u=x}\]
\item
  The PDF is defined in all points where \(F_X(x)\) is continuous and
  thus derivable
\end{itemize}
\end{frame}

\begin{frame}{2 properties of PDF}
\protect\phantomsection\label{properties-of-pdf}
\begin{itemize}
\tightlist
\item
  From the axioms of probability it follows:

  \begin{enumerate}
  \tightlist
  \item
    Integral is 1: \(\int_{-\infty}^{+\infty} f_X(x)dx = 1\)
  \item
    Always non-negative: \(f_X(u) \ge 0\)
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{CDF for continuous RV in terms of PDF}
\protect\phantomsection\label{cdf-for-continuous-rv-in-terms-of-pdf}
\begin{itemize}
\tightlist
\item
  The CDF is defined as:
  \[F_X(x) = \Pr(X \le x) = \int_{-\infty}^x f_X(u) du\] where
  \(f_X(u)\) is the PDF of \(X\)
\end{itemize}
\end{frame}

\begin{frame}{Probability of event in terms of CDF / PDF of continuous
RV}
\protect\phantomsection\label{probability-of-event-in-terms-of-cdf-pdf-of-continuous-rv}
\begin{itemize}
\tightlist
\item
  For an event \(A = [a, b] \subseteq \bbR\): \[
  \Pr(a \le X \le b)
  = \Pr(a < X < b)
  = F_X(b) - F_X(a)
  = \int_a^b f_X(u)du
  \]
\item
  For a generic event \(A \subseteq \bbR\) which corresponds to a subset
  of \(X^{-1}(A) = A_X \subseteq \Omega_X\):
  \[\Pr(A_X) = \Pr(X \in A) = \Pr(A) = \int_A f_X(u)du\]
\end{itemize}
\end{frame}

\begin{frame}{Probability of a single value}
\protect\phantomsection\label{probability-of-a-single-value}
\begin{itemize}
\tightlist
\item
  The probability that a continuous RV takes any particular value
  \(\Pr(X = a)\) is 0
\item
  This is different from a discrete RV
\end{itemize}
\end{frame}

\begin{frame}{Joint CDF for 2 RV: definition}
\protect\phantomsection\label{joint-cdf-for-2-rv-definition}
\begin{itemize}
\tightlist
\item
  The joint CDF of two RVs \(X\) and \(Y\) is defined as:
  \[F_{X,Y}(x, y) = \Pr(X \le x, Y \le y)\]
\end{itemize}
\end{frame}

\begin{frame}{Joint CDF: intuition}
\protect\phantomsection\label{joint-cdf-intuition}
\begin{itemize}
\tightlist
\item
  It is the same as the probability of two events to happen jointly, but
  using the events induced by \(X\) and \(Y\) onto \(\bbR\)
\end{itemize}
\end{frame}

\begin{frame}{Joint CDF for discrete RV in terms of probability}
\protect\phantomsection\label{joint-cdf-for-discrete-rv-in-terms-of-probability}
\begin{itemize}
\tightlist
\item
  The joint CDF of two discrete RVs \(X\) and \(Y\) can be expressed as:
  \[F_{X,Y}(x, y) = \sum_{u \le x, v \le y} \Pr(X = u, Y = v)\]
\end{itemize}
\end{frame}

\begin{frame}{Joint PMF for discrete RV in terms of probability}
\protect\phantomsection\label{joint-pmf-for-discrete-rv-in-terms-of-probability}
\begin{itemize}
\tightlist
\item
  The joint PMF of 2 discrete RVs: \[
  f_{X,Y}(x, y) =
  \begin{cases}
  \Pr(X = x, Y = y) & x \in \{x_1, ..., x_n\}, y \in \{y_1, ..., y_m\}\\
  0 & \text{otherwise}\\
  \end{cases}
  \]
\item
  Note that a joint PMF can be represented with a bidimensional table
\end{itemize}
\end{frame}

\begin{frame}{Joint PMF properties}
\protect\phantomsection\label{joint-pmf-properties}
\begin{itemize}
\tightlist
\item
  The joint PMF has the properties:

  \begin{enumerate}
  \tightlist
  \item
    Always non-negative: \(f_{X,Y}(x, y) \ge 0\)
  \item
    Integral is 1: \(\sum_x \sum_y f_{X,Y}(x, y) = 1\)
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Joint PDF for continuous RV in terms of joint CDF}
\protect\phantomsection\label{joint-pdf-for-continuous-rv-in-terms-of-joint-cdf}
\begin{itemize}
\tightlist
\item
  The joint PDF of 2 RVs \(X\) and \(Y\)
  \[f_{X,Y}(x, y) = \frac{\partial^2 F_{X,Y}(u, v)}{\partial x \partial y}\]
\item
  The joint PDF has all the properties of a PDF (always positive,
  integral is 1)
\end{itemize}
\end{frame}

\begin{frame}{Joint CDF for discrete RV in terms of joint PDF}
\protect\phantomsection\label{joint-cdf-for-discrete-rv-in-terms-of-joint-pdf}
\begin{itemize}
\tightlist
\item
  One can get a joint CDF by integrating the joint PDF:
  \[F_{X,Y}(x, y) = \sum_{u \le x, v \le y} f_{X,Y}(u, v)\]
\end{itemize}
\end{frame}

\begin{frame}{Joint CDF for continuous RV in terms of joint PDF}
\protect\phantomsection\label{joint-cdf-for-continuous-rv-in-terms-of-joint-pdf}
\[F_{X,Y}(x, y) = \int_{-\infty}^{x} \int_{-\infty}^y f_{X,Y}(u, v)du dv\]
- This comes through the definition of joint PDF as partial derivative
\end{frame}

\begin{frame}{Probability of event in terms of joint PDF for continuous
RV}
\protect\phantomsection\label{probability-of-event-in-terms-of-joint-pdf-for-continuous-rv}
\begin{itemize}
\tightlist
\item
  Given a joint PDF \(f_{X,Y}(x, y)\), the probability of event in
  sample space \(A \subseteq \Omega_X \times \Omega_Y\), i.e.,
  \(S = X(\Omega_X) \times Y(\Omega_Y) \subseteq \bbR^2\)
  \[\Pr(A) = \int_{S} f_{X,Y}(u, v) du dv\]
\item
  This is a key relationship for marginal and conditional PDFs
\end{itemize}
\end{frame}

\begin{frame}{Marginal CDF for discrete / continuous RV in terms of
joint CDF}
\protect\phantomsection\label{marginal-cdf-for-discrete-continuous-rv-in-terms-of-joint-cdf}
\begin{itemize}
\tightlist
\item
  One can get a marginal CDF from the joint CDF by setting variables to
  \(+\infty\) \[
  F_X(x) \defeq \Pr(X \le x) = \Pr(X \le x, Y \le \infty) \defeq F_{X,Y}(x, \infty)
  \]
\end{itemize}
\end{frame}

\begin{frame}{Marginal PMF for discrete RV in terms of joint PMF}
\protect\phantomsection\label{marginal-pmf-for-discrete-rv-in-terms-of-joint-pmf}
\begin{itemize}
\tightlist
\item
  Given two discrete RVs \(X\) (defined on \(\Omega_X\)) and \(Y\)
  (defined on \(\Omega_Y\)) one can get a marginal PMF from the joint
  PMF through summing on one variable
\item
  The marginal PMF of \(X\) is defined as: \[
  \begin{aligned}
  f_X(x) & \defeq \Pr(X = x)\\
  &= \Pr(X = x, Y \in \bbR)\\
  &= \sum_{y_i \in Y(\Omega_Y)} \Pr(X = x, Y = y_i)
  \text{ (since all events $X = x \land Y = y_i$ are distinct)}\\
  &= \sum_{y_i \in Y(\Omega_Y)} f_{X,Y}(x, y_i)
  \end{aligned}
  \]
\item
  Note that \(f_X(x)\) is a PMF with all the associated properties
\end{itemize}
\end{frame}

\begin{frame}{Marginal PDF for continuous RV in terms of joint PDF}
\protect\phantomsection\label{marginal-pdf-for-continuous-rv-in-terms-of-joint-pdf}
\begin{itemize}
\tightlist
\item
  Given two continuous RVs \(X\) and \(Y\), one can get a marginal PDF
  from the joint PDF through integrating on one variable \[
  \begin{aligned}
  f_X(x) & \defeq \frac{d F_X(u)}{du} = \frac{d}{du} \Pr(X \le x)
  = \frac{d}{du} \Pr(X \le x, Y \le +\infty)\\
  & = \frac{d}{du} \int_{v=-\infty}^{u} \int_{y=-\infty}^{\infty} f_{X,Y}(v, y) dv dy\\
  & = \int_{y = -\infty}^{\infty} f_{X,Y}(x, y) dy\\
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Def of independent RV}
\protect\phantomsection\label{def-of-independent-rv}
\begin{itemize}
\tightlist
\item
  The RVs \(X\) and \(Y\) are independent \(\iff\) the events
  \(X \le x\) and \(Y
  \le y\) are independent for all \(x\) and \(y\)
  \[\Pr(X \le x, Y \le y) = \Pr(X \le x) \cdot \Pr(Y \le y)\]
\item
  This is equivalent to events \(X \in A\) and \(Y \in B\) being
  independent
\end{itemize}
\end{frame}

\begin{frame}{CDF of independent RV}
\protect\phantomsection\label{cdf-of-independent-rv}
\begin{itemize}
\tightlist
\item
  If RV \(X\) and \(Y\) are independent their joint CDF can be factored
  into the product of marginal CDF: \[
  F_{X,Y}(x,y)
  \defeq \Pr(X \le x, Y \le y)
  = \Pr(X \le x) \cdot \Pr(Y \le y)
  \defeq F_X(x) F_Y(y)
  \; \forall x, y
  \]
\item
  Also the converse it true
\end{itemize}
\end{frame}

\begin{frame}{PDF / PMF of independent RVs}
\protect\phantomsection\label{pdf-pmf-of-independent-rvs}
\begin{itemize}
\tightlist
\item
  If RV \(X\) and \(Y\) are independent their joint PDF (or PMF) can be
  factored into the product of marginal PDF (or PMF): \[
  f_{X,Y}(x,y)
  \defeq \frac{\partial F_{X,Y}(x, y)}{\partial x \partial y}
  = \frac{\partial F_X(x) F_Y(y)}{\partial x \partial y}
  = \frac{\partial F_X(x)}{\partial x} \cdot \frac{\partial F_Y(y)}{\partial y}
  = f_X(x) \cdot f_Y(y) \; \forall x, y
  \]
\item
  Also the converse it true
\end{itemize}
\end{frame}

\begin{frame}{Characterization of PDF and CDF of independent RV}
\protect\phantomsection\label{characterization-of-pdf-and-cdf-of-independent-rv}
\begin{itemize}
\tightlist
\item
  RV \(X\) and \(Y\) are independent \(\iff\) their joint PDF / PMF /
  CDF factors in terms of marginal PDF / PMF / CDF
\end{itemize}
\end{frame}

\begin{frame}{Marginal PDF / PMF / CDF}
\protect\phantomsection\label{marginal-pdf-pmf-cdf}
\begin{itemize}
\tightlist
\item
  It refers to a distribution of a single RV in a set-up where multiple
  RVs exist
\item
  E.g., the marginal PDF \(X\) is the joint PDF of \(X\) and \(Y\)
  integrated over \(Y\) so when we talk about marginal we refer to a
  single RV
\end{itemize}
\end{frame}

\begin{frame}{Def of conditional PDF for RVs}
\protect\phantomsection\label{def-of-conditional-pdf-for-rvs}
\begin{itemize}
\tightlist
\item
  \(X\) and \(Y\) are RVs, the conditional PDF of \(X\) given \(Y\) is
  defined as: \[f_{X|Y}(x, y) \defeq \frac{f_{X,Y}(x, y)}{f_Y(y)}\]
\item
  A similar definition holds for PMFs
\item
  MEM: A PDF / PMF is like a probability of events, so definition of
  conditional prob extends to PDF / PMF in the same way
\end{itemize}
\end{frame}

\begin{frame}{Conditional probability in terms of conditional PDF}
\protect\phantomsection\label{conditional-probability-in-terms-of-conditional-pdf}
\[\Pr(X \in A|Y=y) = \int_{X \in A} f_{X|Y}(x, y) dx\] - Note that - The
conditional probability is a function of \(y\) - \(\Pr(X \in A|Y=y)\) is
intended as the limit \(dy \to 0\) of
\(\Pr(X \in A|y \le Y \le y + dy)\) since the event \(Y = y\) has
probability 0

\begin{itemize}
\tightlist
\item
  It can be proved by writing conditional prob in terms of its
  definition
\end{itemize}
\end{frame}

\begin{frame}{Marginal PDF for continuous RV in terms of conditional
PDF}
\protect\phantomsection\label{marginal-pdf-for-continuous-rv-in-terms-of-conditional-pdf}
\begin{itemize}
\tightlist
\item
  Write PDF of \(X\) in terms of joint PDF of \(X\) and \(Y\)
  \[f_X(x) = \int_{y=-\infty}^{+\infty} f_{X,Y}(x, y)dy\]
\item
  Then express the joint PDF in terms of conditional PDF:
  \[f_X(x) = \int_{y=-\infty}^{+\infty} f_{X|Y}(x, y) f_Y(y) dy\]
\item
  This is similar to the law of total probability, since we express a
  probability as summation of the conditional probability multiplied by
  the probability that we are conditioning on
\end{itemize}
\end{frame}

\begin{frame}{Summary of relationships}
\protect\phantomsection\label{summary-of-relationships}
\begin{itemize}
\tightlist
\item
  \(\Pr(x) \rightarrow\) CDF
\item
  PDF = \(\frac{d}{dx}\) CDF
\item
  PDF / PMF is close to prob (it is like a prob density)
\item
  \(\Pr() = \int\) PDF
\item
  CDF = \(\int_{-\infty}^x\) PDF
\item
  Marginal prob = \(\int\) joint PDF
\item
  Marginal prob = \(\int\) cond PDF \(\times\) marginal PDF (like ``law
  of total probability'')
\end{itemize}
\end{frame}

\begin{frame}{Mean of discrete RV: definition}
\protect\phantomsection\label{mean-of-discrete-rv-definition}
\begin{itemize}
\item
  The mean of a discrete RV is defined as:
  \[\EE[X] \defeq \sum_i x_i f_X(x_i)\]
\item
  We use the PMF in the definition since in this way it is more similar
  to the definition of mean for continuous RVs
\item
  MEM:

  \begin{enumerate}
  \tightlist
  \item
    Sum of each value multiplied by its prob
  \item
    Weighted average of the values of a PDF
  \item
    Dot product of a vector of values and prob of values
  \end{enumerate}
\item
  If \(X\) can take countably infinite values, then the infinite series
  should converge in absolute value
\end{itemize}
\end{frame}

\begin{frame}{Mean of discrete RV in terms of probability}
\protect\phantomsection\label{mean-of-discrete-rv-in-terms-of-probability}
\[\EE[X] = \sum_i x_i \Pr(X = x_i)\]
\end{frame}

\begin{frame}{Alternative names and symbols for mean of a RV}
\protect\phantomsection\label{alternative-names-and-symbols-for-mean-of-a-rv}
\begin{itemize}
\tightlist
\item
  The mean is also called:

  \begin{itemize}
  \tightlist
  \item
    Mathematical expectation
  \item
    Expectation
  \item
    Expected value
  \item
    First moment of a RV
  \end{itemize}
\item
  It is indicated as \(\mu_X\) or \(\EE[X]\)
\item
  Note that the average of values (e.g., sample mean) is indicated as
  \(\overline{X}\)
\end{itemize}
\end{frame}

\begin{frame}{What is the mean of a biased coin?}
\protect\phantomsection\label{what-is-the-mean-of-a-biased-coin}
\begin{itemize}
\tightlist
\item
  \(\Pr(X = 1) = p\) and \(\Pr(X = 0) = 1 - p\), then
  \(\EE[X] = 0 \times (1 - p) + 1 \times p = p\)
\item
  Note that the mean of the RV can be a value that the RV cannot assume
\end{itemize}
\end{frame}

\begin{frame}{Mean of continuous RV: definition}
\protect\phantomsection\label{mean-of-continuous-rv-definition}
\begin{itemize}
\tightlist
\item
  The mean is defined as:
  \[\EE[X] \defeq \int_{- \infty}^{+ \infty} x f_X(x) dx\]
\item
  The mean is well-defined if the integral converges in absolute value
\end{itemize}
\end{frame}

\begin{frame}{Mean as measure of central tendency}
\protect\phantomsection\label{mean-as-measure-of-central-tendency}
\begin{itemize}
\tightlist
\item
  Draw many IID samples from a RV \(X\)
\item
  Compute the (sample) average
  \(\overline{X}_n = \frac{X_1 + ... + X_n}{n}\)
\item
  Then \(\overline{X}_n\) will approximate the mean of \(X\): \[
  \lim_{n \to \infty} \overline{X}_n = \frac{1}{n} \sum_{i=1}^n
  = \frac{X_1 + X_2 + ... + X_n}{n} \approx \EE[X]
  \]
\item
  Thus the mean can be interpreted as the average value of the RV using
  infinite IID samples
\item
  This is the Law of Large Numbers (LNN)
\end{itemize}
\end{frame}

\begin{frame}{Intuition of law of large numbers for discrete vars}
\protect\phantomsection\label{intuition-of-law-of-large-numbers-for-discrete-vars}
\begin{itemize}
\tightlist
\item
  In the case of discrete RV:

  \begin{itemize}
  \tightlist
  \item
    We take an infinite number of samples for \(X\) and sum them
  \item
    Group the samples by values
  \item
    The average is the sum of the values for \(X\) multiplied by the
    probability (since the frequency converges to the probability),
    which is the def of mean
  \end{itemize}
\item
  Note that the mean is not the most frequent value (that's the mode)
\end{itemize}
\end{frame}

\begin{frame}{Mean as center of mass}
\protect\phantomsection\label{mean-as-center-of-mass}
\begin{itemize}
\tightlist
\item
  The mean can be interpreted as the center of mass of the PDF / PMF,
  i.e., where one needs to put a wedge to ``balance'' the PDF / PMF
\end{itemize}
\end{frame}

\begin{frame}{Mean as minimum value for squared errors}
\protect\phantomsection\label{mean-as-minimum-value-for-squared-errors}
\begin{itemize}
\tightlist
\item
  The mean is the value \(y\) that minimizes the quantity
  \(\sum_i (x_i - y)^2\) on an infinite number of trials
\item
  This can be proved either by

  \begin{itemize}
  \tightlist
  \item
    Calculus or
  \item
    Adding and subtracting \(\EE[X]\) and showing that the value is
    minimum when \(y = \EE[X]\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Theorem of the mean}
\protect\phantomsection\label{theorem-of-the-mean}
\begin{itemize}
\item
  Aka theorem of the lazy statistician
\item
  Given a RV \(X\) (discrete or continuous) and a scalar function
  \(g(x)\), then \(Y = g(X)\) is a RV
\item
  \textbf{\emph{Thesis}}
  \[\EE[Y] = \EE[g(X)] = \int_{-\infty}^{+\infty} g(x) f_X(x)dx\]
\item
  \textbf{\emph{Notes}}
\item
  One does not have to compute the PDF of \(Y\) to compute its mean, but
  use only the PDF of \(X\) and the function \(g(x)\)
\item
  This formula holds trivially also for the identity function, since
  \(X = Y = I(X)\)
\item
  This formula holds also for function of multiple RV using the joint
  PDF
\end{itemize}
\end{frame}

\begin{frame}{Theorem of the mean: proof}
\protect\phantomsection\label{theorem-of-the-mean-proof}
\begin{itemize}
\tightlist
\item
  WLOG consider a \textit{discrete} RV
\item
  By definition \[
  \EE[Y]
  \defeq \sum_{y_i \in \Omega_Y} y_i f_Y(y_i)
  = \sum_{y_i \in \Omega_Y} y_i \Pr(Y=y_i)
  \]
\item
  Consider the generic \(y_i\) and express \(\Pr(Y = y_i)\) in terms of
  \(\Pr(X = ...)\) \[
  \begin{aligned}
  \Pr(Y = y_i) &= \Pr(g(X) = y_i)\\
  &= \Pr(X \in g^{-1}(y_i))\\
  &\text{(since a set of points $x_{ij}$ correspond to each $y_i$)}\\
  &= \Pr(X=x_{i1} \cup X=x_{i2} \cup ... \cup X=x_{iN})\\
  &\text{(since all events are distinct)}\\
  &= \Pr(X=x_{i1}) + \Pr(X=x_{i2}) + ... \Pr(X=x_{iN})
  \end{aligned}
  \]
\item
  Thus we can write \(\EE[Y]\):
  \[\EE[Y] = \sum_{y_i \in \Omega_Y} y_i (\Pr(X=x_{i1}) + ... + \Pr(X=x_{iN}))\]
\item
  Now we should note that the previous summation is over all and only
  the possible \(x_j\)
\item
  So we can sum directly over \(x_j\), and the \(y_i\) corresponding to
  each \(x_j\) is \(g(x_j)\), so \[
  \EE[Y]
  = \sum_{x_j \in \Omega_X} g(x_j) \Pr(X=x_j)
  \defeq \sum_{x_j \in \Omega_X} g(x_j) f_X(x_j)
  \]
\end{itemize}
\end{frame}

\begin{frame}{Indicator variable of an event}
\protect\phantomsection\label{indicator-variable-of-an-event}
\begin{itemize}
\tightlist
\item
  Consider a RV \(X\) and an event \(A \subseteq \bbR\) (which
  corresponds to an event in the sample space
  \(X^{-1}(A) \subseteq \Omega_X\))
\item
  The indicator variable of an event \(A\) for RV \(X\) is a RV defined
  as: \[
  I_A(X) \defeq
  \begin{cases}
  1 & \text{ if } X \in A\\
  0 & \text{otherwise}
  \end{cases}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Intuition of indicator RV}
\protect\phantomsection\label{intuition-of-indicator-rv}
\begin{itemize}
\tightlist
\item
  It is a way to synthesize specific events as RV from an already
  exiting RV
\item
  It is a transformed version \(g(X)\) of a RV
\end{itemize}
\end{frame}

\begin{frame}{Example of indicator variable}
\protect\phantomsection\label{example-of-indicator-variable}
\begin{itemize}
\tightlist
\item
  Consider \(X\) the result of a die toss:
  \[\Omega_X = \{1, 2, 3, 4, 5, 6\}\]
\item
  Consider the event
  \(A \subset \Omega_X = \{ \text{die outcome is even} \}\)
\item
  The indicator variable \(I_A(X)\) is a RV that is 1 when the outcome
  die is even
\end{itemize}
\end{frame}

\begin{frame}{Mean of an indicator variable}
\protect\phantomsection\label{mean-of-an-indicator-variable}
\begin{itemize}
\tightlist
\item
  Consider a RV \(X\), an event \(A\), and the indicator variable
  \(I_A(X)\), then \[\EE[I_A(X)] = \Pr(X \in A) = \int x f_{I_A}(x) dx\]
\end{itemize}
\end{frame}

\begin{frame}{Mean of an indicator variable: proof}
\protect\phantomsection\label{mean-of-an-indicator-variable-proof}
\begin{itemize}
\tightlist
\item
  The mean \(\EE[I_A(X)]\) is \[
  \begin{aligned}
  &= \int_{-\infty}^{\infty} I_A(x) f_X(x) dx \\
  &\text{ (because of theorem on the mean of a function of a RV)} \\
  &= \int_A I_A(x) f_X(x) dx \\
  &\text{ (since $I_A$ is 0 outside $A$)} \\
  &= \int_A f_X(x) dx \\
  &\text{ (since $I_A$ is 1 inside $A$)} \\
  &= \Pr(X \in A) \\
  &\text{ (by property of PDF)}
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Linearity of mean}
\protect\phantomsection\label{linearity-of-mean}
\begin{itemize}
\tightlist
\item
  If \(X_1\), \ldots, \(X_n\) are RVs and \(a_1\), \ldots, \(a_n\)
  constant, then \[\EE[\sum_i a_i X_i] = \sum_i a_i \EE[X_i]\]
\item
  It can be proved by theorem of mean of RV
\item
  Note that there is no assumption made on the RVs, i.e., the mean is
  linear even for RVs that are not independent or mutually exclusive
\end{itemize}
\end{frame}

\begin{frame}{Mean of product of independent RVs}
\protect\phantomsection\label{mean-of-product-of-independent-rvs}
\begin{itemize}
\tightlist
\item
  If \(X_1\), \ldots, \(X_n\) are independent RVs, then:
  \[\EE[\prod_i X_i] = \prod_i \EE[X_i]\]
\item
  It can be proved by theorem of mean of RV and factorization of PDFs
\end{itemize}
\end{frame}

\begin{frame}{Conditional mean}
\protect\phantomsection\label{conditional-mean}
\begin{itemize}
\tightlist
\item
  The conditional mean of \(X\) given \(Y\) is defined as:
  \[\EE[X|Y=y] \defeq \int_{-\infty}^{\infty} x f_{X|Y}(x, y) dx\] where
  the conditional PDF of \(X\) given \(Y\) is
  \(f_{X|Y}(x, y) \defeq \frac{f_{X,Y}(x, y)}{f_Y(y)}\)
\item
  Note that the conditional mean \(\EE[X|Y=y]\) is a function of \(y\),
  while \(\EE[X]\) is a number
\end{itemize}
\end{frame}

\begin{frame}{Conditional mean of independent variables}
\protect\phantomsection\label{conditional-mean-of-independent-variables}
\begin{itemize}
\tightlist
\item
  If \(X\) and \(Y\) are independent, then \(\EE[X|Y=y] = \EE[X]\)
\end{itemize}
\end{frame}

\begin{frame}{Law of total expectation}
\protect\phantomsection\label{law-of-total-expectation}
\begin{itemize}
\tightlist
\item
  Aka Law of iterated expectation, Adam's law
\item
  The unconditional mean can be expressed in terms of conditional mean:
  \[\EE[X] = \EE_Y[\EE[X|Y]] = \int_{y=\infty}^{\infty} \EE[X|Y=y] f_Y(y) dy\]
\item
  This is similar to law of total probability
  \[\Pr(X) = \sum_y \Pr(X | Y=y) \Pr(Y = y)\] and for this reason it's
  called law of total expectation
\end{itemize}
\end{frame}

\begin{frame}{Law of total expectation: proof}
\protect\phantomsection\label{law-of-total-expectation-proof}
\begin{itemize}
\tightlist
\item
  It can be proven through: \[
  f_X(x) = \int_{y = -\infty}^{\infty} f_{X,Y}(x, y) dy
  = \int f_{X|Y}(x, y) f_Y(y) dy
  \]
\end{itemize}
\end{frame}

\begin{frame}{Example: random sum of RVs}
\protect\phantomsection\label{example-random-sum-of-rvs}
\begin{itemize}
\item
  Let \(W = X_1 + X_2 + ... + X_N\) where:

  \begin{itemize}
  \tightlist
  \item
    \(X_i\) are IID with mean \(\mu_X\) and variance \(\sigma_X^2\)
  \item
    \(N\) is a RV independent of \(X_i\)
  \end{itemize}
\item
  What is \(\EE[W]\)?
\item
  \textbf{\emph{Solution}}
\item
  The mean is: \[
  \begin{aligned}
  \EE[W]
  &= \EE_N[\EE[W|N]] & \text{ (from law of total expectation)} \\
  &= \EE_N[\sum_{i=1}^N X_i] & \\
  &= \EE_N[N \mu_X] & \text{ (from linearity and IID)} \\
  &= \EE[N] \mu_X & \text{ (from linearity)} \\
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Corollary of law of total expectation}
\protect\phantomsection\label{corollary-of-law-of-total-expectation}
\begin{itemize}
\tightlist
\item
  If \(A_i\) is a partition of the outcome space \(\Omega\), i.e.,
  events are mutually exclusive and exhaustive, then
  \[\EE[X] = \sum_{i=1}^n \EE[X|A_i] \Pr(A_i)\]
\end{itemize}
\end{frame}

\begin{frame}{Corollary of law of total expectation: proof}
\protect\phantomsection\label{corollary-of-law-of-total-expectation-proof}
\begin{itemize}
\tightlist
\item
  Consider an indicator variable for each of the events \(A_i\),
  \(I_{A_i}\)
\item
  We can consider the indicator variable \(A\) given by the sum of all
  the \(I_{A_i}\), which is the ``certain event''
\item
  By the Law of total expectation \[
  \EE[X] = \EE_A[\EE[X | A]
  = \int_a \EE[X|A=a] f_{A_i}(a) da
  = \sum_i \EE[X|A_i] \int_{A_i} f_{A_i}(a) da
  = \sum_i \EE[X|A_i] \Pr(A_i)
  \] since the expected value of an indicator variable is its
  probability
\item
  TODO: not super clear
\end{itemize}
\end{frame}

\begin{frame}{Theorem of mean for joint RVs}
\protect\phantomsection\label{theorem-of-mean-for-joint-rvs}
\begin{itemize}
\tightlist
\item
  Given two RVs \(X\) and \(Y\) and a function \(g(x, y)\) then
  \(g(X, Y)\) is a random variable and:
  \[\EE[g(X, Y)] = \int_x \int_y g(x,y) f_{X,Y}(x,y) dx\]
\end{itemize}
\end{frame}

\begin{frame}{Theorem of conditional mean of a function of RV}
\protect\phantomsection\label{theorem-of-conditional-mean-of-a-function-of-rv}
\begin{itemize}
\item
  Given two RVs \(X\) and \(Y\) and a function \(g(x)\)
\item
  The definition of conditional mean is:
  \[\EE[X|Y=y] \defeq \int_x x f_{X|Y}(x, y) dx\] which is a function of
  \(y\)
\item
  If we transform \(X\) through \(g(x)\) then the theorem of the mean
  applies: \[\EE[g(X)|Y=y] = \int_x g(x) f_{X|Y}(x,y) dx\]
\end{itemize}
\end{frame}

\begin{frame}{Theorem of conditional mean of a function of RVs}
\protect\phantomsection\label{theorem-of-conditional-mean-of-a-function-of-rvs}
\begin{itemize}
\tightlist
\item
  Given two RVs \(X\) and \(Y\) and a function \(g(x, y)\) then
  \(g(X, Y)\) is a random variable and:
  \[\EE[g(X, Y)|Y=y] = \int_{x=-\infty}^{\infty} g(x,y) f_{X|Y}(x,y) dx\]
\item
  This is equivalent to theorem of the mean but applied to the
  conditional mean
\end{itemize}
\end{frame}

\begin{frame}{Theorem of conditional mean of a function of RVs: proof}
\protect\phantomsection\label{theorem-of-conditional-mean-of-a-function-of-rvs-proof}
\begin{itemize}
\tightlist
\item
  By definition of conditional mean
  \(\EE[X|Y] \defeq \int x f_{X|Y}(x) dx\)
\item
  The conditional mean is just a mean of a special RV \(X|Y\)
\item
  The theorem of the mean still applies to \(X|Y\)
\end{itemize}
\end{frame}

\begin{frame}{Variance of a RV}
\protect\phantomsection\label{variance-of-a-rv}
\begin{itemize}
\tightlist
\item
  Let \(X\) be a RV with mean \(\EE[X] < +\infty\)
\item
  The variance of \(X\) is defined as:
  \[\VV[X] \defeq \EE[(X - \EE[X])^2]\]
\item
  The variance is also indicated as \(\sigma_X^2\)
\item
  Note that the variance does not have the same unit of measure of the
  mean, but squared
\end{itemize}
\end{frame}

\begin{frame}{Computing variance using theorem of mean}
\protect\phantomsection\label{computing-variance-using-theorem-of-mean}
\begin{itemize}
\tightlist
\item
  Using the theorem of mean of RV:
  \[\VV[X] = \int_{x=-\infty}^{\infty} (x - \EE[X])^2 f_X(x) dx\]
\end{itemize}
\end{frame}

\begin{frame}{Standard deviation of a RV}
\protect\phantomsection\label{standard-deviation-of-a-rv}
\begin{itemize}
\tightlist
\item
  = the positive square root of the variance:
  \[\sigma_X = \sqrt{\VV[X]}\]
\item
  The standard deviation has the same unit of measure of the mean, while
  the variance has the squared dimension
\end{itemize}
\end{frame}

\begin{frame}{Meaning of variance}
\protect\phantomsection\label{meaning-of-variance}
\begin{itemize}
\tightlist
\item
  It represents the dispersion (or scatter) of the PDF / PMF of the RV
  around the mean
\end{itemize}
\end{frame}

\begin{frame}{Variance of a die toss}
\protect\phantomsection\label{variance-of-a-die-toss}
\begin{itemize}
\tightlist
\item
  Using the definition: \[
  \VV[X]
  \defeq \EE[(X - \mu)^2]
  = (1 - 3.5)^2 \cdot \frac{1}{6} + ... + (6 - 3.5)^2 \cdot \frac{1}{6}
  = 2.92
  \]
\end{itemize}
\end{frame}

\begin{frame}{Variance of a biased coin}
\protect\phantomsection\label{variance-of-a-biased-coin}
\begin{itemize}
\tightlist
\item
  Using the definition: \[
  \VV[X]
  \defeq \EE[(X - \mu)^2]
  = (0 - p)^2 \cdot (1 - p) + (1 - p)^2 \cdot p
  = (1 - p) p
  \]
\end{itemize}
\end{frame}

\begin{frame}{Alternative expression for variance}
\protect\phantomsection\label{alternative-expression-for-variance}
\[\VV[X] = \EE[X^2] - \EE[X]^2 = \EE[X^2] - \mu^2\] - Using the
definition of variance and property of mean
\(\EE[(X - \mu)^2] = \EE[X^2 + \mu^2 - 2 X \mu]\)
\end{frame}

\begin{frame}{Variance of linear combination of 2 RV}
\protect\phantomsection\label{variance-of-linear-combination-of-2-rv}
\begin{itemize}
\tightlist
\item
  If \(a\) and \(b\) are constants \[\VV[aX + b] = a^2 \VV[X]\]
\item
  Variance is not linear with respect to constants
\end{itemize}
\end{frame}

\begin{frame}{Variance of independent RV}
\protect\phantomsection\label{variance-of-independent-rv}
\begin{itemize}
\tightlist
\item
  If \(X_1\), \ldots, \(X_n\) are independent RVs and \(a_1\), \ldots,
  \(a_n\) are constants: \[\VV[\sum a_i X] = \sum a_i^2 \VV[X]\]
\end{itemize}
\end{frame}

\begin{frame}{Variance of the difference of RVs}
\protect\phantomsection\label{variance-of-the-difference-of-rvs}
\begin{itemize}
\tightlist
\item
  If \(X\) and \(Y\) are independent RVs then:
  \[\VV[X - Y] = \VV[X] + \VV[Y]\]
\item
  Note that the variance of the difference of independent RVs is the sum
  of the variances, and not the difference
\end{itemize}
\end{frame}

\begin{frame}{Law of total variance}
\protect\phantomsection\label{law-of-total-variance}
\begin{itemize}
\tightlist
\item
  Aka Conditional variance identity, Eve's Law
\item
  If \(X\) and \(Y\) are two RVs:
  \[\VV[X] = \EE_Y[\VV[X|Y]] + \VV_Y(\EE[X|Y])\]
\item
  MEM: EVVE = Expected Variance + Variance of Expected
\end{itemize}
\end{frame}

\begin{frame}{Law of total variance: proof}
\protect\phantomsection\label{law-of-total-variance-proof}
\[
  \begin{aligned}
  \VV[X]
  &= \EE[X^2] - (\EE[X])^2 \\
  & \text{(variance property})\\
  &= \EE_Y[\EE[X^2|Y]] - (\EE_Y[\EE[X|Y]])^2 \\
  & \text{(law of total expectation to both sides)}\\
  &= \EE_Y[\VV[X|Y] + (\EE[X|Y])^2] - (\EE_Y[\EE[X|Y]])^2 \\
  & \text{(variance property, i.e., summing and subtracting $\EE_Y[(\EE[X|Y])^2]$}\\
  &= \EE_Y[\VV[X|Y]] + \EE_Y[(\EE[X|Y])^2] - (\EE_Y[\EE[X|Y]])^2 \\
  & \text{ (linearity of mean)}\\
  &= \EE_Y[\VV[X|Y]] + \VV_Y[\EE[X|Y]] \text{ (variance property)}\\
  \end{aligned}
  \]

\begin{itemize}
\tightlist
\item
  MEM: It's about applying back and forth the alternative variance
  definition + law of total expectation
\end{itemize}
\end{frame}

\begin{frame}{Law of total variance: example}
\protect\phantomsection\label{law-of-total-variance-example}
\begin{itemize}
\tightlist
\item
  Let \(W = X_1 + X_2 + ... + X_N\) where:

  \begin{itemize}
  \tightlist
  \item
    \(X_i\) are IID with mean \(\mu_X\) and variance \(\sigma_X^2\)
  \item
    \(N\) is a RV independent of \(X_i\)
  \end{itemize}
\item
  What is \(\VV[W]\)?
\end{itemize}
\end{frame}

\begin{frame}{Law of total variance: example solution}
\protect\phantomsection\label{law-of-total-variance-example-solution}
\begin{itemize}
\tightlist
\item
  Given the law of total variance: \[
  \begin{aligned}
  \VV[W]
  &= \VV_N[\EE[W|N]] + \EE_N[\VV[W|N]\\
  &= \VV_N[\EE[\sum X_i|N]] + \EE_N[\VV[\sum X_i|N]\\
  &= \VV_N[N \mu_X] + \EE_N[N \sigma_X^2]\\
  &= \VV[N] \mu_X^2 + \EE[N] \sigma_X^2
  \end{aligned}
  \]

  By using just the law of total expectation: \[

  &= \EE_N[\sum_{i=1}^N \EE[X_i^2]] - ...\\
  &= \EE[N] (\sigma_X^2 - \mu_X^2) - \EE[N]^2 \mu_X^2 \\

\begin{frame}{Covariance of RV}
\protect\phantomsection\label{covariance-of-rv}
\begin{itemize}
\tightlist
\item
  Given two RVs \(X\) and \(Y\), the covariance is defined as:
  \[\Cov[X,Y] \defeq \EE[(X - \mu_X)(Y - \mu_Y)]\]
\item
  It is also indicated as \(\sigma_{X,Y}\)
\end{itemize}
\end{frame}

\begin{frame}{Compute covariance using theorem of mean}
\protect\phantomsection\label{compute-covariance-using-theorem-of-mean}
\begin{itemize}
\tightlist
\item
  Using the theorem of the mean, the covariance can be written in terms
  of the joint PDF:
  \[\Cov[X,Y] = \int_y \int_x (x - \mu_X)(y - \mu_Y)f_{X,Y}(x,y) dx dy\]
\end{itemize}
\end{frame}

\begin{frame}{Intuition of covariance}
\protect\phantomsection\label{intuition-of-covariance}
\begin{itemize}
\tightlist
\item
  It measures the strength of the \textit{linear} relationship between
  random variables \(X\) and \(Y\)
\end{itemize}
\end{frame}

\begin{frame}{Covariance in terms of mean}
\protect\phantomsection\label{covariance-in-terms-of-mean}
\[\Cov[X,Y] = \EE[X \cdot Y] - \EE[X] \cdot \EE[Y]\] - This is a
generalization of the expression of variance in terms of mean
\end{frame}

\begin{frame}{Covariance of independent RV}
\protect\phantomsection\label{covariance-of-independent-rv}
\begin{itemize}
\item
  If \(X\) and \(Y\) are independent RVs, then \(\Cov[X,Y] = 0\) (i.e.,
  uncorrelated)
\item
  Note that the converse is not true

  \begin{itemize}
  \tightlist
  \item
    Uncorrelated variables are not necessarily independent, simply there
    is no linear association
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Variance for sum / difference of RV}
\protect\phantomsection\label{variance-for-sum-difference-of-rv}
\begin{itemize}
\tightlist
\item
  In case of two general RVs:
  \[\VV[X \pm Y] = \VV[X] + \VV[Y] \pm 2 \cdot \Cov[X,Y]\]
\end{itemize}
\end{frame}

\begin{frame}{Relationship between covariance of RV and variance of RV}
\protect\phantomsection\label{relationship-between-covariance-of-rv-and-variance-of-rv}
\[|\Cov[X,Y]| \le \sqrt{\VV[X] \VV[Y]}\] - In other symbols:
\(|\sigma_{X,Y}| \le \sigma_X \sigma_Y\)
\end{frame}

\begin{frame}{Correlation coefficient of RVs}
\protect\phantomsection\label{correlation-coefficient-of-rvs}
\begin{itemize}
\item
  Aka Pearson correlation, Pearson rho
\item
  Given two RVs \(X\) and \(Y\), the correlation coefficient is defined
  as: \[
  \rho(X, Y)
  = \frac{\Cov[X,Y]}{\sqrt{\VV[X] \VV[Y]}}
  = \frac{\sigma_{X,Y}}{\sigma_X \sigma_Y}
  \]
\item
  It is also indicated as \(\rho_{X,Y}\)
\item
  Note that \(-1 \le \rho_{X,Y} \le 1\)
\end{itemize}
\end{frame}

\begin{frame}{Meaning of correlation coefficient}
\protect\phantomsection\label{meaning-of-correlation-coefficient}
\begin{itemize}
\tightlist
\item
  The coefficient of correlation is a \textit{normalized} measure of
  \textit{linear} dependence between RVs
\item
  In fact:

  \begin{itemize}
  \tightlist
  \item
    If \(X\) and \(Y\) are independent (or at least uncorrelated)
    \(\rho(X, Y) = 0\)
  \item
    If they are equal (or proportional) \(\rho(X, Y) = 1\)
  \item
    If they are opposite \(\rho(X, Y) = -1\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Rank of an array of numbers}
\protect\phantomsection\label{rank-of-an-array-of-numbers}
\begin{itemize}
\tightlist
\item
  Consider an array of numbers \(\vx\) (e.g., realizations of a RV)
\item
  The rank of the numbers \(\vx\) is the vector where each number
  \(x_i\) is replaced with its index in the sorted array \(\vx\)
  \[r_X(x_i) = sort(\vx).idx(x_i)\]
\end{itemize}
\end{frame}

\begin{frame}{Rank of an array of numbers: example}
\protect\phantomsection\label{rank-of-an-array-of-numbers-example}
\begin{itemize}
\tightlist
\item
  \(X = (7, 1, 9, 5)\)
\item
  Order the values in increasing order \(1, 5, 7, 9\),
\item
  Assign to \(r_X(x_i)\) the index in the array corresponding to the
  value of \(x_i\) \[r_X = (3, 1, 4, 2)\]
\end{itemize}
\end{frame}

\begin{frame}{Rank of an array of numbers: interpretation}
\protect\phantomsection\label{rank-of-an-array-of-numbers-interpretation}
\begin{itemize}
\tightlist
\item
  Ranking removes the magnitude of the values and retains only
  information about the order of the values and their mutual
  relationship
\end{itemize}
\end{frame}

\begin{frame}{Spearman rho: definition}
\protect\phantomsection\label{spearman-rho-definition}
\begin{itemize}
\tightlist
\item
  Aka rank correlation
\item
  Consider two RVs \(X, Y\) and their realizations \(\vx, \vy\)
\item
  Compute the rank variables \(r_{\vx}, r_{\vy}\) corresponding to the
  realizations of \(\vx\) and \(\vy\)
\item
  Spearman rho is defined as the (Pearson) correlation coefficients
  between the ranks of the realizations of two RVs \(X\) and \(Y\) \[
  \rho_S(X, Y)
  = \rho_P(r_{\vX}, r_{\vY})
  = \frac{\Cov[r_{\vX}, r_{\vY}]}{\sqrt{\VV[r_{\vX}] \cdot \VV[r_{\vY}]}}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Spearman rho: interpretation}
\protect\phantomsection\label{spearman-rho-interpretation}
\begin{itemize}
\tightlist
\item
  It is a non-parametric (i.e., there is no underlying model) measure of
  correlation
\item
  It assesses how well the relationship between two variables can be
  described by a monotonic function
\end{itemize}
\end{frame}

\begin{frame}{Pearson vs Spearman rho}
\protect\phantomsection\label{pearson-vs-spearman-rho}
\begin{itemize}
\tightlist
\item
  Pearson rho measures \textit{linear} relationship
\item
  Spearman rho measures \textit{monotonic non-linear} relationship
\end{itemize}
\end{frame}

\begin{frame}{Mode of a RV}
\protect\phantomsection\label{mode-of-a-rv}
\begin{itemize}
\tightlist
\item
  = the value of the RV that occurs most often (i.e., where PDF or PMF
  have a maximum)
\item
  Note that a RV can have multiple modes and be multimodal (e.g.,
  bimodal, trimodal)
\item
  The mode provides a measure of central tendency, like the mean
\end{itemize}
\end{frame}

\begin{frame}{Mean vs mode of a RV}
\protect\phantomsection\label{mean-vs-mode-of-a-rv}
\begin{itemize}
\item
  They are both measures of central tendency
\item
  The mean is the average value of the RV when doing infinite IID draws
  (LLN)
\item
  The mode is the most common value
\item
  The mean can be a value that the RV does not assume
\item
  The mode is a value of a RV
\item
  A RV has a single mean, but can have many modes
\end{itemize}
\end{frame}

\begin{frame}{Quantile of a RV}
\protect\phantomsection\label{quantile-of-a-rv}
\begin{itemize}
\tightlist
\item
  The \(\alpha\)-th quantile (with \(0 \le \alpha \le 1\)) of a RV \(X\)
  is the value \(x_\alpha \in \bbR\) of the RV such that:

  \begin{itemize}
  \tightlist
  \item
    In terms of probability: \(\Pr(X \le x_\alpha) = \alpha\)
  \item
    In terms of CDF: \(F_X(x_\alpha) = \alpha\) (i.e., the inverse of
    the CDF)
  \item
    In terms of PDF: the portion of the PDF on the left of \(x_\alpha\)
    is equal to \(\alpha\)
  \end{itemize}
\item
  MEM: \(x_{\alpha} = F_X^{-1}(\alpha)\)
\end{itemize}
\end{frame}

\begin{frame}{Quantile of a RV: more general definition}
\protect\phantomsection\label{quantile-of-a-rv-more-general-definition}
\begin{itemize}
\tightlist
\item
  For discrete RVs \(X\) the quantile value \(x_{\alpha}\) might be not
  unique or be undefined
\item
  In this case the definition is:
  \[q_{\alpha} = \inf_x \{ x: \Pr(x) \ge \alpha \}\]
\end{itemize}
\end{frame}

\begin{frame}{Percentile of a RV}
\protect\phantomsection\label{percentile-of-a-rv}
\begin{itemize}
\item
  = the same as quantile where \(\alpha\) is expressed as a percent
  (i.e., in {[}0\%, 100\%{]}) instead of {[}0, 1{]}
\item
  E.g., 10th percentile (also called first decile) corresponds to the
  \(\alpha=0.1\) quantile

  \begin{itemize}
  \tightlist
  \item
    I.e., it gives an area under of the PDF to the left of it equal to
    0.1
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Median of a RV}
\protect\phantomsection\label{median-of-a-rv}
\begin{itemize}
\item
  Aka 50th percentile, 0.5 quantile, or the fifth decile
\item
  The median of a RV \(X\) is the value \(x_{0.5}\) of the RV such that:

  \begin{itemize}
  \tightlist
  \item
    In terms of probability: \(\Pr(X \le x_{0.5}) = 0.5\), i.e., there
    is a 50-50 chance of getting a smaller or larger value of \(X\) than
    \(x_{0.5}\)
  \item
    In terms of CDF: \(F_X(x_{0.5}) = 0.5\)
  \item
    For continuous RV the median separates the PDF into 2 parts with
    equal underlying area 0.5
  \item
    For discrete RV, the median might not exist or might not be unique,
    due to the discreteness of the CDF / PMF
  \end{itemize}
\item
  It is a measure of central tendency (like mean and mode)
\end{itemize}
\end{frame}

\begin{frame}{Median is more robust than mean}
\protect\phantomsection\label{median-is-more-robust-than-mean}
\begin{itemize}
\tightlist
\item
  One outlier can affect the mean, since its effect is squared
\item
  Outliers have a smaller effect on the median, since only the order
  (and not the magnitude) is considered
\end{itemize}
\end{frame}

\begin{frame}{Geometric mean}
\protect\phantomsection\label{geometric-mean}
\begin{itemize}
\tightlist
\item
  Given \(N\) RVs or values with \(X_i \ge 0\)
  \[GM = \sqrt[N]{\prod_{i=1}^N X_i}\]
\item
  MEM: \(AM \ge GM\)
\item
  MEM: AM overestimates the true return, which is the GM
\end{itemize}
\end{frame}

\begin{frame}{Geometric mean in terms of arithmetic mean}
\protect\phantomsection\label{geometric-mean-in-terms-of-arithmetic-mean}
\begin{itemize}
\tightlist
\item
  The geometric can be written in terms of arithmetic mean: \[
  \begin{aligned}
  GM
  & = \sqrt[N]{\prod_{i=1}^N X_i} \\
  & = \exp(\frac{\sum_{i=1}^N \log(X_i)}{N}) \\
  & = \exp(\avg(\log(X_1), ..., \log(X_n)) \\
  \end{aligned}
  \]
\item
  In words, the geometric mean is the exponential of the arithmetic mean
  of the logarithm of the values
\item
  MEM: log, average, exp
\end{itemize}
\end{frame}

\begin{frame}{Harmonic mean}
\protect\phantomsection\label{harmonic-mean}
\[
  HM
  = 1 / \avg(\frac{1}{X_1}, ..., \frac{1}{X_n})
  = \frac{1}{(\frac{1}{N} \sum \frac{1}{X_i})}
  = \frac{n}{\sum \frac{1}{X_i}}
  \] - In words, the harmonic mean is the reciprocal of the arithmetic
mean of the reciprocals - MEM: \(HM \ge GM\)
\end{frame}

\begin{frame}{Interquartile range of a RV}
\protect\phantomsection\label{interquartile-range-of-a-rv}
\begin{itemize}
\tightlist
\item
  = the difference between the 75 and 25 percentile, i.e.,
  \(x_{0.75} - x_{0.25}\)
\item
  It measures how big is the \(x\) range that contains 50\% of the mass
  around the median
\item
  It is a measure of dispersion of a RV, like the variance
\end{itemize}
\end{frame}

\begin{frame}{Mean absolute deviation}
\protect\phantomsection\label{mean-absolute-deviation}
\begin{itemize}
\tightlist
\item
  Aka MAD
\item
  It is defined as: \[MAD \defeq \EE[|X - \mu|]\]
\item
  It is a measure of dispersion of a RV, like the variance, but it
  weights the outliers less heavily than variance
\item
  It is not differentiable
\end{itemize}
\end{frame}

\begin{frame}{Semi-variance}
\protect\phantomsection\label{semi-variance}
\begin{itemize}
\tightlist
\item
  Sometimes we want to differentiate between upward and downward
  deviation

  \begin{itemize}
  \tightlist
  \item
    E.g., in case of returns for an asset, we are more concerned in
    downward deviations
  \end{itemize}
\item
  Downward semi-variance is defined:
  \[\frac{\sum_{X_i < \mu} (X_i - \mu)^2}{\sum_{X_i < \mu} 1}\]
\end{itemize}
\end{frame}

\begin{frame}{Skewness}
\protect\phantomsection\label{skewness}
\begin{itemize}
\item
  Skewness measures which side of the distribution is ``heavier'', and
  it is defined as: \[\EE[\left(\frac{X - \mu}{\sigma}\right)^3]\]
\item
  MEM: It is the mean of the cube of the z-score of the RV
\item
  \textbf{\emph{Notes}}
\item
  For symmetric distributions skewness = 0
\item
  Positive skewness means that the distribution has a longer tail to the
  right, and the peak is towards left

  \begin{itemize}
  \tightlist
  \item
    MEM: The skewness \(>\) or \(< 0\) points to where is the heavier
    tail
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Skewness: interpretation}
\protect\phantomsection\label{skewness-interpretation}
\begin{itemize}
\tightlist
\item
  A distribution can be not symmetric and have one side with more mass
  than the other
\item
  MEM: Think of a Gaussian, keep it centered, then move part of the peak
  towards the left, so the extra mass goes in the right tail (the mass
  needs to go somewhere)
\end{itemize}
\end{frame}

\begin{frame}{Kurtosis}
\protect\phantomsection\label{kurtosis}
\begin{itemize}
\item
  Kurtosis measures the peaked-ness of the distribution, and it is
  defined as: \[\EE[\left(\frac{X - \mu}{\sigma}\right)^4]\]
\item
  MEM: It is the mean of the 4th power of the z-score of the RV
\item
  \textbf{\emph{Notes}}
\item
  High kurtosis means sharper peak and fatter tails

  \begin{itemize}
  \tightlist
  \item
    MEM: High kurtosis is bad!
  \end{itemize}
\item
  Low kurthosis means rounder peak and thinner tails
\item
  MEM: High kurtosis means sharp peak (Kurt is very thin)
\end{itemize}
\end{frame}

\begin{frame}{Excess kurtosis}
\protect\phantomsection\label{excess-kurtosis}
\begin{itemize}
\tightlist
\item
  A Gaussian has kurtosis = 3
\item
  For this reason excess kurtosis refers to a Gaussian as baseline:
  \[\text{excess kurtosis = kurtosis} - 3\]
\end{itemize}
\end{frame}

\begin{frame}{Kurtosis: interpretation}
\protect\phantomsection\label{kurtosis-interpretation}
\begin{itemize}
\tightlist
\item
  A distribution can have values concentrated near the mean or on the
  tails so that it has

  \begin{itemize}
  \tightlist
  \item
    Thick peak and shallow tails, or
  \item
    Thin peak and fat tails
  \end{itemize}
\item
  MEM: One can start with a Gaussian and then make the peak thinner, the
  mass needs to go somewhere, and it goes in the tails
\end{itemize}
\end{frame}

\begin{frame}{Markov inequality}
\protect\phantomsection\label{markov-inequality}
\begin{itemize}
\item
  \textbf{\emph{Hypothesis}}
\item
  Given \(X\) discrete or continuous RV
\item
  \(X\) is a non-negative RV (i.e., \(X \ge 0\), PDF is all after 0)
\item
  \(X\) has finite mean: \(\EE[X] < \infty\)
\item
  \textbf{\emph{Thesis}}
\item
  The probability that \(X\) is larger than a certain value is bounded
  by the mean \[\Pr(X \ge x) \le \frac{\EE[X]}{x}\]
\end{itemize}
\end{frame}

\begin{frame}{Markov inequality: geometric interpretation}
\protect\phantomsection\label{markov-inequality-geometric-interpretation}
\begin{itemize}
\item
  Given a RV \(X \ge 0\) with a finite mean
\item
  The ``flipped CDF'' \(1 - F_X(x)\) is dominated by an hyperbole
  passing by \((y, x) = (\EE[X], 1)\)
\item
  This is also related to the fact that a PDF needs to sum to 1 and thus
  needs to decrease at least like \(1/n\)
\end{itemize}
\end{frame}

\begin{frame}{Chebyshev inequality}
\protect\phantomsection\label{chebyshev-inequality}
\begin{itemize}
\item
  \textbf{\emph{Hypothesis}}
\item
  Given \(X\) discrete or continuous RV
\item
  \(X\) with finite mean \(\mu\) and variance \(\sigma^2\)
\item
  \textbf{\emph{Thesis}}
\item
  The probability that \(X\) is far from the mean is bound by the
  variance:
  \[\Pr(|X - \mu| \ge \varepsilon) \le \frac{\sigma^2}{\varepsilon^2}\]
\end{itemize}
\end{frame}

\begin{frame}{Chebyshev inequality in terms of z-scores}
\protect\phantomsection\label{chebyshev-inequality-in-terms-of-z-scores}
\begin{itemize}
\item
  \textbf{\emph{Hypothesis}}
\item
  Given \(X\) discrete or continuous RV
\item
  \(X\) with finite mean \(\mu\) and variance \(\sigma^2\)
\item
  \textbf{\emph{Thesis}}
\item
  Expressing the distance from the mean in terms of standard deviation
  \(\varepsilon = k \sigma\):
  \[\Pr(\frac{|X - \mu|}{\sigma} \ge k) \le \frac{1}{k^2}\]
\item
  The probability that the z-score of a RV is far away from 0 at least a
  certain number \(k\) is bounded by \(\frac{1}{k ^ 2}\)
\end{itemize}
\end{frame}

\begin{frame}{Comparing Markov and Chebyshev inequalities}
\protect\phantomsection\label{comparing-markov-and-chebyshev-inequalities}
\begin{itemize}
\tightlist
\item
  Markov assumes \(X \ge 0\)
\item
  Chebyshev makes no assumptions
\item
  Both inequalities have a similar form:
  \[\Pr(X \ge x) \le \frac{\mu}{x}\]
  \[\Pr(|X - \mu| \ge x) \le \frac{\sigma^2}{x^2}\]
\end{itemize}
\end{frame}

\begin{frame}{Hoeffding inequality}
\protect\phantomsection\label{hoeffding-inequality}
\begin{itemize}
\tightlist
\item
  Given a Bernoulli RV with probability of success \(\mu\)
\item
  We want to estimate \(\mu\) using \(N\) samples:
  \[\nu = \frac{1}{N} \sum_{i=1}^N X_i\]
\item
  Then \[\Pr(|\nu - \mu| > \varepsilon) \le 2 e^{-2 \varepsilon^2 N}\]
\item
  Since \(\nu\) is bound in \([\mu - \varepsilon, \mu + \varepsilon]\),
  we want a small \(\varepsilon\) with a large probability
\end{itemize}
\end{frame}

\begin{frame}{Statistical inference}
\protect\phantomsection\label{statistical-inference}
\begin{itemize}
\tightlist
\item
  = process of generating a conclusion on a large population of objects
  from a small sample of the population
\end{itemize}
\end{frame}

\begin{frame}{Population vs sample}
\protect\phantomsection\label{population-vs-sample}
\begin{itemize}
\tightlist
\item
  Population = the entire group of objects
\item
  Sample = small part of the population
\end{itemize}
\end{frame}

\begin{frame}{Examples of statistical inference}
\protect\phantomsection\label{examples-of-statistical-inference}
\begin{itemize}
\tightlist
\item
  Draw a conclusion about:

  \begin{itemize}
  \tightlist
  \item
    The fairness of a coin by tossing it repeatedly
  \item
    The weights (or heights) of 12,000 students, selecting only 100
    students
  \item
    Defective bolts produced in a factory, by looking at 20 bolts
    manufactured during each day in a 6 day week (sample size = 120)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sampling with / without replacement}
\protect\phantomsection\label{sampling-with-without-replacement}
\begin{itemize}
\tightlist
\item
  If we draw an element from a set, we have the choice of replacing it
  or not before drawing again
\end{itemize}
\end{frame}

\begin{frame}{IID samples}
\protect\phantomsection\label{iid-samples}
\begin{itemize}
\tightlist
\item
  Given a RV \(X \sim F\), we can draw \(N\) times from its distribution
  (i.e., without replacement) getting \(N\) samples \(X_i \sim F\)
\item
  These samples \(X_i\) are Independent Identically Distributed (IID)
\end{itemize}
\end{frame}

\begin{frame}{IID samples as idealized condition}
\protect\phantomsection\label{iid-samples-as-idealized-condition}
\begin{itemize}
\tightlist
\item
  We can sample without replacement from a distribution \(F\)

  \begin{itemize}
  \tightlist
  \item
    \(F\) is a distribution so there are infinite samples and not a
    finite collection of objects
  \end{itemize}
\item
  If sampling was done with replacement, \(X_i\) and \(X_j\) could be
  the same and thus \(X_i\) could not be independent
\end{itemize}
\end{frame}

\begin{frame}{Sample statistics}
\protect\phantomsection\label{sample-statistics}
\begin{itemize}
\tightlist
\item
  A sample statistics \(Y\) is a deterministic function of given samples
  \(X_1, ..., X_N\) of a population: \[Y = g(X_1, ..., X_N)\]
\item
  In general we are interested in functions that ``summarize''
  properties of the samples

  \begin{itemize}
  \tightlist
  \item
    E.g., \(g()\) can be mean, variance
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sample statistics is a RV}
\protect\phantomsection\label{sample-statistics-is-a-rv}
\begin{itemize}
\tightlist
\item
  A sample statistics \(Y = g(X_1, ..., X_N)\) is a RV, since it is a
  function of RVs \(X_i\)
\item
  In other words we can draw samples \(x^{(k)}_i\) of \(X_i\) and get a
  different realization \(y^{(k)}\) of the sample statistic \(Y\)
  \[y^{(k)} = g(x^{(k)}_1, ..., x^{(k)}_N)\]
\end{itemize}
\end{frame}

\begin{frame}{Example of sample statistics}
\protect\phantomsection\label{example-of-sample-statistics}
\begin{itemize}
\tightlist
\item
  \(X\) is a RV modeling the height of a student population \(P\)
\item
  Pick 100 students randomly and have \(X_1, ..., X_{100}\) RVs from the
  population \(P\)
\item
  In one sample we have a realization for each student height
  \(x_1, ..., x_{100}\)
\item
  Then we compute a sample statistic \(h_1 = g(x_1, ..., x_{100})\):
  this is a realization of the RV sample statistics \(H\)
\end{itemize}
\end{frame}

\begin{frame}{Example of sample statistics: OLS beta}
\protect\phantomsection\label{example-of-sample-statistics-ols-beta}
\begin{itemize}
\tightlist
\item
  Assume \(Y = \alpha + \beta X + \varepsilon\)
\item
  Estimate \(\beta\) through OLS, thus \(\hat{\beta}\) is

  \begin{itemize}
  \tightlist
  \item
    A sample statistics
  \item
    A RV, since it is function of the specific samples of \(x_i\) and
    \(y_i\) \[
    \hat{\beta}
    = \frac{\overline{\Cov}(Y, X)}{\overline{\VV}[X]}
    = \frac{\frac{1}{N} \sum (y_i - \overline{y})(x_i - \overline{x})}
    {\frac{1}{N} \sum (x_i - \overline{x})^2}
    \]
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sampling distribution of a sample statistics}
\protect\phantomsection\label{sampling-distribution-of-a-sample-statistics}
\begin{itemize}
\tightlist
\item
  Since a sample statistics is a RV, it has a probability distribution
\item
  This distribution is called ``sampling distribution of the (sample)
  statistics''
\end{itemize}
\end{frame}

\begin{frame}{How to evaluate sampling distribution?}
\protect\phantomsection\label{how-to-evaluate-sampling-distribution}
\begin{enumerate}
\tightlist
\item
  Closed form

  \begin{itemize}
  \tightlist
  \item
    Sometimes one knows the distribution of the sample statistics, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Average of Gaussians is Gaussian
    \end{itemize}
  \end{itemize}
\item
  Enumeration

  \begin{itemize}
  \tightlist
  \item
    Consider all the possible samples, e.g., 100 samples from a
    population of 1,000, i.e., \(1000 \choose 100\)
  \item
    Compute the probability distribution of the sample statistic
  \end{itemize}
\item
  Approximation

  \begin{itemize}
  \tightlist
  \item
    Estimate the distribution of the sample statistics by sampling,
    e.g.,

    \begin{itemize}
    \tightlist
    \item
      Empirical distribution
    \item
      Bootstrap
    \end{itemize}
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Estimator properties}
\protect\phantomsection\label{estimator-properties}
\begin{itemize}
\tightlist
\item
  One can estimate different statistics of a RV (e.g., mean, variance,
  skewness, PDF) with an estimator, which is a sample statistics
\item
  The estimator is a RV which has:

  \begin{itemize}
  \tightlist
  \item
    A mean (ideally equal to the mean of the estimated, aka
    un-biased-ness)
  \item
    A std dev (std err of sample statistics)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Estimator properties: examples}
\protect\phantomsection\label{estimator-properties-examples}
\begin{itemize}
\tightlist
\item
  An estimator of:

  \begin{itemize}
  \tightlist
  \item
    The mean of \(X\) has a std dev, which is not the std dev of \(X\)
    although it is related to it
  \item
    The std dev of \(X\) has a std dev in turn
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Sample mean}
\protect\phantomsection\label{sample-mean}
\begin{itemize}
\tightlist
\item
  Draw \(n\) IID samples \(X_1, ..., X_n\) from a population
\item
  The sample mean (or ``mean of the sample'') is the RV: \[
  \overline{X}
  = \overline{X}_n
  \defeq \frac{1}{n} \sum_{i=1}^n X_i
  = \overline{\EE}[X_i]
  \]
\end{itemize}
\end{frame}

\protect\phantomsection\label{standard-error-of-a-sample-statistics}
\begin{itemize}
\tightlist
\item
  = standard deviation of the sample distribution of a sample statistic
  (e.g., mean, variance, \ldots), e.g.,

  \begin{itemize}
  \tightlist
  \item
    Standard error of the mean
  \item
    Standard error of the variance
  \item
    Standard error of OLS regression coefficients
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Standard error of the mean}
\protect\phantomsection\label{standard-error-of-the-mean}
\begin{itemize}
\tightlist
\item
  = standard deviation of the sample mean of \(n\) IID samples
\item
  Indicated with \(\sigma_{\overline{X}}\), \(SE_{\overline{X}}\), SEM
\end{itemize}
\end{frame}

\begin{frame}{Sample mean is an unbiased estimator}
\protect\phantomsection\label{sample-mean-is-an-unbiased-estimator}
\begin{itemize}
\tightlist
\item
  Assume that we want to estimate the mean \(\mu\) of a population
\item
  We take \(n\) IID samples of the population \(X_1, ..., X_n\) (\(n\)
  RVs)
\item
  The sample mean is defined as:
  \[\overline{X} = \frac{1}{n} \sum_{i=1}^n X_i\]
\item
  Different \(n\) samples drawn from the same population will give
  different values of the sample mean, so the sample mean is a RV
\item
  The sample mean is an unbiased estimator of the population mean, since
  \[
  \EE[\overline{X}]
  = \EE[ \frac{1}{n} \sum_{i=1}^n X_i ]
  = \frac{1}{n} \sum_{i=1}^n \EE[X_i]
  = \frac{1}{n} \sum_{i=1}^n \EE[X]
  = \EE[X]
  \]
\end{itemize}
\end{frame}

\begin{frame}{Standard error of the mean}
\protect\phantomsection\label{standard-error-of-the-mean-1}
\begin{itemize}
\item
  Assume that we want to estimate the SEM of a population
\item
  We take \(n\) IID samples of the population \(X_1, ..., X_n\) (\(n\)
  RVs),
\item
  The sample mean is defined as: \[\overline{X} = \frac{1}{n} \sum X_i\]
  and it is a RV
\item
  The standard error of the mean is the standard deviation of
  \(\overline{X}\) and it is equal to \[
  \begin{aligned}
  \VV[\overline{X}]
  &\defeq \EE[(\overline{X} - \EE[\overline{X}]]
  = \EE[(\frac{1}{n} \sum_i X_i - \mu)^2] \\
  &= \EE[(\frac{1}{n} (\sum_i X_i - \mu))^2]
  = (\frac{1}{n^2} \EE[(\sum_i X_i - \mu))^2] \\
  &= (\frac{1}{n^2} \EE[(\sum_i (X_i - \mu))^2]
  = (\frac{1}{n^2} n \EE[(X - \mu))^2] \\
  &= \frac{1}{n^2} n \VV[X]
  = \frac{1}{n} \VV[X] \\
  \end{aligned}
  \]
\item
  In other terms \[
  \sigma_{\overline{X}}
  = SE_{\overline{X}}
  = \text{SEM}
  = \frac{\sigma_X}{\sqrt{n}}
  \] where \(\sigma_X\) is the std dev of the population \(X\) and \(n\)
  is the number of observations
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of the formula for std err of the mean}
\protect\phantomsection\label{interpretation-of-the-formula-for-std-err-of-the-mean}
\begin{itemize}
\tightlist
\item
  This formula makes sense since:

  \begin{itemize}
  \tightlist
  \item
    If the variation of the underlying distribution being estimated
    (i.e., \(\sigma_X\)) is larger, the SEM is also larger
  \item
    If the number of samples \(n\) is larger, the SEM is smaller
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Estimate of standard error of the mean}
\protect\phantomsection\label{estimate-of-standard-error-of-the-mean}
\begin{itemize}
\tightlist
\item
  We know that
  \[\VV[\overline{X}] = \frac{\VV[X]}{n} = \frac{\sigma_X^2}{n}\] but we
  might not know \(\sigma_X\)
\item
  In many formulas, if we don't know the std dev of the population
  \(\sigma_X\), we can use the sample standard deviation \(S\)
  \[\VV[\overline{X}] \approx \frac{S^2}{n}\]
\end{itemize}
\end{frame}

\begin{frame}{Summary of properties for sample mean}
\protect\phantomsection\label{summary-of-properties-for-sample-mean}
\begin{itemize}
\item
  Draw \(n\) IID samples \(X_1, ..., X_n\) from a population
\item
  Compute sample mean \(\overline{X}\) from the samples
\item
  What is the relationship between the probability distribution of the
  sample mean \(\overline{X}\) and \(X\)?
\item
  \textbf{\emph{Expected value}}
\item
  The population mean \(\EE[X]\) is the center of mass of the population
  distribution
\item
  The sample mean \(\overline{X}\) is the center of mass of the observed
  data distribution
\item
  The sample mean is an unbiased estimate of the population mean, i.e.,
  \(\EE[\overline{X}] = \EE[X]\)
\item
  \textbf{\emph{Variance}}
\item
  The more data \(n\) is used to compute the sample mean
  \(\overline{X}\), the more concentrated is the PDF / PMF of the sample
  mean around the population mean
\item
  If

  \begin{itemize}
  \tightlist
  \item
    Samples are IID (e.g., the population is infinite), or
  \item
    The population is finite and sampling is \text{with} replacement
    then \[\VV[\overline{X}] = \frac{\VV[X]}{n}\]
  \end{itemize}
\item
  If the population is finite (with size \(N\)) and sampling (with size
  \(n\)) is \textit{without} replacement, then:
  \[\VV[\overline{X}] = \frac{\VV[X]}{n}\frac{N - n}{N - 1}\]
\end{itemize}
\end{frame}

\begin{frame}{Unbiased estimator of population variance knowing
population mean \(\mu_X\)}
\protect\phantomsection\label{unbiased-estimator-of-population-variance-knowing-population-mean-mu_x}
\begin{itemize}
\item
  If we know the mean \(\mu_X\) of the underlying population \(X\), then
  \[S^2 = \frac{1}{n} \sum_i (X_i - \mu_X)^2\] is an unbiased estimation
  of the population variance \(\VV[X]\), i.e., \[\EE[S^2] = \VV[X]\]
\item
  Aka sample variance
\item
  \textbf{\emph{Proof}} \[
  \EE[S^2]
  \defeq \EE[\frac{1}{n} \sum_i (X_i - \mu_X)^2]
  = \frac{1}{n} \sum_i \EE[(X_i - \mu)^2]
  = \frac{1}{n} \sum_i \VV[X]
  = \VV[X]
  \]
\end{itemize}
\end{frame}

\begin{frame}{Unbiased estimator of population variance not knowing
\(\mu_X\)}
\protect\phantomsection\label{unbiased-estimator-of-population-variance-not-knowing-mu_x}
\begin{itemize}
\item
  If we need to estimate the mean of the underlying population from the
  data, then the sample variance:
  \[S^2 = \frac{1}{n-1} \sum_i (X_i - (\frac{1}{n} \sum_j X_j))^2\] is
  an unbiased estimate of the variance of \(X\), i.e.,
  \(\EE[S^2] = \VV[X]\)
\item
  Note that we need to divide by \(n - 1\), instead of \(n\) to get an
  unbiased estimate
\item
  This is because the mean is also estimate from the data and it is
  using a degree of freedom
\end{itemize}
\end{frame}

\begin{frame}{Sample variance as RV}
\protect\phantomsection\label{sample-variance-as-rv}
\begin{itemize}
\tightlist
\item
  Since the sample variance \(S^2\) is a function of the data, then
  \(S^2\)

  \begin{itemize}
  \tightlist
  \item
    Is a RV
  \item
    Has a population distribution
  \end{itemize}
\item
  The expected value of the population distribution of the sample
  variance \(\EE[S^2]\) is the variance of the population that we are
  estimating \(\VV[X]\) (unbiased estimate)
\item
  The more data \(n\) we have

  \begin{itemize}
  \tightlist
  \item
    The more concentrated the distribution of \(S^2\) is
  \item
    We don't have a relationship for it in general (it is function of
    higher moments), so we can use numerical techniques (e.g.,
    bootstrap)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Asymptotics}
\protect\phantomsection\label{asymptotics}
\begin{itemize}
\tightlist
\item
  = behavior of sample statistics as the sample size \(n\) goes to
  infinity
\item
  E.g., Law of Large Numbers (LLN) and Central Limit Theorem (CLT)
\end{itemize}
\end{frame}

\begin{frame}{LLN vs CLT}
\protect\phantomsection\label{lln-vs-clt}
\begin{itemize}
\tightlist
\item
  Both are statements about the sample mean

  \begin{itemize}
  \tightlist
  \item
    LLN: sample mean is consistent
  \item
    CLT: sample mean is asymptotically Gaussian
  \end{itemize}
\item
  Both are about asymptotic behaviors \(n \to \infty\)
\item
  Both apply to continuous and discrete RVs
\end{itemize}
\end{frame}

\begin{frame}{Consistent estimator}
\protect\phantomsection\label{consistent-estimator}
\begin{itemize}
\tightlist
\item
  An estimator is consistent \(\iff\) its value converges to what should
  estimate as the amount of collected data goes to infinity
\end{itemize}
\end{frame}

\begin{frame}{Consistent vs unbiased estimator}
\protect\phantomsection\label{consistent-vs-unbiased-estimator}
\begin{itemize}
\tightlist
\item
  An \textit{unbiased} estimate refers to averaging an infinite number
  \(\EE\) of times a fixed number \(n\) of samples
  \[\EE[g(X_1, ..., X_n)]\]
\item
  A \textit{consistent} estimate refers to doing a single average of a
  diverging number of samples \(n\)
  \[\lim_{n \to \infty} g(X_1, ..., X_n)\]
\end{itemize}
\end{frame}

\begin{frame}{Law of Large Numbers (LLN) in few words}
\protect\phantomsection\label{law-of-large-numbers-lln-in-few-words}
\begin{itemize}
\tightlist
\item
  The LLN is about consistency of the sample mean, i.e., it tells us
  what happens to the sample mean when we collect an infinite amount of
  samples
\end{itemize}
\end{frame}

\begin{frame}{Law of Large Numbers (LLN)}
\protect\phantomsection\label{law-of-large-numbers-lln}
\begin{itemize}
\tightlist
\item
  The sample mean of IID samples:
  \[\overline{X} = \frac{1}{n} \sum_{i=1}^n X_i\] is a consistent
  estimator for the population mean, i.e.,
  \[\lim_{n \to \infty} \frac{1}{n} \sum_{i=1}^n X_i = \EE[X]\]
\item
  The convergence is in probability
\end{itemize}
\end{frame}

\begin{frame}{LLN for other estimators}
\protect\phantomsection\label{lln-for-other-estimators}
\begin{itemize}
\tightlist
\item
  LLN applies also to other estimators that rely on the mean, e.g.,
  variance and std dev:
  \[\frac{1}{n} \sum_j h(X_i) \rightarrow \int g(y)dF_X(y) = \EE[g(X)]\]
  where the convergence is in probability
\item
  TODO: who is h vs g?
\end{itemize}
\end{frame}

\begin{frame}{LLN for variance}
\protect\phantomsection\label{lln-for-variance}
\begin{itemize}
\tightlist
\item
  For the variance: \[
  \VV[\overline{X}]
  = \EE[(\overline{X} - \EE[\overline{X}])^2]
  = \EE[\overline{X}^2] - (\EE[\overline{X}])^2
  \]
\item
  Using LLN: \[
  \lim_{n \to \infty} \VV[\overline{X}]
  = \EE[X^2] - (\EE[X])^2
  = \VV[X]
  \] where we used \(g(x) = x^2\) in the first term and the previous
  result, and applying the LLN for the mean to the second term
\end{itemize}
\end{frame}

\begin{frame}{Example of LLN: Bernoulli distribution}
\protect\phantomsection\label{example-of-lln-bernoulli-distribution}
\begin{itemize}
\tightlist
\item
  Consider IID draws \(X_i\) from a Bernoulli variable with a certain
  \(p\)
\item
  From LLN
  \[\lim_{n \to \infty} \frac{1}{n} \sum_{i=1}^n X_i = \EE[X] = p\]
\item
  In words the fraction of times that the coin comes up heads (i.e., the
  relative frequency) approximates the probability of success for
  \(n \to \infty\)

  \begin{itemize}
  \tightlist
  \item
    This is the basis for the frequentist interpretation of probability
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Central Limit Theorem (CLT) in few words}
\protect\phantomsection\label{central-limit-theorem-clt-in-few-words}
\begin{itemize}
\tightlist
\item
  The CLT tells that the shape of the distribution of the sample mean
  \(\overline{X}\) for large \(n\) is Gaussian, independently from the
  PDF of the sampled distribution \(X\)
\end{itemize}
\end{frame}

\begin{frame}{Central Limit Theorem (CLT)}
\protect\phantomsection\label{central-limit-theorem-clt}
\begin{itemize}
\item
  Consider the sample mean of IID samples
  \[\overline{X} = \frac{1}{n} \sum_{i=1}^n X_i\] then for large \(n\)
  \(\overline{X}\) is Gaussian, independently from the PDF of the
  sampled distribution \(X\)
\item
  Note that CLT does not tell anything about the rate of convergence to
  a Gaussian in terms of the value of \(n\)
\end{itemize}
\end{frame}

\begin{frame}{CLT + LLN}
\protect\phantomsection\label{clt-lln}
\begin{itemize}
\item
  Combining all the asymptotic results for \(\overline{X}\): a) CLT
  about the shape of \(\overline{X}\) b) LLN for the mean of
  \(\overline{X}\) c) variance of \(\overline{X} = \VV[x] / n\)
\item
  We obtain that for large \(n\)

  \begin{enumerate}
  \item
    Sample mean is Gaussian centered on the population mean and with a
    variance related to the population variance
    \[\overline{X} \sim N(\EE[X], \frac{\VV[X]}{n})\]
  \item
    Z-scoring 1)
    \[\frac{\overline{X} - \mu_X}{\sigma_X / \sqrt{n}} \sim N(0, 1)\]
  \item
    Using the sample estimates of the unknown quantities \[
    \frac{\overline{X} - \mu_{\overline{X}}}{\sigma_{\overline{X}}}
    \sim N(0, 1)
    \]
  \item
    In words 3) \[
    \frac{\text{estimate} - \text{mean of estimate}}
    {\text{std err of estimate}}
    \sim N(0, 1)
    \]
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Example of CLT for fair dice}
\protect\phantomsection\label{example-of-clt-for-fair-dice}
\begin{itemize}
\item
  Let \(X_i\) be the outcome of rolling a fair die

  \begin{itemize}
  \tightlist
  \item
    We know that \(\EE[X] = 3.5\) and \(\VV[X] = 2.92\), so we don't
    have to estimate anything from the data
  \end{itemize}
\item
  We roll \(n\) dice and take the average \(\overline{X}\), i.e.,
  compute the sample mean
\item
  The sample mean is a discrete RV
\item
  The CLT tells that:
  \[\frac{\overline{X} - \mu}{\sigma / \sqrt{n}} \to Z\] tends to a
  standard normal for large \(n\)
\item
  We can verify this numerically by:

  \begin{itemize}
  \tightlist
  \item
    Performing many experiments of averaging \(n\) die outcomes
  \item
    Standardizing the resulting distribution
  \item
    Plotting the histogram against a standard normal
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Example of CLT for biased coin flip}
\protect\phantomsection\label{example-of-clt-for-biased-coin-flip}
\begin{itemize}
\item
  Let \(X_i\) be the outcome of a biased coin with unknown probability
  of success \(p\)
\item
  We know that \(\EE[X] = p\) and \(\VV[X] = p (1 - p)\)
\item
  How to estimate \(p\)?
\item
  \textbf{\emph{Solution}}
\item
  Let's call \(\hat{p}\) the sample proportion of successes:
  \[\hat{p} = \frac{1}{n} \sum_{i=1}^n X_i\]
\item
  The CLT tells us that:
  \[\frac{\hat{p} - p}{\sqrt{\frac{p (1 - p)}{n}}}\] tends to a standard
  normal for large \(n\), so we can estimate \(\hat{p}\)
\item
  Note that the approximation gets better for larger \(n\), but there is
  no information on the rate of convergence, e.g., for different values
  of the params the rate of convergence can be different
\end{itemize}
\end{frame}

\begin{frame}{Confidence intervals for a statistic}
\protect\phantomsection\label{confidence-intervals-for-a-statistic}
\begin{itemize}
\tightlist
\item
  Given a statistic \(Y\) (e.g., mean, median) of a RV \(X\), the
  \(\alpha\)-confidence interval \(I\) is the interval that contains the
  true value of the statistic with a certain probability \(\alpha\):
  \[\Pr(Y \in I) = \alpha\]
\end{itemize}
\end{frame}

\begin{frame}{Confidence intervals for the sample mean}
\protect\phantomsection\label{confidence-intervals-for-the-sample-mean}
\begin{itemize}
\tightlist
\item
  The \(\alpha\)-confidence interval for the sample mean
  \(\overline{X}\) is the interval \(I\) around \(\mu_X\) such that
  \(\Pr(\mu \in I) = \alpha\)
\end{itemize}
\end{frame}

\begin{frame}{Confidence intervals for the mean using sample mean}
\protect\phantomsection\label{confidence-intervals-for-the-mean-using-sample-mean}
\begin{itemize}
\tightlist
\item
  Every time we know the distribution of the sample mean
  \(\overline{X}\), we can compute confidence intervals for the mean of
  the underlying population \(\mu_X\)
\end{itemize}
\end{frame}

\begin{frame}{Correct interpretation of confidence intervals}
\protect\phantomsection\label{correct-interpretation-of-confidence-intervals}
\begin{itemize}
\item
  Assume we estimate the confidence interval of average height of female
  population in US using the sample mean and standard error
\item
  We claim: ``the national mean female height is between 63 and 65
  inches with 95\% probability''
\item
  \textbf{\emph{Incorrect interpretation}}
\item
  We have no way to assess the probability of the confidence interval to
  contain the population mean, since it's unknown
\item
  This is a misinterpretation of the meaning of confidence intervals
\item
  The statement is not a Bayesian statement
\item
  \textbf{\emph{Correct interpretation}}
\item
  The statement needs to be interpreted in a frequentist sense
\item
  If we compute the confidence intervals many times (e.g., extracting
  different data sets), in 95\% of the cases the confidence interval
  will capture the true population mean
\end{itemize}
\end{frame}

\begin{frame}{z-confidence intervals for the mean}
\protect\phantomsection\label{z-confidence-intervals-for-the-mean}
\begin{itemize}
\item
  \(X\) is not a Gaussian RV
\item
  We use a large number \(n\) of IID samples from \(X\)
\item
  We know \(\sigma_X\) variance of the underlying population
\item
  \textbf{\emph{Thesis}}
\item
  We want to use the realization of \(\overline{X}\) we have to estimate
  the unknown \(\mu_X = \EE[X]\)
\item
  \textbf{\emph{Algorithm}}
\item
  We know that:

  \begin{itemize}
  \tightlist
  \item
    \(\overline{X}\) is Gaussian (because of CLT)
  \item
    \(\overline{X}\) has mean \(\EE[X]\) (because the sample mean is
    unbiased estimator)
  \item
    The std err of \(\overline{X}\) is \(\frac{\sigma_X}{\sqrt{n}}\)
  \end{itemize}
\item
  We can build \(\alpha\) (e.g., 95\%) confidence interval for
  \(\overline{X}\) in the form: \[
  \Pr(\overline{X} \text{ inside } \mu_X \pm Z_{\alpha}
  \frac{\sigma_X}{\sqrt{n}}) = \alpha
  \] where \(Z_{\alpha}\) is a two-sided standard normal quantile (e.g.,
  \(\alpha = 68\%, 95\%, 99\%\) we have \(Z_\alpha = 1, 2, 3\),
  respectively)
\item
  This means that if we take multiple samples \(\overline{X}\)
  (frequentist interpretation), then the sample mean must be close to
  the real mean
\item
  \textbf{\emph{Flipping the roles}}
\item
  The issue is that we have a realization of \(\overline{X}\) but we
  don't know \(\mu_X\), which is what we are interested in bounding
\item
  We can reverse the statement and claim that: \[
  \Pr(\mu_X \text{ inside } \overline{X} \pm Z_{\alpha}
  \frac{\sigma_X}{\sqrt{n}}) = \alpha
  \]
\item
  The interpretation of this statement is that if we build repeatedly
  the confidence interval estimating \(\overline{X}\) from the
  population, the true \(\mu_X\) will be inside the interval with the
  given probability \(\alpha\)
\end{itemize}
\end{frame}

\begin{frame}{t-confidence intervals for the mean}
\protect\phantomsection\label{t-confidence-intervals-for-the-mean}
\begin{itemize}
\item
  \(X\) is Gaussian
\item
  We use a small number of \(n\) of IID samples from \(X\)
\item
  We don't know \(\sigma_X\)
\item
  \textbf{\emph{Thesis}}
\item
  We have a realization of \(\overline{X}\) and we want to use this
  information to estimate the unknown \(\mu_X = \EE[X]\)
\item
  \textbf{\emph{Algorithm}}
\item
  Take \(n\) IID samples \(X_i\) of a Gaussian and compute:

  \begin{itemize}
  \tightlist
  \item
    The (unbiased) sample mean:
    \(\overline{X} = \frac{1}{n} \sum_{i=1}^n X_i\)
  \item
    The (unbiased) sample variance:
    \(S^2 = \frac{1}{n - 1} \sum_{i=1}^n (X_i - \overline{X})^2\)
  \end{itemize}
\item
  We know that:

  \begin{itemize}
  \tightlist
  \item
    \(\overline{X}\) is Gaussian, since it is a linear combination of
    Gaussians
  \item
    \(S^2\) is chi-square with \(n-1\) degrees of freedom (multiplied by
    a constant), since it is sum of squared IID standard Gaussians
  \item
    \(T = \frac{\overline{X} - \EE[\overline{X}]}{S / \sqrt{n}}\) has a
    t-distribution with \(\nu = n - 1\) degrees of freedom after
    \(\sqrt{n}\) and \(\sqrt{n - 1}\) cancel each other
  \end{itemize}
\item
  Note that this case will be reduced to the previous one if we used
  \(\sigma_X\) instead of \(S\), because \(T\) would be Gaussian since
  it is a linear combination of Gaussian with (deterministic) constants
\end{itemize}
\end{frame}

\begin{frame}{z- vs t-confidence intervals for the mean}
\protect\phantomsection\label{z--vs-t-confidence-intervals-for-the-mean}
\begin{itemize}
\tightlist
\item
  \textbf{\emph{z-confidence intervals}}
\item
  The hypotheses for z-confidence intervals for the mean are:

  \begin{itemize}
  \tightlist
  \item
    \(X\) with any distribution
  \item
    \(\VV[X]\) is known
  \item
    Large sample size \(n\)
  \end{itemize}
\item
  The sample mean \(\overline{X}\) is Gaussian
\item
  We can build z-confidence intervals for \(\mu_X\) in the form:
  \[\mu_X \in \overline{X} \pm Z_{\alpha} \times \frac{\sigma_X}{\sqrt{n}}\]
  where:

  \begin{itemize}
  \tightlist
  \item
    \(Z_{\alpha}\) is a 2-sided z-quantile
  \end{itemize}
\item
  \textbf{\emph{t-confidence intervals}}
\item
  The hypotheses for t-confidence intervals for the mean are:

  \begin{itemize}
  \tightlist
  \item
    \(X\) is Gaussian
  \item
    \(\VV[X]\) is unknown
  \item
    Small sample size \(n\)
  \end{itemize}
\item
  The sample mean \(\overline{X}\) is a student t-distribution with
  \(n - 1\) degrees of freedom
\item
  We can build t-confidence intervals for \(\mu_X\) in the form:
  \[\mu_X \in \overline{X} \pm T_{\alpha} \times \frac{s}{\sqrt{n}}\]
  where:

  \begin{itemize}
  \tightlist
  \item
    \(s\) is the sample standard deviation
  \item
    \(T_{\alpha}\) is a 2-sided t-quantile
  \end{itemize}
\item
  Note that the t-confidence interval is going to be larger than the
  z-confidence interval since \(T_{\alpha} \ge Z_{\alpha}\) because
  t-distribution has heavier tails than a Gaussian distribution
\end{itemize}
\end{frame}

\begin{frame}{When to use t-confidence intervals}
\protect\phantomsection\label{when-to-use-t-confidence-intervals}
\begin{itemize}
\tightlist
\item
  In general it is always better to use t-intervals when using numerical
  methods

  \begin{itemize}
  \tightlist
  \item
    For back of the envelope calculations z-intervals can be easier to
    compute
  \end{itemize}
\item
  The t-confidence intervals assume that data is from IID normal
  distribution

  \begin{itemize}
  \tightlist
  \item
    It still works as long as distribution is roughly symmetric and
    mound-shaped
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Confidence intervals for asymmetric distributions}
\protect\phantomsection\label{confidence-intervals-for-asymmetric-distributions}
\begin{itemize}
\tightlist
\item
  For skewed distributions:

  \begin{itemize}
  \tightlist
  \item
    Cannot use t-distributions (in fact the confidence intervals will
    not be symmetric around the mean)
  \item
    Take logs of the observations to make distributions more symmetrical
  \item
    Use bootstrap
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{T-confidence intervals for paired observations}
\protect\phantomsection\label{t-confidence-intervals-for-paired-observations}
\begin{itemize}
\tightlist
\item
  For paired observations \(X\) and \(Y\)

  \begin{itemize}
  \tightlist
  \item
    Take the difference of paired observations to get a new RV \(X - Y\)
  \item
    Use t-intervals for the mean of difference \(\mu_{X - Y}\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{T-confidence intervals for groups in randomized trial (A/B
test)}
\protect\phantomsection\label{t-confidence-intervals-for-groups-in-randomized-trial-ab-test}
\begin{itemize}
\item
  We want to compare the measures from two groups in a randomized trial,
  also known as A/B test
\item
  E.g., receiving a medicine vs a placebo
\item
  We randomize the trials before assigning to A and B to balance
  covariates in the two groups, that might contaminate the results
\item
  We cannot use paired observations since the groups are independent
\item
  We assume that:

  \begin{itemize}
  \tightlist
  \item
    The variance in the two groups is the same
  \item
    The number of samples for the groups are \(n_x\) and \(n_y\)
  \end{itemize}
\item
  The \(\alpha\) (e.g., 95\%) confidence interval for \(\mu_Y - \mu_X\)
  is: \[
  \overline{Y} - \overline{X} \pm t_{\nu, \alpha} S_p
  \left( \frac{1}{n_x} + \frac{1}{n_y} \right) ^{1/2}
  \] where:

  \begin{itemize}
  \tightlist
  \item
    The degrees of freedom of the t-distribution are
    \(\nu = n_x + n_y - 2\)
  \item
    \(S_p^2\) is the pooled variance defined as:
    \[S_p^2 = \frac{(n_x - 1) S_x^2 + (n_y - 1) S_y^2}{(n_x + n_y - 2)}\]
  \end{itemize}
\item
  If the variance is not equal one can use t-distribution with
  fractional degrees of freedom (given by a formula) to approximate the
  underlying distribution
\end{itemize}
\end{frame}

\begin{frame}{What is hypothesis testing?}
\protect\phantomsection\label{what-is-hypothesis-testing}
\begin{itemize}
\tightlist
\item
  We have a statement \(H_a\) about a phenomenon
\item
  We want to quantify the statistical evidence supporting it
\end{itemize}
\end{frame}

\begin{frame}{Hypothesis testing set-up}
\protect\phantomsection\label{hypothesis-testing-set-up}
\begin{itemize}
\item
  We consider two hypotheses to explain the phenomenon under study:

  \begin{enumerate}
  \tightlist
  \item
    The alternative hypothesis \(H_a\): what we want to test
  \item
    The null hypothesis \(H_0\) (read ``h-nought''): the phenomenon is
    just the result of random fluctuations
  \end{enumerate}
\item
  We assume \(H_0\) is true unless the evidence strongly suggests that
  \(H_a\) is true and that \(H_0\) should be rejected
\item
  MEM: It's like a legal trial where one is assumed innocent (\(H_0\))
  until proved guilty (\(H_a\)) beyond reasonable doubt (statistical
  evidence)
\end{itemize}
\end{frame}

\begin{frame}{Test statistic, rejection region, and decision}
\protect\phantomsection\label{test-statistic-rejection-region-and-decision}
\begin{itemize}
\tightlist
\item
  We compute the distribution of the test statistics under \(H_0\)
\item
  Compute rejection region for null hypothesis from confidence level
  \(\alpha\)
\item
  Observed data are used to compute a test statistics
\item
  One reject either the null or alternative hypothesis based on the
  value of the test statistics compared to the rejection region of the
  null hypothesis
\end{itemize}
\end{frame}

\begin{frame}{Accepting null / alternative hypothesis}
\protect\phantomsection\label{accepting-null-alternative-hypothesis}
\begin{itemize}
\item
  In statistics / physics we should always use the term ``rejecting an
  hypothesis'', instead of ``proving an hypothesis'' since:

  \begin{enumerate}
  \tightlist
  \item
    We cannot find evidence that proves a theory right
  \item
    We can only find evidence that falsifies an hypothesis
  \end{enumerate}
\item
  E.g., we cannot prove the statement ``all swans are white'', since we
  should examine all swans and make sure they are all white
\item
  Instead if we find a single non-white swan we can reject the statement
  ``all swans are white''
\end{itemize}
\end{frame}

\begin{frame}{Type I and II errors}
\protect\phantomsection\label{type-i-and-ii-errors}
\begin{itemize}
\item
  There are 4 possible outcomes of our statistical decision process

  \begin{enumerate}
  \tightlist
  \item
    True negative: correctly accept null hypothesis
  \item
    True positive: correctly accept alternative hypothesis
  \item
    Type I error (false positive): accept \(H_a\) when \(H_0\) is true
  \item
    Type II error (false negative): accept \(H_0\) when \(H_a\) is true
  \end{enumerate}
\item
  MEM: P in FP has 1 downstroke -\textgreater{} type 1, N in FN has 2
  down-strokes -\textgreater{} type 2
\end{itemize}
\end{frame}

\begin{frame}{Probabilities of type I and II in hypothesis testing}
\protect\phantomsection\label{probabilities-of-type-i-and-ii-in-hypothesis-testing}
\begin{itemize}
\tightlist
\item
  The probability of

  \begin{itemize}
  \tightlist
  \item
    Type I error (i.e., false positive) is called \(\alpha\), i.e., the
    confidence level of the test
  \item
    Type II error (i.e., false negative) is called \(\beta\) (related to
    power of test)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Confidence level vs confidence interval}
\protect\phantomsection\label{confidence-level-vs-confidence-interval}
\begin{itemize}
\item
  There is a little confusion between \(\alpha\) in the context of:

  \begin{enumerate}
  \tightlist
  \item
    Confidence intervals

    \begin{itemize}
    \tightlist
    \item
      E.g., ``an interval with \(\alpha = 95\%\) confidence interval''
    \end{itemize}
  \item
    Hypothesis testing

    \begin{itemize}
    \tightlist
    \item
      E.g., ``a test with \(\alpha = 5\%\) confidence level''
    \end{itemize}
  \end{enumerate}
\end{itemize}

\begin{frame}{One-sided vs two-sided test}
\protect\phantomsection\label{one-sided-vs-two-sided-test}
\begin{itemize}
\item
  One-sided test: \(H_0: \Theta = \Theta_0\) vs
  \(H_a: \Theta > \Theta_0\)
\item
  Two-sided test: \(H_0: \Theta = \Theta_0\) vs
  \(H_a: \Theta \ne \Theta_0\) \(\iff\)
  \(H_a: \Theta > \Theta_0 \text{ or } \Theta < \Theta_0\)
\item
  Note that this implies different rejection regions for the same
  confidence level \(\alpha\)
\end{itemize}
\end{frame}

\begin{frame}{One-sided hypothesis test: example of reasoning for sample
mean}
\protect\phantomsection\label{one-sided-hypothesis-test-example-of-reasoning-for-sample-mean}
\begin{itemize}
\tightlist
\item
  We have a RV \(X\) with a mean assumed to be \(\mu_0\)
\item
  We sample \(X\) and get a \(\overline{x} > \mu_0\) (i.e., a
  realization of the sample mean \(\overline{X}\))
\item
  Did we get the value \(\overline{x}\) because the mean of \(X\) is:

  \begin{itemize}
  \tightlist
  \item
    Truly \(\mu_0\) and there are random fluctuations due to the
    stochastic nature of \(X\) (null hypothesis); or
  \item
    Larger than \(\mu_0\) (alternative hypothesis)?
  \end{itemize}
\item
  We don't know the answer, but we can require that the probability to
  reject the null hypothesis by mistake (i.e., false positive) is a
  certain confidence level \(\alpha\)

  \begin{itemize}
  \tightlist
  \item
    We assume \(H_0\)
  \item
    We compute the interval of values \(C\) that would make us reject
    \(H_0\) (rejection region)
  \item
    Check whether \(\overline{x} > C\) or not
  \end{itemize}
\item
  Note that if \(\overline{x} > C\)

  \begin{itemize}
  \tightlist
  \item
    We still don't know if we just witness a rare event or \(H_0\) is
    false
  \item
    We only know that if there was a rare event, it had a probability
    less than \(\alpha\) to happen
  \end{itemize}
\item
  To perform the test we need to know the sample statistics under the
  null hypothesis (e.g., normal or t-distribution)
\end{itemize}
\end{frame}

\begin{frame}{Hypothesis testing algorithm}
\protect\phantomsection\label{hypothesis-testing-algorithm}
\begin{itemize}
\tightlist
\item
  Assume \(H_0\)
\item
  Set a confidence level \(\alpha\)
\item
  Compute the rejection region under \(H_0\) for the test statistic at a
  given confidence level \(\alpha\)
\item
  Compare the test statistics computed from the data with the rejection
  region
\item
  Report the binary outcome ``reject / retain \(H_0\)''
\end{itemize}
\end{frame}

\begin{frame}{One-sided hypothesis test: example}
\protect\phantomsection\label{one-sided-hypothesis-test-example}
\begin{itemize}
\item
  Consider testing the hypothesis about the mean \(\mu_X\) of \(X\):
  \[H_0: \mu_X = \mu_0\] \[H_a: \mu_X > \mu_0\]
\item
  The idea is to reject the null hypothesis if \(\overline{X}\) is
  larger than a constant \(C\) chosen so that the probability of a type
  I error (i.e., false positive) is \(\alpha\):
  \[\Pr(\overline{X} > C \; | \; H_0) = \alpha\] where \(\overline{X}\)
  has a certain sample statistics (e.g., normal or t-distribution)
\item
  In other terms from \(\alpha\) we come up with the constant \(C\) and
  then we verify if \(\overline{x}\) is \(> C\) or not
\item
  \textbf{\emph{Numerical example}}
\item
  We need to find \(C: \Pr(\overline{X} > C | H_0) = \alpha\)
\item
  Assume \(\overline{X} \sim N(\mu_0, \sigma_X^2 / n)\) and
  \(\alpha = 0.05\) then
  \[C = \mu_0 + 1.645 \times \sigma_X / \sqrt{n}\]
\item
  Often we prefer to express the previous equation in terms of Z-scores:
  \[\frac{\overline{X} - \mu_0}{\sigma_X / \sqrt{n}} > Z_{1-\alpha}\]
\item
  If we need to estimate \(\sigma_X\) from the data, then we need to use
  a t-statistics for \(\overline{X}\) with the appropriate degrees of
  freedom \(n - 1\) and use quantile for t-distribution (t-test instead
  of a z-test)
\end{itemize}
\end{frame}

\begin{frame}{Two-sided hypothesis test: example}
\protect\phantomsection\label{two-sided-hypothesis-test-example}
\begin{itemize}
\tightlist
\item
  Consider testing the hypothesis about the mean \(\mu_X\) of \(X\):
  \[H_0: \mu_X = \mu_0\] \[H_a: \mu_X \ne \mu_0\]
\item
  The idea is still to find an interval so that one would mistakenly
  reject \(H_0\) with a probability \(\alpha\)
\item
  In this case we reject the null hypothesis if the test statistic is
  either too large or too small \[
  \Pr(\big|\frac{\overline{X} - \mu_0}{\sigma_X / \sqrt{n}} \big| > C) = \alpha
  \] so we need to consider the area under both tails of the PDF
\item
  E.g., for \(\alpha = 0.05\) we need \(C = 2\), which is a more
  stringent check than \(C = 1.645\) needed for a 1-sided test, since
  the prior is weaker
\end{itemize}
\end{frame}

\begin{frame}{P-value}
\protect\phantomsection\label{p-value}
\begin{itemize}
\tightlist
\item
  = probability under the null hypothesis of obtaining evidence as or
  more extreme than what observed
  \[\text{p-value = } \Pr(\text{seeing evidence } \ge H_a | H_0)\]
\item
  It can be one-sided or two-sided
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of p-values}
\protect\phantomsection\label{interpretation-of-p-values}
\begin{itemize}
\tightlist
\item
  P-values answer the question: ``suppose nothing is going on: how
  unusual it is to see the estimate we got?''
\item
  If the p-value is small, then either ``\(H_0\) is true and we have
  observed a rare event'' or ``\(H_0\) is false''
\end{itemize}
\end{frame}

\begin{frame}{Example of p-value}
\protect\phantomsection\label{example-of-p-value}
\begin{itemize}
\tightlist
\item
  Testing \(H_0: \mu = \mu_0\) vs \(H_a: \mu > \mu_0\), we get a test
  statistic (t-score in this case) of 2.5 for 15 df
\item
  What's the probability of getting a t-score \(\ge 2.5\) by chance?
\item
  \textcolor{blue}{\texttt{pt(2.5, 15, lower.tail=FALSE) = 0.01225 = 1\%}}
\end{itemize}
\end{frame}

\begin{frame}{P-value vs hypothesis testing}
\protect\phantomsection\label{p-value-vs-hypothesis-testing}
\begin{itemize}
\item
  P-value and hypothesis testing are related but look at the problem in
  different ways

  \begin{enumerate}
  \tightlist
  \item
    In hypothesis testing

    \begin{itemize}
    \tightlist
    \item
      We compute the rejection region for \(H_0\) that gives the desired
      significance level \(\alpha\)
    \item
      We compare the test statistic with the rejection region
    \item
      The answer is binary, i.e., reject / accept null hypothesis
    \end{itemize}
  \item
    With p-values

    \begin{itemize}
    \tightlist
    \item
      The result is a probability, i.e., the probability of getting the
      evidence under the null hypothesis
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{P-value in terms of confidence level of hypothesis
testing}
\protect\phantomsection\label{p-value-in-terms-of-confidence-level-of-hypothesis-testing}
\begin{itemize}
\item
  We can think of the p-value as the smallest value of confidence level
  \(\alpha\) for which we would still reject the null-hypothesis
\item
  The rejection region is bounded by the value that has a p-value equal
  to the confidence level \(\alpha\), e.g.,

  \begin{itemize}
  \tightlist
  \item
    If p-value is \(3\% = 0.03\) we can reject the null hypothesis up to
    a confidence level of \(0.03\)
  \item
    We reject \(H_0\) at \(\alpha = 0.05, 0.04, 0.03\) but not at
    \(\alpha = 0.02\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Trade-off between confidence level and power of a test}
\protect\phantomsection\label{trade-off-between-confidence-level-and-power-of-a-test}
\begin{itemize}
\tightlist
\item
  We want to avoid false positives

  \begin{itemize}
  \tightlist
  \item
    Thus we limit the false positive rate \(\Pr(H_a | H_0)\) using a low
    significance level \(\alpha\)
  \end{itemize}
\item
  On one hand, if all we cared was to not make mistakes

  \begin{itemize}
  \tightlist
  \item
    We could set \(\alpha\) to a very low level
  \item
    Then the test would not detect any positives at all
  \end{itemize}
\item
  On the other hand we are also interested in rejecting \(H_0\) when it
  is false

  \begin{itemize}
  \tightlist
  \item
    This is related to power of a test \(\Pr(H_a | H_a)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Power of a test}
\protect\phantomsection\label{power-of-a-test}
\begin{itemize}
\item
  The power of a test is the probability of rejecting the
  null-hypothesis when it is false \[
  \text{power} \defeq \Pr(\text{Reject $H_0$ $|$ $H_0$ is false}) = \Pr(H_a | H_a)
  \]
\item
  One wants tests to have tests with high power
\item
  Typically one designs an experiment (e.g., needed number of samples)
  so that it is possible to reject the null-hypothesis if it is false
\end{itemize}
\end{frame}

\begin{frame}{Power of a test as function of \(\beta\)}
\protect\phantomsection\label{power-of-a-test-as-function-of-beta}
\begin{itemize}
\tightlist
\item
  \(\beta = \Pr(H_0 | H_a)\) is the probability of type II error (i.e.,
  false negative)
\item
  Power of a test is defined as \(\Pr(H_a | H_a)\)
\item
  Thus the power of a test is equal to \(1 - \beta\)
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Example of calculating power for z-test}
\protect\phantomsection\label{example-of-calculating-power-for-z-test}
\begin{itemize}
\item
  Assume that we know \(\sigma\) and we use a z-test
\item
  \(H_0: \overline{X} \sim N(\mu_0, \sigma^2 / n)\) vs
  \(H_a: \overline{X} \sim N(\mu_a, \sigma^2 / n)\)
\item
  The power of the test is defined as
  \(\Pr(\text{Reject $H_0$ $|$ $H_a$ is true})\)
\item
  Since \(H_a\) is assumed true, then there is a distribution for
  \(\overline{X}\) under \(H_a\)
\item
  In hypothesis testing we reject \(H_0\) at confidence level \(\alpha\)
  if \(\frac{\overline{X} - \mu_0}{\sigma / \sqrt{n}} > Z_{1 - \alpha}\)
\item
  The formula for power of z-test is: \[
  \Pr(\frac{\overline{X} - \mu_0}{\sigma / \sqrt{n}} > Z_{1-\alpha} \;|\;
  \overline{X} \sim N(\mu_a, \sigma / \sqrt{n}))
  \]
\item
  Note that the power is function of:

  \begin{itemize}
  \tightlist
  \item
    The value of the test statistic that we want to detect in \(H_a\)
    (\(\mu_a\))
  \item
    The value of the test statistic assumed in \(H_0\) (\(\mu_0\))
  \item
    The significance level \(\alpha\)
  \item
    \(\sigma\) and \(n\) through the sample variance
  \end{itemize}
\item
  The power of the test to detect the real \(\mu_a\) is:

\begin{frame}{T-test power}
\protect\phantomsection\label{t-test-power}
\begin{itemize}
\tightlist
\item
  If we don't know \(\sigma\) we need to use a t-test
\item
  We always prefer to use t-test instead of z-test, since it is more
  accurate: \[
  \Pr(\frac{\overline{X} - \mu_0}{S / \sqrt{n}} > t_{n-1, 1-\alpha}
  \;|\; \overline{X} \text{ has } \mu = \mu_a)
  \]
\item
  Note that the t-distribution for \(\mu = \mu_a\) is a non-central
  t-distribution (i.e., it is not centered around 0)
\item
  In R there is a function \textcolor{blue}{\texttt{power.t.test}} to
  compute the power
\end{itemize}
\end{frame}

\begin{frame}{Example of data mining (jelly beans and acne)}
\protect\phantomsection\label{example-of-data-mining-jelly-beans-and-acne}
\begin{itemize}
\item
  One believes that jelly beans cause acne

  \begin{itemize}
  \tightlist
  \item
    Get data about consumption of jelly beans and occurrence of acne
  \item
    This relationship is tested and nothing is found at 5\% significance
  \item
    Then one might start testing jelly bean of each color one at the
    time
  \item
    After 20 attempts one finds out that pink jelly beans correlate with
    acne with a p-value of 5\%
  \end{itemize}
\item
  By running 20 experiments each with 5\% probability of being incorrect
  by chance there is almost certainty to find something
\item
  \textbf{\emph{Correct approach}}
\item
  Come up with hypotheses ahead of time
\item
  Adjust the p-value to account for data mining bias / multiple
  hypothesis testing
\item
  Hold out data to verify the relationship we have found
\end{itemize}
\end{frame}

\begin{frame}{Example of multiple-comparison problem using coins}
\protect\phantomsection\label{example-of-multiple-comparison-problem-using-coins}
\begin{itemize}
\item
  A procedure to determine if a coin is unfair consists in

  \begin{itemize}
  \tightlist
  \item
    Flipping a coin 10 times
  \item
    Checking if it lands heads 10 times
  \end{itemize}
\item
  The null hypothesis is that the coin is fair
\item
  \textbf{\emph{Single test}}
\item
  Assume the coin is fair
\item
  The p-value of the test under \(H_0\) is
  \(1 / 2^{10} = 1 / 1024 \approx 0.001\), which results in rejecting
  \(H_0\) with a p-value \(< 0.05\)
\item
  \textbf{\emph{Multiple tests}}
\item
  The probability that at least one coin out of 1000 is not fair (by
  luck) is almost 1, since it is 1 - probability that are all fair:
  \(1 - (1 - 0.001)^{1000} \approx 1\)
\item
  A multiple comparison problem arises if we want to use this test to
  check the unfairness of many coins
\end{itemize}
\end{frame}

\begin{frame}{Nomenclature for multiple testing}
\protect\phantomsection\label{nomenclature-for-multiple-testing}
\begin{itemize}
\item
  Consider the 4 possible scenarios for:

  \begin{itemize}
  \tightlist
  \item
    Decision: accepting \(H_0\) or \(H_a\)
  \item
    Ground truth: \(H_0\) or \(H_a\) is true \[
    \begin{tabular}{ l | c | c | c }
    & $H_0$ true & $H_a$ true & \\
    \hline
    Accept $H_0$ & TN & FN & \\
    \hline
    Accept $H_a$ & FP & TP & \\
    \hline
    \end{tabular}
    \]
  \end{itemize}
\item
  These scenarios are mutually exclusive and cover all the
  possibilities, so their sum is equal to the number of experiments
  \(m\)
\item
  Let's call according to standard nomenclature:

  \begin{itemize}
  \tightlist
  \item
    \(m\): the total number of hypotheses tested
  \item
    \(m_0\): the number of true null hypotheses
  \item
    \(V\): the number of rejected null hypotheses when the null was true
    (i.e., false positives)
  \item
    \(R\): the total number of rejected null hypotheses (i.e.,
    discoveries)

    \begin{itemize}
    \tightlist
    \item
      MEM: \(R\) is discoveRy
    \end{itemize}
  \end{itemize}
\item
  The 4 quantities \(m, m_0, R, V\) allow to compute the entire
  confusion matrix \[
  \begin{tabular}{ l | c | c | c }
  & $H_0$ true & $H_a$ true & \\
  \hline
  Accept $H_0$ & & & \\
  \hline
  Accept $H_a$ & $V$ & & $R$ \\
  \hline
  & $m_0$ & & \\
  \end{tabular}
  \]
\item
  Note that:

  \begin{itemize}
  \tightlist
  \item
    \(m\) and \(R\) are known or observable
  \item
    \(V\) and \(m_0\) are not observable
  \item
    \(V\) and \(R\) are RVs since we could run multiple experiments and
    we might get different results
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Probability of false positive}
\protect\phantomsection\label{probability-of-false-positive}
\begin{itemize}
\tightlist
\item
  The probability of a false positive is defined as: \[
  \Pr(FP)
  = \Pr(\text{Accept } H_a | H_0 \text{ is true})
  = \frac{\Pr(\text{Accept } H_a \text { and } H_0 \text{ is true})}
  {\Pr(H_0 \text{ is true})}
  = \lim_{m \to \infty} \frac{V}{m_0}
  \]
\item
  In words, we do infinite experiments and then compute the probability
  of false positive
\item
  The problem is that it is not observable quantity
\end{itemize}
\end{frame}

\begin{frame}{False Positive Rate}
\protect\phantomsection\label{false-positive-rate}
\begin{itemize}
\item
  Aka FPR
\item
  The false positive rate is defined as \[FPR = \EE_m[ \frac{V}{m_0}]\]
\item
  Expectation is over repeating the \(m\) experiments
\item
  Note that probability of false positive and expectation of the ratio
  is different, since in first case we consider a single experiment, in
  the second we consider the ensemble of experiments (???)
\end{itemize}
\end{frame}

\begin{frame}{Controlling FPR in a single experiment}
\protect\phantomsection\label{controlling-fpr-in-a-single-experiment}
\begin{itemize}
\tightlist
\item
  Discarding all discoveries with p-value \(< \alpha\) controls the
  false positive rate at level \(\alpha\) \textit{on average} for a
  \textit{single} experiment

  \begin{itemize}
  \tightlist
  \item
    On average: in the sense that we could do the same experiments over
    and over
  \item
    For a single experiment: this might be not enough when doing lots of
    tests (e.g., 10,000) because of the multiple-comparison problem
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Family Wise Error Rate}
\protect\phantomsection\label{family-wise-error-rate}
\begin{itemize}
\tightlist
\item
  Aka FWER
\item
  Defined as: \[FWER \defeq \Pr(V \ge 1)\] i.e., the probability of at
  least one false positive running \(m\) experiments
\item
  Family-wise seems to refer to a ``family'' of experiments (i.e., a
  multiple comparison)
\end{itemize}
\end{frame}

\begin{frame}{Bonferroni correction to control FWER}
\protect\phantomsection\label{bonferroni-correction-to-control-fwer}
\begin{itemize}
\tightlist
\item
  Suppose you do \(m\) tests
\item
  You want to control FWER so that \(\Pr(V \ge 1) \le \alpha\)
\item
  Calculate p-values
\item
  Call significant any experiment for which p-value \(< \alpha / m\),
  i.e., \(\Pr(H_a^{(i)} | H_0^{(i)}) < \alpha / m\)
\end{itemize}
\end{frame}

\begin{frame}{Bonferroni correction to control FWER: proof}
\protect\phantomsection\label{bonferroni-correction-to-control-fwer-proof}
\begin{itemize}
\item
  We can use the union bound to show that the p-value over multiple
  tests is less than \(\alpha\) \[
  \begin{aligned}
  \Pr(V \ge 1)
  & = \Pr(\text{reject falsely at least one } H_0) & \text{(by def)} \\
  & = \Pr(FP_1 \cup FP_2 \cup ... \cup FP_n) & \text{(expanding "at least")}\\
  & \le \sum_i \Pr(FP_i) & \text{(union bound)} \\
  & \le \sum \frac{\alpha}{m} & \text{(confidence level)} \\
  & = \alpha
  \end{aligned}
  \]
\item
  Note that no assumption of independence between tests is made
\item
  ??? If there is independence the bound can be improved since \[
  \Pr(V \ge 1)
  = 1 - \Pr(V = 0)
  = 1 - \Pr(\neg FP_1 \land \neg FP_2 ... \land \neg FP_n)
  = 1 - (\Pr(\neg FP))^{m}
  = 1 - (1 - \alpha)^m
  \]
\end{itemize}
\end{frame}

\begin{frame}{False Discovery Rate}
\protect\phantomsection\label{false-discovery-rate}
\begin{itemize}
\tightlist
\item
  Aka FDR
\item
  Defined as: \[FDR = \EE \left[ \frac{V}{R} \right]\] i.e., the average
  fraction of false positives \(V\) with respect to the discoveries
  \(R\)
\item
  This is an interesting metric since we know how many discoveries we
  made

  \begin{itemize}
  \tightlist
  \item
    Number of discoveries \(R\) are an observable variable
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{FDR-controlling procedures}
\protect\phantomsection\label{fdr-controlling-procedures}
\begin{itemize}
\tightlist
\item
  FDR-controlling procedures, when conducting multiple comparison, are
  designed to \textit{control} the \textit{expected} proportion of:

  \begin{itemize}
  \tightlist
  \item
    False discoveries (which is an observable variable)
  \item
    Not false positives (which is a variable we cannot observe)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{FPR vs FWER vs FDR}
\protect\phantomsection\label{fpr-vs-fwer-vs-fdr}
\begin{itemize}
\item
  False-Positive Rate is: \[FPR = \EE_m[ \frac{V}{m_0}]\] the expected
  fraction of false discoveries (\(V\)) with respect to to the number of
  null hypotheses to reject (\(m_0\))

  \begin{itemize}
  \tightlist
  \item
    It is the ratio of two not observable quantities
  \end{itemize}
\item
  Family-Wise Error Rate is: \[FWER = \Pr(V \ge 1)\] the probability of
  having at least a false discovery (\(V\))
\item
  Family Discovery Rate is: \[FDR = \EE \left[ \frac{V}{R} \right]\] the
  expected fraction of false discoveries (\(V\)) with respect to the
  number of discoveries (\(R\))

  \begin{itemize}
  \tightlist
  \item
    It is the ratio of an observable and an unobservable quantities
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{BH to control FDR}
\protect\phantomsection\label{bh-to-control-fdr}
\begin{itemize}
\item
  Benjamini-Hochberg is one of the most popular correction methods
\item
  \textbf{\emph{Algorithm}}
\item
  Suppose you do \(m\) (independent) tests
\item
  You want to control FDR so that \(\EE[V / R] < \alpha\)
\item
  Calculate p-values of all experiments
\item
  Order the p-values from smallest to largest \(p_1, p_2, ..., p_m\)
\item
  Find the smallest index corresponding to the p-value that falls under
  the sloped line \(\alpha \frac{i}{m}\)

  \begin{itemize}
  \tightlist
  \item
    This p-value \(p_T\) is called the BH rejection threshold
  \end{itemize}
\item
  Call significant any experiment with \(p_i < p_T\)
\end{itemize}
\end{frame}

\begin{frame}{BH correction: pros and cons}
\protect\phantomsection\label{bh-correction-pros-and-cons}
\begin{itemize}
\tightlist
\item
  Pros

  \begin{itemize}
  \tightlist
  \item
    Easy to calculate
  \end{itemize}
\item
  Cons

  \begin{itemize}
  \tightlist
  \item
    Allows for more false positives than Bonferroni correction
  \item
    Might not work well when the hypotheses are not independent
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{BY to control FDR}
\protect\phantomsection\label{by-to-control-fdr}
\begin{itemize}
\tightlist
\item
  This is an extension of BH method when tests are dependent
\item
  The sloped line is:
  \[l_i = \alpha \frac{i}{m} \frac{1}{\sum_{j=1}^m \frac{1}{j}}\]
\item
  Since we are dividing for a value that is larger than 1, the sloped
  line becomes lower and the threshold is more stringent
\end{itemize}
\end{frame}

\begin{frame}{Empirical CDF}
\protect\phantomsection\label{empirical-cdf}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Consider \(X\) with an unknown CDF \(F(x)\)
  \item
    We want to estimate \(F(x)\) from \(n\) samples \(X_1, ..., X_n\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Using the frequentist interpretation:
    \[F(x) = \Pr(X \le x) \approx \frac{\#(X \le x)}{\# \text{attempts}}\]
  \item
    The empirical CDF is defined as:
    \[\hat{F}_n(x) = \frac{\sum_{i=1}^n I(X_i \le x)}{n}\]
  \item
    In words, the empirical CDF is a discrete RV putting mass
    \(\frac{1}{n}\) at each value \(X_i\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Convergence of empirical CDF}
\protect\phantomsection\label{convergence-of-empirical-cdf}
\begin{itemize}
\item
  For any value \(x\) it holds that the empirical CDF is unbiased
  estimator \begin{align*}
  & \EE[\hat{F}_n(x)] = F(x) \\
  & \VV[\hat{F}_n(x)] = \frac{F(x) (1 - F(x))}{n} \\
  \end{align*}
\item
  This implies that the empirical CDF \(\hat{F}_n\) converges in
  probability to the true CDF \(F\)
\end{itemize}
\end{frame}

\begin{frame}{Proof of mean of empirical CDF}
\protect\phantomsection\label{proof-of-mean-of-empirical-cdf}
\begin{itemize}
\tightlist
\item
  Consider the mean of the empirical CDF \(\EE[\hat{F}_n(x)]\) \[
  \begin{aligned}
  \EE[\hat{F}_n(x)]
  &= \EE[\frac{1}{n} \sum_i I(X_i \le x)]
  \text{ (because of def of empirical CDF)} \\
  &= \frac{1}{n} \sum_i \EE[I(X_i \le x)]
  \text{ (by linearity of mean)} \\
  &= \frac{1}{n} \sum_i \Pr(X_i \le x)
  \text{ (because of mean of indicator var $\EE[I(A)] = \Pr(A)$ )} \\
  &= \frac{1}{n} \cdot n \Pr(X_i \le x)
  \text{ (by definition of CDF)} \\
  &= F(x)
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Proof of variance of empirical CDF}
\protect\phantomsection\label{proof-of-variance-of-empirical-cdf}
\begin{itemize}
\tightlist
\item
  Consider the variance of the empirical CDF \(\VV[\hat{F}_n(x)]\) \[
  \begin{aligned}
  \VV[\hat{F}_n(x)]
  &= \EE[(\hat{F}_n(x) - F(x))^2]
  \text{ (by def of variance and unbias-ndess of empirical CDF) } \\
  &= \EE[(\frac{1}{n} \sum I(X_i \le x) - F(x))^2]
  \text{ (by def of empirical CDF) }\\
  &= \frac{1}{n^2} \EE[ \sum I((X_i \le x) - F(x))^2 ] \\
  &= \frac{1}{n^2} \EE[ \sum (I(X_i \le x) - F(x))^2 ]
  \text{ (since all $X_i$ are independent and thus uncorrelated) } \\
  &= \frac{1}{n^2} \cdot n \EE[ (I(X_i \le x) - F(x))^2 ] \\
  &= \frac{1}{n} \EE[ (I(X_i \le x))^2 + (F(x))^2 - 2 I(X_i \le x) F(x) ]
  \text{ (developing the square) } \\
  &= \frac{1}{n} (\EE[ (I(X_i \le x))^2] + (F(x))^2 - 2 \EE[ (I(X_i \le x))^ 2] F(x)) \\
  &= \frac{1}{n} (\EE[ I(X_i \le x) ] + ... )
  \text{ (since the square of indicator var $I^2 = I$) } \\
  &= \frac{1}{n} (F(x) + (F(x))^2 - 2 (F(x))^2)
  \text{ (since $\EE[I(X_i \le x)] = \Pr(X \le x) = F(x)$) } \\
  &= \frac{1}{n} (F(x) - (F(x))^2) \\
  &= \frac{1}{n} (F(x) (1 - F(x)) \\
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Statistical functional}
\protect\phantomsection\label{statistical-functional}
\begin{itemize}
\tightlist
\item
  Given a RV \(F\), a statistical functional \(T(F)\) is any function of
  the CDF of \(F\)
\end{itemize}
\end{frame}

\begin{frame}{Statistical functional: number or RV}
\protect\phantomsection\label{statistical-functional-number-or-rv}
\begin{itemize}
\tightlist
\item
  The statistical functional \(T(F)\):

  \begin{itemize}
  \tightlist
  \item
    Is a number (e.g., the mean, the median, the variance) if we use a
    distribution \(F\)
  \item
    Is a RV if we use a sample distribution \(F\) (since different
    samples will give different values)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Statistical functional: example}
\protect\phantomsection\label{statistical-functional-example}
\begin{itemize}
\tightlist
\item
  The following quantities are statistical functionals since they are
  function of the CDF of \(F\)

  \begin{itemize}
  \tightlist
  \item
    Mean since \(\mu = \int x dF\)
  \item
    Variance since \(\sigma^2 = \int (x - \mu)^2 dF\)
  \item
    Median since it is \(= F^{-1}(\frac{1}{2})\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Plug-in principle}
\protect\phantomsection\label{plug-in-principle}
\begin{itemize}
\item
  Given a statistical functional \(T(F)\), the plug-in estimator of
  \(T(F)\) is defined by: \[\hat{T}_n = T(\hat{F}_n)\]
\item
  In words, to estimate a functional through its samples, we ``plug in''
  the empirical CDF \(\hat{F}_n\) into the functional
\item
  Note that it is not a theorem but simply a common sense guideline
\end{itemize}
\end{frame}

\begin{frame}{Linear statistical functional}
\protect\phantomsection\label{linear-statistical-functional}
\begin{itemize}
\tightlist
\item
  A statistical functional \(T(F)\) is linear \(\iff\) it is in the
  form: \[T(F) = \int r(x) dF(x)\] where \(r(x)\) is a weighting
  function
\item
  In words \(T\) is a linear combination of values from the PDF \(dF\)
\item
  MEM: It is the same form as the theorem of the lazy statistician but
  using the CDF
\end{itemize}
\end{frame}

\begin{frame}{Linear statistical functional: examples and non-examples}
\protect\phantomsection\label{linear-statistical-functional-examples-and-non-examples}
\begin{itemize}
\tightlist
\item

  \begin{itemize}
  \tightlist
  \item
    Mean
  \item
    Variance
  \item
    Skewness
  \end{itemize}
\item
  Non-examples:

  \begin{itemize}
  \tightlist
  \item
    Median
  \item
    Trimmed mean (i.e., the mean without a percent of extreme values)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Plug-in estimator for linear statistical functional}
\protect\phantomsection\label{plug-in-estimator-for-linear-statistical-functional}
\begin{itemize}
\tightlist
\item
  Applying the plug-in principle for linear statistical functional: \[
  \begin{aligned}
  \hat{T}_n
  &= T(\hat{F}_n) & \text{ (def of plug-in principle)} \\
  &= \int r(x) d\hat{F}_n(x) & \text{ (def of linear statistical functional)} \\
  &= \frac{1}{n} \sum_i r(x_i) \hat{F}_n(x_i)
  & \text{ (PMF has mass $\frac{1}{n} F_n(x_i)$ in each point)} \\
  \end{aligned}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap in brief}
\protect\phantomsection\label{bootstrap-in-brief}
\begin{itemize}
\tightlist
\item
  Bootstrap is used to estimate the distribution of a sampling statistic
  \(T = g(X_1, ..., X_n)\) given a finite amount of samples
  \(X_i \sim F\)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(g(\cdot)\) can be the mean, median, standard deviation, OLS
    coefficient, etc.
  \item
    Bootstrap is a generalization of the plug-in principle
  \end{itemize}
\item
  Useful when:

  \begin{itemize}
  \tightlist
  \item
    The theoretical distribution of the statistic is unknown; or
  \item
    The sample size is too small for traditional parametric methods
  \end{itemize}
\item
  Bootstrap is a \textbf{\textcolor{red}{non-parametric}} method

  \begin{itemize}
  \tightlist
  \item
    We don't make any assumption on the distribution we need to estimate
  \item
    In a parametric method we assume that there is a model and we only
    need to estimate some parameters of the model
  \end{itemize}
\item
  Applications

  \begin{itemize}
  \tightlist
  \item
    Estimating distribution of sample statistics: e.g.,
    \(F_X(x), f_x(x)\)
  \item
    Constructing confidence intervals: e.g., \(\mu \pm \epsilon\)
  \item
    Calculating standard errors: e.g., \(\sigma(\hat{T}_n)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap procedure}
\protect\phantomsection\label{bootstrap-procedure}
\begin{itemize}
\tightlist
\item
  Given \(X \sim F\)

  \begin{itemize}
  \tightlist
  \item
    Draw \(n\) IID samples \(X_i\) from \(X\)
  \item
    Consider a statistic \(T = g(X_1, ..., X_n)\) of the data
  \item
    We want to approximate the distribution of \(T\) or estimate a
    statistic of \(T\) (e.g., mean, std err)
  \end{itemize}
\end{itemize}

\textbf{\textcolor{red}{Algorithm}}

\begin{itemize}
\item
  Use the observed data \(X_1, ..., X_n\) to construct an estimated
  population distribution \(\hat{F}_n\)

  \begin{itemize}
  \tightlist
  \item
    We pretend that the empirical distribution is the real one
  \end{itemize}
\item
  Repeat \(B\) times:

  \begin{itemize}
  \tightlist
  \item
    Draw \(n\) samples with replacement from \(\hat{F}_n\)
  \item
    Compute the sample statistics from the \(n\) samples:
    \(T^{(i)} = g(X^{(i)}_1, ..., X^{(i)}_n)\)
  \end{itemize}
\item
  Use \(B\) samples of \(T^{(i)}\) to estimate its empirical
  distribution \(\hat{T}\)
\item
  Compute statistics (e.g., confidence interval, standard error) of the
  statistics \(T\) from the empirical distribution of \(\hat{T}\)
\end{itemize}
\end{frame}

\begin{frame}{Sample with replacement as computational shortcut in
bootstrap}
\protect\phantomsection\label{sample-with-replacement-as-computational-shortcut-in-bootstrap}
\begin{itemize}
\tightlist
\item
  In theory we need to

  \begin{itemize}
  \tightlist
  \item
    compute the empirical PMF of \(\hat{F}\)
  \item
    draw from it
  \end{itemize}
\item
  In practice drawing an observation from \(\hat{F}\) is equivalent to
  drawing one point at random from \(X_1, ..., X_n\) (i.e., sample with
  replacement)
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap: pros}
\protect\phantomsection\label{bootstrap-pros}
\begin{itemize}
\tightlist
\item
  Tremendously useful tool
\item
  Fewer assumptions

  \begin{itemize}
  \tightlist
  \item
    It does not need simplifying assumptions required to get closed
    formulas
  \item
    E.g., the underlying data does not need to be Gaussian
  \end{itemize}
\item
  Greater accuracy

  \begin{itemize}
  \tightlist
  \item
    It does not rely on large sample sizes, in contrast with asymptotics
    from CLT / LLN
  \end{itemize}
\item
  Generality

  \begin{itemize}
  \tightlist
  \item
    The same method applies to any sample statistics, even difficult
    non-linear ones (e.g., median)
  \end{itemize}
\item
  Simulation vs math

  \begin{itemize}
  \tightlist
  \item
    Bootstrap liberated data scientists from performing lots of complex
    mathematics, approximations, and asymptotics
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap: example of die rolls}
\protect\phantomsection\label{bootstrap-example-of-die-rolls}
\begin{itemize}
\item
  We want to compute the distribution of the sum of rolling a die 50
  times \[T = \sum_{i=1}^{50} X_i = g(X_1, ..., X_{50})\]
\item
  We know the PMF of the die, including the probability of head \(p\)
\end{itemize}

\begin{enumerate}
\tightlist
\item
  By math

  \begin{itemize}

  \item

    \tightlist
    \item
      Compute PDF or use theorem of lazy statistician
    \end{itemize}
  \end{itemize}
\item
  By sampling (real or simulated)

  \begin{itemize}
  \tightlist
  \item
    Repeat the procedure enough to get convergence of the PDF

    \begin{itemize}
    \tightlist
    \item
      Roll the die 50 times
    \item
      Compute the sample statistic \(T\)
    \end{itemize}
  \item
    Plot the approximate distribution of \(T\)
  \end{itemize}

\begin{itemize}
\item
  What if we don't know anything else than just 50 samples of the die?
\item
  Bootstrap

  \begin{itemize}
  \tightlist
  \item
    Sample from the empirical distribution with replacement
  \item
    Build the distribution of \(T\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Pseudo-code for bootstrap of the median}
\protect\phantomsection\label{pseudo-code-for-bootstrap-of-the-median}
\begin{lstlisting}
def bootstrap_median(x, n_boot):
    # Compute n_boot sample statistics.
    median_boot = [0.0] * n_boot
    for i in range(n_boot):
        # Sample with replacement.
        x_star = sample with replacements from x
        # Compute median for bootstrapped samples.
        median_boot[i] = median(x_star)
    # Compute mean and std err from approximation of sample statistics.
    m_median = numpy.mean(median_boot)
    se_median = numpy.std(median_boot)
    return m_median, se_median
\end{lstlisting}
\end{frame}

\begin{frame}{Bootstrap for variance of sample statistics: explanation}
\protect\phantomsection\label{bootstrap-for-variance-of-sample-statistics-explanation}
\begin{itemize}
\item
  Assume:

  \begin{itemize}
  \tightlist
  \item
    \(X \sim F\)
  \item
    \(n\) IID samples \(X_i\) from \(F\)
  \item
    Compute a statistic of the data \(T = g(X_1, ..., X_n)\)
  \end{itemize}
\item
  We want to compute \(\VV_F[T]\) (variance of sample statistics), where
  subscript \(F\) indicates that it depends on the unknown distribution
  \(F\)
\item
  There are 2 approximations to compute \(\VV_F[T]\)
\item
  First approximation

  \begin{itemize}
  \tightlist
  \item
    We don't have \(F\), but only samples drawn from it
  \item
    We can approximate the distribution of \(F\) with the distribution
    of \(\hat{F}\), since we know that the empirical CDF \(\hat{F}\)
    converges to the true CDF \(F\) (plug-in principle)
    \[\VV_F[T] \approx \VV_{\hat{F}}[T]\]
  \item
    This approximation

    \begin{itemize}
    \tightlist
    \item
      Is not so small
    \item
      Depends on the number of samples and shape of \(F\)
    \end{itemize}
  \end{itemize}
\item
  Second approximation

  \begin{itemize}
  \tightlist
  \item
    Even if we had the exact \(\hat{F}\), there might not be a closed
    formula to compute the variance \(\VV_{\hat{F}}[T]\)
  \item
    We can use LLN and simulation to approximate the desired variance:
    \[
    \VV_{\hat{F}}[T]
    \approx v_{boot}
    = \frac{1}{B} \sum_i (T_i - \overline{T}) ^ 2
    \]
  \item
    This approximation is as small as we want by taking \(B\) large
    enough
  \item
    We know that \(v_{boot} \to \VV_{\hat{F}}[T]\) when \(B \to \infty\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap for variance of sample statistics (2/3)}
\protect\phantomsection\label{bootstrap-for-variance-of-sample-statistics-23}
\textbf{\textcolor{red}{First approximation}} (Plug-in Principle)

\begin{itemize}
\item
  We don't know \(F\), but we observe samples drawn from it
\item
  Approximate \(F\) with the empirical distribution \(\hat{F}\):

  \begin{itemize}
  \tightlist
  \item
    \(\hat{F}\) assigns probability mass \(1/n\) to each observed
    \(X_i\)
  \item
    The empirical CDF \(\hat{F} \to F\) converges to the true CDF \(F\)
    uniformly almost surely
  \end{itemize}
\item
  Using the plug-in principle: \[
  \VV_F[T] \approx \VV_{\hat{F}}[T]
  \]
\item
  This approximation error depends on:

  \begin{itemize}
  \tightlist
  \item
    The number of samples \(n\)
  \item
    The shape of the true distribution \(F\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap for variance of sample statistics (3/3)}
\protect\phantomsection\label{bootstrap-for-variance-of-sample-statistics-33}
\textbf{\textcolor{red}{Second approximation}} (Monte Carlo Simulation)

\begin{itemize}
\item
  Even with \(\hat{F}\) known, \(\VV_{\hat{F}}[T]\) might not have a
  closed-form expression
\item
  Use bootstrap resampling: draw \(B\) samples \(T_1, ..., T_B\) by
  resampling with replacement from the data
\item
  Then compute: \[
  \overline{T} = \frac{1}{B} \sum_{i=1}^B T_i
  \] \[
  v_{\text{boot}} = \frac{1}{B} \sum_{i=1}^B (T_i - \overline{T})^2
  \]
\item
  This estimate converges: \[
  v_{\text{boot}} \xrightarrow{P} \VV_{\hat{F}}[T] \quad \text{as } B \to \infty
  \]
\item
  The accuracy of \(v_{\text{boot}}\) improves with larger \(B\)
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap for variance of sample statistics: analytical
formula}
\protect\phantomsection\label{bootstrap-for-variance-of-sample-statistics-analytical-formula}
\begin{itemize}
\item
  Assume:

  \begin{itemize}
  \tightlist
  \item
    \(X \sim F\)
  \item
    \(n\) IID samples \(X_i\) from \(F\)
  \item
    Compute a statistic of the data: \(T = g(X_1, ..., X_n)\)
  \end{itemize}
\item
  We want to compute \(\VV_F[T]\) (variance of sample statistics), where
  subscript \(F\) indicates that it depends on the unknown distribution
  \(F\)
\item
  In some special cases we have a formula for \(\VV_F[T]\) using \(F\)

  \begin{itemize}
  \tightlist
  \item
    E.g., for sample statistic \(T = \frac{1}{n} \sum_i X_i\) (sample
    mean) and \(X\) Gaussian: \[\VV_F[T] = \frac{\VV[F]}{n}\]
  \end{itemize}
\item
  If we don't have \(\VV[F]\) we can use \(\hat{F}\) to compute an
  approximation of \(\VV_{\hat{F}}[F]\)
  \[S^2 = \frac{1}{n} \sum_i (X_i - \overline{X})^2\]
\item
  This is a case where we replace the first approximation of bootstrap
  with a closed formula
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap confidence intervals}
\protect\phantomsection\label{bootstrap-confidence-intervals}
\begin{itemize}
\tightlist
\item
  In the general case:

  \begin{itemize}
  \tightlist
  \item
    We compute the empirical distribution of \(T\) by bootstrapping
  \item
    Estimate the confidence intervals from the CDF using percentiles
  \end{itemize}
\item
  In some cases by CLT we know that \(T = g(X_1, ..., X_n)\) tends to a
  Gaussian
\item
  We can assume that \(T\) is Gaussian and thus confidence intervals
  are: \[\overline{T} \pm Z_{a/2} se_{boot}\] where \(se_{boot}\) is the
  sqrt of variance estimated through bootstrap
\end{itemize}
\end{frame}

\begin{frame}{Bootstrap hypothesis testing: example}
\protect\phantomsection\label{bootstrap-hypothesis-testing-example}
\begin{itemize}
\item
  We have two samples of data \(A\) and \(B\) of different lengths
\item
  Check if a sample statistics (e.g., the median) of \(A\) and \(B\) are
  different
\item
  \textbf{\emph{Algorithm}}
\item
  We cannot do a paired test since \(A\) and \(B\) are not paired
\item
  We can test if the difference of sample statistics is different enough
  from 0
\item
  In other words the test statistic is the difference of medians
\item
  We bootstrap the difference of the medians from the data
\end{itemize}
\end{frame}
