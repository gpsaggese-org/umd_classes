\begin{frame}{Overfitting: Definition}
\protect\phantomsection\label{overfitting-definition}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Overfitting}} occurs when the model fits the
  data more than what is warranted
\item
  Surpass point where \(E_{out}\) is minimal (optimal fit)

  \begin{itemize}
  \tightlist
  \item
    Model complexity too high for data/noise
  \item
    Noise in training set mistaken for signal
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Fitting noise instead of signal}} is not
  useless but harmful

  \begin{itemize}
  \tightlist
  \item
    Model infers in-sample pattern that, when extrapolated
    out-of-sample, deviates from target function \(\implies\) poor
    generalization
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{tmp.notes_to_pdf.preprocess_notes.txt.figs/tmp.notes_to_pdf.render_image.1.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Optimal Fit}
\protect\phantomsection\label{optimal-fit}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  The opposite of overfitting is \textbf{\textcolor{red}{optimal fit}}

  \begin{itemize}
  \tightlist
  \item
    Train a model with proper complexity for the data
  \end{itemize}
\item
  The optimal fit:

  \begin{itemize}
  \tightlist
  \item
    Implies \(E_{out}\) is minimal
  \item
    Does not imply generalization error \(E_{out} - E_{in}\) is minimal

    \begin{itemize}
    \tightlist
    \item
      E.g., no training implies generalization error equal to 0
    \end{itemize}
  \end{itemize}
\item
  The \textbf{\textcolor{blue}{generalization error}} is the additional
  error \(E_{out} - E_{in}\) when going from in-sample to out-of-sample
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{msml610/lectures_source/figures/L05.2.Optimal_fit.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Overfitting: Diamond Price Example}
\protect\phantomsection\label{overfitting-diamond-price-example}
\begin{itemize}
\item
  Predict diamond price as a function of carat size (regression problem)
\item
  \textbf{\textcolor{red}{True relationship}} \[
  \text{price} \sim (\text{carat size})^2 + \varepsilon
  \] where:

  \begin{itemize}
  \tightlist
  \item
    Square function: price increases more with rarity
  \item
    Noise \(\varepsilon\): e.g., market noise, missing features
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Fit with:}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Line}

    \begin{itemize}
    \tightlist
    \item
      Underfit
    \item
      High bias (large error)
    \item
      Low variance (stable model)
    \end{itemize}
  \item
    \emph{Polynomial of degree 2}

    \begin{itemize}
    \tightlist
    \item
      right fit
    \end{itemize}
  \item
    \emph{Polynomial of degree 10}

    \begin{itemize}
    \tightlist
    \item
      Overfit (wiggly curve)
    \item
      Low bias
    \item
      High variance (many degrees of freedom)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Overfitting: Classification Example}
\protect\phantomsection\label{overfitting-classification-example}
\begin{itemize}
\tightlist
\item
  Assume:

  \begin{itemize}
  \tightlist
  \item
    You want to separate 2 classes using 2 features \(x_1, x_2\)
  \item
    The true class boundary has a parabola shape
  \end{itemize}
\item
  You can use logistic regression and a decision boundary equal to:

  \begin{itemize}
  \tightlist
  \item
    A line \(\text{logit}(w_0 + w_1 x + w_2 y)\)

    \begin{itemize}
    \tightlist
    \item
      Underfit
    \item
      High bias, low variance
    \end{itemize}
  \item
    A parabola
    \(\text{logit}(w_0 + w_1 x + w_2 x^2 + w_3 x y + w_4 y^2)\)

    \begin{itemize}
    \tightlist
    \item
      Right fit
    \end{itemize}
  \item
    A wiggly decision boundary
    \(\text{logit}(w_0 + \text{high powers of } x_1, x_2)\)

    \begin{itemize}
    \tightlist
    \item
      Overfit
    \item
      Low bias, high variance
    \end{itemize}

\begin{frame}{Hypothesis Set and Bias-Variance Analysis}
\protect\phantomsection\label{hypothesis-set-and-bias-variance-analysis}

\item

\begin{frame}{Decomposing Error in Bias-Variance (1/4)}
\protect\phantomsection\label{decomposing-error-in-bias-variance-14}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Regression set-up: target is a real-valued function
  \item
    Hypothesis set \(\calH = \{ h_1(\vx), h_2(\vx), ... h_n(\vx) \}\)
  \item
    Training data \(\calD\) with \(N\) examples
  \item
    Squared error \(E_out = \EE[(g(\vx) - f(\vx))^2]\)
  \item
    Choose the best function \(g \in \calH\) that approximates unknown
    \(f\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Question}}

  \begin{itemize}
  \tightlist
  \item
    What is the out-of-sample error \(E_{out}(g)\) as function of
    \(\calH\) for a training set of \(N\) examples?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Decomposing Error in Bias-Variance (2/4)}
\protect\phantomsection\label{decomposing-error-in-bias-variance-24}
\begin{itemize}
\item
  The final hypothesis \(g\) depends on training set \(D\), so make the
  dependency explicit \(g^{(D)}\): \[
  E_{out}(g^{(D)})
  \defeq \EE_{\vx}[ ( g^{(D)}(\vx) - f(\vx)) ^ 2 ]
  \]
\item
  Interested in:

  \begin{itemize}
  \tightlist
  \item
    Hypothesis set \(\calH\) rather than specific \(h\)
  \item
    Training set \(D\) of \(N\) examples, rather than a specific \(D\)
  \end{itemize}
\item
  Remove dependency from \(D\) by averaging over all possible training
  sets \(D\) with \(N\) examples: \[
  E_{out}(\calH)
  \defeq \EE_{D}[ E_{out}(g^{(D)}) ]
  = \EE_{D}[\EE_{\vx}[ ( g^{(D)}(\vx) - f(\vx)) ^ 2 ]]
  \]
\end{itemize}
\end{frame}

\begin{frame}{Decomposing Error in Bias-Variance (3/4)}
\protect\phantomsection\label{decomposing-error-in-bias-variance-34}
\begin{itemize}
\item
  Switch the order of the expectations since the quantity is
  non-negative: \[
  E_{out}(\calH)
  = \EE_{\vx}[ \EE_{D}[ ( g^{(D)}(\vx) - f(\vx)) ^ 2 ]
  \]
\item
  Focus on \(\EE_D [( g^{(D)}(\vx) - f(\vx) ) ^ 2 ]\) which is a
  function of \(\vx\)
\item
  Define the \textit{average hypothesis} over all training sets as: \[
  \overline{g}(\vx) \defeq \EE_D [ g^{(D)} (\vx) ]
  \]
\item
  Add and subtract it inside the \(\EE_D\) expression: \[
  \begingroup \small
  \begin{aligned}
  E_{out}(\calH)
  = & \EE_{\vx} \left[ \EE_D \left[ \left(
  g^{(D)}(\vx) - f(\vx)
  \right) ^ 2 \right] \right]\\
  %
  = & \EE_{\vx} \EE_D [ (
  g^{(D)} - \overline{g} +
  \overline{g} - f
  ) ^ 2 ]\\
  = & \EE_{\vx} \EE_D [
  (g^{(D)} - \overline{g}) ^ 2 +
  (\overline{g} - f ) ^ 2 +
  2 (g^{(D)} - \overline{g}) (\overline{g} - f)
  ]\\
  %
  & (\EE_D \text{ is linear and } (\overline{g} - f )
  \text{ doesn't depend on } D) \\
  %
  = & \EE_{\vx} \left[
  \EE_D[(g^{(D)} - \overline{g}) ^ 2] +
  (\overline{g} - f ) ^ 2 +
  2 \EE_D[(g^{(D)} - \overline{g})] (\overline{g} - f)
  \right]\\
  \end{aligned}
  \endgroup
  \]
\end{itemize}
\end{frame}

\begin{frame}{Decomposing Error in Bias-Variance (4/4)}
\protect\phantomsection\label{decomposing-error-in-bias-variance-44}
\begin{itemize}
\item
  The cross term: \[
  \EE_D[(g^{(D)} - \overline{g})] (\overline{g} - f)
  \] disappears since applying the expectation on \(D\), it is equal to:
  \[
  (g^{(D)} - \EE_D[\overline{g}]) (\overline{g} - f)
  = 0 \cdot (\overline{g} - f)
  = 0 \cdot \text{constant}
  \]
\item
  Finally: \begin{alignat*}{3}
  E_{out}(\calH)
  & = \EE_{\vx} [ \EE_D [ ( g^{(D)} - \overline{g} ) ^ 2 ] +
  ( \overline{g}(\vx) - f(\vx) ) ^ 2 ]
  &
  \\
  & = \EE_{\vx} [ \EE_D [ ( g^{(D)} - \overline{g} ) ^ 2 ] ] +
  \EE_{\vx}[( \overline{g} - f) ^ 2] 
  & (\EE_{\vx} \text{ is linear}) \\
  & = \EE_{\vx} [ \text{var}(\vx) ] + \EE_{\vx} [ \text{bias}(\vx)^2 ]
  &
  \\
  & = \text{variance} + \text{bias}
  &
  \\
  \end{alignat*}
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of Average Hypothesis}
\protect\phantomsection\label{interpretation-of-average-hypothesis}
\begin{itemize}
\tightlist
\item
  The \textbf{\textcolor{red}{average hypothesis}} over all training
  sets \[
  \overline{g}(\vx) \defeq \EE_D [ g^{(D)} (\vx) ]
  \] can be interpreted as the ``best'' hypothesis from \(\calH\)
  training on \(N\) samples

  \begin{itemize}
  \tightlist
  \item
    Note: \(\overline{g}\) is not necessarily \(\in \calH\)
  \end{itemize}
\item
  In fact it's like \textbf{\textcolor{blue}{ensemble learning}}:

  \begin{itemize}
  \tightlist
  \item
    Consider all the possible data sets \(D\) with \(N\) samples
  \item
    Learn \(g\) from each \(D\)
  \item
    Average all the hypotheses
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Interpretation of Variance and Bias Terms}
\protect\phantomsection\label{interpretation-of-variance-and-bias-terms}
\begin{itemize}
\tightlist
\item
  The out-of-sample error can be decomposed as: \[
  E_{out}(\calH) = \text{bias}^2 + \text{variance}
  \]
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Bias term}} \[
  \text{bias}^2 = \EE_{\vx} [ ( \overline{g}(\vx) - f(\vx) ) ^ 2 ]
  \]

  \begin{itemize}
  \tightlist
  \item
    Does not depend on learning as it is not a function of the data set
    \(D\)
  \item
    Measures how limited \(\calH\) is

    \begin{itemize}
    \tightlist
    \item
      I.e., the ability of \(\calH\) to approximate the target with
      infinite training sets
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Variance term}} \[
  \text{variance} = \EE_{\vx} \EE_D [ ( g^{(D)}(\vx) - \overline{g}(\vx) ) ^ 2]
  \]

  \begin{itemize}
  \tightlist
  \item
    Measures variability of the learned hypothesis from \(D\) for any
    \(\vx\)

    \begin{itemize}
    \tightlist
    \item
      With infinite training sets, we could focus on the ``best'' \(g\),
      which is \(\overline{g}\)
    \item
      But we have only one data set \(D\) at a time, incurring a cost
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Bias-Variance Decomposition with a Noisy Target}
\protect\phantomsection\label{bias-variance-decomposition-with-a-noisy-target}
\begin{itemize}
\item
  Extend bias-variance decomposition to
  \textbf{\textcolor{red}{a noisy target}} \[
  y = f(\vx; \vw) + \varepsilon = \vw^T \vx + \varepsilon
  \]
\item
  With similar hypothesis and analysis conclude that: \[
  \begin{aligned}
  E_{out}(\calH)
  &= \EE_{D, \vx} \left[
  ( g^{(D)} - \overline{g} )^2
  \right]
  + \EE_{\vx} \left[
  ( \overline{g} - f) ^ 2
  \right]
  + \EE_{\varepsilon, \vx} \left[
  ( f - y) ^ 2
  \right]\\
  &= \text{variance + bias + stochastic noise}\\
  \end{aligned}
  \]
\item
  The out-of-sample error is the sum of 3 contributions

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Variance}}: from the set of hypotheses
    to the centroid of the hypothesis set
  \item
    \textbf{\textcolor{blue}{Bias}}: from the centroid of the hypothesis
    set to the noiseless function
  \item
    \textbf{\textcolor{violet}{Noise}}: from the noiseless function to
    the real function
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Bias as Deterministic Noise}
\protect\phantomsection\label{bias-as-deterministic-noise}
\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\item
  The bias term can be interpreted as
  \textbf{\textcolor{red}{deterministic noise}}
\item
  \textbf{\textcolor{blue}{Bias}} is the part of the target function
  that hypothesis set cannot capture: \[
  h^*(\vx) - f(\vx)
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \(h^*()\) is the best approximation of \(f(\vx)\) in the hypothesis
    set \(\calH\) (e.g., \(\overline{g}(x)\))
  \end{itemize}
\item
  The hypothesis set \(\calH\) cannot learn the deterministic noise
  since it is outside of its ability, and thus it behaves like ``noise''
\end{itemize}
\end{column}

\begin{frame}{Deterministic vs Stochastic Noise in Practice}
\protect\phantomsection\label{deterministic-vs-stochastic-noise-in-practice}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Deterministic noise:}}

  \begin{itemize}
  \tightlist
  \item
    Fixed for a particular \(\vx\)
  \item
    Depends on \(\calH\)
  \item
    Independent of \(\varepsilon\) or \(D\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Stochastic noise:}}

  \begin{itemize}
  \tightlist
  \item
    Not fixed for \(\vx\)
  \item
    Independent of \(D\) or \(\calH\)
  \end{itemize}
\item
  In a machine learning problem, no difference exists between stochastic
  and deterministic noise, as \(\calH\) and \(D\) are fixed

  \begin{itemize}
  \tightlist
  \item
    From the training set alone, you cannot determine if data is from a
    \emph{noiseless complex} target or a \emph{noisy simple} target
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Deterministic vs Stochastic Noise Example}
\protect\phantomsection\label{deterministic-vs-stochastic-noise-example-1}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  Noisy low-order target:

  \begin{itemize}
  \tightlist
  \item
    Fit 2nd to 10th order polynomial
  \item
    \(\downarrow E_{in}\) (more degrees of freedom),
    \(\uparrow\uparrow E_{out}\) (fits noise)
  \end{itemize}
\item
  Noiseless high-order target

  \begin{itemize}
  \tightlist
  \item
    Fit 2nd to 10th order polynomial
  \item
    Same phenomenon
  \item
    \(\downarrow E_{in}\) (more degrees of freedom),
    \(\uparrow\uparrow E_{out}\) (fits noise)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Wrong approach}}

  \begin{itemize}
  \tightlist
  \item
    Target is 10th order polynomial
  \item
    Use 10-order hypothesis to fit target perfectly
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Right approach}}

  \begin{itemize}
  \tightlist
  \item
    Consider number of data points, i.e., 15 points
  \item
    Rule of thumb from VC analysis \[
    \text{degrees of freedom of model = number of data points / 10}
    \]
  \item
    Use 1 or 2 degrees of freedom
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Amount of Data and Model Complexity}
\protect\phantomsection\label{amount-of-data-and-model-complexity}
\begin{itemize}
\tightlist
\item
  One must match the \emph{model complexity} to

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Data resources}}
  \item
    \textbf{\textcolor{darkgreen}{Signal to noise ratio}}
  \item
    \textbf{\textcolor{blue}{Not target complexity}}
  \end{itemize}
\item
  The rule of thumb is: \[
  d_{VC} (\text{degrees of freedom of the model})
  = N (\text{number of data points}) / 10
  \]

  \begin{itemize}
  \tightlist
  \item
    In other words, 10 data points needed to fit a degree of freedom
  \item
    If the data is noisy, you need even more data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Overfitting as a Function of Data Resources, Model
Complexity, Noise}
\protect\phantomsection\label{overfitting-as-a-function-of-data-resources-model-complexity-noise}
\begin{itemize}
\item
  You can measure overfitting in terms of
  \textbf{\textcolor{red}{generalization error}}
  \(\frac{E_{out} - E_{in}}{E_{out}}\)

  \begin{itemize}
  \tightlist
  \item
    \(\uparrow\) data resources \(N \implies \downarrow\) overfitting
  \item
    \(\uparrow\) model complexity \(d_{VC} \implies \uparrow\)
    overfitting
  \item
    \(\uparrow\) deterministic noise (target complexity)
    \(\implies \uparrow\) overfitting
  \item
    \(\uparrow\) stochastic noise \(\sigma^2 \implies \uparrow\)
    overfitting
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Learning Curves vs Bias-Variance Curves}
\protect\phantomsection\label{learning-curves-vs-bias-variance-curves}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Learning curves}} are the dual of the
  bias-variance curves
\item
  For \textbf{\textcolor{darkgreen}{bias-variance curves}} \[
  E_{in}, E_{out} = f(d_{VC} \vert N)
  \]

  \begin{itemize}
  \tightlist
  \item
    Keep the complexity of training set fixed (number of examples \(N\))
  \item
    Vary the model in terms of:

    \begin{itemize}
    \tightlist
    \item
      Model complexity \(d\)
    \item
      Number of features \(p\)
    \item
      Regularization amount \(\lambda\)
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  For \textbf{\textcolor{blue}{learning curves}} \[
  E_{in}, E_{out} = f(N \vert d_{VC})
  \]

  \begin{itemize}
  \tightlist
  \item
    Fix the model
  \item
    Vary the size \(N\) of training set
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Typical Form of Learning Curves}
\protect\phantomsection\label{typical-form-of-learning-curves}
\begin{itemize}
\tightlist
\item
  Learning curves plot \(E_{in}\) and \(E_{out}\) as a function of data
  amount \(N\), given the model \(h()\)

  \begin{itemize}
  \tightlist
  \item
    \(E_{out} \ge E_{in}\) for any \(N\)
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Small $N$}}

  \begin{itemize}
  \tightlist
  \item
    \(E_{in}\) is small (even 0)
  \item
    The model is likely overfitted, memorizing and generalizing poorly
  \item
    \(E_{out}\) is large
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Increasing $N$}}

  \begin{itemize}
  \tightlist
  \item
    \(E_{in} \uparrow\) as the model cannot fit all data
  \item
    \(E_{out} \downarrow\) as the model fits better and generalizes
    better
  \item
    Generalization error \(E_{out} - E_{in} \downarrow\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{$N \to \infty$}}

  \begin{itemize}
  \tightlist
  \item
    \(E_{in}\) reaches a maximum (irreducible error)
  \item
    \(E_{out}\) reaches a minimum
  \item
    \(E_{out} - E_{in}\) depends on the complexity of the model
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{High-Bias vs High-Variance Regime}
\protect\phantomsection\label{high-bias-vs-high-variance-regime}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  From the learning curve you can see the two regimes:

  \begin{itemize}
  \item
    \textbf{\textcolor{red}{High-variance regime}} for small \(N\)

    \begin{itemize}
    \tightlist
    \item
      \(E_{in}\) is small
    \item
      Small data set \(D\) \(\implies\) high dependency on \(D\)
    \end{itemize}
  \item
    More data helps reduce gap between \(E_{in}\) and \(E_{out}\)
  \item
    \textbf{\textcolor{blue}{High-bias regime}} for large \(N\)

    \begin{itemize}
    \tightlist
    \item
      \(E_{in}\) is large; flattens for large \(N\)
    \item
      Best model fitted; more data won't help
    \item
      \(E_{out}\) can be close to \(E_{in}\) (good generalization) or
      not
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}
