\begin{frame}{Machine Learning Paradigms with Examples (1/3)}
\protect\phantomsection\label{machine-learning-paradigms-with-examples-13}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Supervised Learning}}

  \begin{itemize}
  \tightlist
  \item
    Learn from labeled data to predict labels for new inputs
  \item
    E.g., image classification using ResNet on ImageNet
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Unsupervised Learning}}

  \begin{itemize}
  \tightlist
  \item
    Discover hidden patterns or structure in unlabeled data
  \item
    E.g., K-means clustering for customer segmentation
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Reinforcement Learning}}

  \begin{itemize}
  \tightlist
  \item
    Learn through interaction with an environment, receiving
    rewards/punishments
  \item
    E.g., deep Q-Learning for playing Atari games
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Self-Supervised Learning}}

  \begin{itemize}
  \tightlist
  \item
    Generate pseudo-labels from unlabeled data to pre-train models
  \item
    E.g., BERT (Masked Language Modeling)
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Semi-Supervised Learning}}

  \begin{itemize}
  \tightlist
  \item
    Combine small labeled data with large unlabeled data to improve
    performance
  \item
    E.g., named entity recognition (NER) using annotated sentences with
    entity tags combined with many raw text documents
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Machine Learning Paradigms with Examples (2/3)}
\protect\phantomsection\label{machine-learning-paradigms-with-examples-23}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Online Learning}}

  \begin{itemize}
  \tightlist
  \item
    Learn incrementally from a stream of data in real time
  \item
    E.g., online logistic regression for click-through rate prediction
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Multi-Task Learning}}

  \begin{itemize}
  \tightlist
  \item
    Train simultaneously a model to perform multiple related tasks
  \item
    E.g., learn sentiment analysis and question answering
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Meta-Learning}}

  \begin{itemize}
  \tightlist
  \item
    ``Learning to learn'': adapt quickly to new tasks using prior
    experience
  \item
    E.g., a model can be fine-tuned quickly on a new task using just a
    few gradient steps
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Zero-Shot / Few-Shot Learning}}

  \begin{itemize}
  \tightlist
  \item
    Generalize to new tasks with no or few labeled examples
  \item
    E.g., GPT-4 solving tasks with zero-shot prompting
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Active Learning}}

  \begin{itemize}
  \tightlist
  \item
    The model selects the most informative samples to be labeled by an
    oracle (e.g., a human)
  \item
    E.g., pick samples where the model is least confident to get more
    examples
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Machine Learning Paradigms with Examples (3/3)}
\protect\phantomsection\label{machine-learning-paradigms-with-examples-33}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Federated Learning}}

  \begin{itemize}
  \tightlist
  \item
    Train models across decentralized devices without sharing raw data
  \item
    E.g., fraud detection or credit scoring across banks
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Evolutionary Learning}}

  \begin{itemize}
  \tightlist
  \item
    Optimize model structures or parameters using evolutionary
    algorithms inspired by natural selection and genetics
  \item
    Gradient free, global search, variable length inputs
  \item
    E.g., genetic algorithms
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Curriculum Learning}}

  \begin{itemize}
  \tightlist
  \item
    Train models on easier tasks first, gradually increasing difficulty
  \item
    E.g., training in robotic control simulations
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Multi-Agent Learning}}

  \begin{itemize}
  \tightlist
  \item
    Multiple agents learn and interact in shared environments, often in
    game-theoretic settings (e.g., competition, collaboration)
  \item
    E.g., AlphaStar to play StarCraft II
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Supervised Learning}
\protect\phantomsection\label{supervised-learning}
\begin{itemize}
\tightlist
\item
  Learn a function \(f: X \to Y\) that maps inputs to correct outputs

  \begin{itemize}
  \tightlist
  \item
    Training examples \((\vx, y)\) with pairs of inputs and correct
    outputs
  \item
    Requires labeled data for training
  \item
    Measure performance with error on a separate test set
  \end{itemize}
\item
  \textbf{\textcolor{red}{Classification}}: output is a discrete label,
  e.g.,

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{Spam}} vs
    \textcolor{blue}{\texttt{Not Spam}}
  \item
    Digit recognition \textcolor{blue}{\texttt{0}},
    \textcolor{blue}{\texttt{1}}, \ldots{}
  \item
    Sentiment analysis \textcolor{blue}{\texttt{Pos}},
    \textcolor{blue}{\texttt{Neg}}, \textcolor{blue}{\texttt{Neutral}}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Regression}}: output is a continuous
  value, e.g.,

  \begin{itemize}
  \tightlist
  \item
    House prices given features like size and location
  \item
    House demand
  \item
    Stock prices
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Common algorithms}}:

  \begin{itemize}
  \tightlist
  \item
    Linear Regression
  \item
    Decision Trees
  \item
    K-nearest neighbors
  \item
    Neural Networks
  \item
    \ldots{}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Unsupervised Learning}
\protect\phantomsection\label{unsupervised-learning}
\begin{itemize}
\tightlist
\item
  Learn from data \textbf{\textcolor{red}{without}} labeled outputs

  \begin{itemize}
  \tightlist
  \item
    Goal: discover patterns or structure in the data
  \item
    No explicit feedback signal
  \item
    Evaluation can be qualitative
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Main techniques}}:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Clustering}}: Group similar examples,
    e.g.,

    \begin{itemize}
    \tightlist
    \item
      Customer segmentation
    \item
      Grouping news articles by topic without knowing the topics
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Dimensionality Reduction}}: Reduce number
    of variables with PCA while preserving structure

    \begin{itemize}
    \tightlist
    \item
      E.g., visualize high-dimensional data in 2D
    \end{itemize}
  \item
    \textbf{\textcolor{cyan}{Density Estimation}}: Estimate probability
    distribution of data

    \begin{itemize}
    \tightlist
    \item
      E.g., anomaly detection in server logs
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Association Rule Learning}}: Discover
    interesting relations between variables

    \begin{itemize}
    \tightlist
    \item
      E.g., market basket analysis (e.g., ``people who buy X also buy
      Y'')
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Common algorithms:}}

  \begin{itemize}
  \tightlist
  \item
    K-means
  \item
    PCA
  \item
    Autoencoders
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Reinforcement Learning}
\protect\phantomsection\label{reinforcement-learning}
\begin{itemize}
\tightlist
\item
  Learn by \textbf{\textcolor{red}{interacting with an environment}} to
  \textbf{\textcolor{darkgreen}{maximize cumulative reward}}

  \begin{itemize}
  \tightlist
  \item
    Learn policy \(\pi(s) \to a\) that maximizes expected reward
  \item
    Trade-off between exploration (trying new actions) and exploitation
    (using known good actions)
  \item
    Environments provide clear rules and feedback (win/loss/reward)
  \item
    Often involve physical simulation or real-world interaction
  \end{itemize}

\begin{frame}{Question}
\protect\phantomsection\label{question}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Make the question concrete and precise}}

  \begin{itemize}
  \tightlist
  \item
    Define the problem clearly
  \item
    Specify inputs and expected outputs
  \item
    Align question with business or research objectives
  \item
    E.g.,:

    \begin{itemize}
    \tightlist
    \item
      \textbf{\textcolor{darkgreen}{Bad}}: \emph{``How can we improve
      sales?''}
    \item
      \textbf{\textcolor{blue}{Good}}: \emph{``What factors most
      significantly impact sales of product X in region Y during season
      Z?''}
    \end{itemize}
  \end{itemize}
\item
  Formulating question is
  \textbf{\textcolor{violet}{the most important part}} of the machine
  learning problem

  \begin{itemize}
  \tightlist
  \item
    Misunderstanding leads to:

    \begin{itemize}
    \tightlist
    \item
      Solving the wrong problem
    \item
      Collecting wrong data
    \item
      \ldots{}
    \end{itemize}
  \end{itemize}
\item
  \emph{``If I were given one hour to save the planet, I would spend 59
  minutes defining the problem and one minute resolving it''} (Einstein)
\end{itemize}
\end{frame}

\begin{frame}{Input Data}
\protect\phantomsection\label{input-data}
\begin{itemize}
\tightlist
\item
  Ensure \textbf{\textcolor{red}{data is specific to prediction}} goal

  \begin{itemize}
  \tightlist
  \item
    E.g., use known movie ratings to predict unseen movie ratings from
    the same population
  \item
    Training set \(\approx\) test set
  \end{itemize}
\item
  Relationship between data and prediction goal is
  \textbf{\textcolor{darkgreen}{not always direct}}

  \begin{itemize}
  \tightlist
  \item
    E.g., interested in prices but predict supply and demand instead
  \end{itemize}
\item
  Poor-quality data leads to inaccurate predictions

  \begin{itemize}
  \tightlist
  \item
    \emph{``Garbage in - garbage out''}
  \end{itemize}
\item
  Recognize \textbf{\textcolor{blue}{when data is insufficient}} for
  valid answers

  \begin{itemize}
  \tightlist
  \item
    \emph{``Combination of data and desire for an answer does not ensure
    a reasonable answer can be extracted''} (John Tukey)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{More data vs better models}}

  \begin{itemize}
  \tightlist
  \item
    Meta-studies show difference between generic and best model is like
    5\%
  \item
    \emph{``It's not who has the best algorithm that wins. It's who has
    the most data''} (Google researcher)
  \item
    \emph{``Every time I fire a linguist, the performance of the speech
    recognizer goes up''} (IBM researcher in speech recognition)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Features}
\protect\phantomsection\label{features}
\begin{itemize}
\tightlist

  about inputs

  \begin{itemize}
  \tightlist
  \item

  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Characteristics of good features}}:

  \begin{enumerate}
  \tightlist
  \item
    Perform data compression
  \item
    Retain relevant information
  \item
    Often created with expert knowledge
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Common mistakes in feature building}}:

  \begin{enumerate}
  \tightlist
  \item
    Automating feature selection may lead to overfitting

    \begin{itemize}
    \tightlist
    \item
      Black box predictions can be accurate but stop working anytime
    \item
      E.g., Google Flu's unclear feature-model link
    \end{itemize}
  \item
    Ignoring data-specific quirks

    \begin{itemize}
    \tightlist
    \item
      E.g., mislabeling outliers
    \end{itemize}
  \item
    Unnecessarily discarding information
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Models}

    \textbf{\textcolor{red}{Interpretable}}

    \item

    \end{itemize}
  \item
    \textbf{\textcolor{cyan}{Scalable}}

    \begin{itemize}
    \tightlist
    \item
      Can handle large datasets efficiently (growing data and user

\begin{frame}{How Are Machine Learning Systems Organized?}
\protect\phantomsection\label{how-are-machine-learning-systems-organized}
\begin{itemize}
\tightlist
\item
  Machine learning systems organized in a
  \textbf{\textcolor{red}{pipeline}}

  \begin{enumerate}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Break down}} problem into sub-problems
  \item
    \textbf{\textcolor{blue}{Solve}} problems one at a time
  \item
    \textbf{\textcolor{violet}{Combine}} solutions to solve initial
    problem
  \end{enumerate}
\item
  Performance \(p\) of a pipeline: \[
  p_{system} = \sum_i p_i \cdot \alpha_i
  \] where:

  \begin{itemize}
  \tightlist
  \item
    \(p_i\): Performance of each stage
  \item
    \(\alpha_i\): Importance of each stage
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Example of Photo OCR System}
\protect\phantomsection\label{example-of-photo-ocr-system}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}: build systems to read text in a
  picture

  \begin{itemize}
  \tightlist
  \item
    OCR = ``Optical Character Recognition''
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Stages of ML pipeline for OCR:}}

  \begin{itemize}
  \tightlist
  \item
    Text detection: find text areas in the picture
  \item
    Character segmentation: split text into letter boxes

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{h e l l o}}
    \end{itemize}
  \item
    Character classification: classify characters individually
  \item
    Spelling correction: fix text errors using context

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{hell0}} corrected to
      \textcolor{blue}{\texttt{hello}}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Sliding window approach}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{violet}{Problem}}: Unknown text location and size
  \item
    Text detection

    \begin{itemize}
    \tightlist
    \item
      Train classifier to recognize letters vs non-letters
    \item
      Scan image in two directions with different sizes for text
    \item
      Evaluate classifier cheaper than training
    \item
      Create text likelihood map (e.g., heatmap) using classifier
      probabilities
    \item
      Enclose text areas in boxes
    \item
      Discard boxes not fitting aspect ratio (valid text width
      \textgreater{} height)
    \end{itemize}
  \item
    Character segmentation

    \begin{itemize}
    \tightlist
    \item
      Use sliding window classifiers to find ``breaks'' between
      characters
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

    \end{itemize}
  \item
    Collect and label by hand

    \begin{itemize}
    \tightlist
    \item
      E.g., crowd sourcing like Amazon Mechanical Turk
    \end{itemize}
  \end{enumerate}
\end{itemize}

\begin{frame}{Ceiling Analysis for ML Pipeline}
\protect\phantomsection\label{ceiling-analysis-for-ml-pipeline}
\begin{itemize}
\item
  The most valuable resource is time

  \begin{itemize}
  \tightlist
  \item
    You work on optimization for months
  \item
    Find out that the optimization doesn't make much difference
  \end{itemize}
\item
  \textbf{\textcolor{red}{Problem}}: On which part of the pipeline
  should time/resource be spent?
\item
  \textbf{\textcolor{blue}{Solution}}: Ceiling analysis to Analyze
  pipeline performance

  \begin{itemize}
  \tightlist
  \item
    Single number for system performance

    \begin{itemize}
    \tightlist
    \item
      E.g., accuracy for OCR
    \end{itemize}
  \item
    For each component:

    \begin{itemize}
    \tightlist
    \item
      Mock component with ``oracle'' (always give correct output)
    \item
      Leave others untouched
    \item
      Compute pipeline performance
    \end{itemize}
  \item
    Identify critical components by estimating performance improvement
    with 10\% component improvement
  \item
    Measure, don't guess!
  \end{itemize}
\end{itemize}
\end{frame}
