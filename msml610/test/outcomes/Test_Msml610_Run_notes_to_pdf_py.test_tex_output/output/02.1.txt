\begin{frame}{Machine Learning Paradigms}
\protect\phantomsection\label{machine-learning-paradigms}
\begin{itemize}
\item
  How do you set up the learning problem?

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{Supervised learning}}

    \begin{itemize}
    \tightlist
    \item
      The dataset includes inputs with corresponding outputs
    \item
      Develop an input-output relationship
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Unsupervised learning}}

    \begin{itemize}
    \tightlist
    \item
      The data is unlabeled, discover structure within the data
    \item
      E.g., anomaly detection, clustering
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Reinforcement learning}}

    \begin{itemize}
    \tightlist
    \item
      The correct answer is not immediately available
    \item
      Evaluate actions based on final outcomes
    \end{itemize}
  \item
    \textbf{\textcolor{violet}{Active learning}}

    \begin{itemize}
    \tightlist
    \item
      Not all examples are available initially
    \item
      Request outputs for specific inputs
    \end{itemize}
  \item
    \ldots{}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Machine Learning Theory}
\protect\phantomsection\label{machine-learning-theory}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{VC theory}}

  \begin{itemize}
  \tightlist
  \item
    Measure model capacity and generalize based on hypothesis space
    complexity
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Bias-variance decomposition}}

  \begin{itemize}
  \tightlist
  \item
    Prediction error is the sum of:

    \begin{itemize}
    \tightlist
    \item
      Bias: Error from simplistic model assumptions
    \item
      Variance: Error due to sensitivity to training data fluctuations
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Computation complexity}}

  \begin{itemize}
  \tightlist
  \item
    Related to information theory and compression
  \item
    E.g., Minimum Description Length (MDL) measures model complexity via
    efficient model and data description
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Bayesian approach}}

  \begin{itemize}
  \tightlist
  \item
    Treat ML as probability
  \item
    Combine prior knowledge with observed data to update belief about a
    model
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Problem in ML theory}}

  \begin{itemize}
  \tightlist
  \item
    Assumptions may not align with practical problems
  \end{itemize}
\end{itemize}
\end{frame}
