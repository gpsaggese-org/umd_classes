\begin{frame}{Bernoulli distribution: definition}
\protect\phantomsection\label{bernoulli-distribution-definition}
\begin{itemize}
\tightlist
\item
  The Bernoulli distribution \[X \sim \text{Bernoulli}(p)\] represents
  flipping a coin that has probability \(0 \le p \le 1\) of coming up
  heads: \[\Pr(X=1) = p\] \[\Pr(X=0) = 1 - p\]
\end{itemize}
\end{frame}

\begin{frame}{Bernoulli distribution: PDF}
\protect\phantomsection\label{bernoulli-distribution-pdf}
\begin{itemize}
\tightlist
\item
  The PDF can be written in terms of a function:
  \[f_X(x) = \Pr(X = x) = p^x (1 - p)^{1-x} \text{ for } x \in \{0, 1\}\]
\end{itemize}
\end{frame}

\begin{frame}{Bernoulli distribution: mean and variance}
\protect\phantomsection\label{bernoulli-distribution-mean-and-variance}
\begin{itemize}
\tightlist
\item
  Given \(X \sim \text{Bernoulli}(p)\)
\item
  Mean: \(\EE[X] = p\)
\item
  Variance: \(\VV[X] = p(1 - p)\)
\end{itemize}
\end{frame}

\begin{frame}{Binomial distribution: definition}
\protect\phantomsection\label{binomial-distribution-definition}
\begin{itemize}
\tightlist
\item
  The Binomial distribution \[X \sim \text{Binomial}(n, p)\] represents
  the number of heads when tossing \(n\) times a coin with probability
  of heads \(p\):
  \[\Pr(X = x) = \Pr(\text{getting $x$ heads}), 0 \le x \le n\]
\end{itemize}
\end{frame}

\begin{frame}{Binomial distribution: PDF}
\protect\phantomsection\label{binomial-distribution-pdf}
\begin{itemize}
\tightlist
\item
  The PDF is
  \[f_X(x) = {n \choose x} p^x (1-p)^{n-x} \text{ for } x \in \{0, ..., n\}\]
\item
  In fact one can evaluate this probability as the sum of the
  probability of all possible events, since they are all mutually
  exclusive:

  \begin{itemize}
  \tightlist
  \item
    Each event has probability \(p^x (1 - p)^{n-x}\) (because it is a
    Bernoulli trial)
  \item
    There are \({n \choose x}\) possible ways of choosing a set of \(x\)
    objects (in this case heads) out of \(n\) identical objects (the
    binomial coefficient)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Relationship between Binomial and Bernoulli variables}
\protect\phantomsection\label{relationship-between-binomial-and-bernoulli-variables}
\begin{itemize}
\tightlist
\item
  A Binomial variable \(Y \sim \text{Binomial}(n, p)\) can be written as
  sum of \(n\) IID Bernoulli variables \(X_i \sim \text{Bernoulli}(p)\):
  \[Y = \sum_{i=1}^n X_i\]
\end{itemize}
\end{frame}

\begin{frame}{Mean and variance of Binomial distribution}
\protect\phantomsection\label{mean-and-variance-of-binomial-distribution}
\begin{itemize}
\tightlist
\item
  By using the relationship between Binomial and Bernoulli variables
  \[X \sim \text{Binomial}(n, p) = \sum_{i=1}^n X_i\]
\item
  Mean: \(\EE[X] = n p\)
\item
  Variance: \(\VV[X] = n p(1 - p)\)
\end{itemize}
\end{frame}

\begin{frame}{Example of binomial distribution (7 girls)}
\protect\phantomsection\label{example-of-binomial-distribution-7-girls}
\begin{itemize}
\item
  A friend has 8 children, 7 of which are girls
\item
  What's the probability of this event, if each birth is independent and
  each gender has 50\%-50\%?
\item
  \textbf{\emph{Solution}}
\item
  We can look at that as 8 configurations (boy in \(i\)-st position) out
  of \(2^8\), which is \(\frac{1}{2^5}\)
\item
  Using binomial distribution, head is girl then
  \[\Pr = {8 \choose 7} (1/2)^{7} (1/2) = \frac{1}{2^5}\]
\item
  What's the probability of having at least 7 girls?
\item
  It is the probability of having 7 or 8 girls, i.e. \[
  \Pr
  = {8 \choose 7} (1/2)^{7} (1/2) + {8 \choose 8} (1/2)^{8}
  = ({8 \choose 7} + {8 \choose 8}) (1/2)^8
  = {9 \choose 8} (1/2)^8
  = {9 \choose 1} (1/2)^8
  = 9 / 256
  \]
\end{itemize}
\end{frame}

\begin{frame}{1-sided and 2-sided quantiles for symmetric distributions}
\protect\phantomsection\label{sided-and-2-sided-quantiles-for-symmetric-distributions}
\begin{itemize}
\tightlist
\item
  For symmetric distributions the 1-sided and 2-sided quantiles are
  related by the relationship:
  \[q_{2s}(\alpha) = q_{1s}(\frac{1 + \alpha}{2})\]
\end{itemize}
\end{frame}

\begin{frame}{1-sided and 2-sided quantiles for symmetric distributions:
proof}
\protect\phantomsection\label{sided-and-2-sided-quantiles-for-symmetric-distributions-proof}
\begin{itemize}
\tightlist
\item
  This proof holds for any symmetric around 0 PDF
\item
  Consider that \[CDF_{2s}(x) = CDF_{1s}(x) - CDF_{1s}(-x)\] given the
  definition of CDF in terms of integral of a PDF
\item
  Given the symmetry of the PDF it holds:
  \[CDF_{1s}(-x) = 1 - CDF_{1s}(x)\]
\item
  Thus \[CDF_{2s}(x) = 2 \cdot CDF_{1s}(x) - 1\]
\item
  Now we need to express the 2-sided quantile in terms of 1-sided
  quantile where the quantile is the inverse of the corresponding CDF
\item
  If \(f(x) = g(h(x))\) with both \(g\) and \(h\) invertible, then
  \[f^{-1}(x) = h^{-1}(g^{-1}(x))\]
\item
  The inverse of \(y = 2 x - 1\) is \(x = (y + 1) / 2\) and the inverse
  of \(CDF_{1s}\) is \(q_{1s}\), therefore
  \[q_{2s}(x) = q_{1s}((x + 1) / 2)\]
\end{itemize}
\end{frame}

\begin{frame}{Example of computing probabilities with Gaussian}
\protect\phantomsection\label{example-of-computing-probabilities-with-gaussian}
\begin{itemize}
\item
  The daily ad-clicks for a company are approximately distributed as
  \(X \sim N(\mu=1020, \sigma^2=50^2)\)

\end{itemize}

\begin{frame}{Poisson distribution: interpretation}

\item
  A RV \(X \sim \Poisson(\lambda)\) models counts or arrivals per unit

  \begin{itemize}
  \tightlist
  \item

\begin{frame}{Poisson distribution: PDF}
\protect\phantomsection\label{poisson-distribution-pdf}
\begin{itemize}
\tightlist
\item
  Its PDF is: \[
  f_X(k)
  = \Pr(X = k) =
  \frac{\lambda^k e^{-\lambda}}{k!}
  \] where \(k\) is an integer \(\ge 0\)
\item
  MEM:

  \begin{itemize}
  \tightlist
  \item
    It is like a prob of occurrence of \(k\) events (\(\lambda^k\))
  \item
    Divided by the number of permutations (\(k!\))
  \item
    Finally \(e^{-\lambda}\) is to get the area to 1
  \end{itemize}
\item
  The PDF starts at \(e^{-\lambda}\), increases, and then decreases
\end{itemize}
\end{frame}

\begin{frame}{Poisson distribution: mean and variance}
\protect\phantomsection\label{poisson-distribution-mean-and-variance}
\begin{itemize}
\item
  Mean: \(\EE[X] = \lambda\)
\item
  Variance: \(\VV[X] = \lambda\)
\item
  Thus to model something as a Poisson, mean and variance need to be
  equal

  \begin{itemize}
  \tightlist
  \item
    This assumption can be checked using the data
  \end{itemize}
\end{itemize}

\begin{frame}{Poisson as approximation to the Binomial}
\protect\phantomsection\label{poisson-as-approximation-to-the-binomial}
\begin{itemize}
\item
  When \(n \gg 1\) and \(p \ll 1\) (i.e., many attempts of a unlikely
  event), then \[\text{Binomial}(n, p) \approx \Poisson(n p)\] \[
  Pr(X = x)
  = {n \choose x} p^x (1 - p)^{n - x}
  \approx \frac{(np)^x e^{-np}}{x!}
  \]
\item
  Note that \(n p\) is the average number of successes in \(n\)
  attempts, since
  \[p = \frac{n_\text{success}}{n} \implies n p = n_\text{success}\] so
  it is like considering successes as number of arrivals in a given
  period
\item
  Poisson is simpler to compute since it has only one factorial, instead
  of two factorials like the Binomial
\end{itemize}
\end{frame}

\begin{frame}{Student's t-distribution: definition in terms of

\begin{frame}{Markov inequality}
\protect\phantomsection\label{markov-inequality}
\begin{itemize}
\item
  \textbf{\emph{Hypothesis}}
\item
  Given \(X\) discrete or continuous RV
\item
  \(X\) is a non-negative RV (i.e., \(X \ge 0\), PDF is all after 0)
\item
  \(X\) has finite mean: \(\EE[X] < \infty\)
\item
  \textbf{\emph{Thesis}}
\item
  The probability that \(X\) is larger than a certain value is bounded
  by the mean \[\Pr(X \ge x) \le \frac{\EE[X]}{x}\]
\end{itemize}
\end{frame}

\begin{frame}{Markov inequality: geometric interpretation}
\protect\phantomsection\label{markov-inequality-geometric-interpretation}
\begin{itemize}
\item
  Given a RV \(X \ge 0\) with a finite mean
\item
  The ``flipped CDF'' \(1 - F_X(x)\) is dominated by an hyperbole
  passing by \((y, x) = (\EE[X], 1)\)
\item
  This is also related to the fact that a PDF needs to sum to 1 and thus
  needs to decrease at least like \(1/n\)
\end{itemize}
\end{frame}

\begin{frame}{Chebyshev inequality}
\protect\phantomsection\label{chebyshev-inequality}
\begin{itemize}
\item
  \textbf{\emph{Hypothesis}}
\item
  Given \(X\) discrete or continuous RV
\item
  \(X\) with finite mean \(\mu\) and variance \(\sigma^2\)
\item
  \textbf{\emph{Thesis}}
\item
  The probability that \(X\) is far from the mean is bound by the
  variance:
  \[\Pr(|X - \mu| \ge \varepsilon) \le \frac{\sigma^2}{\varepsilon^2}\]
\end{itemize}
\end{frame}

\begin{frame}{Chebyshev inequality in terms of z-scores}
\protect\phantomsection\label{chebyshev-inequality-in-terms-of-z-scores}
\begin{itemize}
\item
  \textbf{\emph{Hypothesis}}
\item
  Given \(X\) discrete or continuous RV
\item
  \(X\) with finite mean \(\mu\) and variance \(\sigma^2\)
\item
  \textbf{\emph{Thesis}}
\item
  Expressing the distance from the mean in terms of standard deviation
  \(\varepsilon = k \sigma\):
  \[\Pr(\frac{|X - \mu|}{\sigma} \ge k) \le \frac{1}{k^2}\]
\item
  The probability that the z-score of a RV is far away from 0 at least a
  certain number \(k\) is bounded by \(\frac{1}{k ^ 2}\)
\end{itemize}
\end{frame}

\begin{frame}{Hoeffding inequality}
\protect\phantomsection\label{hoeffding-inequality}
\begin{itemize}
\tightlist
\item
  Given a Bernoulli RV with probability of success \(\mu\)
\item
  We want to estimate \(\mu\) using \(N\) samples:
  \[\nu = \frac{1}{N} \sum_{i=1}^N X_i\]
\item
  Then \[\Pr(|\nu - \mu| > \varepsilon) \le 2 e^{-2 \varepsilon^2 N}\]
\item
  Since \(\nu\) is bound in \([\mu - \varepsilon, \mu + \varepsilon]\),
  we want a small \(\varepsilon\) with a large probability
\end{itemize}
\end{frame}
