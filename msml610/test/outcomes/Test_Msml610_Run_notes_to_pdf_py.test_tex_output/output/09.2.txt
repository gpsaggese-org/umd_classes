\begin{frame}{Algorithms for Specific Models}
\protect\phantomsection\label{algorithms-for-specific-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{General temporal probabilistic reasoning}}
  makes minimal assumptions:

  \begin{itemize}
  \tightlist
  \item
    Markov property for transitions
  \item
    Markov property for sensor model
  \item
    No constraints on:

    \begin{itemize}
    \tightlist
    \item
      Mathematical form of transition/sensor models
    \item
      Nature of state and evidence variables (discrete or continuous)
    \end{itemize}
  \end{itemize}
\item
  Improve efficiency and accuracy by exploiting specific model
  structures:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Hidden Markov Models (HMMs)}}:

    \begin{itemize}
    \tightlist
    \item
      State is a single discrete variable
    \item
      Transition and observation models are discrete probability tables
    \item
      Enables fast algorithms like Viterbi, forward-backward, etc
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Kalman Filters}}:

    \begin{itemize}
    \tightlist
    \item
      State variables are continuous and normally distributed
    \item
      Linear Gaussian models for transitions and observations
    \item
      Allows exact, efficient updates using matrix operations
    \end{itemize}
  \end{itemize}
\item
  Tailored algorithms can be orders of magnitude faster and more
  accurate than general methods
\end{itemize}
\end{frame}

\begin{frame}{Hidden Markov Model: Formulation}
\protect\phantomsection\label{hidden-markov-model-formulation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Hidden Markov Model (HMM)}}: Temporal model
  with simplified structure for efficiency

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{State model}}:

    \begin{itemize}
    \tightlist
    \item
      System state at time \(t\) is a discrete random variable
      \(X_t \in \{1, \dots, S\}\)
    \item
      E.g., in umbrella world, \(X_t = Rain_t\) with states \{\(Rain\),
      \(Sunny\)\}
    \item
      Can combine multiple variables into one ``mega-state'' variable
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Transition model}}
    \(\Pr(X_t | X_{t-1})\):

    \begin{itemize}
    \tightlist
    \item
      Transition matrix \(\vT\) of size \(S \times S\)
    \item
      Entry \(T_{ij} = \Pr(X_t = j | X_{t-1} = i)\): probability of
      transitioning from state \(i\) to \(j\)
    \end{itemize}
  \item
    \textbf{\textcolor{teal}{Sensor model}}:

    \begin{itemize}
    \tightlist
    \item
      Defined as \(\Pr(E_t | X_t = i)\) for each state \(i\)
    \item
      Representable as a vector or diagonal matrix \(\vO\)
    \item
      No assumptions about number or type (discrete/continuous) of
      observation variables
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Benefit}}

  \begin{itemize}
  \tightlist
  \item
    Enables efficient algorithms like forward, backward, and Viterbi
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Hidden Markov Model: Example}
\protect\phantomsection\label{hidden-markov-model-example}
\begin{itemize}
\item
  E.g., if \(Rain = T\) is state 1 and \(Rain = F\) is state 2, then the
  transition matrix for the umbrella world

  \begingroup \scriptsize

  {\def\LTcaptype{none} % do not increment counter
  \begin{longtable}[]{@{}ll@{}}
  \toprule\noalign{}
  \(R_{t-1}\) & \(\Pr(R_t | R_{t-1})\) \\
  \midrule\noalign{}
  \endhead
  T & 0.7 \\
  F & 0.3 \\
  \bottomrule\noalign{}
  \end{longtable}
  }

  \endgroup

  becomes the transition model

  \[
  \mT =
  \begin{pmatrix}
  0.7 & 0.3 \\
  0.3 & 0.7
  \end{pmatrix}
  \]
\item
  On day 1 we observe \(U_1 = T\) and on day 3, \(U_3 = F\), we have the
  observation matrices

  \[
  \mO_1 =
  \begin{pmatrix}
  0.9 & 0 \\
  0 & 0.2
  \end{pmatrix}
  \quad
  \mO_3 =
  \begin{pmatrix}
  0.1 & 0 \\
  0 & 0.8
  \end{pmatrix}
  \]
\end{itemize}
\end{frame}

\begin{frame}{Hidden Markov Model: Algorithms}
\protect\phantomsection\label{hidden-markov-model-algorithms}
\begin{itemize}
\tightlist
\item
  Using matrix representation all the forward / backward computations
  become matrix operations: \begin{align*}
  & \mathbf{f}_{1:t+1} = \alpha \mathbf{O}_{t+1} \mathbf{T}^\top \mathbf{f}_{1:t} \\
  & \mathbf{b}_{k+1:t} = \mathbf{T} \mathbf{O}_{k+1} \mathbf{b}_{k+2:t} \\
  \end{align*}

  \begin{itemize}
  \tightlist
  \item
    Express inference tasks (e.g., filtering, smoothing) as efficient
    matrix multiplication
  \end{itemize}
\item
  Specialized algorithms to improve time and space complexity
\end{itemize}
\end{frame}

\begin{frame}{Markov Random Fields}
\protect\phantomsection\label{markov-random-fields}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{Markov Random Field}} is a probabilistic
  graphical model

  \begin{itemize}
  \tightlist
  \item
    Represents a joint distribution using an undirected graph

    \begin{itemize}
    \tightlist
    \item
      Nodes = random variables
    \item
      Edges = relationships (dependencies) between variables
    \end{itemize}
  \item
    Key idea: \textbf{\textcolor{orange}{Markov property}}

    \begin{itemize}
    \tightlist
    \item
      Each variable is conditionally independent of non-neighbors given
      its neighbors
    \end{itemize}
  \item
    Model
    \textbf{\textcolor{darkgreen}{spatial and contextual dependencies}}

    \begin{itemize}
    \tightlist
    \item
      Capture local interactions that combine into a global structure
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Example}}: Image de-noising

  \begin{itemize}
  \tightlist
  \item
    Each pixel tends to have similar intensity to its neighbors
  \item
    Noise introduces local inconsistencies
  \item
    MRF models smoothness while respecting observed data
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Example}}: Social networks

  \begin{itemize}
  \tightlist
  \item
    Friends influence each other's behavior
  \item
    Dependencies exist only among connected individuals
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Markov Random Fields: Model Form}
\protect\phantomsection\label{markov-random-fields-model-form}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \(Pr(X) = \frac{1}{Z} \prod_{C \in \mathcal{C}} \psi_C(X_C)\)
\item
  \(X\): set of all random variables in the model
\item
  \(\mathcal{C}\): set of cliques in the graph (fully connected subset
  of nodes)
\item
  \(\psi_C(X_C)\): \textbf{\textcolor{red}{potential function}} for
  clique \(C\)

  \begin{itemize}
  \tightlist
  \item
    Assigns a positive score to each possible configuration of variables
    in \(C\)
  \item
    Clique potentials \(\psi_C\) encode preferences or constraints
  \item
    Intuition: measures ``compatibility'' of values
  \item
    High \(\psi_C\) = compatible configuration
  \item
    Low \(\psi_C\) = unlikely configuration
  \end{itemize}
\item
  \(Z\): \textbf{\textcolor{blue}{partition function}}

  \begin{itemize}
  \tightlist
  \item
    Ensures probabilities sum to \(1\)
  \item
    Usually very hard to compute for large graphs
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Markov Logic Networks: Basics}
\protect\phantomsection\label{markov-logic-networks-basics}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{Markov Logic Network}} (MLN) combines:

  \begin{itemize}
  \tightlist
  \item
    First-order logic (expressing knowledge with rules and quantifiers)
  \item
    Markov Random Fields (modeling uncertainty with probabilities)
  \end{itemize}
\item
  Each element is a pair \((F_i, w_i)\)

  \begin{itemize}
  \tightlist
  \item
    \(F_i\): a first-order logic formula

    \begin{itemize}
    \tightlist
    \item
      E.g., ``\(Friends(x,y) \implies Similar(x,y)\)''
    \end{itemize}
  \item
    \(w_i\): a weight measuring the strength of belief in \(F_i\)

    \begin{itemize}
    \tightlist
    \item
      Higher \(w_i\) = formula more important in shaping the probability
      distribution
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Semantics}}:

  \begin{itemize}
  \tightlist
  \item
    An MLN defines a probability distribution over \emph{possible
    worlds}
  \item
    A world = a complete assignment of truth values to all ground atoms
  \item
    If a world satisfies many high-weight formulas, it becomes
    \emph{more probable}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Joint distribution}}:

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(X=x) = \frac{1}{Z} \exp\left(\sum_i w_i n_i(x)\right)\)
  \item
    \(n_i(x)\) = number of \textbf{\textcolor{teal}{true groundings}} of
    formula \(F_i\) in world \(x\)

    \begin{itemize}
    \tightlist
    \item
      Example: If \(F_i\) = ``Friends(x,y) → Similar(x,y)'' and in world
      \(x\) this holds for 7 pairs \((x,y)\) out of 10, then
      \(n_i(x)=7\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Special cases}}:

  \begin{itemize}
  \tightlist
  \item
    If all weights \(w_i \to \infty\): only worlds where all formulas
    are satisfied have nonzero probability \(\to\) recovers
    \emph{classical logic}
  \item
    If all weights are finite: allows some violations but assigns them
    lower probability
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{EM Algorithm: Intuition and Applications}
\protect\phantomsection\label{em-algorithm-intuition-and-applications}
\begin{itemize}
\tightlist
\item
  Expectation-Maximization (EM) is a method for learning with hidden or
  missing data

  \begin{itemize}
  \tightlist
  \item
    Useful when some variables influencing the data are not directly
    observed
  \item
    Works by iteratively improving parameter estimates
  \item
    Alternates between estimating missing data and optimizing parameters
  \end{itemize}
\item
  Two main steps:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{red}{E-step (Expectation)}}: Estimate
    distribution over hidden variables using current parameters
  \item
    \textbf{\textcolor{blue}{M-step (Maximization)}}: Update parameters
    to maximize expected log-likelihood from the E-step
  \end{itemize}
\item
  Used in diverse settings:

  \begin{itemize}
  \tightlist
  \item
    Unsupervised clustering (e.g., Gaussian Mixture Models)
  \item
    Learning with incomplete data in Bayesian networks
  \item
    Hidden Markov Models (HMMs)
  \end{itemize}
\item
  Key property: EM increases data likelihood at each iteration
\item
  Converges to a local maximum of the likelihood function
\item
  No need for a step size parameter unlike gradient descent
\end{itemize}
\end{frame}

\begin{frame}{EM Algorithm: Mechanics and Example in Gaussian Mixture
Models}
\protect\phantomsection\label{em-algorithm-mechanics-and-example-in-gaussian-mixture-models}
\begin{itemize}
\tightlist
\item
  Goal: Recover parameters of Gaussian components from unlabeled data
\item
  \textbf{\textcolor{red}{E-step}}:

  \begin{itemize}
  \tightlist
  \item
    Compute \(p_{ij} = P(C=i \mid x_j)\) using Bayes' rule
  \item
    \(p_{ij} \propto P(x_j \mid C=i) P(C=i)\)
  \item
    Calculate effective count: \(n_i = \sum_j p_{ij}\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{M-step}}:

  \begin{itemize}
  \tightlist
  \item
    Update means: \(\mu_i \leftarrow \sum_j p_{ij} x_j / n_i\)
  \item
    Update covariances:
    \(\Sigma_i \leftarrow \sum_j p_{ij} (x_j - \mu_i)(x_j - \mu_i)^T / n_i\)
  \item
    Update weights: \(w_i \leftarrow n_i / N\)
  \end{itemize}
\item
  Intuition: Softly assign points to components, then re-estimate the
  components
\item
  Example scenario:

  \begin{itemize}
  \tightlist
  \item
    500 data points from a mix of 3 Gaussians
  \item
    EM reconstructs original distribution closely after iterations
  \end{itemize}
\item
  Limitations:

  \begin{itemize}
  \tightlist
  \item
    Sensitive to initialization
  \item
    May converge to poor local optima
  \item
    Component collapse or merging can occur
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Introduction to the Expectation--Maximization (EM)
Algorithm}
\protect\phantomsection\label{introduction-to-the-expectationmaximization-em-algorithm}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Purpose of EM Algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Iterative method for finding maximum likelihood or maximum a
    posteriori (MAP) estimates in statistical models with latent
    variables
  \item
    Particularly useful when data is incomplete or has missing values
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Key Concepts}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Observed Data ($\mathbf{X}$)}}: The
    data we can directly observe
  \item
    \textbf{\textcolor{teal}{Latent Variables ($\mathbf{Z}$)}}: Hidden
    or unobserved variables that influence the observed data
  \item
    \textbf{\textcolor{cyan}{Parameters ($\boldsymbol{\theta}$)}}:
    Unknown parameters to be estimated
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Challenge Addressed}}

  \begin{itemize}
  \tightlist
  \item
    Direct maximization of the likelihood function
    \(p(\mathbf{X}|\boldsymbol{\theta})\) is often intractable due to
    the presence of latent variables
  \end{itemize}
\item
  \textbf{\textcolor{violet}{EM Algorithm Overview}}

  \begin{itemize}
  \tightlist
  \item
    Alternates between estimating the expected value of the
    log-likelihood (E-step) and maximizing this expectation (M-step)
  \end{itemize}
\item
  \textbf{\textcolor{magenta}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Widely used in clustering (e.g., Gaussian Mixture Models), natural
    language processing, and image reconstruction
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The EM Algorithm: Step-By-Step}
\protect\phantomsection\label{the-em-algorithm-step-by-step}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Initialization}}

  \begin{itemize}
  \tightlist
  \item
    Start with initial guesses for the parameters
    \(\boldsymbol{\theta}^{(0)}\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{E-Step (Expectation Step)}}

  \begin{itemize}
  \tightlist
  \item
    Compute the expected value of the log-likelihood function, with
    respect to the conditional distribution of the latent variables
    given the observed data and current parameter estimates:

    \begin{itemize}
    \tightlist
    \item
      \(Q(\boldsymbol{\theta}|\boldsymbol{\theta}^{(t)}) = \mathbb{E}_{\mathbf{Z}|\mathbf{X},\boldsymbol{\theta}^{(t)}}[\log p(\mathbf{X}, \mathbf{Z}|\boldsymbol{\theta})]\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{M-Step (Maximization Step)}}

  \begin{itemize}
  \tightlist
  \item
    Maximize the expected log-likelihood found in the E-step to update
    the parameters:

    \begin{itemize}
    \tightlist
    \item
      \(\boldsymbol{\theta}^{(t+1)} = \arg\max_{\boldsymbol{\theta}} Q(\boldsymbol{\theta}|\boldsymbol{\theta}^{(t)})\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Iteration}}

  \begin{itemize}
  \tightlist
  \item
    Repeat E and M steps until convergence, i.e., until the parameters
    stabilize or the increase in likelihood is below a threshold
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Mathematical Foundation of EM}
\protect\phantomsection\label{mathematical-foundation-of-em}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Likelihood with Latent Variables}}

  \begin{itemize}
  \tightlist
  \item
    The marginal likelihood of the observed data is:

    \begin{itemize}
    \tightlist
    \item
      \(p(\mathbf{X}|\boldsymbol{\theta}) = \int p(\mathbf{X}, \mathbf{Z}|\boldsymbol{\theta}) d\mathbf{Z}\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Intractability}}

  \begin{itemize}
  \tightlist
  \item
    The integral is often difficult to compute due to the complexity
    introduced by the latent variables
  \end{itemize}
\item
  \textbf{\textcolor{blue}{EM Solution}}

  \begin{itemize}
  \tightlist
  \item
    EM circumvents this by iteratively applying the E and M steps to
    find parameter estimates that locally maximize the likelihood
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Convergence}}

  \begin{itemize}
  \tightlist
  \item
    Each iteration of EM is guaranteed to increase the likelihood
    function, ensuring convergence to a local maximum
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Example: Gaussian Mixture Models (GMM)}
\protect\phantomsection\label{example-gaussian-mixture-models-gmm}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem Setup}}

  \begin{itemize}
  \tightlist
  \item
    Data is assumed to be generated from a mixture of Gaussian
    distributions, each with its own mean and covariance
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Latent Variables}}

  \begin{itemize}
  \tightlist
  \item
    Each data point is associated with a latent variable indicating the
    Gaussian component from which it was generated
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{E-Step in GMM}}

  \begin{itemize}
  \tightlist
  \item
    Compute the posterior probabilities (responsibilities) that each
    data point belongs to each Gaussian component
  \end{itemize}
\item
  \textbf{\textcolor{teal}{M-Step in GMM}}

  \begin{itemize}
  \tightlist
  \item
    Update the parameters (means, covariances, and mixing coefficients)
    of each Gaussian component using the responsibilities computed in
    the E-step
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Iteration}}

  \begin{itemize}
  \tightlist
  \item
    Repeat E and M steps until the parameters converge
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Properties and Limitations of EM}
\protect\phantomsection\label{properties-and-limitations-of-em}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Advantages}}

  \begin{itemize}
  \tightlist
  \item
    Can handle missing or incomplete data effectively
  \item
    Provides a framework for parameter estimation in complex models
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Limitations}}

  \begin{itemize}
  \tightlist
  \item
    Converges to a local maximum, which may not be the global maximum
  \item
    Sensitive to initial parameter estimates; poor initialization can
    lead to suboptimal solutions
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Extensions and Variants}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{teal}{Variational Bayes}}: Provides a fully
    Bayesian approach by estimating distributions over parameters
  \item
    \textbf{\textcolor{cyan}{Generalized EM (GEM)}}: Relaxes the
    requirement of fully maximizing the expected log-likelihood in the
    M-step
  \item
    \textbf{\textcolor{blue}{Expectation Conditional Maximization (ECM)}}:
    Breaks the M-step into several conditional maximization steps
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Practical Considerations}}

  \begin{itemize}
  \tightlist
  \item
    Multiple runs with different initializations can help in finding
    better solutions
  \item
    Monitoring the increase in likelihood can help in determining
    convergence
  \end{itemize}
\end{itemize}
\end{frame}
