\begin{frame}{From Predictions to Decisions}
\protect\phantomsection\label{from-predictions-to-decisions}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Prediction}} answers: \emph{``What will
  happen?''}

  \begin{itemize}
  \tightlist
  \item
    Input: features \(X\)
  \item
    Output: estimate \(\hat{y} = \EE[Y | X]\)
  \item
    Optimized for accuracy on a fixed distribution
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Decision}} answers: \emph{``What should
  I do?''}

  \begin{itemize}
  \tightlist
  \item
    Input: current state, set of possible actions \(\mathcal{A}\)
  \item
    Output: action \(a^* \in \mathcal{A}\) that maximizes expected
    utility
  \item
    Requires reasoning about \emph{what would change} under each action
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Question}}: If we already have a highly
  accurate predictor, isn't the decision obvious?

  \begin{itemize}
  \tightlist
  \item
    No: a good predictor tells us what \emph{correlates} with good
    outcomes, not what \emph{causes} them
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Prediction Pipeline: Standard Supervised Learning}
\protect\phantomsection\label{prediction-pipeline-standard-supervised-learning}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Typical ML system}} is optimized
  \emph{end-to-end} for prediction:

  \begin{enumerate}
  \tightlist
  \item
    Collect observational data \((X_i, Y_i)\)
  \item
    Fit a model \(\hat{f}(X) \approx \EE[Y | X]\)
  \item
    Deploy system to score new units
  \end{enumerate}
\item
  \textbf{\textcolor{darkgreen}{Assumption}}: the joint distribution
  \(\Pr(X, Y)\) at deployment is the same as at training

  \begin{itemize}
  \tightlist
  \item
    No intervention changes how \(X\) is generated
  \item
    No agent \emph{acts} on the prediction
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: a spam filter trained on past
  emails to predict \(\Pr(spam | X)\)

  \begin{itemize}
  \tightlist
  \item
    Accurate predictor is useful as long as spammers do not adapt
  \end{itemize}
\end{itemize}

\begin{frame}{Decision Pipeline: What Changes}
\protect\phantomsection\label{decision-pipeline-what-changes}
\begin{itemize}
\tightlist
\item
  A \textbf{\textcolor{red}{decision pipeline}} explicitly represents:

  \begin{itemize}
  \tightlist
  \item
    The set of available actions \(\mathcal{A}\)
  \item
    A utility function \(U(y, a)\) over outcomes \(y\) and actions
    \(a \in \mathcal{A}\)
  \item
    A causal model \(\Pr(Y | do(a))\) of how actions affect outcomes
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Key distinction}}:

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(Y | X = x)\) is observational
  \item
    \(\Pr(Y | do(X = x))\) is interventional
  \item
    These can differ drastically when confounders are present
  \item
    Decisions depend on the interventional quantity
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: a hospital wants to decide
  \emph{whether to prescribe drug A}

  \begin{itemize}
  \tightlist
  \item
    Observational: \(\Pr(recovery | took drug A)\)
  \item
    Interventional: \(\Pr(recovery | do(give drug A))\)
  \item
    The first is biased by who chose to take the drug
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Confounding: The Root Cause}
\protect\phantomsection\label{confounding-the-root-cause}
\begin{itemize}
\item
  A \textbf{\textcolor{red}{confounder}} \(Z\) is a common cause of both
  the proposed action \(X\) and the outcome \(Y\)

  \begin{itemize}
  \tightlist
  \item
    Induces a non-causal association \(X \leftrightarrow Y\) through
    \(Z\)
  \item
    Observational \(\Pr(Y | X) \neq \Pr(Y | do(X))\)
  \end{itemize}
\item
  Without adjusting for \(Z\), any purely predictive model will encode
  the spurious association
\item
  \textbf{\textcolor{darkgreen}{Example}}: ice cream sales and drowning
  deaths are highly correlated

  \begin{itemize}
  \tightlist
  \item
    Predictor: \emph{``more ice cream \(\to\) predict more drownings''}
    (accurate)
  \item
    Decision: \emph{``ban ice cream to reduce drownings''} (disastrous)
  \item
    Confounder: hot weather causes both
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Question}}: Can we avoid this by collecting
  more features?

  \begin{itemize}
  \tightlist
  \item
    Only if the missing confounder is among features \emph{and} properly
    adjusted for
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Policy Reversal}
\protect\phantomsection\label{policy-reversal}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: A \emph{policy reversal} is when
  an optimal decision based on predictions becomes harmful once enacted

  \begin{itemize}
  \tightlist
  \item
    Common in medicine, economics, operations research
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Mechanism}}: acting on \(X\) breaks the
  correlation with the unobserved confounder \(Z\), so the observed
  association no longer holds
\item
  \textbf{\textcolor{blue}{Example}}: emergency room triage

  \begin{itemize}
  \tightlist
  \item
    Predictor: patients with asthma + pneumonia have lower mortality
  \item
    Naive policy: send them home first since they're low-risk
  \item
    Reality: they had lower mortality \emph{because} they received
    aggressive care

    \begin{itemize}
    \tightlist
    \item
      Removing the care removes the protection
    \end{itemize}
  \end{itemize}
\item
  Predictive patterns in historical data reflect the \emph{policy that
  generated the data}, not what happens when we change it
\end{itemize}
\end{frame}

\begin{frame}{Causal Models for Decisions}
\protect\phantomsection\label{causal-models-for-decisions}
\begin{itemize}
\item
  A \textbf{\textcolor{red}{causal model}} specifies, for each variable
  \(X_i\):

  \begin{itemize}
  \tightlist
  \item
    Its direct causes (parents): \(\text{Pa}(X_i)\)
  \item
    A structural mechanism: \(X_i = f_i(\text{Pa}(X_i), U_i)\) with
    noise \(U_i\)
  \end{itemize}
\item
  Enables reasoning about:

  \begin{itemize}
  \tightlist
  \item
    \emph{Observation}: \(\Pr(Y | X = x)\) (seeing \(x\))
  \item
    \emph{Intervention}: \(\Pr(Y | do(X = x))\) (setting \(x\))
  \item
    \emph{Counterfactual}: \(\Pr(Y_{X=x} | X = x')\) (had \(x\) been
    different)
  \end{itemize}
\item
  For decision-making we primarily need the interventional layer
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    A marketing team can observe that users who open an email purchase
    more
  \item
    Only a causal model can say whether \emph{sending} the email causes
    purchases
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Utility Functions}
\protect\phantomsection\label{utility-functions}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: A \emph{utility function}
  \(U: \mathcal{Y} \to \bbR\) maps each outcome \(y \in \mathcal{Y}\) to
  a real-valued preference score

  \begin{itemize}
  \tightlist
  \item
    Higher is better
  \item
    Encodes the decision-maker's goals on a single scale
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Properties}}:

  \begin{itemize}
  \tightlist
  \item
    Utilities are \emph{subjective}: different stakeholders can have
    different \(U\)
  \item
    Utilities are \emph{cardinal}: differences (not just rank) carry
    meaning

    \begin{itemize}
    \tightlist
    \item
      Defined up to positive affine transformations \(U \to a U + b\),
      \(a > 0\)
    \item
      The zero point and unit are arbitrary, but differences are
      meaningful
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: a hospital utility

  \begin{itemize}
  \tightlist
  \item
    Full recovery: \(U = 1.0\)
  \item
    Recovery with complications: \(U = 0.7\)
  \item
    Death: \(U = 0.0\)
  \item
    A \(10\%\) complication reduction is worth \(0.03\) in utility on
    average
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Expected Utility Principle}
\protect\phantomsection\label{expected-utility-principle}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Fact}}: A \emph{rational agent} chooses the
  action \(a\) that maximizes expected utility under uncertainty: \[
  a^*
    = \arg\max_{a \in \mathcal{A}}
      \EE_{Y \sim \Pr(Y | a)}[U(Y)]
  \]
\item
  The justification for maximizing expected utility comes from the VNM
  theorem
\item
  \textbf{\textcolor{darkgreen}{Definition}}: \emph{Lottery} = uncertain
  prospect that assigns probabilities \(p_i\) to possible outcomes
  \(x_i\)

  \begin{itemize}
  \tightlist
  \item
    E.g., \(L = (0.5, \$100;\; 0.5, \$0)\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Theorem}} (Von Neumann-Morgenstern theorem,
  1944):

  \begin{itemize}
  \tightlist
  \item
    If preferences over lotteries satisfy 4 axioms

    \begin{itemize}
    \tightlist
    \item
      \emph{Completeness}: for any two lotteries, \(A \succeq B\) or
      \(B \succeq A\) (any pair can be compared)
    \item
      \emph{Transitivity}:
      \(A \succeq B, B \succeq C \implies A \succeq C\)
    \item
      \emph{Continuity}: If \(A \succ B \succ C\) then there exists some
      probability \(p \in (0,1)\) such that the individual is
      indifferent between \(B\) and a lottery that gives \(A\) with
      probability \(p\) and \(C\) with probability \(1-p\)
    \item
      \emph{Independence}: Adding the same ``background gamble'' to both
      options should not reverse preferences. If \(A \succeq B\), then
      for any lottery \(C\) and probability \(\alpha \in (0,1)\),
      \(\alpha A + (1-\alpha)C \succeq \alpha B + (1-\alpha)C\)
    \end{itemize}
  \item
    Then there exists a utility function \(U\) such that the agent acts
    \emph{as if} maximizing expected utility
    \(\EE[U(L)] = \sum_i p_i U(x_i)\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Expected Utility: Buying Insurance Example}
\protect\phantomsection\label{expected-utility-buying-insurance-example}
\begin{itemize}
\item
  Assume

  \begin{itemize}
  \tightlist
  \item
    \(W\) = current wealth
  \item
    \(c\) = insurance premium (paid for sure)
  \item
    \(L\) = potential loss (without insurance) with probability \(p\)
  \item
    \(U(\cdot)\) = utility preference
  \end{itemize}
\item
  Decision: buy insurance if
  \[U(W - c) > p \cdot U(W - L) + (1 - p) \cdot U(W)\]
\item
  Risk-averse utilities (concave, i.e., \(U(\EE[W]) > \EE[U(W)]\)) make
  this inequality easier to satisfy

  \begin{itemize}
  \tightlist
  \item
    Concavity means that losing money hurts more than gaining the same
    amount
  \item
    Insurance also removes outcome variance, which is valuable to a
    risk-averse agent
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Expected Utility: Two Treatment Example}
\protect\phantomsection\label{expected-utility-two-treatment-example}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Setup}}: a doctor chooses between two
  treatments

  \begin{itemize}
  \tightlist
  \item
    Treatment 1: cures with probability \(0.9\), no side effect
  \item
    Treatment 2: cures with probability \(0.95\), mild side effect
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Utility}}: Need to quantify side effect
  through \(U(\cdot)\)

  \begin{itemize}
  \tightlist
  \item
    \(U\)(\emph{``Cure with no side effect''}) = \(1.0\)
  \item
    \(U\)(\emph{``Cure with side effect''}) = \(0.8\)
  \item
    \(U\)(\emph{``No cure''}) = \(0.0\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Computation}}: \begin{align*}
  \EE[U | T_1] &= 0.9 \cdot 1.0 + 0.1 \cdot 0.0 = 0.90 \\
  \EE[U | T_2] &= 0.95 \cdot 0.8 + 0.05 \cdot 0.0 = 0.76
  \end{align*}
\item
  \textbf{\textcolor{violet}{Decision}}:

  \begin{itemize}
  \tightlist
  \item
    \(T_1\) is better
  \item
    Even though \(T_2\) is more effective, its side-effect penalty
    outweighs the gain
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{From Observational to Interventional Expectation}
\protect\phantomsection\label{from-observational-to-interventional-expectation}
\begin{itemize}
\item
  \emph{Classical decision theory} uses \(\Pr(Y | a)\) as the
  distribution of \(Y\) given action \(a\)

  \begin{itemize}
  \tightlist
  \item
    If the data is observational, \(\Pr(Y | a)\) includes confounding
    bias
  \item
    Plugging it into expected utility yields a biased action ranking
  \end{itemize}
\item
  \emph{Causal decision theory} replaces \(\Pr(Y | a)\) with the
  interventional distribution \(\Pr(Y | do(a))\) \[
  a^*
    = \arg\max_{a \in \mathcal{A}}
      \EE_{Y \sim \Pr(Y | do(a))}[U(Y)]
  \]
\item
  \textbf{\textcolor{red}{Property}}: They are equivalent under:

  \begin{itemize}
  \tightlist
  \item
    Randomized experiments (no confounders)
  \item
    Successful back-door or front-door adjustment (i.e., complete
    knowledge of the causal graph + sufficient data)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Intervention in an Decision Network}
\protect\phantomsection\label{intervention-in-an-decision-network}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Do-operator}} \(do(X = x)\) mutates the causal
  graph:

  \begin{itemize}
  \tightlist
  \item
    Remove all edges \emph{into} \(X\)
  \item
    Fix \(X\) to the value \(x\)
  \item
    Propagate through the remaining structural equations
  \item
    Resulting distribution reflects a policy, not the observed world
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: \(do(Drug = 1)\)

  \begin{itemize}
  \tightlist
  \item
    Drug is no longer chosen by the physician
  \item
    All causal arrows from \(Gender\), \(Age\), etc. into \(Drug\) are
    cut
  \end{itemize}
\end{itemize}

\begin{frame}{Expected Outcome of an Intervention}
\protect\phantomsection\label{expected-outcome-of-an-intervention}
\begin{itemize}
\item
  Given a causal graph and estimated mechanisms, compute the expected
  outcome under each candidate action: \[
  \EE[Y | do(a)] = \sum_{y} y \cdot \Pr(Y = y | do(A = a))
  \]
\item
  Using back-door adjustment formula, compute interventional
  distribution \[
  \Pr(Y | do(A)) = \sum_{Z} \Pr(Y | A, Z) \Pr(Z)
  \] where \(Z\) is a back-door adjustment set
\item
  \textbf{\textcolor{red}{Example}}: drug decision

  \begin{itemize}
  \tightlist
  \item
    Observational: \(\Pr(\text{recovery} | \text{drug}) = 0.78\)
  \item
    Adjusted:
    \(\sum_z \Pr(\text{recovery} | \text{drug}, z) \Pr(z) = 0.83\)
  \item
    The adjusted value is what plugs into expected utility
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Multi-Criteria Trade-offs}

\item
  E.g.,

  \tightlist
  \item
    In clinical settings: efficacy, side effects, cost, patient burden

  \item

  \item
    Useful when stakeholder preferences are not yet clear
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}: recommending a cancer therapy

  \begin{itemize}
  \tightlist

\begin{frame}{Utility Theory: Pros and Cons}
\protect\phantomsection\label{utility-theory-pros-and-cons}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist

  \begin{itemize}

    If the optimal action is stable across a plausible range of
    utilities, the decision is robust
  \end{itemize}

\begin{frame}{Solving an Decision Network}
\protect\phantomsection\label{solving-an-decision-network}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Input}}: decision diagram with CPTs for chance
  nodes and a utility table
\item
  \textbf{\textcolor{orange}{Output}}: optimal policy \(\pi^*\), i.e., a
  function from information to actions
\item
  \textbf{\textcolor{darkgreen}{Steps}} (variable elimination):

  \begin{enumerate}
  \tightlist
  \item
    Eliminate chance nodes by summing them out
  \item
    Eliminate decision nodes by maximizing over their domain
  \item
    Process in reverse topological order
  \item
    Record the arg-max at each decision node as the optimal policy
  \end{enumerate}
\item
  \textbf{\textcolor{teal}{Complexity}}:

  \begin{itemize}
  \tightlist
  \item
    Time: \(O(|\mathcal{A}|^d \cdot \prod_i |X_i|)\) in the worst case
  \item
    Tractable when the graph has low treewidth
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Key insight}}: the same graphical machinery
  used for Bayesian inference also supports optimal decision-making
\end{itemize}
\end{frame}

\begin{frame}{Posteriors to Optimal Actions}
\protect\phantomsection\label{posteriors-to-optimal-actions}
\begin{itemize}
\tightlist
\item
  Bayesian decision-making is composed of two steps:

  \begin{enumerate}
  \tightlist
  \item
    \emph{Inference}: compute the posterior
    \(\Pr(\theta | \mathcal{D})\) from data
  \item
    \emph{Decision}: choose the action that maximizes \emph{posterior}
    expected utility \[
    a^* = \arg\max_{a \in \mathcal{A}}
          \EE_{\theta \sim \Pr(\theta | \mathcal{D})}
          \left[
            \EE_{Y \sim \Pr(Y | do(a), \theta)}[U(Y)]
          \right]
    \] where:

    \begin{itemize}
    \tightlist
    \item
      The \emph{inner expectation} is over outcome uncertainty \(Y\)
      given \(\theta\) (aleatoric)
    \item
      The \emph{outer expectation} is over parameter uncertainty
      \(\theta\) (epistemic)
    \end{itemize}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Risk Function: Definition}
\protect\phantomsection\label{risk-function-definition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: The \emph{risk function}
  \(R(\theta, \delta)\) is the expected loss of decision rule \(\delta\)
  when the true state of nature is \(\theta\) \[
  R(\theta, \delta) \defeq \EE_{X \sim P_\theta}[L(\theta, \delta(X))]
  \]

  \begin{itemize}
  \tightlist
  \item
    \(L(\theta, \delta(X))\) = loss incurred by choosing action
    \(\delta(X)\) when the true parameter is \(\theta\)
  \item
    The expectation is over data \(X\) drawn from \(P_\theta\)
  \item
    Risk is a function of both the rule \(\delta\) and the parameter
    \(\theta\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Intuition}}: \(R(\theta, \delta)\) tells
  you how bad rule \(\delta\) is on average for each possible value of
  \(\theta\)

  \begin{itemize}
  \tightlist
  \item
    Lower risk is better
  \item
    No single rule is guaranteed to have the lowest risk at every
    \(\theta\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}: a decision rule with risks
  \((2, 3, 4)\) over \(\theta \in \{0, 1, 2\}\) means:

  \begin{itemize}
  \tightlist
  \item
    \(R(0, \delta) = 2\), \(R(1, \delta) = 3\), \(R(2, \delta) = 4\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Admissible Decision Rules: Definition}
\protect\phantomsection\label{admissible-decision-rules-definition}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: In decision theory, a decision
  rule \(\delta_1\) \emph{dominates} \(\delta_2\) when:

  \begin{itemize}
  \tightlist
  \item
    For every parameter \(\theta\),
    \(R(\theta, \delta_1) \le R(\theta, \delta_2)\)
  \item
    For at least one \(\theta\),
    \(R(\theta, \delta_1) < R(\theta, \delta_2)\)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Intuition}}: \(\delta_1\) is never worse
  and sometimes strictly better
\item
  \textbf{\textcolor{darkgreen}{Example}}: the rule with risks
  \((2, 3, 4)\) from the previous slide is dominated by \((2, 2, 4)\),
  since it is better at \(\theta=1\), ties elsewhere
\item
  \textbf{\textcolor{teal}{Definition}}: In decision theory, a decision
  rule \(\delta\) is \emph{admissible} if no other rule \emph{dominates}
  it
\item
  \textbf{\textcolor{cyan}{Intuition}}: An admissible rule cannot be
  uniformly beaten

  \begin{itemize}
  \tightlist
  \item
    If another rule were at least as good everywhere and better
    somewhere, you could replace the original
  \item
    Admissibility is a \emph{minimal} standard: a rule that \emph{is}
    dominated should never be used
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Why Bayes Rules Are Admissible}
\protect\phantomsection\label{why-bayes-rules-are-admissible}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Theorem}}: A Bayes decision rule, i.e.,
  minimizing Bayes risk
  \[r(\delta) = \int R(\theta, \delta)\,\pi(\theta)\,d\theta\] is
  admissible
\item
  \textbf{\textcolor{darkgreen}{Proof}} (by contradiction)

  \begin{enumerate}
  \tightlist
  \item
    Assume a Bayes rule \(\delta_B\) were not admissible

    \begin{itemize}
    \tightlist
    \item
      Then some \(\delta'\) dominates it:
      \(R(\theta, \delta') \le R(\theta,
      \delta_B)\;\forall\theta\), with strict \(<\) for some \(\theta\)
    \end{itemize}
  \item
    Integrate both sides with respect to the prior \(\pi(\theta)\): \[
    \int R(\theta, \delta')\,\pi(\theta)\,d\theta
      \;<\;
    \int R(\theta, \delta_B)\,\pi(\theta)\,d\theta
    \]

    \begin{itemize}
    \tightlist
    \item
      The inequality is strict because the integrand is strictly smaller
      on a set of positive prior mass
    \end{itemize}
  \item
    Contradiction: \(r(\delta') < r(\delta_B)\), but \(\delta_B\) was
    chosen to minimize Bayes risk

    \begin{itemize}
    \tightlist
    \item
      Therefore no dominating rule can exist, and \(\delta_B\) is
      admissible
    \end{itemize}
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Key insight}}:

  \begin{itemize}
  \tightlist
  \item
    A prior assigns weights to parameter values
  \item
    A rule that is optimal on average cannot be uniformly improved
  \item
    If it could, the weighted average would also be better
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Decision Rule: Example}
\protect\phantomsection\label{bayesian-decision-rule-example}
\begin{itemize}
\tightlist
\item
  A/B test of a new checkout flow

  \begin{itemize}
  \tightlist
  \item
    Parameter \(\theta\): conversion rate lift of variant B over A
  \item
    Prior: \(\theta \sim \mathcal{N}(0, 0.02^2)\)
  \item
    Data \(\mathcal{D}\): \(n = 5000\) users per arm; observed lift
    \(0.01\)
  \item
    Posterior: \(\theta | \mathcal{D} \sim \mathcal{N}(0.008, 0.004^2)\)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Utility of launching}}

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(\theta > 0 | \mathcal{D}) \approx 0.977\)
  \item
    \(U(\theta) = \theta - c\)

    \begin{itemize}
    \tightlist
    \item
      where \(c = 0.002\) represents the cost of maintaining the new
      code
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Expected utility}}:

  \begin{itemize}
  \tightlist
  \item
    \(\EE[U | \text{launch}] = 0.008 - 0.002 = 0.006 > 0\)
  \item
    \(\EE[U | \text{keep A}] = 0\)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Decision}}: launch B
\end{itemize}
\end{frame}

\begin{frame}{Bayesian vs.~Frequentist Decisions}
\protect\phantomsection\label{bayesian-vs.-frequentist-decisions}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\textbf{\textcolor{red}{Bayesian}}

\begin{itemize}
\item
  Posterior \(\Pr(\theta | \mathcal{D})\)
\item
  Acts on full distribution
\item
  Incorporates prior beliefs
\item
  Natural for sequential updating
\item
  Connects directly to utility
\item
  \emph{Pros}

  \begin{itemize}
  \tightlist
  \item
    Coherent under uncertainty
  \item
    Works for small samples
  \end{itemize}
\item
  \_Cons\_\_

  \begin{itemize}
  \tightlist
  \item
    Requires priors
  \item
    Computationally heavier
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\textbf{\textcolor{darkgreen}{Frequentist}}

\begin{itemize}
\item
  Point estimate \(\hat{\theta}\) + CI
\item
  Acts on test outcomes
\item
  No priors
\item
  Decision via hypothesis test
\item
  Utility often implicit
\item
  \emph{Pros}

  \begin{itemize}
  \tightlist
  \item
    Fewer subjective inputs
  \item
    Standardized in many fields
  \end{itemize}
\item
  \emph{Cons}

  \begin{itemize}
  \tightlist
  \item
    Thresholds (e.g., \(p < 0.05\)) are not utility-optimal
  \item
    Harder to combine with new data
  \end{itemize}
\end{itemize}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Bottom line}}: Bayesian framing slots
  naturally into causal decision-making because both use joint
  distributions end-to-end
\end{itemize}
\end{frame}

\begin{frame}{Prior Elicitation: What and Why}
\protect\phantomsection\label{prior-elicitation-what-and-why}
\begin{itemize}
\tightlist
\item
  In causal decision problems, the prior encodes beliefs about the
  \emph{causal effect} \(\theta\) before seeing data

  \begin{itemize}
  \tightlist
  \item
    E.g., size, sign, and uncertainty of \(\theta\)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Prior elicitation}} is especially important
  when:

  \begin{itemize}
  \tightlist
  \item
    Data is scarce or expensive (medicine, policy)
  \item
    Experiments are unethical or impossible
  \item
    Stakeholders disagree
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Question}}: Where do priors come from?

  \begin{itemize}
  \tightlist
  \item
    Literature meta-analyses
  \item
    Expert judgment
  \item
    Past experiments or pilot studies
  \item
    Regulatory or legal constraints
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Methods for Eliciting Priors}
\protect\phantomsection\label{methods-for-eliciting-priors}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Quantile elicitation}}: ask experts for
  10/50/90 percentiles of \(\theta\), then fit a distribution

  \begin{itemize}
  \tightlist
  \item
    E.g., \emph{``10\% chance lift is below \(0.5\%\), 50\% below
    \(1\%\), 90\% below \(3\%\)''}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pairwise judgments}}: elicit relative
  probabilities of outcomes, combine via a log-linear model
\item
  \textbf{\textcolor{blue}{Historical calibration}}: compare past
  predictions against actuals to adjust for bias and overconfidence

  \begin{itemize}
  \tightlist
  \item
    E.g., a clinical trial team estimates a 20\% prior probability that
    a new drug is at least as effective as standard of care, based on
    meta-analysis of 14 similar compounds
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Reference priors}}: use weakly informative
  priors (e.g., Jeffreys) when knowledge is vague

  \begin{itemize}
  \tightlist
  \item
    Avoid flat priors in high dimensions, they are not uninformative
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Pitfalls and Best Practices}
\protect\phantomsection\label{pitfalls-and-best-practices}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pitfalls}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Anchoring}: experts cling to a first-mentioned number
  \item
    \emph{Overconfidence}: intervals are consistently too narrow
  \item
    \emph{Framing effects}: elicit the \emph{same} quantity two ways but
    get different answers
  \item
    \emph{Confusing aleatoric vs epistemic uncertainty}:
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Best practices}}

  \begin{itemize}
  \tightlist
  \item
    Elicit from multiple experts independently, then aggregate
  \item
    Cross-validate with subsets of historical data
  \item
    Perform \emph{prior sensitivity analysis}

    \begin{itemize}
    \tightlist
    \item
      \emph{``Does the decision change under plausible alternative
      priors?''}
    \end{itemize}
  \item
    Document the elicitation process to defend decisions
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Value of Information: Motivation}
\protect\phantomsection\label{value-of-information-motivation}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Problem}}:
\item
  Before committing to a costly action, a decision-maker may gather more
  data

  \begin{itemize}
  \tightlist
  \item
    E.g., diagnostic test, survey, pilot experiment, additional A/B arm
  \end{itemize}
\item
  Is the extra information worth its cost?
\item
  \textbf{\textcolor{darkgreen}{Definition}}: \emph{Value of information
  (VOI)} quantifies the expected gain in utility from obtaining new
  evidence before making a decision

  \begin{itemize}
  \tightlist
  \item
    A central concept in decision analysis
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    A company considers an expensive market survey before a \$10M
    product launch
  \item
    Launch without: expected utility \(\EE[U]\)
  \item
    Launch with survey outcome \(E\)

    \begin{itemize}
    \tightlist
    \item
      \(\EE_E[\max_a \EE[U(Y) | do(a), E]]\)
    \end{itemize}
  \item
    Run the survey only if the difference exceeds its cost
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Expected Value of Perfect Information}
\protect\phantomsection\label{expected-value-of-perfect-information}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Definition}}: \emph{Expected Value of Perfect
  Information} (EVPI) is the upper bound on what any
  information-gathering activity could be worth: \[
  \text{EVPI}
    = \EE_{\theta}
      \left[
        \max_a U(a, \theta)
      \right]
      - \max_a \EE_\theta[U(a, \theta)]
  \]
\item
  \textbf{\textcolor{darkgreen}{Intuition}}:

  \begin{itemize}
  \tightlist
  \item
    First term: utility if we could learn \(\theta\) \emph{perfectly}
    before deciding
  \item
    Second term: utility of the best action under current uncertainty
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key properties}}:

  \begin{itemize}
  \tightlist
  \item
    \(\text{EVPI} \geq 0\) always
  \item
    \(\text{EVPI} = 0\) iff the current best action is optimal for
    \emph{every} \(\theta\)
  \item
    Any test costing more than EVPI is not worth running, under any
    accuracy
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Expected Value of Sample Information (EVSI)}
\protect\phantomsection\label{expected-value-of-sample-information-evsi}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Problem}}: Real information sources are
  imperfect
\item
  \textbf{\textcolor{darkgreen}{Definition}}: \emph{Expected Value of
  Sample Information} (EVSI) relaxes EVPI to consider noisy evidence
  \(E\)
  \[\text{EVSI}(E) = \EE_E[\max_a \EE[U | a, E]] - \max_a \EE[U | a]\]
\item
  \textbf{\textcolor{blue}{Key properties}}

  \begin{itemize}
  \tightlist
  \item
    \begin{itemize}
    \tightlist
    \item
      \(0 \leq \text{EVSI}(E) \leq \text{EVPI}\)
    \end{itemize}
  \item
    Noisier evidence \(\to\) smaller EVSI
  \item
    Tighter prior \(\to\) smaller EVSI (less to learn)
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    Diagnostic test with sensitivity \(0.9\) and specificity \(0.85\)
  \item
    Compute EVSI by integrating over all possible \((E = +, E = -)\)
    outcomes
  \item
    Run the test if \(\text{EVSI} - \text{cost} > 0\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Exploration vs Exploitation}
\protect\phantomsection\label{exploration-vs-exploitation}
\begin{itemize}
\tightlist
\item
  A sequential decision-maker faces a tension between:

  \begin{itemize}
  \tightlist
  \item
    \emph{Exploit}: pick the action that currently looks best
  \item
    \emph{Explore}: try other actions to reduce uncertainty about their
    effects
  \end{itemize}
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Pure exploitation can lock in a suboptimal choice based on early
    noise
  \item
    Pure exploration wastes samples on clearly bad options
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \tightlist
  \item
    A doctor who always prescribes the first drug that worked early in
    her career will never learn about newer, better options
  \end{itemize}
\item
  Causal models add richness: exploration can target structural
  uncertainty, not just parameter uncertainty
\end{itemize}
\end{frame}

\begin{frame}{Causal Multi-Armed Bandits}
\protect\phantomsection\label{causal-multi-armed-bandits}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Classic multi-armed bandit}}:

  \begin{itemize}
  \tightlist
  \item
    \(K\) arms with unknown reward distributions
  \item
    At each step, pull one arm, observe reward, update beliefs
  \item
    Objective: minimize \textbf{\textcolor{darkgreen}{regret}} relative
    to the best arm
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Causal bandit}}:

  \begin{itemize}
  \tightlist
  \item
    Arms are interventions \(do(a)\) on a known (or partly known) causal
    graph
  \item
    Observing one arm provides information about \emph{other} arms via
    the graph
  \item
    Much lower regret than a standard bandit in many settings
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example}}: advertising

  \begin{itemize}
  \tightlist
  \item
    Arms: combinations of channel, audience, creative
  \item
    Causal structure: audience and creative \emph{interact}, channel has
    additive effect
  \item
    Exploiting this structure cuts samples needed to find the best
    combination
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bayesian Optimization}
\protect\phantomsection\label{bayesian-optimization}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: \emph{Bayesian optimization}
  (BO) targets problems where:

  \begin{itemize}
  \tightlist
  \item
    The objective \(f(x)\) is \emph{expensive} to evaluate in terms of
    time, money, or resources

    \begin{itemize}
    \tightlist
    \item
      E.g., training a deep model, running a chemistry experiment,
      testing a web layout
    \end{itemize}
  \item
    No gradients available \(\nabla f(x)\)

    \begin{itemize}
    \tightlist
    \item
      You can't do gradient descent on \(f(x)\)
    \end{itemize}
  \item
    \(x\) lies in a low-to-moderate dimensional space (not
    high-dimensional)

    \begin{itemize}
    \tightlist
    \item
      E.g., \textless{} 20 dimensions
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Algorithm}}:

  \begin{enumerate}
  \tightlist
  \item
    Place a prior over \(f(x)\) (typically a Gaussian process)
  \item
    At each round, fit the posterior on observed \((x_i, f(x_i))\)
  \item
    Select \(x_{next}\) by maximizing an \emph{acquisition function}
    \(\alpha(x)\)
  \item
    Query \(f(x_{next})\), update, repeat
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Key insight}}:

  \begin{itemize}
  \tightlist
  \item
    Posterior (e.g., Gaussian Process) gives both a mean estimate
    \(\mu(x)\) and an uncertainty estimate \(\sigma(x)\) at every point
  \item
    You don't just chase the current best, but also explore regions
    where you're uncertain
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Examples}}:

  \begin{itemize}
  \tightlist
  \item
    Hyperparameter tuning for a deep model
  \item
    Tuning web page layout
  \item
    Picking reagent concentrations in a chemistry lab
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Acquisition Functions}
\protect\phantomsection\label{acquisition-functions}
\begin{itemize}
\tightlist
\item
  The acquisition function \(\alpha(x)\) is the decision policy

  \begin{itemize}
  \tightlist
  \item
    Encodes how you trade off exploration vs exploitation
  \end{itemize}
\item
  There are several strategies

  \begin{itemize}
  \tightlist
  \item
    \emph{Expected improvement (EI)}:
    \[\alpha_{EI}(x) = \EE[\max(f(x) - f^*, 0)]\]

    \begin{itemize}
    \tightlist
    \item
      \emph{``How much better than current best \(f^*\) might this point
      be, in expectation?''}
    \end{itemize}
  \item
    \emph{Upper confidence bound (UCB)}:
    \[\alpha_{UCB}(x) = \mu(x) + \kappa \sigma(x)\]

    \begin{itemize}
    \tightlist
    \item
      \emph{``Optimistically bias toward high-uncertainty regions;
      \(\kappa\) controls how much''}
    \end{itemize}
  \item
    \emph{Thompson sampling}: sample
    \(\tilde f \sim \Pr(f | \mathcal{D})\), maximize \(\tilde f\)

    \begin{itemize}
    \tightlist
    \item
      \emph{``Act as if a random plausible world is the real one''}
    \end{itemize}
  \item
    \emph{Entropy search}: pick \(x\) that most reduces entropy of
    \(\arg\max f\)

    \begin{itemize}
    \tightlist
    \item
      \emph{``Which point tells me most about where the optimum is?''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{red}{Example}}: tuning a learning rate

  \begin{itemize}
  \tightlist
  \item
    Posterior mean peaks at \(10^{-3}\), variance is largest at
    \(10^{-5}\)
  \item
    EI may query \(10^{-5}\) if improvement potential \textgreater{}
    current best
  \item
    UCB with high \(\kappa\) behaves similarly
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key takeaway}}: the acquisition function
  \emph{is} the policy, derived from the posterior and a utility for
  information
\end{itemize}
\end{frame}

\begin{frame}{Thompson Sampling}
\protect\phantomsection\label{thompson-sampling}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: \emph{Thompson sampling} is a
  Bayesian exploration strategy

  \begin{itemize}
  \tightlist
  \item
    Maintain a posterior over each arm's mean reward
  \item
    At each step

    \begin{itemize}
    \tightlist
    \item
      Sample a parameter from each posterior
    \item
      Pull the arm with the highest sampled mean
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Algorithm}}:

  \begin{enumerate}
  \tightlist
  \item
    Start with prior \(\Pr(\theta_a)\) for each arm \(a\)
  \item
    Sample \(\tilde\theta_a \sim \Pr(\theta_a | \mathcal{D})\) for each
    \(a\)
  \item
    Pull \(a^* = \arg\max_a \tilde\theta_a\)
  \item
    Observe reward, update posterior, repeat
  \end{enumerate}
\item
  \textbf{\textcolor{blue}{Properties}}:

  \begin{itemize}
  \tightlist
  \item
    Naturally balances exploration and exploitation
  \item
    Near-optimal regret bounds \(O(\sqrt{T K \log T})\)
  \item
    Easy to adapt to causal structure by sampling from posteriors over
    causal effects
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Causal Bayesian Optimization}
\protect\phantomsection\label{causal-bayesian-optimization}
\begin{itemize}
\tightlist
\item
  Causal BO combines BO with causal graphs:

  \begin{itemize}
  \tightlist
  \item
    Decision variable \(x\) is an intervention \(do(X = x)\)
  \item
    Surrogate model is a structural causal model (not just a Gaussian
    Process on \(y\))
  \item
    Leverages the graph to share information across interventions
  \end{itemize}
\item
  \textbf{\textcolor{red}{Example}}: drug dose optimization

  \begin{itemize}
  \tightlist
  \item
    Standard BO models \(y = f(\text{dose})\) directly
  \item
    Causal BO models intermediate biomarkers that mediate dose \(\to\)
    outcome
  \item
    Fewer trials needed when the causal mechanism is partly known
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    Sample efficient
  \item
    Principled exploration
  \item
    Uncertainty-aware
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Requires accurate causal structure
  \item
    Misspecification can hurt more than help
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Putting Everything Together}
\protect\phantomsection\label{putting-everything-together}
\begin{itemize}
\tightlist
\item
  Sequential decision-making:

  \begin{itemize}
  \tightlist
  \item
    \emph{Value of Information} tells you whether to explore
  \item
    \emph{Acquisition functions} tell you where to explore
  \item
    \emph{Causal structure} tells you how to share information across
    different interventions

    \begin{itemize}
    \tightlist
    \item
      Makes exploration vastly more efficient than a black-box approach
    \end{itemize}
  \end{itemize}
\item
  Prediction alone can't guide decisions
\item
  A causal model + Bayesian uncertainty + a principled exploration
  policy gives you a system that learns which actions work with minimal
  cost
\end{itemize}
\end{frame}

\begin{frame}{Aleatoric vs Epistemic Uncertainty}
\protect\phantomsection\label{aleatoric-vs-epistemic-uncertainty}
\begin{itemize}
\tightlist
\item
  There are different types of uncertainty

  \begin{itemize}
  \tightlist
  \item
    Aleatoric (related to physical reality of the system)
  \item
    Epistemic (related to lack of knowledge)
  \end{itemize}
\item
  A decision-maker needs to know

  \begin{itemize}
  \tightlist
  \item
    how much of the outcome uncertainty can I learn away (epistemic)
  \item
    how much is just irreducible randomness (aleatoric)?
  \item
    You can't compute the Value of Information correctly
  \item
    You might over-pay for data that can't actually reduce the
    uncertainty you care about
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Aleatoric Uncertainty: Irreducible Randomness}
\protect\phantomsection\label{aleatoric-uncertainty-irreducible-randomness}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Definition}}: \emph{Aleatoric uncertainty}
  comes from the inherent randomness in the outcome

  \begin{itemize}
  \tightlist
  \item
    Aka \_statistical\_or \emph{risk} uncertainty, stochastic
    uncertainty
  \item
    Cannot be reduced by collecting more data
  \item
    Captured by \(\Pr(Y | do(a), \theta)\) given known \(\theta\) (i.e.,
    the noise term in the structural equation.)
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Example}}: flipping a known-fair coin

  \begin{itemize}
  \tightlist
  \item
    \(\Pr(\text{heads}) = 0.5\) exactly
  \item
    More flips don't add information so we can predict better
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Example}}: medical treatment

  \begin{itemize}
  \tightlist
  \item
    Even if we knew the drug's exact mechanism, individual patient
    outcomes would vary due to biological stochasticity
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    A recommender's click-through rate varies user-to-user
  \item
    Even with perfect knowledge of
    \(\Pr(\text{click} | \text{user, item})\), specific clicks are
    random
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Implication}}: aleatoric uncertainty sets the
  \emph{floor} on risk

  \begin{itemize}
  \tightlist
  \item
    No amount of data or learning removes uncertainty
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Representing Aleatoric Uncertainty}
\protect\phantomsection\label{representing-aleatoric-uncertainty}
\begin{itemize}
\item
  Consider a structural causal model: \begin{align*}
  Y = f(\text{Pa}(Y), U_Y)\\
  U_Y \sim \Pr(U_Y)
  \end{align*} where \(U_Y\) is the exogenous noise capturing aleatoric
  variability

  \begin{itemize}
  \tightlist
  \item
    Common assumptions for \(U_Y\)

    \begin{itemize}
    \tightlist
    \item
      Gaussian: \(U_Y \sim \mathcal{N}(0, \sigma^2)\) for continuous
      \(Y\)
    \item
      Heavy-tailed (Student-\(t\), Laplace) for robust modeling
    \item
      Bernoulli / categorical for discrete outcomes
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{red}{Question}}: How do you separate signal from
  noise when you only see \(Y\)?
\item
  \textbf{\textcolor{darkgreen}{Problem}}: you only observe the sum as
  the realized \(Y\), which mixes

  \begin{itemize}
  \tightlist
  \item
    Signal: the systematic part \(f(Pa(Y))\) driven by the parents of
    \(Y\)
  \item
    Noise: the random shock \(U_Y\)
  \end{itemize}
\item
  Without additional assumptions, infinitely many splits into signal +
  noise are equally consistent with the data
\item
  \textbf{\textcolor{blue}{Solution}}:

  \begin{itemize}
  \tightlist
  \item
    If you have

    \begin{enumerate}
    \tightlist
    \item
      A tractable parametric form for \(U_Y\) (e.g., assume \(U_Y \sim
      \mathcal{N}(0, \sigma^2)\))

      \begin{itemize}
      \tightlist
      \item
        Constrains how much of \(Y\)'s variation could plausibly be
        noise.
      \end{itemize}
    \item
      Causal structure: know (or estimate) the parents \(Pa(Y)\) and
      function \(f\)
    \end{enumerate}

    \begin{itemize}
    \tightlist
    \item
      Measure the parents, estimate \(f\), compute expected \(Y\)
      (signal), and attribute the residual to noise:
      \(\hat{U}_Y = Y - \hat{f}(Pa(Y))\)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Example:}}

  \begin{itemize}
  \tightlist
  \item
    Suppose \(Y\) is blood pressure
  \item
    Causal model says

    \begin{itemize}
    \tightlist
    \item
      Parents \(Pa(Y) = \{Age, BMI, DrugDose\}\)
    \item
      \(f\) is linear:
      \(Y = \beta_0 + \beta_1 Age + \beta_2 BMI + \beta_3 DrugDose + U_Y\)
    \end{itemize}
  \item
    If you have data on \(Age\), \(BMI\), and \(DrugDose\), fit the
    regression and get \(\hat{f}(Pa(Y))\)
  \item
    Then the residual \(Y - \hat{f}(Pa(Y))\) is your estimate of the
    noise \(U_Y\)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Epistemic Uncertainty: Model Misspecification}
\protect\phantomsection\label{epistemic-uncertainty-model-misspecification}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Epistemic uncertainty}} stems from \emph{lack
  of knowledge}

  \begin{itemize}
  \tightlist
  \item
    Also called \emph{systematic} or \emph{knowledge} uncertainty
  \item
    Reducible in principle by collecting more/better data or improving
    the model
  \item
    Captured by the posterior \(\Pr(\theta | \mathcal{D})\)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Sources}}:

  \begin{itemize}
  \tightlist
  \item
    Limited or biased data
  \item
    Wrong functional form
  \item
    Missing variables or wrong causal graph
  \item
    Confounders not adjusted for
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}:

  \begin{itemize}
  \tightlist
  \item
    A small clinical trial with \(n = 40\) patients
  \item
    Wide posterior on treatment effect \(\theta\)
  \item
    Gathering more data shrinks this posterior
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Variance Decomposition of Uncertainty}
\protect\phantomsection\label{variance-decomposition-of-uncertainty}
\begin{itemize}
\item
  A Bayesian predictive distribution decomposes as: \[
  \Pr(Y | do(a), \mathcal{D})
    = \int \Pr(Y | do(a), \theta) \, \Pr(\theta | \mathcal{D}) \, d\theta
  \]
\item
  \textbf{\textcolor{red}{Variance decomposition}}: \[
  \VV[Y | do(a), \mathcal{D}]
    = \underbrace{\EE_\theta[\VV[Y | a, \theta]]}_{\text{aleatoric}}
      + \underbrace{\VV_\theta[\EE[Y | a, \theta]]}_{\text{epistemic}}
  \]
\item
  \textbf{\textcolor{darkgreen}{Example}}: deep ensemble for sales
  forecasting

  \begin{itemize}
  \tightlist
  \item
    Within-model variance \(\approx\) aleatoric
  \item
    Between-model variance \(\approx\) epistemic
  \item
    Report both to the decision-maker
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Warning}}: most off-the-shelf ML models
  report only a point prediction, neither kind of uncertainty is visible
\end{itemize}
\end{frame}

\begin{frame}{Communicating Uncertainty to Stakeholders}
\protect\phantomsection\label{communicating-uncertainty-to-stakeholders}
\begin{itemize}
\item
  Stakeholders often want a single number

  \begin{itemize}
  \tightlist
  \item
    \emph{``How long does it take to do X?''}
  \item
    \emph{``How much does it cost''}
  \end{itemize}
\item
  Giving one is a disservice
\item
  \textbf{\textcolor{red}{Best practices}}:

  \begin{itemize}
  \tightlist
  \item
    Report intervals (90\% posterior credible intervals) alongside point
    estimates
  \item
    Distinguish aleatoric from epistemic in words and visuals
  \item
    Frame probabilities in natural frequencies: \emph{``about 2 in
    100''}
  \item
    Use \emph{visual aids}: fan charts, violin plots, calibration plots
  \item
    Link uncertainty to \emph{actions}:

    \begin{itemize}
    \tightlist
    \item
      E.g., \emph{``if we get 200 more samples, the interval shrinks
      from \$5M to \$1M''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Anti-patterns}}:

  \begin{itemize}
  \tightlist
  \item
    Reporting only point predictions
  \item
    Combining aleatoric and epistemic into one opaque bar
  \item
    Hiding priors in appendices
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Communicating Uncertainty: A Worked Example}
\protect\phantomsection\label{communicating-uncertainty-a-worked-example}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Setting}}: pricing team evaluating a new fee
  structure
\item
  \textbf{\textcolor{darkgreen}{Bad}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{The model predicts \$2.3M in additional revenue}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Good}}:

  \begin{itemize}
  \tightlist
  \item
    \_Our best estimate is \$2.3M
  \item
    \emph{90\% of the uncertainty sits between \$0.8M and \$3.7M}
  \item
    \emph{Of this, about \$1.5M of the width is intrinsic customer
    variability (aleatoric), a pilot won't narrow that much}
  \item
    \emph{The remaining \$1.4M is parameter uncertainty (epistemic) and
    would shrink by roughly 60\% if we ran the proposed 4-week pilot''}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Decision impact}}: the pilot discussion now
  has a clear expected value, not a gut feeling
\end{itemize}
\end{frame}

\begin{frame}{Summary: Decision-Making with Causal Models}
\protect\phantomsection\label{summary-decision-making-with-causal-models}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Prediction alone is insufficient}} for
  choosing actions

  \begin{itemize}
  \tightlist
  \item
    E.g., Simpson's paradox, confounding, and policy reversal
  \item
    Causal reasoning is needed
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Expected utility under interventional distributions}}
  is the right target \[
  a^*
    = \arg\max_a
      \EE_{Y \sim \Pr(Y | do(a))}[U(Y)]
  \]
\item
  \textbf{\textcolor{darkgreen}{Influence diagrams}} extend causal DAGs
  with decisions and utilities, giving a graphical, solvable decision
  problem
\item
  \textbf{\textcolor{teal}{Sequential decisions}} need exploration

  \begin{itemize}
  \tightlist
  \item
    VOI tells us \emph{when} to gather information
  \item
    Thompson sampling and Bayesian optimization tell us \emph{how}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Uncertainty}} decomposes into aleatoric
  (irreducible) and epistemic (reducible)

  \begin{itemize}
  \tightlist
  \item
    Both must be modeled and communicated
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key takeaway}}: causal reasoning, decision
  theory, and Bayesian inference turn predictions into \emph{robust}
  actions
\end{itemize}
\end{frame}
