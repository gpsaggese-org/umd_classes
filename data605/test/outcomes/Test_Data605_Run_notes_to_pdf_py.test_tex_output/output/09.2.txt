    \end{itemize}
  \item
    Record results as ``lineage''

    \begin{itemize}
    \tightlist
    \item
      The sequence of stages is rearranged and optimized without
      changing the results
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Actions}}

  \begin{itemize}
  \tightlist
  \item
    Trigger computation evaluation
  \item
    E.g.,
    \textcolor{blue}{\texttt{show(), take(), count(), collect(), save()}}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Spark Transformations: 1 / 3}
\protect\phantomsection\label{spark-transformations-1-3}
\begin{itemize}
\tightlist
\item
  \textcolor{blue}{\texttt{map(func)}}

  \begin{itemize}
  \tightlist
  \item
    Return a new RDD by applying \textcolor{blue}{\texttt{func()}} to
    each element
  \end{itemize}

  \textcolor{blue}{\texttt{flatmap(func)}}

  \begin{itemize}
  \tightlist

  \item
    \textcolor{blue}{\texttt{func()}} returns a sequence
  \end{itemize}
\item
  \textcolor{blue}{\texttt{filter(func)}}

  \textcolor{blue}{\texttt{union(otherDataset)}}

  \begin{itemize}
  \tightlist
  \item
    Return a new RDD with the union of elements in the source dataset
    and the argument
  \end{itemize}
\item
  \textcolor{blue}{\texttt{intersection(otherDataset)}}

  \begin{itemize}
  \tightlist
  \item
    Return a new RDD with the intersection of elements in the source
    dataset and the argument
  \end{itemize}
\end{itemize}

\begin{frame}{Spark Transformations: 2 / 3}
\protect\phantomsection\label{spark-transformations-2-3}
\begin{itemize}
\tightlist
\item
  \textcolor{blue}{\texttt{join(otherDataset, [numTasks])}}

  \begin{itemize}
  \tightlist
  \item
    On RDDs \textcolor{blue}{\texttt{(K, V)}} and
    \textcolor{blue}{\texttt{(K, W)}}, return dataset of
    \textcolor{blue}{\texttt{(K, (V, W))}} pairs for each key
  \item
    Support outer joins: \textcolor{blue}{\texttt{leftOuterJoin}},
    \textcolor{blue}{\texttt{rightOuterJoin}},
    \textcolor{blue}{\texttt{fullOuterJoin}}
  \end{itemize}
\item
  \textcolor{blue}{\texttt{groupByKey([numPartitions])}}

  \begin{itemize}
  \tightlist
  \item
    On RDD of \textcolor{blue}{\texttt{(K, V)}} pairs, returns
    \textcolor{blue}{\texttt{(K, Iterable<V>)}} pairs
  \item
    For aggregation (e.g., sum, average), use
    \textcolor{blue}{\texttt{reduceByKey}} for better performance

    \begin{itemize}
    \tightlist
    \item
      Process data in place instead of iterators
    \end{itemize}
  \item
    Output parallelism depends on parent RDD partitions

    \begin{itemize}
    \tightlist
    \item
      Use \textcolor{blue}{\texttt{numPartitions}} to set tasks
    \end{itemize}
  \end{itemize}
\item
  \textcolor{blue}{\texttt{sortByKey([ascending], [numPartitions])}}

  \begin{itemize}
  \tightlist
  \item
    Returns \textcolor{blue}{\texttt{(K, V)}} pairs sorted by keys in
    ascending or descending order
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Spark: RDD Persistence}
\protect\phantomsection\label{spark-rdd-persistence}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Users explicitly cache an RDD}}

  \begin{itemize}
  \tightlist
  \item
    I.e., \textcolor{blue}{\texttt{persist()}},
    \textcolor{blue}{\texttt{unpersist()}}
  \item
    Cache if RDD is expensive to compute

    \begin{itemize}
    \tightlist
    \item
      E.g., filtering large data
    \end{itemize}
  \item
    When you persist an RDD, each node:

    \begin{itemize}
    \tightlist
    \item
      Stores partitions of the RDD (in memory or on disk)
    \item
      Reuses cached partitions on derived datasets
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_9_1/lec_9_1_slide_25_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Cache}}

  \begin{itemize}
  \tightlist
  \item
    Enhances speed of future actions (often by \textgreater10x)
  \item
    Managed by Spark using LRU policy + garbage collector
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Users can choose the storage level}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{MEMORY\_ONLY}} (default)
  \item
    \textcolor{blue}{\texttt{DISK\_ONLY}} (e.g., Python Pickle)
  \item
    \textcolor{blue}{\texttt{MEMORY\_AND\_DISK}}

    \begin{itemize}
    \tightlist
    \item
      If an RDD doesn't fit in memory, store it on disk
    \end{itemize}
  \item
    Caching on disk can be more costly than not caching
  \item
    Caching everything is often a bad idea
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Spark: RDD Persistence and Fault-tolerance}
\protect\phantomsection\label{spark-rdd-persistence-and-fault-tolerance}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\item
  Spark unifies persistence and fault-tolerance through RDD lineage
\item
  \textbf{\textcolor{red}{Caching and Persistence}}

  \begin{itemize}
  \tightlist
  \item
    Store RDDs in memory or disk to avoid repeated computation
  \item
    Useful for iterative algorithms and interactive queries
  \item
    E.g., \textcolor{blue}{\texttt{cache()}} (memory only) or
    \textcolor{blue}{\texttt{persist()}} (custom storage level)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_9_1/lec_9_1_slide_26_image_1.png}}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Fault-Tolerance Mechanism}}

  \begin{itemize}
  \tightlist
  \item
    RDDs are immutable and record lineage of transformations
  \item
    If a partition is lost, Spark reconstructs it using lineage
  \item
    No need to checkpoint unless lineage is too long
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Persistence is Fault-Tolerant}}

  \begin{itemize}

  \item

  \item

  \end{itemize}

  \textbf{\textcolor{blue}{Data shuffle is expensive}} because of:

  \begin{itemize}
  \tightlist
  \item
    Data serialization (pickle)
  \item
    Disk I/O (saving to disk)
  \item
    Network I/O (copying across Executors)
  \item
    Deserialization and memory allocation
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Spark schedules general task graphs}}

  \begin{itemize}
  \tightlist
  \item
    Automatic function pipelining
  \item
    Data locality aware
  \item
    Partitioning aware to avoid shuffles
  \end{itemize}
\end{itemize}
