\begin{frame}{DB Taxonomy}
\protect\phantomsection\label{db-taxonomy}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{At least five DB genres}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Relational} (e.g., PostgreSQL)
  \item
    \emph{Key-value} (e.g., Redis)
  \item
    \emph{Document} (e.g., MongoDB)
  \item
    \emph{Columnar} (e.g., Apache Parquet)
  \item
    \emph{Graph} (e.g., Neo4j)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Criteria to differentiate DBs}}

  \begin{itemize}
  \tightlist
  \item
    Data model
  \item
    Trade-off with CAP theorem
  \item
    Querying capability
  \item
    Replication scheme
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Relational DB}
\protect\phantomsection\label{relational-db}
\begin{itemize}
\item
  E.g., \emph{Postgres}, MySQL, Oracle, SQLite
\item
  \textbf{\textcolor{red}{Data model}}

  \begin{itemize}
  \tightlist
  \item
    Set-theory, relational algebra
  \item
    Data as tables with rows and columns
  \item
    Many attribute types (e.g., numeric, strings, dates, arrays, blobs)
  \item
    Strictly enforced attribute types
  \item
    SQL query language
  \item
    ACID compliance
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Good for}}

  \begin{itemize}
  \tightlist
  \item
    Known data layout, unknown access pattern
  \item
    Schema complexity for query flexibility
  \item
    Regular data
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Not so good for}}

  \begin{itemize}
  \tightlist
  \item
    Hierarchical data (not easily represented as rows in tables)
  \item
    Variable/heterogeneous data (record-to-record variation)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Key-Value Store}
\protect\phantomsection\label{key-value-store}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\item
  E.g., Redis, DynamoDB, \emph{Git}, AWS S3, filesystem
\item
  \textbf{\textcolor{red}{Data model}}

  \begin{itemize}
  \tightlist
  \item
    Map keys (e.g., strings) to complex values (e.g., binary blob)
  \item
    Support get, put, delete operations on a primary key
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Cache data
  \item
    Store users' session data in web applications
  \item
    Store shopping carts in e-commerce applications
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Good for}}

  \begin{itemize}
  \tightlist
  \item
    Unrelated data (e.g., no joins)
  \item
    Fast lookups
  \item
    Easy horizontal scaling using partitioning
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Not so good for}}

  \begin{itemize}
  \tightlist
  \item
    Data queries
  \item
    Lacking secondary indexes and scanning
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Document Store}
\protect\phantomsection\label{document-store}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\item
  E.g., \emph{MongoDB}, \emph{CouchBase}
\item
  \textbf{\textcolor{red}{Data model}}

  \begin{itemize}
  \tightlist
  \item
    Key-value with document as value (nested dict)
  \item
    Unique ID for each document (e.g., hash)
  \item
    Any number of fields per document, including nested

    \begin{itemize}
    \tightlist
    \item
      E.g., JSON, XML, dict data
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Semi-structured data
  \end{itemize}
\end{itemize}
\end{column}

  \textbf{\textcolor{blue}{Good for}}

\begin{frame}{Columnar Store}
\protect\phantomsection\label{columnar-store}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\item
  E.g., \emph{HBase}, Cassandra, \emph{Parquet}
\item
  \textbf{\textcolor{red}{Data model}}

  \begin{itemize}
  \tightlist
  \item
    Store data by columns, not rows
  \item
    Similar to key-value and relational DBs

    \begin{itemize}
    \tightlist
    \item
      Use keys to query values
    \item
      Values are groups of columns
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Application}}

  \begin{itemize}
  \tightlist
  \item
    Store web pages
  \item
    Store time series data
  \item
    OLAP workloads
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Good for}}

  \begin{itemize}
  \tightlist
  \item
    Horizontal scalability
  \item
    Enable compression and versioning
  \item
    Sparse tables without extra storage cost
  \item
    Inexpensive to add columns
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Not so good for}}

  \begin{itemize}
  \tightlist
  \item
    Designing schema based on query plans
  \item
    No native joins; applications handle joins
  \end{itemize}
\end{itemize}
\end{column}

\end{frame}

\begin{frame}{Graph DB}
\protect\phantomsection\label{graph-db}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\item
  E.g., \emph{Neo4j}, GraphX
\item
  \textbf{\textcolor{red}{Data model}}

  \begin{itemize}
  \tightlist
  \item
    Interconnected data: nodes, relationships
  \item
    Nodes and edges have properties (key-value pairs)
  \item
    Queries traverse nodes and relationships
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Social data
  \item
    Recommendation engines
  \item
    Geographical data
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Good for}}

  \begin{itemize}
  \tightlist
  \item
    Networked data, hard to model with a relational model
  \item
    Matches object-oriented (OO) systems
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Not so good for}}

  \begin{itemize}
  \tightlist
  \item
    Poor scalability, hard to partition graph across different nodes

    \begin{itemize}
    \tightlist
    \item
      Store graph in graph DB, relations in key-value store
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Taxonomy by CAP}
\protect\phantomsection\label{taxonomy-by-cap}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{CA (Consistent, Available) systems}}

  \begin{itemize}
  \tightlist
  \item
    Struggle with partitions, use replication
  \item
    Traditional RDBMSs (PostgreSQL, MySQL)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{CP (Consistent, Partition-Tolerant) systems}}

  \begin{itemize}
  \tightlist
  \item
    Struggle with availability, maintain consistency across partitions
  \item
    BigTable (column-oriented/tabular)
  \item
    HBase (column-oriented/tabular)
  \item
    Redis (key-value)
  \item
    Berkeley DB (key-value)
  \end{itemize}
\end{itemize}
\end{column}
