\begin{frame}{Data Models}
\protect\phantomsection\label{data-models}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Data modeling}}

  \begin{itemize}
  \tightlist
  \item
    Represents and captures structure and properties of real-world
    entities
  \item
    Abstraction: real-world \(\to\)
    \textbf{\textcolor{darkgreen}{representation}}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Data model}}

  \begin{itemize}
  \tightlist
  \item
    Describes how data is \emph{represented} (e.g., relational,
    key-value) and \emph{accessed} (e.g., insert operations, query)
  \item
    Schema in a DB describes a specific data collection using a data
    model
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Why need data model?}}

  \begin{itemize}
  \tightlist
  \item
    Know data structure to write general-purpose code
  \item
    Share data across programs, organizations, systems
  \item
    Integrate information from multiple sources
  \item
    Preprocess data for efficient access (e.g., building an index)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Multiple Layers of Data Modeling}
\protect\phantomsection\label{multiple-layers-of-data-modeling}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Physical layer}}

  \begin{itemize}
  \tightlist
  \item
    How is the data physically stored
  \item
    How to represent complex data structures (e.g., B-trees for
    indexing)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Logical layer}}

  \begin{itemize}
  \tightlist
  \item
    Entities
  \item
    Attributes
  \item
    Type of information stored
  \item
    Relationships among the above
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Views}}

  \begin{itemize}
  \tightlist
  \item
    Restrict information flow
  \item
    Security and/or ease-of-use
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Good Data Models}

  \begin{itemize}
  \tightlist
  \item
    Expressive

    \begin{itemize}
    \tightlist
    \item
      Capture real-world data
    \end{itemize}
  \item
    Easy to use
  \item
    Perform well
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Trade-off between characteristics}}

  \begin{itemize}
  \tightlist
  \item
    E.g., more powerful models

    \begin{itemize}
    \tightlist
    \item
      Represent more datasets
    \item
      Harder to use/query
    \item
      Less efficient (e.g., more memory, time)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Evolution of data models}} captures data
  structure

  \begin{itemize}
  \tightlist
  \item
    Structured data \(\to\) Relational DBs
  \item
    Semi-structured web data \(\to\) XML, JSON
  \item
    Unstructured data \(\to\) NoSQL DBs
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{A Brief History of Databases (1960s)}

  \textbf{\textcolor{red}{1960s, Hierarchical and Network Model}}

  \begin{itemize}
  \tightlist
  \item
    Connect records of different types
  \item
    Example: connect accounts with customers
  \item
    Network model aimed for generality and flexibility
  \end{itemize}

  IBM's IMS Hierarchical Database (1966)

  \begin{itemize}
  \tightlist
  \item
    Designed for Apollo space program
  \item
    Predates hard disks
  \item
    Used by over 95\% of top Fortune 1000 companies
  \item
    Processes 50 billion transactions daily, manages 15 million GBs of
    data
  \end{itemize}

  \begin{itemize}
  \tightlist
  \item
    Exposed too much internal data (structures/pointers)
  \item
    Leaky abstraction
  \end{itemize}
\end{itemize}
\end{frame}

    \end{itemize}

\begin{frame}{Object-Oriented}
\protect\phantomsection\label{object-oriented}
\begin{itemize}
\tightlist
\item
  OOP is a data model

  \begin{itemize}
  \tightlist
  \item
    Object behavior described through data (fields) and code (methods)
  \end{itemize}
\item
  \textbf{\textcolor{red}{Composition}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{has-a}} relationships
  \item
    E.g., \textcolor{blue}{\texttt{Employee}} class has an
    \textcolor{blue}{\texttt{Address}} class
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Inheritance}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{is-a}} relationships
  \item
    E.g., \textcolor{blue}{\texttt{Employee}} class derives from
    \textcolor{blue}{\texttt{Person}} class
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Polymorphism}}

  \begin{itemize}
  \tightlist
  \item
    Code executed depends on the class of the object
  \item
    One interface, many implementations
  \item
    E.g., \textcolor{blue}{\texttt{draw()}} method on a
    \textcolor{blue}{\texttt{Circle}} vs
    \textcolor{blue}{\texttt{Square}} object, both descending from
    \textcolor{blue}{\texttt{Shape}} class
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Encapsulation}}

  \begin{itemize}
  \tightlist
  \item
    E.g., private vs public fields/members
  \item
    Prevents external code from accessing inner workings of an object
  \end{itemize}
\end{itemize}
\end{frame}
