\begin{frame}{Software Testing}
\protect\phantomsection\label{software-testing}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Goal}}

  \begin{itemize}
  \tightlist
  \item
    Evaluate functionality, reliability, performance, and security of a
    product to ensure it meets requirements
  \item
    Software testing is critical in development
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Adages}}

  \begin{itemize}
  \tightlist
  \item
    \emph{``If it's not tested, it doesn't work''}
  \item
    \emph{``Debugging is 2x harder than writing code''}

    \begin{itemize}
    \tightlist
    \item
      Corollary: \emph{``If you do your best to write code, how can you
      debug it?''}
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Many different types of testing}}

  \begin{itemize}
  \tightlist
  \item
    What do you test?
  \item
    From what point of view?
  \item
    \ldots{}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{What Are You Testing?}
\protect\phantomsection\label{what-are-you-testing}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Unit testing}}

  \begin{itemize}
  \tightlist
  \item
    Test individual components for correct functionality in isolation
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Integration testing}}

  \begin{itemize}
  \tightlist
  \item
    Ensure components work together as expected
  \item
    Detect interface defects
  \end{itemize}
\item
  \textbf{\textcolor{blue}{System testing}}

  \begin{itemize}
  \tightlist
  \item
    Evaluate a fully integrated system's compliance with requirements
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How Are You Testing?}
\protect\phantomsection\label{how-are-you-testing}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Smoke/sanity testing}}

  \begin{itemize}
  \tightlist
  \item
    Quick check of functionalities to ensure main functions work
  \item
    E.g., decide if a new build is stable
  \item
    E.g., application doesn't crash on launch
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Regression testing}}

  \begin{itemize}
  \tightlist
  \item
    Ensure new changes don't affect existing functionality
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Acceptance testing}}

  \begin{itemize}
  \tightlist
  \item
    Final testing phase before release
  \item
    More common in waterfall than Agile
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Performance testing}}

  \begin{itemize}
  \tightlist
  \item
    Load, stress, and spike testing
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Security testing}}

  \begin{itemize}
  \tightlist
  \item
    Identify vulnerabilities, threats, and risks
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Usability testing}}

  \begin{itemize}
  \tightlist
  \item
    Assess ease of use for end-users
  \item
    E.g., uI/UX
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Compatibility testing}}

  \begin{itemize}
  \tightlist
  \item
    Check compatibility with browsers, database versions, OS, mobile
    devices
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{CI / CD}
\protect\phantomsection\label{ci-cd}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Continuous integration (CI)}}

  \begin{itemize}
  \tightlist
  \item
    Merge code changes into a central repository multiple times a day
  \item
    Automate build and test after each change
  \item
    Add code with unit tests
  \item
    \emph{Goal}: Detect and fix integration errors quickly
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Continuous deployment (CD)}}

  \begin{itemize}
  \tightlist
  \item
    Automatically deploy code changes to production

    \begin{itemize}
    \tightlist
    \item
      Without human intervention
    \item
      After build and test phases pass
    \end{itemize}
  \item
    \emph{Goal}: Deliver features, bug fixes, and updates continuously
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Examples}}

  \begin{itemize}
  \tightlist
  \item
    GitHub Actions, GitLab Workflows, AWS Code, Jenkins
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{RESTful API}

\begin{itemize}
\tightlist

  \textbf{\textcolor{red}{REST API}}

    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{GET}},
      \textcolor{blue}{\texttt{POST}}, \textcolor{blue}{\texttt{PUT}},
      \textcolor{blue}{\texttt{DELETE}}
    \end{itemize}
  \item
    Naming convention, link format
  \item
    Response (XML or JSON)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Stateless}}

  \begin{itemize}
  \tightlist
  \item
    Each request contains all necessary information
  \item
    No shared state
  \item
    Inspired by HTTP (modulo cookies)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Stages of Deployment}
\protect\phantomsection\label{stages-of-deployment}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Software is deployed through several environments}}

  \begin{itemize}
  \tightlist
  \item
    Each environment tests, validates, and prepares software for release
    to end user
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Development environment (Dev)}}

  \begin{itemize}
  \tightlist
  \item
    Individual for each developer or feature team
  \item
    \emph{Goal}: Developers write and initially test code
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Testing}}

  \begin{itemize}
  \tightlist
  \item
    Aka ``Quality Assurance (QA)''
  \item
    Mirrors production environment to perform under similar conditions
  \item
    \emph{Goal}: Systematic testing to uncover defects and ensure
    quality
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Pre-prod}}

  \begin{itemize}
  \tightlist
  \item
    Aka staging
  \item
    Final testing phase before deployment to production
  \item
    Replica of production environment for final checks and stakeholder
    review
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Production (Prod)}}

  \begin{itemize}
  \tightlist
  \item
    Live environment where software is available to end users
  \item
    Optimized for security, performance, and scalability
  \item
    Focus on uptime, user experience, and data integrity
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Semantic Versioning}
\protect\phantomsection\label{semantic-versioning}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Semantic versioning}} conveys meaning about
  changes to software

  \begin{itemize}
  \tightlist
  \item
    Systematic approach
  \item
    Communicate to users potential impact of updating to a new version
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Increment version}}

  \begin{itemize}
  \tightlist
  \item
    Major Version \textcolor{blue}{\texttt{X.y.z}}

    \begin{itemize}
    \tightlist
    \item
      Incompatible API changes
    \item
      Significant updates that may break backward compatibility
    \end{itemize}
  \item
    Minor Version \textcolor{blue}{\texttt{x.Y.z}}

    \begin{itemize}
    \tightlist
    \item
      Backward-compatible enhancements
    \item
      Significant new features that don't break existing functionalities
    \end{itemize}
  \item
    Patch Version \textcolor{blue}{\texttt{x.y.Z}}

    \begin{itemize}
    \tightlist
    \item
      Backward-compatible bug fixes that address incorrect behavior
    \end{itemize}
  \item
    Pre-release Version:

    \begin{itemize}
    \tightlist
    \item
      Denote a pre-release version that might not be stable
    \item
      E.g., \textcolor{blue}{\texttt{1.0.0-alpha}},
      \textcolor{blue}{\texttt{1.0.0-beta}}
    \item
      Releases for testing and feedback, not for production use
    \end{itemize}
  \item
    Build Metadata

    \begin{itemize}
    \tightlist
    \item
      Optional metadata to denote build information or environment
      specifics
    \item
      E.g., \textcolor{blue}{\texttt{1.0.0+20210313120000}} or
      \textcolor{blue}{\texttt{1.0.0+f8a34b3228c}}
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Monolithic Architecture}
\protect\phantomsection\label{monolithic-architecture}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Monolith}} = all features in one deployable
  unit
\item
  \textbf{\textcolor{darkgreen}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Simplicity}: Simpler to develop, test, deploy, and scale as a
    single unit
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Tightly coupled components}: Components run in the same
    process, leading to scalability and resilience issues
  \item
    \emph{Technology stack uniformity}: Developed with a single
    technology stack, limiting flexibility
  \item
    \emph{Deployment complexity}: Updates require redeploying the entire
    application
  \item
    \emph{Single point of failure}: Issues in any module can affect the
    entire application
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Microservice Architecture}
\protect\phantomsection\label{microservice-architecture}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Microservices}} = many small services
  collaborating over the network
\item
  \textbf{\textcolor{darkgreen}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Complex deployment
  \item
    Requires tooling
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_7_1/lec_7_1_slide_50_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Pros}}:

  \begin{itemize}
  \tightlist
  \item
    \emph{Modularity}: Small, independently deployable services with
    specific business functionality
  \item
    \emph{Scalability}: Scale services independently for efficient
    resource use based on demand
  \item
    \emph{Technology diversity}: Use the best technology stack for each
    service's functionality
  \item
    \emph{Deployment flexibility}: Supports continuous delivery and
    deployment for faster updates
  \item
    \emph{Resilience}: Isolate and address faults; one service failure
    doesn't affect the entire system
  \end{itemize}
\end{itemize}
\end{frame}
