\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Advantages of cloud computing}}

  \begin{itemize}
  \tightlist
  \item
    Device agnostic: seamless computation across devices
  \item
    On demand
  \item
    Efficiency / scalability

    \begin{itemize}
    \tightlist
    \item
      Programming frameworks (e.g., Hadoop, Spark, Dask) enable

    \end{itemize}
  \item
    Reliability
  \item
    Cost: ``pay-as-you-go'' for resources

    \begin{itemize}
    \tightlist
    \item
      Cheaper than building own systems
    \item
      Computing as a commodity (like electricity)
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{frame}{Buying Infrastructure}
\protect\phantomsection\label{buying-infrastructure}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{To buy or to rent}} computers and
  infrastructure?

  \begin{itemize}
  \tightlist
  \item
    Trade-off between time and money
  \item
    Very difficult to decide objectively
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Buying / building infrastructure}}

  \begin{itemize}
  \tightlist
  \item
    Requires time and capital investments (Capex)

    \begin{itemize}
    \tightlist
    \item
      Especially at the beginning without revenues
    \item
      Smooth cash flow (constant \$/mo) is better than a lumpy one (big
      one-time purchase)
    \end{itemize}
  \item
    Buy hardware (e.g., computers, storage, network)

    \begin{itemize}
    \tightlist
    \item
      Estimate current hardware size
    \item
      Difficult to estimate future demands
    \end{itemize}
  \item
    Update obsolete hardware
  \item
    Cost of owning hardware (Opex)

    \begin{itemize}
    \tightlist
    \item
      Data center, electricity, cooling, handling faults
    \end{itemize}
  \item
    Administering

    \begin{itemize}
    \tightlist
    \item
      Install, update, maintain software stack
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

  \end{itemize}

  \begin{itemize}

  \item
    Hosting websites and applications
  \item

\begin{frame}{Software-as-a-Service (SaaS)}
\protect\phantomsection\label{software-as-a-service-saas}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Cloud provides the application}}

  \begin{itemize}
  \tightlist
  \item
    No need to install on your machine
  \item
    Use directly from the cloud
  \item
    Examples include:

    \begin{itemize}
    \tightlist
    \item
      \emph{Dropbox}: Access and share files without local storage
    \item
      \emph{Salesforce}: Manage customer relationships online
    \item
      Any app running in a browser: Google Docs or Microsoft 365
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Benefits}}

  \begin{itemize}
  \tightlist
  \item
    Accessibility from any device with internet connectivity
  \item
    Automatic updates and maintenance by the provider
  \item
    Scalability for growing user needs
  \item
    Cost-effectiveness by reducing physical hardware and software
    installations
  \end{itemize}
\end{itemize}

  \end{itemize}
