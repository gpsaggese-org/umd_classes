\begin{frame}{Data Wrangling and Data Cleaning}
\protect\phantomsection\label{data-wrangling-and-data-cleaning}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Data wrangling and cleaning}}

  \begin{itemize}
  \tightlist
  \item
    Turn \emph{messy, inconsistent, incomplete} data into a
    \emph{coherent} dataset
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Most effort}} in data science is spent
  here, not on modeling

  \begin{itemize}
  \tightlist
  \item
    It is iterative work
  \item
    Clean \(\to\) Model \(\to\) Clean \(\to\) \ldots{}
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_7_1/lec_7_1_slide_18_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Data wrangling}}

  \begin{itemize}
  \tightlist
  \item
    Transform data into a useful structure
  \item
    Reshape tables, merge sources, create new variables
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Data cleaning}}

  \begin{itemize}
  \tightlist
  \item
    Correct or remove incorrect, corrupt, or irrelevant data
  \item
    Handle missing values, outliers, inconsistent formats
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Often done together}}
\end{itemize}
\end{frame}

\begin{frame}{Typical Data Wrangling Workflow}
\protect\phantomsection\label{typical-data-wrangling-workflow}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Inspect data}}

  \begin{itemize}
  \tightlist
  \item
    Look at column names, data types
  \item
    Look at a few rows
  \item
    Basic summaries
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Diagnose problems}}

  \begin{itemize}
  \tightlist
  \item
    Missing values, unusual codes, inconsistent units, strange
    distributions
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Decide strategy}}, based on project

  \begin{itemize}
  \tightlist
  \item
    What to keep
  \item
    What to fix
  \item
    What to drop
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_7_1/lec_7_1_slide_19_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{teal}{Apply transformations}}

  \begin{itemize}
  \tightlist
  \item
    Filter, join, reshape, recode, aggregate
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Validate results}}

  \begin{itemize}
  \tightlist
  \item
    Check that changes make sense and that key statistics are reasonable
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Data Extraction}
\protect\phantomsection\label{data-extraction}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Web scraping}}

  \begin{itemize}
  \tightlist
  \item
    Use APIs or scrape data explicitly
  \item
    Scraping is challenging

    \begin{itemize}
    \tightlist
    \item
      Fragile (page changes can break your parser)
    \item
      Throttling/rate limits
    \item
      Cat-and-mouse game with websites
    \end{itemize}
  \item
    Set up pipelines for periodic scraping
  \item
    Tools for automated scraping

    \begin{itemize}
    \tightlist
    \item
      E.g., \textcolor{blue}{\texttt{import.io}},
      \textcolor{blue}{\texttt{portia}}, ..
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Various sources of data}}

\begin{frame}{Outlier Detection}
\protect\phantomsection\label{outlier-detection}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{For Gaussian data}}

  \begin{itemize}
  \tightlist
  \item
    Data points 1.5x MAD from median
  \item
    Plot histogram to confirm
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{For non-Gaussian data}}

  \begin{itemize}
  \tightlist
  \item
    Estimate generating distribution (parametric or not)
  \item
    Distance-based: find points with few neighbors
  \item
    Density-based:

    \begin{itemize}
    \tightlist
    \item
      Define \emph{density} as the average distance to \emph{k} nearest
      neighbors
    \item
      Use relative density to identify outliers
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{column}{0.5\linewidth}

  \item
    Techniques break down as data dimensionality increases
  \item
    Need \(O(e^n)\) points with \(n\) dimensions to estimate
  \item
    Project data into lower-dimensional space to find outliers

\begin{frame}{Multivariate Outliers}
\protect\phantomsection\label{multivariate-outliers}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Mean/covariance not robust}}

  \begin{itemize}
  \tightlist
  \item
    Sensitive to outliers
  \item
    Use robust statistics
  \end{itemize}
\item
  Assume \textbf{\textcolor{blue}{multivariate Gaussian data}}

  \begin{itemize}
  \tightlist
  \item
    Defined by \emph{mean} \(\vmu\) and \emph{covariance matrix}
    \(\mSigma\)
  \item
    \emph{Iterative approach}

    \begin{itemize}
    \tightlist
    \item
      Mahalanobis distance:
      \(\sqrt{(\vx - \vmu)^T\mSigma^{-1}(\vx - \vmu)}\)
    \item
      Measures distance of point \(x\) from distribution
    \item
      Identify outliers: points too far by Mahalanobis distance
    \item
      Remove outliers
    \item
      Recompute mean and covariance
    \end{itemize}
  \end{itemize}
\item
  Data volume often large

  \begin{itemize}
  \tightlist
  \item
    Use approximation techniques
  \end{itemize}
\item
  Try different techniques based on data
\end{itemize}
\end{column}
