\begin{frame}{SQL Overview}
\protect\phantomsection\label{sql-overview}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Relational algebra}}: mathematical language to
  manipulate \emph{relations}
\item
  \textbf{\textcolor{darkgreen}{SQL}}: language to describe and
  transform data in a relational DB

  \begin{itemize}
  \tightlist
  \item
    Originally Sequel
  \item
    Changed to Structured Query Language
  \end{itemize}
\item
  \textbf{\textcolor{blue}{SQL statements grouped by goal}}

  \begin{itemize}
  \tightlist
  \item
    Data definition language (DDL)

    \begin{itemize}
    \tightlist
    \item
      Define schema (tables, attributes, indices)
    \item
      Specify integrity constraints (primary key, foreign key, not null)
    \end{itemize}
  \item
    Data modification language (DML)

    \begin{itemize}
    \tightlist
    \item
      Modify data in tables
    \item
      Insert, Update, Delete
    \end{itemize}
  \item
    Query data (DQL)
  \item
    Control transactions

    \begin{itemize}
    \tightlist
    \item
      Specify beginning and end, control isolation level
    \end{itemize}
  \item
    Define views
  \item
    Authorization

    \begin{itemize}
    \tightlist
    \item
      Specify access and security constraints
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Select}
\protect\phantomsection\label{select}
\begin{Shaded}
\begin{Highlighting}[]
\KeywordTok{SELECT}\NormalTok{ A\_1, A\_2, }\OperatorTok{..}\NormalTok{., A\_n}
    \KeywordTok{FROM}\NormalTok{ r\_1, r\_2, }\OperatorTok{..}\NormalTok{., r\_m}
    \KeywordTok{WHERE}\NormalTok{ P;}
\end{Highlighting}
\end{Shaded}

\begin{itemize}
\tightlist
\item
  \textcolor{blue}{\texttt{SELECT}}: select the attributes to list
  (i.e., projection)
\item
  \textcolor{blue}{\texttt{FROM}}: list of tables to be accessed

  \begin{itemize}
  \tightlist
  \item
    Define a Cartesian product of the tables
  \item
    The query is going to be optimized to avoid to enumerate tuples that
    will be eliminated
  \end{itemize}
\item
  \textcolor{blue}{\texttt{WHERE}}: predicate involving attributes of
  the relations in the \textcolor{blue}{\texttt{FROM}} clause (i.e.,
  selection)
\item
  In \textcolor{blue}{\texttt{SELECT}} or
  \textcolor{blue}{\texttt{WHERE}} clauses, might need to use the table
  names as prefix to qualify the attribute name

  \begin{itemize}
  \tightlist
  \item
    E.g., \textcolor{blue}{\texttt{instructor.ID}} vs
    \textcolor{blue}{\texttt{teaches.ID}}
  \end{itemize}
\item
  A \textcolor{blue}{\texttt{SELECT}} statement can be expressed in
  terms of relational algebra

  \begin{itemize}
  \tightlist
  \item
    Cartesian product \(\to\) selection \(\to\) projection
  \item
    Difference: SQL allows duplicate values, relational algebra works
    with mathematical sets
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Null Values}
\protect\phantomsection\label{null-values}
\begin{itemize}
\tightlist
\item
  An arithmetic operation with \textcolor{blue}{\texttt{NULL}} yields
  \textcolor{blue}{\texttt{NULL}}
\item
  Comparison with NULL

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{1 < NULL}}
  \item
    \textcolor{blue}{\texttt{NOT(1 < NULL)}}
  \item
    SQL yields \textcolor{blue}{\texttt{UNKNOWN}} when comparing with
    \textcolor{blue}{\texttt{NULL}} value
  \item
    There are 3 logical values: \textcolor{blue}{\texttt{True}},
    \textcolor{blue}{\texttt{False}}, \textcolor{blue}{\texttt{Unknown}}
  \end{itemize}
\item
  Boolean operators

  \begin{itemize}
  \tightlist
  \item
    Can be extended according to common sense, e.g.,
  \item
    \textcolor{blue}{\texttt{True AND UNKNOWN = UNKNOWN}}
  \item
    \textcolor{blue}{\texttt{False AND Unknown = False}}
  \end{itemize}
\item
  In a \textcolor{blue}{\texttt{WHERE}} clause, if the result is
  \textcolor{blue}{\texttt{UNKNOWN}} it's not included
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Having}
\protect\phantomsection\label{having}
\begin{itemize}
\item
  State a condition that applies to groups instead of tuples (like
  WHERE)
\item
  Any attribute in the HAVING clause must appear in the GROUP BY clause
\item
  E.g., find departments with avg salary of instructors \textgreater{}
  42k

\begin{Shaded}
\begin{Highlighting}[]
    \KeywordTok{SELECT}\NormalTok{ dept\_name, }\FunctionTok{AVG}\NormalTok{(salary) }\KeywordTok{AS}\NormalTok{ avg\_salary}
        \KeywordTok{FROM}\NormalTok{ instructor}
        \KeywordTok{GROUP} \KeywordTok{BY}\NormalTok{ dept\_name}
        \KeywordTok{HAVING} \FunctionTok{AVG}\NormalTok{(salary) }\OperatorTok{\textgreater{}} \DecValTok{42000}\NormalTok{;}
\end{Highlighting}
\end{Shaded}
\end{itemize}

\vspace{0.5cm}

\begin{itemize}
\tightlist
\item
  How does it work

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{FROM}} is evaluated to create a relation
  \item
    (optional) \textcolor{blue}{\texttt{WHERE}} is used to filter
  \item
    \textcolor{blue}{\texttt{GROUP BY}} collects tuples into groups
  \item
    (optional) \textcolor{blue}{\texttt{HAVING}} is applied to each
    group and groups are filtered
  \item
    \textcolor{blue}{\texttt{SELECT}} generates tuples of the results,
    applying aggregate functions to get a single result for each group
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Multi-Table Queries}
\protect\phantomsection\label{multi-table-queries-1}
\begin{itemize}
\item
  Consider the query:

\begin{Shaded}
\begin{Highlighting}[]
    \KeywordTok{SELECT}\NormalTok{ title, }\DataTypeTok{year}\NormalTok{, producerC\#, }\FunctionTok{count}\NormalTok{(starName)}
        \KeywordTok{FROM}\NormalTok{ movies, starsIn}
        \KeywordTok{WHERE}\NormalTok{ title }\OperatorTok{=}\NormalTok{ starsIn.movieTitle }\KeywordTok{AND}
              \DataTypeTok{year} \OperatorTok{=}\NormalTok{ starsIn.movieYear}
        \KeywordTok{GROUP} \KeywordTok{BY}\NormalTok{ title, }\DataTypeTok{year}\NormalTok{, producerC\#}
\end{Highlighting}
\end{Shaded}

  \begin{itemize}
  \tightlist
  \item
    What about movies with no stars?
  \item
    Need to use \textbf{\textcolor{red}{outer joins}}
  \end{itemize}

\begin{Shaded}
\begin{Highlighting}[]
    \KeywordTok{SELECT}\NormalTok{ title, }\DataTypeTok{year}\NormalTok{, producerC\#, }\FunctionTok{count}\NormalTok{(starName)}
        \KeywordTok{FROM}\NormalTok{ movies }\KeywordTok{LEFT} \KeywordTok{OUTER} \KeywordTok{JOIN}\NormalTok{ starsIn}
        \KeywordTok{ON}\NormalTok{ title }\OperatorTok{=}\NormalTok{ starsIn.movieTitle }\KeywordTok{AND} \DataTypeTok{year} \OperatorTok{=}\NormalTok{ starsIn.movieYear}
        \KeywordTok{GROUP} \KeywordTok{BY}\NormalTok{ title, }\DataTypeTok{year}\NormalTok{, producerC\#}
\end{Highlighting}
\end{Shaded}

  \begin{itemize}
  \tightlist
  \item
    All tuples from `movies' that have no matches in starsIn are
    included with NULLs
  \item
    So if a tuple (m1, 1990) has no match in starsIn, we get (m1, 1990,
    NULL) in the result
  \item
    The count(starName) works correctly then
  \item
    Note: count(*) would not work correctly (NULLs can have unintuitive
    behavior)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Other SQL Constructs}
\protect\phantomsection\label{other-sql-constructs-1}
\begin{itemize}
\tightlist
\item
  NULLs

  \begin{itemize}
  \tightlist
  \item
    Value of any attribute can be NULL
  \item
    Because: value is unknown, or it is not applicable, or hidden, etc.
  \item
    Can lead to counterintuitive behavior
  \item
    For example, the following query does not return movies where length
    = NULL
  \end{itemize}

\begin{Shaded}
\begin{Highlighting}[]
      \KeywordTok{SELECT} \OperatorTok{*} \KeywordTok{FROM}\NormalTok{ movies}
      \KeywordTok{WHERE} \FunctionTok{length} \OperatorTok{\textgreater{}=} \DecValTok{120} \KeywordTok{OR} \FunctionTok{length} \OperatorTok{\textless{}=} \DecValTok{120}
\end{Highlighting}
\end{Shaded}

  \begin{itemize}
  \tightlist
  \item
    Aggregate operations can be especially tricky
  \end{itemize}
\item
  Transactions

  \begin{itemize}
  \tightlist
  \item
    A transaction is a sequence of queries and update statements
    executed as a single unit
  \item
    For example, transferring money from one account to another

    \begin{itemize}
    \tightlist
    \item
      Both the \emph{deduction} from one account and \emph{credit} to
      the other account should happen, or neither should
    \end{itemize}
  \end{itemize}
\item
  Triggers

  \begin{itemize}
  \tightlist
  \item
    A trigger is a statement that is executed automatically by the
    system as a side effect of a modification to the database
  \end{itemize}
\end{itemize}
\end{frame}

  \begin{itemize}
  \tightlist

  \end{itemize}
