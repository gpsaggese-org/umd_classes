\begin{frame}{(Apache) HBase}
\protect\phantomsection\label{apache-hbase}
\begin{columns}[T]
\begin{column}{0.8\linewidth}
\begin{itemize}
\tightlist
\item
  HBase = \textbf{\textcolor{red}{H}}adoop Data@Base@

  \begin{itemize}
  \tightlist
  \item
    Supports large tables on commodity hardware clusters
  \item
    Column-oriented DB
  \item
    Part of Apache Hadoop ecosystem
  \item
    Uses Hadoop filesystem (HDFS)

    \begin{itemize}
    \tightlist
    \item
      HDFS modeled after Google File System (GFS)
    \item
      HBase based on Google BigTable
    \item
      Google BigTable runs on GFS, HBase runs on HDFS
    \end{itemize}
  \item
    Used by Google, Airbnb, eBay
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{When to use HBase}}

  \begin{itemize}
  \tightlist
  \item
    For large DBs (e.g., many 100 GBs or TBs)
  \item
    With at least 5 nodes in production
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist

\begin{frame}{From HDFS to HBase}
\protect\phantomsection\label{from-hdfs-to-hbase}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Different types of workloads for DB backends}}

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{orange}{OLTP}} (On-Line Transactional Processing)

    \begin{itemize}
    \tightlist
    \item
      Read and write individual data items in a large table
    \item
      E.g., update inventory and price as orders come in
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{OLAP}} (On-Line Analytical Processing)

    \begin{itemize}
    \tightlist
    \item
      Read large data amounts and process it
    \item
      E.g., analyze item purchases over time
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Hadoop FileSystem (HDFS) supports OLAP workloads}}

  \begin{itemize}
  \tightlist
  \item
    Provide a filesystem with large files
  \item
    Read data sequentially, end-to-end
  \item
    Rarely updated
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{HBase supports OLTP interactions}}

  \begin{itemize}
  \tightlist
  \item
    Built on HDFS
  \item
    Use additional storage and memory to organize tables
  \item
    Write tables back to HDFS as needed
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{HBase Data Model}
\protect\phantomsection\label{hbase-data-model}
\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Warning}}: HBase uses names similar to
  relational DB concepts, but with different meanings
\item
  A \textbf{\textcolor{orange}{database}} consists of multiple tables
\item
  Each \textbf{\textcolor{darkgreen}{table}} consists of multiple rows,
  sorted by row key
\item
  Each row contains a \emph{row key} and one or more column families
\item
  Each \textbf{\textcolor{teal}{column family}}

  \begin{itemize}
  \tightlist
  \item
    Contains multiple columns (family:column)
  \item
    Defined when the table is created
  \end{itemize}
\item
  A \textbf{\textcolor{cyan}{cell}}

  \begin{itemize}
  \tightlist
  \item
    Uniquely identified by (table, row, family:column)
  \item
    Contains metadata (e.g., timestamp) and an uninterpreted array of
    bytes (blob)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Versioning}}

  \begin{itemize}
  \tightlist
  \item
    New values don't overwrite old ones
  \item
    \textcolor{blue}{\texttt{put()}} and
    \textcolor{blue}{\texttt{get()}} allow specifying a timestamp
    (otherwise uses current time)
  \end{itemize}
\end{itemize}
\end{column}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{A row in HBase is like a mini-database}}

  \begin{itemize}
  \tightlist
  \item
    A cell has many values
  \item
    Data stored sparsely
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Rows in HBase are "deeper" than in relational DBs}}

  \begin{itemize}
  \tightlist
  \item
    Relational DBs: rows have many column values (fixed array with
    types)
  \item
    HBase: rows like a two-level nested dictionary with metadata (e.g.,
    timestamp)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Store versioned website data
  \item
    Store a wiki
  \end{itemize}
\end{itemize}

\begin{frame}{Column Family vs Column}
\protect\phantomsection\label{column-family-vs-column}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Adding a column}}

  \begin{itemize}
  \tightlist
  \item
    Cheap
  \item
    Done at run-time
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Adding a column family}}

  \begin{itemize}
  \tightlist
  \item
    Not at run-time
  \item
    Requires table copy (expensive)
  \item
    Indicates data storage method

    \begin{itemize}
    \tightlist
    \item
      Easy to add: map
    \item
      Hard to add: static array
    \item
      E.g., mongoDB document vs Relational DB column
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Why differentiate column families vs columns?}}

  \begin{itemize}
  \tightlist
  \item
    Why not store all row data in one column family?
  \item
    Each column family configured independently, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Compression
    \item
      Performance tuning
    \item
      Stored together in files
    \end{itemize}
  \item
    Designed for specific data types

    \begin{itemize}
    \tightlist
    \item
      E.g., timestamped web data for search engine
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Consistency Model}
\protect\phantomsection\label{consistency-model}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Atomicity}}

  \begin{itemize}
  \tightlist
  \item
    Update entire rows atomically or not at all
  \item
    Independent of column count
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Consistency}}

  \begin{itemize}
  \tightlist
  \item
    GET returns a complete row from the table's history

    \begin{itemize}
    \tightlist
    \item
      Weak/eventual consistency
    \item
      Check timestamp for certainty
    \end{itemize}
  \item
    SCAN

    \begin{itemize}
    \tightlist
    \item
      Includes all data written before scan
    \item
      May include updates since start
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Isolation}}

  \begin{itemize}
  \tightlist
  \item
    Concurrent vs sequential semantics
  \item
    Not guaranteed beyond a single row
  \item
    Row is the atom of information
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Durability}}

  \begin{itemize}
  \tightlist
  \item
    Successful writes are durable on disk
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Checking for Row or Column Existence}
\protect\phantomsection\label{checking-for-row-or-column-existence}
\begin{itemize}
\tightlist
\item
  HBase uses Bloom filters to check row or column existence

  \begin{itemize}
  \tightlist
  \item
    Acts like a cache for keys
  \item
    Track presence without querying
  \end{itemize}
\item
  \textbf{\textcolor{red}{Hashset complexity}}

  \begin{itemize}
  \tightlist
  \item
    Unbounded space for data storage
  \item
    No false positives
  \item
    O(1) average/amortized time
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Bloom filter implementation}}

  \begin{itemize}
  \tightlist
  \item
    Probabilistic hash set
  \item
    Array of bits initially set to 0
  \item
    Hash new data, set bits to 1
  \item
    To test data presence, compute hash, check bits

    \begin{itemize}
    \tightlist
    \item
      All bits 0: data not seen
    \item
      All bits 1: likely seen (false positive possible)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bloom filter complexity}}

  \begin{itemize}
  \tightlist
  \item
    Constant space usage
  \item
    False positives possible (no false negatives)
  \item
    O(1) time complexity
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Write-Ahead Log (WAL)}
\protect\phantomsection\label{write-ahead-log-wal}
\begin{itemize}
\item
  Technique used by DBs

  \begin{itemize}
  \tightlist
  \item
    Provide atomicity and durability
  \item
    Protect against node failures
  \item
    Equivalent to journaling in file systems
  \end{itemize}
\item
  HBase and Postgres use WAL
\item
  \textbf{\textcolor{red}{WAL mechanics}}
\item
  Updated state of tables:

  \begin{itemize}
  \tightlist
  \item
    Not written to disk immediately
  \item
    Buffered in memory
  \item
    Written to disk as checkpoints periodically
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Server crash during this period loses state
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Use append-only disk-resident data structure
  \item
    Log operations since last checkpoint in WAL
  \item
    Clear WAL when tables stored to disk
  \item
    Use WAL to recover state if server crashes
  \end{itemize}
\item
  Disable WAL during big import jobs to improve performance

  \begin{itemize}
  \tightlist
  \item
    Trade off disaster recovery protection for speed
  \end{itemize}
\end{itemize}
\end{frame}

  \textbf{\textcolor{red}{How to store the web on disk?}}
\item
  \textbf{\textcolor{darkgreen}{HBase is backed by HDFS}}

  \tightlist
  \item
    Store each table (e.g., Wikipedia) in one file
  \item
    ``One file'' means one gigantic file stored in HDFS

    \end{itemize}
  \end{itemize}

  Idea in several steps:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{blue}{Idea 1: Put entire table in one file}}

    \tightlist
    \item
      Overwrite file with any cell change
    \item
      Too slow
    \end{itemize}
  \item
    \textcolor{blue}{\texttt{Idea 2: One file + WAL}}

    \begin{itemize}
    \tightlist
    \item
      Better, but doesn't scale to large data
    \end{itemize}
  \item
    \textcolor{blue}{\texttt{Idea 3: One file per column family + WAL}}

    \begin{itemize}
    \tightlist
    \item
      Getting better!
    \end{itemize}
  \item
    \textcolor{blue}{\texttt{Idea 4: Partition table into regions by key}}

    \begin{itemize}
    \tightlist
    \item
      Region = chunk of rows {[}a, b)
    \item
      Regions never overlap
    \end{itemize}
  \end{itemize}
