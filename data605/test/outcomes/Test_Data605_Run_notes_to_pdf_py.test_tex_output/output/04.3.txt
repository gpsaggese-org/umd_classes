\begin{frame}[fragile]{Storage Characteristics}
\protect\phantomsection\label{storage-characteristics}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Storage media trade-offs}}:

  \begin{itemize}
  \tightlist
  \item
    Speed of access (e.g., 500-3,500 MB/sec)
  \item
    Cost per data unit (e.g., 50 USD/TB)
  \item
    Medium reliability
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Volatile vs non-volatile storage}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Volatile}: loses contents when power is switched off
  \item
    \emph{Non-volatile}: retains contents even after power is switched
    off
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Sequential vs random access}}

  \begin{itemize}
  \item
    \emph{Sequential}: read the data contiguously \small

\begin{verbatim}
SELECT * FROM employee
\end{verbatim}
  \item
    \emph{Random}: read the data from anywhere at any time \small

\begin{verbatim}
SELECT * FROM employee
WHERE name LIKE '__a__b'
\end{verbatim}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Storage Hierarchy (by Speed and Cost)}
\protect\phantomsection\label{storage-hierarchy-by-speed-and-cost}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\footnotesize

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Cache}}

  \begin{itemize}
  \tightlist
  \item
    Fastest, most costly
  \item
    \textasciitilde MBs on chip
  \item
    DB developers consider cache effects
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Main memory}}

  \begin{itemize}
  \tightlist
  \item
    Up to 100s of GBs
  \item
    Typically can't store entire DB
  \item
    Volatile
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_4_2/lec_4_2_slide_5_image_1.png}}
\end{column}
\end{columns}

\footnotesize

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Flash memory / SSDs}}

  \begin{itemize}
  \tightlist
  \item
    Less expensive than RAM, more than magnetic disk
  \item
    Non-volatile, random access
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Magnetic disk}}

  \begin{itemize}
  \tightlist
  \item
    Long-term online storage
  \item
    Non-volatile
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Optical disk (CD, Blu-ray)}}

  \begin{itemize}
  \tightlist
  \item
    Mainly read-only
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Magnetic tapes}}

  \begin{itemize}
  \tightlist
  \item
    Backup, archival data
  \item
    Stored long-term, e.g., for legal reasons
  \item
    Sequential-access
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{How Important Is Memory Hierarchy?}
\protect\phantomsection\label{how-important-is-memory-hierarchy}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Trade-offs have shifted}} over the last 10-15
  years
\item
  \textbf{\textcolor{darkgreen}{Innovations}}

  \begin{itemize}
  \tightlist
  \item
    Fast networks, SSDs, large memories
  \item
    Data volume is growing rapidly
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Observations}}

  \begin{itemize}
  \tightlist
  \item
    It is faster to access another computer's memory through a network

  \end{itemize}

  \textbf{\textcolor{violet}{Disk controller}}

  \begin{itemize}
  \tightlist
  \item
    Accepts commands to read/write a sector
  \item
    Operates arm/heads
  \item
    Remaps bad sectors to a different location
  \end{itemize}

\begin{frame}{Accessing Data Speed}

  \item
    Time to read a random sector

    It has 3 components

    \begin{itemize}
    \tightlist
    \item
      \emph{Seek time}: Time to seek to the track
      (\textasciitilde4-10ms)
    \item
      \emph{Rotational latency}: Waiting for the sector to get under the
      head (\textasciitilde4-11ms)
    \item
      \emph{Transfer time}: Time to transfer the data (Very low)
    \end{itemize}
  \item
    About 10ms per access

    \begin{itemize}
    \tightlist
    \item
      Randomly accessed blocks
    \item
      100 block transfers (100/sec x 4 KB/block = 400 KB/s)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Serial data transfer rates}}

  \begin{itemize}
  \tightlist
  \item
    Data transfer rate without seek
  \item
    30-50MB/s to 200MB/s
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Seeks are bad!}}
\end{itemize}

  \item

  \item
    Requires reading an entire ``page'' of data (typically 4KB)

    \begin{itemize}
    \tightlist
    \item
      Equivalent to a block in magnetic disks
    \end{itemize}
  \end{itemize}

\begin{frame}{RAID}
\protect\phantomsection\label{raid}
\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{RAID}} = Redundant Array of Independent Disks
\item
  \textbf{\textcolor{darkgreen}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Storage capacity is growing exponentially
  \item
    Data-storage needs are growing even faster
  \item
    There is a need for more disks
  \item
    Mean Time To Failure (MTTF) between disk failures is shrinking
    (e.g., days)

    \begin{itemize}
    \tightlist
    \item
      A single data copy leads to an unacceptable frequency of data loss
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Observations}}

  \begin{itemize}
  \tightlist
  \item
    Disks are cheap
  \item
    Failures are costly
  \item
    Use extra disks for reliability

    \begin{itemize}
    \tightlist
    \item
      Store data redundantly
    \item
      Data survives disk failure
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Goal}}

  \begin{itemize}
  \tightlist
  \item
    Present a logical view of a large, reliable disk from many
    unreliable disks
  \item
    Different RAID levels balance reliability and performance
  \end{itemize}
\end{itemize}

\begin{frame}{Error Correction Codes}
\protect\phantomsection\label{error-correction-codes}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  = a technique used for controlling errors in data transmission over
  unreliable communication channels
\item
  \textbf{\textcolor{red}{Idea}}:

  \begin{itemize}
  \tightlist
  \item
    the sender encodes the message in a redundant way
  \item
    the receiver can detect errors and correct (a limited number of)
    errors
  \end{itemize}
\item
  1940-1960s: Hamming, Reed-Solomon, Shannon, Viterbi
\item
  E.g., triple redundancy

  \begin{itemize}
  \tightlist
  \item
    Send the same bit 3 times, receiver does majority voting
  \item
    Detect and correct one bit errors
  \end{itemize}
\item
  E.g. parity bit

  \begin{itemize}
  \tightlist
  \item
    Add an extra bit representing the number of 1s
  \item
    Detect (but not correct) one bit errors
  \end{itemize}
\end{itemize}

\begin{frame}{Choosing a RAID Level}
\protect\phantomsection\label{choosing-a-raid-level}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\item
  Main choice between RAID 0, RAID 1, and RAID 5
\item
  \textbf{\textcolor{red}{RAID 0 (striping)}}

  \begin{itemize}
  \tightlist
  \item
    Better performance, no reliability
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{RAID 1 (mirroring)}}

  \begin{itemize}
  \tightlist
  \item
    Better performance and reliability
  \item
    High cost
  \item
    E.g., to write a single block

    \begin{itemize}
    \tightlist
    \item
      RAID 1: 2 block writes
    \item
      RAID 5: 2 block reads, 2 block writes
    \end{itemize}
  \item
    Preferred for high update rate, small data (e.g., log disks)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{RAID 5 (interleaved parity)}}

  \begin{itemize}
  \tightlist
  \item
    Lower storage cost
  \item
    Preferred for low update rate, large data (e.g., analytics)
  \end{itemize}
\end{itemize}
