\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Different DB types make different trade-offs}}

  \begin{itemize}
  \tightlist
  \item
    Different worldviews
  \item
    Schema vs schema-less
  \item
    Rich vs fast query ability
  \item
    Strong consistency (ACID), weak, eventual consistency
  \item
    APIs (SQL, JS, REST)
  \item
    Horizontal vs vertical scaling, sharding, replication
  \item
    Indexing vs no indexing
  \item
    Tuned for reads or writes, control over tuning
  \end{itemize}
\item

  \item

  \item

    (\textcolor{blue}{\texttt{List[Obj]}})
  \end{itemize}

  \textbf{\textcolor{darkgreen}{Within relational DB}}

  \tightlist

    \tightlist
    \item
      No schema concerns when writing data

  \item
    Cons

    \begin{itemize}

    \item

    \item

    \end{itemize}
  \end{itemize}
\end{itemize}

  \item

  \end{itemize}

\begin{frame}{CAP Theorem}
\protect\phantomsection\label{cap-theorem}
\begin{columns}[T]
\begin{column}{0.45\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{CAP theorem}}: Any distributed DB can have at
  most two of the following

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{Consistency}}:

    \begin{itemize}
    \tightlist
    \item
      All clients see the same data
    \item
      Writes are atomic; subsequent reads retrieve the new value
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Availability}}: Returns a value if at least
    one server is running
  \item
    \textbf{\textcolor{violet}{Partition tolerance}}: The system works
    even if communication is temporarily lost (network is partitioned)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.5\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_5_0/lec_5_0_slide_12_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  Originally a conjecture (Eric Brewer)

  \begin{itemize}
  \tightlist
  \item
    Proved formally (Gilbert, Lynch, 2002)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{CAP Corollary}
\protect\phantomsection\label{cap-corollary}
\begin{columns}[T]
\begin{column}{0.45\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{CAP Theorem}}: pick 2 among consistency,
  availability, partition tolerance
\item
  \textbf{\textcolor{darkgreen}{Network partitions}}

  \begin{itemize}
  \tightlist
  \item
    Cannot be prevented in large-scale distributed systems
  \item
    Can be reduced in probability using redundancy and fault tolerance
  \end{itemize}
\item
  You must sacrifice either:

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{blue}{Availability}}

    \begin{itemize}
    \tightlist
    \item
      Allow system downtime
    \item
      E.g., banking system
    \end{itemize}
  \item
    \textbf{\textcolor{violet}{Consistency}}

    \begin{itemize}
    \tightlist
    \item
      Allow different system views
    \item
      E.g., social network
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{CAP Theorem: Intuition}
\protect\phantomsection\label{cap-theorem-intuition}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  Consider:

  \begin{itemize}
  \tightlist
  \item
    Client (\emph{Node0})
  \item
    Two DB replicas (\emph{Node1}, \emph{Node2})
  \end{itemize}
\item
  \textbf{\textcolor{red}{Network partition occurs}}

  \begin{itemize}
  \tightlist
  \item
    DB servers (\emph{Node1}, \emph{Node2}) can't communicate
  \item
    Users (\emph{Node0}) access only one (\emph{Node2})
  \item
    \emph{Reads}: Access data on the same partition
  \item
    \emph{Writes}: Can't update due to potential inconsistency
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.25\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_5_0/lec_5_0_slide_14_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\item
  \textbf{\textcolor{darkgreen}{CAP theorem}}: Sacrifice consistency or
  availability
\item
  \textbf{\textcolor{blue}{Available, not consistent}}

  \begin{itemize}
  \tightlist
  \item
    Inconsistency acceptable (e.g., social networking)
  \item
    Allow updates on accessible replica
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Consistent, not available}}

  \begin{itemize}
  \tightlist
  \item
    Inconsistency unacceptable (e.g., banking)
  \item
    Stop service to maintain consistency
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Replication Schemes}
\protect\phantomsection\label{replication-schemes}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Replication schemes}}: Organize multiple
  servers for a distributed DB
\item
  \textbf{\textcolor{darkgreen}{Primary-secondary replication}}

  \begin{itemize}
  \tightlist
  \item
    Application communicates with the primary
  \item
    Replicas require the primary for updates
  \item
    Single point of failure
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Update-anywhere replication}}

  \begin{itemize}
  \tightlist
  \item
    Aka ``multi-master replication''
  \item
    Every replica can update data, which is propagated to others
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Quorum-based replication}}

  \begin{itemize}
  \tightlist
  \item
    \emph{N}: Total replicas
  \item
    Write to \emph{W} replicas
  \item
    Read from \emph{R} replicas, pick the latest update (timestamps)
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Synchronous Replication}
\protect\phantomsection\label{synchronous-replication}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Synchronous replication}}: updates propagate
  to replicas in a single transaction
\item
  Implementations

  \begin{itemize}
  \tightlist
  \item
    \textbf{\textcolor{darkgreen}{2-Phase Commit (2PC)}}: original
    method

    \begin{itemize}
    \tightlist
    \item
      Single point of failure
    \item
      Can't handle primary server failure
    \end{itemize}
  \item
    \textbf{\textcolor{blue}{Paxos}}: widely used

    \begin{itemize}
    \tightlist
    \item
      No primary required
    \item
      More fault tolerant
    \end{itemize}
  \item
    Both are complex/expensive
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}

\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{violet}{CAP theorem}}: only one of Consistency or
  Availability can be guaranteed during a network partition

  \begin{itemize}
  \tightlist
  \item
    Many systems use relaxed consistency models
  \end{itemize}
\end{itemize}

\begin{frame}{Asynchronous Replication}
\protect\phantomsection\label{asynchronous-replication}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Asynchronous replication}}

  \begin{itemize}
  \tightlist
  \item
    Primary node updates replicas
  \item
    Transaction completes before replicas update
  \item
    Quick commits, less consistency
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Eventual consistency}}

  \begin{itemize}
  \tightlist
  \item
    Popularized by AWS DynamoDB
  \item
    Consistency only on eventual outcome
  \item
    ``Eventual'' may mean after server/network fix
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_5_0/lec_5_0_slide_17_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{"Freshness" property}}

  \begin{itemize}
  \tightlist
  \item
    Read from replica may not be latest
  \item
    Request version with specific ``freshness''

    \begin{itemize}
    \tightlist
    \item
      E.g., ``data from not more than 10 minutes ago''
    \item
      E.g., show airplane ticket price a few minutes old
    \end{itemize}
  \item
    Replicas use timestamps for data versioning
  \item
    Use local replica if fresh, else request primary node
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{4) Scalability Issues with RDMS: Problem}
\protect\phantomsection\label{scalability-issues-with-rdms-problem}
\begin{itemize}
\tightlist
\item
  Sources of SQL DB scalability issues:
\end{itemize}

\begin{enumerate}
\tightlist
\item
  \textbf{\textcolor{red}{Locking data}}

  \begin{itemize}
  \tightlist
  \item
    DB engine locks rows/tables for ACID
  \item
    When locked higher latency \(\to\) Fewer updates/second \(\to\)
    Slower application
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Worse in distributed set-up}}
\end{enumerate}

\begin{itemize}
\tightlist
\item
  Requires data replication over multiple servers (scaling out)
\item
  Slower application due to:

  \begin{itemize}
  \tightlist
  \item
    Network delays
  \item
    Locks across networks for DB consistency
  \item
    Overhead of replica consistency (2PC, Paxos)
  \end{itemize}
\end{itemize}
\end{frame}
