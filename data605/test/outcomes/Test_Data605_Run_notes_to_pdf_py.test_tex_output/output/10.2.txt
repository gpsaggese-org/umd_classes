\begin{frame}{Data Streams: Motivation}
\protect\phantomsection\label{data-streams-motivation}
\begin{itemize}
\item
  Big Data is generated as a
  \textbf{\textcolor{red}{continuous, unbounded stream}}
\item
  Applications generate data at
  \textbf{\textcolor{darkgreen}{high velocity}}

  \begin{itemize}
  \tightlist
  \item
    Financial transactions and market feeds
  \item
    Sensor instrumentation, RFID, IoT telemetry
  \item
    Network and system monitoring
  \item
    Continuous media (video, audio)
  \end{itemize}
\item
  A \textbf{\textcolor{blue}{data stream}} is a time-ordered sequence of
  events

  \begin{itemize}
  \tightlist
  \item
    Stream processing treats streams as first-class computational
    objects
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Requirements}}

  \begin{itemize}
  \tightlist
  \item
    Ingest and handle high-throughput event streams
  \item
    Low-latency, near-real-time operations (e.g., time-series analytics)
  \item
    Efficient dissemination of relevant subsets to consumers
  \item
    Distributed processing to scale beyond a single machine
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Data Streams: Examples}
\protect\phantomsection\label{data-streams-examples}
\begin{itemize}
\tightlist
\item
  Continuous queries

  \begin{itemize}
  \tightlist
  \item
    Any SQL query can be continuous
  \item
    E.g., \emph{``compute moving average over last hour every 10 mins''}
  \end{itemize}
\item
  Anomaly detection, pattern recognition

  \begin{itemize}
  \tightlist
  \item
    E.g., \emph{``alert me when A occurs and then B within 10 mins''}
  \item
    Correlate events from different streams
  \end{itemize}
\item
  Statistical tasks

  \begin{itemize}
  \tightlist
  \item
    E.g., de-noising measured readings
  \item
    Build an online machine learning model
  \end{itemize}
\item
  Process multimedia data

  \begin{itemize}
  \tightlist
  \item
    E.g., online object detection, activity detection
  \end{itemize}
\end{itemize}
\end{frame}

  \begin{itemize}

  \end{itemize}

  \textbf{\textcolor{darkgreen}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Insert arriving items into a relational table
  \item

  \end{itemize}

\begin{itemize}
\tightlist
\item
  Modern distributed systems use
  \textbf{\textcolor{red}{small, independent components}}

\begin{frame}{Delivery Semantics: At-most once}
\protect\phantomsection\label{delivery-semantics-at-most-once}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{At-most once}}: \emph{message may be lost, not
  redelivered}
\end{itemize}

\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_12_1/lec_12_1_slide_7_image_1.png}}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Small implementation overhead, high-performance
  \item
    Easy to implement: ``fire-and-forget''
  \end{itemize}
\item
  Works when occasional loss is acceptable

  \begin{itemize}
  \tightlist
  \item
    E.g., monitoring metrics of a website
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Delivery Semantics: At-least once}
\protect\phantomsection\label{delivery-semantics-at-least-once}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{At-least once}}: messages are retried until
  acknowledged
\end{itemize}

\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_12_1/lec_12_1_slide_7_image_2.png}}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Ensures no loss, but duplicates are possible
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Requires idempotent operations or deduplication
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Delivery Semantics: Exactly once}
\protect\phantomsection\label{delivery-semantics-exactly-once}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Exactly once}}: each message is processed once
  globally
\end{itemize}

\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_12_1/lec_12_1_slide_7_image_3.png}}

\begin{itemize}
\tightlist
\item
  Most consumer-friendly but hardest to guarantee

  \begin{itemize}
  \tightlist
  \item
    Complicated by distributed coordination limits (e.g., ``Two
    Generals' Problem'')
  \end{itemize}
\item
  Used in financial and mission-critical systems

  \begin{itemize}
  \tightlist
  \item
    E.g., payment, trading, accounting systems
  \end{itemize}
\end{itemize}
\end{frame}

  \begin{itemize}

  \item

    \begin{itemize}
    \tightlist
    \item
      Time when each record is generated
    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Processing time}}

    \begin{itemize}

    \item
      Time when each record is received
    \item

      processed
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Problems with events}}

  \tightlist

    Events may arrive late or out of order
  \item
    Determining how long to wait for stragglers is difficult
  \item

\begin{frame}{Record-at-a-time Processing}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Distributed processing over multiple nodes}}

  \begin{itemize}
  \tightlist
  \item
    Nodes organized in a DAG
  \item
    Each node continuously:

    \begin{itemize}
    \tightlist
    \item
      Receives a single record
    \item
      Processes the record immediately
    \item
      Forwards the output to the next node
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Achieves extremely low latency

    \begin{itemize}
    \tightlist
    \item
      E.g., sub-millisecond response times
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Poor fault tolerance

    \begin{itemize}
    \tightlist
    \item
      Requires extra nodes or redundant paths for failover
    \end{itemize}
  \item
    Sensitive to stragglers

    \begin{itemize}
    \tightlist
    \item
      Slow nodes can delay the entire pipeline
    \end{itemize}
  \end{itemize}

\begin{frame}{Triggering Modes}
\protect\phantomsection\label{triggering-modes}
\begin{itemize}
\item
  Indicate when to process newly available streaming data
\item
  \textbf{\textcolor{red}{Default}}

  \begin{itemize}
  \tightlist
  \item
    Process micro-batch after previous completes
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Trigger interval}}

  \begin{itemize}
  \tightlist
  \item
    Specify fixed interval for each micro-batch
  \item
    E.g., ``every 10 minutes''
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Once}}

  \begin{itemize}
  \tightlist
  \item
    Wait for external trigger
  \item
    E.g., ``at end of day''
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Continuous (experimental)}}

  \begin{itemize}
  \tightlist
  \item
    Process data continuously
  \item
    Not all operations available
  \item
    Lower latency
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Saving Modes}
\protect\phantomsection\label{saving-modes}
\begin{itemize}
\tightlist
\item
  Indicate when to save results and where

  \begin{itemize}
  \tightlist
  \item
    Each time the result table updates, write to an external file system

    \begin{itemize}
    \tightlist
    \item
      E.g., HDFS, AWS S3, or a database (e.g., MySQL, Cassandra)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{red}{Append mode}}

  \begin{itemize}
  \tightlist
  \item
    Append new rows since the last trigger
  \item
    Use when existing rows don't change
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Update mode}}

  \begin{itemize}
  \tightlist
  \item
    Write updated rows since the last trigger
  \item
    Update in place
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Complete mode}}

  \begin{itemize}
  \tightlist
  \item
    Write the entire updated result table
  \item
    General but expensive
  \end{itemize}
\end{itemize}
\end{frame}
