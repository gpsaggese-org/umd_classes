\begin{frame}{Key-Value Store vs Document DBs}
\protect\phantomsection\label{key-value-store-vs-document-dbs}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Key-value stores}}

  \begin{itemize}
  \tightlist
  \item
    Function as a map or dictionary

    \begin{itemize}
    \tightlist
    \item
      Examples: HBase, Redis
    \end{itemize}
  \item
    Primarily retrieve values using keys
  \item
    Occasionally search within value fields using patterns
  \item
    Store uninterpreted values (e.g., binary blobs) linked to keys
  \item
    Use a single namespace for all key-value pairs
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Document DBs}}

  \begin{itemize}
  \tightlist
  \item
    Group key-value pairs into \emph{documents}

    \begin{itemize}
    \tightlist
    \item
      Examples: MongoDB, CouchDB
    \end{itemize}
  \item
    Documents formatted in JSON, XML, or BSON (binary JSON)
  \item
    Documents are part of \emph{collections}

    \begin{itemize}
    \tightlist
    \item
      Comparable to \emph{tables} in relational databases

  \item
    Client machine ID
  \item
    Client process ID
  \item
    3-byte auto-incremented counter
  \end{itemize}

\begin{frame}{Indexes}
\protect\phantomsection\label{indexes}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Primary index}}

  \begin{itemize}
  \tightlist
  \item
    Automatically created on the \textcolor{blue}{\texttt{\_id}} field
  \item
    B+ tree indexes
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Secondary index}}

  \begin{itemize}
  \tightlist
  \item
    Improve query performance
  \item
    Can enforce unique values for a field
  \end{itemize}
\item
  Single field index and compound index (like SQL)

  \begin{itemize}
  \tightlist
  \item
    Order of fields in a compound index matters
  \end{itemize}
\item
  Sparse property of an index

  \begin{itemize}
  \tightlist
  \item
    Index contains entries only for documents with the indexed field
  \item
    Ignores records without the field
  \end{itemize}
\item
  Rejects records with duplicate keys if the index is unique and sparse
\item
  Details \href{https://www.mongodb.com/docs/manual/indexes/}{here}
\end{itemize}
