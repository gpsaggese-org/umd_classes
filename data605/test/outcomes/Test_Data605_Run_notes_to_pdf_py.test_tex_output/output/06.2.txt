\begin{frame}[fragile]{Update Operations}
\protect\phantomsection\label{update-operations}
\begin{itemize}
\tightlist
\item
  \textcolor{blue}{\texttt{db.collection.insert(<document>)}}

  \begin{itemize}
  \item
    Omit the \textcolor{blue}{\texttt{\_id}} field to have MongoDB
    generate a unique key

  \end{itemize}

  \textcolor{blue}{\texttt{db.collection.save(<document>)}}

    Updates an existing record or creates a new record
  \end{itemize}
\item
  \textcolor{blue}{\texttt{db.collection.update(<query>, <update>, {upsert: true})}}

  \begin{itemize}
  \tightlist
  \item
    Will update 1 or more records in a collection satisfying query
  \end{itemize}

  \textcolor{blue}{\texttt{db.collection.findAndModify(<query>, <sort>, <update>, <new>, <fields>, <upsert>)}}

  \begin{itemize}

  \item
    Retrieve old or new version of the record
  \end{itemize}

\begin{frame}{MongoDB Features}
\protect\phantomsection\label{mongodb-features}
\begin{itemize}
\item
  Document-oriented NoSQL store
\item
  Rich querying

  \begin{itemize}
  \tightlist
  \item
    Full index support (primary and secondary)
  \end{itemize}
\item
  Fast in-place updates
\item
  Agile and scalable

  \begin{itemize}
  \tightlist
  \item
    Replication and high availability
  \item
    Auto-sharding
  \item
    Map-reduce functionality
  \end{itemize}
\item
  Scale horizontally over commodity hardware

  \begin{itemize}
  \tightlist
  \item
    Horizontally = add more machines
  \item
    Commodity hardware = inexpensive servers
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{MongoDB vs Relational DBs}
\protect\phantomsection\label{mongodb-vs-relational-dbs}
\begin{itemize}
\tightlist
\item
  Keep the functionality that works well in RDBMSs

  \begin{itemize}
  \tightlist
  \item
    Ad-hoc queries
  \item
    Fully featured indexes
  \item
    Secondary indexes
  \end{itemize}
\item
  Do not offer RDBMS functionalities that don't scale up

  \begin{itemize}
  \tightlist
  \item
    Long running multi-row transactions
  \item
    ACID consistency
  \item
    Joins
  \end{itemize}
\end{itemize}
\end{frame}
