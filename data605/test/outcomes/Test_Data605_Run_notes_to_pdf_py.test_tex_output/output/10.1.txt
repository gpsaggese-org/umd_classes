\begin{columns}[T]
\begin{column}{0.6\linewidth}

  \textbf{\textcolor{red}{Client-server}}: Model for distributed

  \begin{itemize}
  \tightlist
  \item
    \emph{Clients}: Request services (e.g., dashboard, GUI, client
    applications)
  \item

\begin{frame}{Parallel vs Distributed Computing}
\protect\phantomsection\label{parallel-vs-distributed-computing}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Parallel computing}}

  \begin{itemize}
  \tightlist
  \item
    One computer, multiple CPUs
  \item
    Cluster: many computers, each with multiple CPUs
  \item
    Homogeneous, geographically close nodes
  \item
    Work on one task simultaneously
  \end{itemize}
\end{itemize}

\begin{frame}{Parallel Systems}
\protect\phantomsection\label{parallel-systems}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Parallel systems}} consist of:

  \begin{itemize}
  \tightlist
  \item
    Multiple processors
  \item
    Multiple memories
  \item
    Multiple disks
  \item
    Fast interconnection network
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Coarse-grain parallel machine}}

  \begin{itemize}
  \tightlist
  \item
    Small number of powerful processors
  \item
    E.g., your laptop with multiple CPUs
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Fine-grain parallel machine}}

  \begin{itemize}
  \tightlist
  \item
    Aka massively parallel
  \item
    Thousands of smaller processors
  \item
    Larger degree of parallelism
  \item
    With or without shared memory
  \item
    E.g., GPUs, The Connection Machine
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Speed-Up and Scale-Up: Intuition}
\protect\phantomsection\label{speed-up-and-scale-up-intuition}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{You have a workload to execute}}

  \begin{itemize}
  \tightlist
  \item
    Change workload \(M\)

    \begin{itemize}
    \tightlist
    \item
      Number of DB transactions
    \item
      Amount of DB data to query
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{You need to execute the workload on a machine}}

  \begin{itemize}
  \tightlist
  \item
    Change computing power \(N\)

    \begin{itemize}
    \tightlist
    \item
      Better CPU (scale vertically, scale up)
    \item
      More CPUs (scale horizontally, scale out)
    \end{itemize}
  \end{itemize}
\item
  Two \textbf{\textcolor{blue}{ways to measure efficiency}} when
  increasing workload and computing power

  \begin{itemize}
  \tightlist
  \item
    \emph{Speed-up}

    \begin{itemize}
    \tightlist
    \item
      Keep constant problem size \(M\)
    \item
      Increase machine power \(N\)
    \end{itemize}
  \item
    \emph{Scale-up}

    \begin{itemize}
    \tightlist
    \item
      Increase problem size \(M\)
    \item
      Increase machine power \(N\)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Speed-Up vs Scale-Up}
\protect\phantomsection\label{speed-up-vs-scale-up}
\begin{itemize}
\tightlist
\item
  The amount of computing power \(N\) can be changed
\item
  The amount of work \(M\) can be changed
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Speed-up}}: fixed-sized problem on a small
  system given to a system \(N\)-times larger
  \[\text{speed-up} = \frac{\text{small system elapsed time}}
      {\text{large system elapsed time}}\]

  \begin{itemize}
  \tightlist
  \item
    Speed-up is linear if equation equals \(N\)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_10_2/lec_10_2_slide_9_image_1.png}}
\end{column}
\end{columns}

\vspace{0.5cm}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Scale-up}}: increase size of both problem
  \(M\) and system \(N\)

  \begin{itemize}
  \tightlist
  \item
    \(N\)-times larger system to perform \(M\)-times larger job
    \[\text{scale-up} = \frac{\text{small system-problem time}}
      {\text{big system-problem time}}\]
  \item
    Scale-up is linear if equation equals 1
  \end{itemize}
\end{itemize}
\end{column}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{Factors Limiting Speed-up and Scale-up}
\protect\phantomsection\label{factors-limiting-speed-up-and-scale-up}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Speed-up and scale-up are typically sub-linear}}
  due to several issues

  \begin{itemize}
  \tightlist
  \item
    E.g., not all computation is parallelizable
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Amdahl's Law}}

  \begin{itemize}
  \tightlist
  \item
    \(p\) = fraction parallelizable
  \item
    \(s\) = number of nodes
  \item
    \(T\) = execution time serially
  \item
    \(T(p)\) = execution time on s nodes = \((1-p)T + (p / s)T\) \[
    Speedup(s)
    = \frac{T}{T(s)}
    = \frac{1}{(1 - p) + \frac{p}{s}}
    \]
  \end{itemize}
\item
  E.g.,

  \begin{itemize}
  \tightlist
  \item
    90\% parallelizable \(\to\) max speed-up 10x
  \item
    50\% parallelizable \(\to\) max speed-up 2x (even with infinite
    nodes)
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_10_2/lec_10_2_slide_10_image_1.png}}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Factors Limiting Speed-up and Scale-up}
\protect\phantomsection\label{factors-limiting-speed-up-and-scale-up-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Startup costs}}

  \begin{itemize}
  \tightlist
  \item
    Starting processes may dominate computation time
  \item
    E.g., dBs create thread pool at startup
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Interference}}

  \begin{itemize}
  \tightlist
  \item
    Processes compete for shared resources (e.g., system bus, disks,
    locks)
  \item
    Time spent waiting on other processes
  \item
    E.g., devs touching same code create merge conflicts
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cost of synchronization}}

  \begin{itemize}
  \tightlist
  \item
    Smaller work pieces increase synchronization complexity
  \item
    E.g., hiring many developers in a company
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Skew}}

  \begin{itemize}
  \tightlist
  \item
    Splitting work increases variance in task response time
  \item
    Difficult to split tasks equally
  \item
    Execution time determined by slowest task
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Consistency Issues in Distributed DB Systems}
\protect\phantomsection\label{consistency-issues-in-distributed-db-systems}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Parallel and distributed DBs}}

  \begin{itemize}
  \tightlist
  \item
    Efficient for query processing (read-only)
  \item
    Requires consistency enforcement
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Atomicity issues}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Problem}: Transaction is all-or-nothing across nodes
  \item
    Two-phase commit (2PC) is centralized

    \begin{itemize}
    \tightlist
    \item
      Commit decision by a single coordinator node
    \item
      Each node executes transaction, reaching ``ready state''
    \item
      If all nodes reach ready state, coordinator commits
    \item
      If a node fails in ready state, it can recover (e.g., write-ahead
      logs)
    \item
      If a node aborts, coordinator aborts transaction
    \end{itemize}
  \item
    Distributed consensus, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Paxos
    \item
      Blockchain
    \end{itemize}
  \end{itemize}
\end{itemize}
