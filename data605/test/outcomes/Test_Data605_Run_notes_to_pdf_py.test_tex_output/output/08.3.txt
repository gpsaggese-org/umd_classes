\begin{frame}{Hadoop Distributed File System (HDFS)}
\protect\phantomsection\label{hadoop-distributed-file-system-hdfs}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  HDFS is a \textbf{\textcolor{red}{distributed file system}}

  \begin{itemize}
  \tightlist
  \item
    Designed to store large data sets reliably
  \item
    Part of the Apache Hadoop ecosystem
  \item
    Inspired by the Google File System (GFS)
  \end{itemize}
\end{itemize}

\begin{enumerate}
\tightlist
\item
  Optimized for \textbf{\textcolor{darkgreen}{high-throughput access}}
  to large files

  \begin{itemize}
  \tightlist
  \item
    Suitable for batch processing
  \item
    Not low-latency access
  \end{itemize}
\end{enumerate}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_8_3/lec_8_3_slide_1_image_1.png}}
\end{column}
\end{columns}

\begin{enumerate}
\setcounter{enumi}{1}
\tightlist
\item
  Designed for
  \textbf{\textcolor{blue}{fault tolerance and scalability}}

  \begin{itemize}
  \tightlist
  \item
    Ensures fault tolerance through replication

    \begin{itemize}
    \tightlist
    \item
      Blocks are stored on different nodes and racks
    \item
      Provides data availability even if some nodes fail
    \end{itemize}
  \item
    Follows a primary-secondary architecture
  \item
    Replication strategy improves read performance
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{HDFS Architecture}
\protect\phantomsection\label{hdfs-architecture}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{NameNode}}

  \begin{itemize}
  \tightlist
  \item
    Store file/dir hierarchy
  \item
    Store file metadata

    \begin{itemize}
    \tightlist
    \item
      E.g., block location, size, permissions
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{DataNodes}}

  \begin{itemize}
  \tightlist
  \item
    Store actual data blocks
  \item
    Split file into 16-256MB blocks
  \item
    Replicate chunks (2x or 3x) across multiple \emph{DataNodes}
  \item
    Keep replicas in different racks
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Client}}

  \begin{itemize}
  \tightlist
  \item
    API (e.g., Python, Java) to library
  \item
    Mount HDFS on local filesystem
  \end{itemize}
\end{itemize}

\begin{frame}{HDFS: Read / Write Protocols}
\protect\phantomsection\label{hdfs-read-write-protocols}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Read}}

  \begin{itemize}
  \tightlist
  \item
    Contact \emph{NameNode} for \emph{DataNode} and block pointer
  \item
    Choose nearest \emph{DataNode} for each block
  \item
    Connect to \emph{DataNode} for data access
  \item
    Read blocks in parallel to improve performance
  \item
    Client reassembles data in correct order
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Write}}

  \begin{itemize}
  \tightlist
  \item
    \emph{NameNode} creates blocks
  \item
    Assign blocks to multiple \emph{DataNodes}
  \item
    Client sends data to \emph{DataNodes}
  \item
    \emph{DataNodes} store data
  \item
    Pipeline blocks to other replicas
  \item
    Write successful after all replicas acknowledge
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Fault Tolerance and Recovery}
\protect\phantomsection\label{fault-tolerance-and-recovery}
\begin{itemize}
\tightlist
\item
  \emph{NameNode} monitors \emph{DataNode} heartbeat signals

  \begin{itemize}
  \tightlist
  \item
    On failure, blocks are re-replicated to maintain replication factor
  \end{itemize}
\item
  \emph{NameNode} itself is a single point of failure

  \begin{itemize}
  \tightlist
  \item
    Solved with HDFS High Availability
  \end{itemize}
\item
  Data integrity ensured using checksums
\end{itemize}
\end{frame}

\begin{frame}{HDFS vs Traditional File Systems}
\protect\phantomsection\label{hdfs-vs-traditional-file-systems}
\begin{itemize}
\item
  Best for
  \textbf{\textcolor{red}{storing and processing large-scale files}}

  \begin{itemize}
  \tightlist
  \item
    E.g., logs, media, sensor data
  \item
    Commonly used in data lakes and ETL pipelines
  \item
    Supports very large files and directories
  \item
    Performance degrades with many small files
  \end{itemize}
\item
  Optimized for \textbf{\textcolor{darkgreen}{write-once, read-many}}
  access pattern
\item
  Lacks low-latency access, but provides
  \textbf{\textcolor{blue}{high throughput}}

  \begin{itemize}
  \tightlist
  \item
    Good for analytics (OLAP)
  \item
    Not suitable for transactional systems (OLTP)

    \begin{itemize}
    \tightlist
    \item
      E.g., bank
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}
