\begin{frame}{How to Store Big Data?}
\protect\phantomsection\label{how-to-store-big-data}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Many different solutions to storing big data}}

  \begin{enumerate}
  \tightlist
  \item
    \emph{Distributed file systems}
  \item
    \emph{Sharding across multiple DBs}
  \item
    \emph{Parallel and distributed DBs}
  \item
    \emph{Key-value stores}
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{1) Distributed File Systems}
\protect\phantomsection\label{distributed-file-systems}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Files stored across machines yet single file-system view to clients}}

  \begin{itemize}
  \tightlist
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Google File System (GFS)
    \item
      Hadoop File System (HDFS)
    \item
      AWS S3
    \end{itemize}
  \item
    Files are:

    \begin{itemize}
    \tightlist
    \item
      Broken into blocks
    \item
      Blocks partitioned across machines
    \item
      Blocks often replicated
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Goals}}

  \begin{itemize}
  \tightlist
  \item
    Store data not fitting on one machine
  \item
    Increase performance
  \item
    Increase reliability/availability/fault tolerance
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{2) Sharding Across Multiple DBs}
\protect\phantomsection\label{sharding-across-multiple-dbs}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Sharding}}: Partition records across multiple
  DBs or machines

  \begin{itemize}
  \tightlist
  \item
    Shard keys, aka partitioning keys / partition attributes
  \item
    Range partition (e.g., timeseries)
  \item
    Hash partition
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Scale beyond centralized DB for more users, storage, processing
    speed
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Replication needed for failures
  \item
    Ensuring consistency is challenging
  \item
    Relational DBs struggle with constraints (e.g., foreign key) and
    transactions on multiple machines
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{3) Parallel and Distributed DBs}
\protect\phantomsection\label{parallel-and-distributed-dbs}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Store and process data on multiple machines (cluster)}}
\item
  \textbf{\textcolor{darkgreen}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    From programmer viewpoint

    \begin{itemize}
    \tightlist
    \item
      Traditional relational DB interface
    \item
      Appears as a single-machine DB
    \end{itemize}
  \item
    Approach scales to 10s-100s of machines
  \item
    Data replication enhances performance and reliability

    \begin{itemize}
    \tightlist
    \item
      Frequent failures with 100s of machines
    \item
      Queries can restart on different machines
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Incremental query execution is complex
  \item
    Scalability limits
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{4) (Parallel) Key-value Stores}
\protect\phantomsection\label{parallel-key-value-stores}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Applications store billions of small records
  \item
    Relational DBs lack multi-machine constraints and transactions
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Key-value stores / Document / NoSQL systems
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Redis
    \item
      Apache HBase (open source of Google BigTable)
    \item
      \ldots{}
    \item
      AWS Dynamo, S3
    \item
      Azure cloud storage (Microsoft)
    \item
      MongoDB cluster
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Partition data across machines
  \item
    Support replication and consistency
  \item
    Balance workload, add machines
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Cons}}

  \begin{itemize}

  \item

    \begin{itemize}

    \item

    \item

    \end{itemize}

\begin{frame}{Storage Infrastructure}

  \end{itemize}

  \item

  \item

    \item
      Broken into blocks
    \item
      Partitioned across machines
    \item
      Replicated across machines
    \end{itemize}
  \item
    Provide a single file-system view to clients
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Distributed File System}
\protect\phantomsection\label{distributed-file-system}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Reliable distributed file system}}

  \begin{itemize}
  \tightlist
  \item
    Data in ``chunks'' across machines
  \item
    Each chunk replicated on different machines
  \item
    Seamless recovery from disk or machine failure
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Bring computation directly to the data}}

  \begin{itemize}
  \tightlist
  \item
    ``Chunk servers'' also serve as ``compute servers''
  \end{itemize}
\item
  Hadoop and HDFS implement all these ideas
\end{itemize}
\end{frame}
