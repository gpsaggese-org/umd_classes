\begin{frame}[fragile]{MapReduce: Overview}
\protect\phantomsection\label{mapreduce-overview}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{MapReduce programming model}}

  \begin{itemize}
  \tightlist
  \item
    Inspired by functional programming (e.g., Lisp)
  \item
    Common pattern of parallel programming to process large number of
    records
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Basic algorithm}}

  \begin{itemize}
  \tightlist
  \item
    Apply \textcolor{blue}{\texttt{map()}} to each record
  \item
    Group results by key
  \item
    Apply \textcolor{blue}{\texttt{reduce()}} to results of
    \textcolor{blue}{\texttt{map()}}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Example}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Goal}: Sum length of all tuples in a document

    \begin{itemize}
    \item
      E.g.,

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{[() (a,) (a, b) (a, b, c)]}
\end{Highlighting}
\end{Shaded}
    \end{itemize}
  \item
    \emph{map(function, set of values)}

    \begin{itemize}
    \item
      Apply function to each value (e.g., len)

\begin{Shaded}
\begin{Highlighting}[]
\BuiltInTok{map}\NormalTok{(}\BuiltInTok{len}\NormalTok{, [(), (a), (a, b), (a, b, c))]) }\OperatorTok{{-}\textgreater{}}\NormalTok{ [}\DecValTok{0}\NormalTok{, }\DecValTok{1}\NormalTok{, }\DecValTok{2}\NormalTok{, }\DecValTok{3}\NormalTok{]}
\end{Highlighting}
\end{Shaded}
    \end{itemize}
  \item
    \emph{reduce(function, set of values)}

  \item

    \begin{itemize}
    \tightlist
    \item
      E.g., key is a file, value is occurrences
    \item
      ``One a penny'' \(\to\)
      \textcolor{blue}{\texttt{[("One", 1), ("a", 1), ("penny", 1)]}}
    \end{itemize}
  \item
    One \textcolor{blue}{\texttt{Map}} call for every
    \textcolor{blue}{\texttt{(k, v)}} pair
  \end{itemize}
\item
  \textbf{\textcolor{teal}{GroupBy}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{GroupBy(List[Tuple[k, v]]) $\rightarrow$ List[Tuple[k, List[v]]]]}}
  \item
    Group and optionally sort records with the reduce key
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Reduce}}

  \begin{itemize}
  \tightlist
  \item
    \textcolor{blue}{\texttt{Reduce(Tuple[k, List[v]]) $\rightarrow$ Tuple[k, v]}}
  \item
    Reduce all values \textcolor{blue}{\texttt{v'}} with same key
    \textcolor{blue}{\texttt{k'}}
  \item
    One \textcolor{blue}{\texttt{Reduce}} call per unique key
    \textcolor{blue}{\texttt{*k'}}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Output}}: Write key-value pairs
  \textcolor{blue}{\texttt{List[Tuple[k, v]]}}
\end{itemize}

\begin{frame}{How Many \textcolor{blue}{\texttt{Map}} and
\textcolor{blue}{\texttt{Reduce}} Jobs?}
\protect\phantomsection\label{how-many-and-jobs}
\begin{itemize}
\item
  Number of:

  \begin{itemize}
  \tightlist
  \item
    Map tasks: \(M\)
  \item
    Reduce tasks: \(R\)
  \item
    Worker nodes: \(N\)
  \end{itemize}
\item
  Typically \(M \gg N\)

  \begin{itemize}
  \tightlist
  \item
    Pros:

    \begin{itemize}
    \tightlist
    \item
      Improve dynamic load balancing
    \item
      Speed up recovery from worker failures
    \end{itemize}
  \item
    Cons:

    \begin{itemize}
    \tightlist
    \item
      More communication between \emph{MasterNode} and
      \emph{WorkerNodes}
    \item
      Lots of smaller files
    \end{itemize}
  \end{itemize}
\item
  Typically \(R > N\)
\item
  Usually \(R < M\), output is spread across fewer files
\end{itemize}
\end{frame}

\begin{frame}{Refinements: Backup Tasks}
\protect\phantomsection\label{refinements-backup-tasks}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \tightlist
  \item
    Slow workers significantly lengthen the job completion time
  \item
    Slow workers due to:

    \begin{itemize}
    \tightlist
    \item
      Older processor
    \item
      Not enough RAM
    \item
      Other jobs on the machine
    \item
      Bad disks
    \item
      OS thrashing / virtual memory hell
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Near the end of \textcolor{blue}{\texttt{Map}} /
    \textcolor{blue}{\texttt{Reduce}} phase

    \begin{itemize}
    \tightlist
    \item
      Spawn backup copies of tasks
    \item
      Whichever one finishes first ``wins''
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Result}}

  \begin{itemize}
  \tightlist
  \item
    Shorten job completion time
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Refinement: Combiners}
\protect\phantomsection\label{refinement-combiners}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Problem}}

  \begin{itemize}
  \item
    Often a \textcolor{blue}{\texttt{Map}} task produces many pairs for
    the same key \textcolor{blue}{\texttt{k}}

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{[(k1, v1), (k1, v2), ...]}
\end{Highlighting}
\end{Shaded}
  \item
    E.g., common words in the word count example
  \item
    Increase complexity of the \textcolor{blue}{\texttt{GroupBy}} stage
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Solution}}

  \begin{itemize}
  \item
    Pre-aggregate values in the \textcolor{blue}{\texttt{Map}} with a
    \textcolor{blue}{\texttt{Combine}}

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{[k1, (v1, v2, ...), k2, ([...])]}
\end{Highlighting}
\end{Shaded}
  \item
    \textcolor{blue}{\texttt{Combine}} is usually the same as the
    \textcolor{blue}{\texttt{Reduce}} function
  \item
    Works only if \textcolor{blue}{\texttt{Reduce}} function is
    commutative and associative
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Result}}

  \begin{itemize}
  \tightlist
  \item
    Better data locality
  \item
    Less shuffling and reordering
  \item
    Less network / disk traffic
  \end{itemize}
\end{itemize}
\end{frame}

  \tightlist
  \item

    input file splits
  \item
    Default partition function:
    \textcolor{blue}{\texttt{hash(key) mod R}}
  \item
    Ensure records with the same intermediate key go to the same worker
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Solution}}

  \begin{itemize}
  \tightlist
  \item
    Override hash function:
  \item
    E.g., \textcolor{blue}{\texttt{hash(hostname(URL)) mod R}} ensures
    URLs from a host end up in the same output file
  \end{itemize}
\end{itemize}
\end{frame}
