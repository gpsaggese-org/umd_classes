\begin{frame}{Data Science}
\protect\phantomsection\label{data-science}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Promises of data science}}

  \begin{itemize}
  \tightlist
  \item
    Give a competitive advantages
  \item
    Make better strategic and tactical business decisions
  \item
    Optimize business processes
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Data science is not new}}, it was
  called:

  \begin{itemize}
  \tightlist
  \item
    Operation research (\textasciitilde1970-80s)
  \item
    Decision support, business intelligence (\textasciitilde1990s)
  \item
    Predictive analytics (Early 2010s)
  \item
    \ldots{}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{What has changed}}

  \begin{itemize}
  \tightlist
  \item
    Now learning and applying data science is \emph{easy}

    \begin{itemize}
    \tightlist
    \item
      No need for hiring a consulting company
    \end{itemize}
  \item
    Tools are \emph{open-source}

    \begin{itemize}
    \tightlist
    \item
      E.g., Python + pydata stack (numpy, scipy, Pandas, sklearn)
    \end{itemize}
  \item
    \emph{Large data sets} available
  \item
    \emph{Cheap computing}

    \begin{itemize}
    \tightlist
    \item
      E.g., cloud computing (AWS, Google Cloud), GPUs
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Motivation: Data Overload}
\protect\phantomsection\label{motivation-data-overload}
\begin{itemize}
\tightlist
\item
  \emph{``Data science is the number one catalyst for economic growth''}
  (McKinsey, 2013)
\item
  \textbf{\textcolor{red}{Explosion of data in every domain}}

  \begin{itemize}
  \tightlist
  \item
    Sensing devices/networks monitor processes 24/7

    \begin{itemize}
    \tightlist
    \item
      E.g., temperature of your room, your vital signs, pollution in the
      air
    \end{itemize}
  \item
    Sophisticated smart-phones

    \begin{itemize}
    \tightlist
    \item
      80\%+ of the world population has a smart-phone
    \end{itemize}
  \item
    Internet and social networks make it easy to publish data
  \item
    Internet of Things (IoT): everything is connected to the internet

    \begin{itemize}
    \tightlist
    \item
      E.g., power supply, toasters
    \end{itemize}
  \item
    Datafication turns all aspects of life into data

    \begin{itemize}
    \tightlist
    \item
      E.g., what you like/enjoy turned into a stream of your ``likes''
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Challenges}}

  \begin{itemize}
  \tightlist
  \item
    How to handle the increasing amount data?
  \item
    How to extract actionable insights and scientific knowledge from
    data?
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Scale of Data Size}
\protect\phantomsection\label{scale-of-data-size}
\begin{columns}[T]
\begin{column}{0.45\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Megabyte}} = \(2^{20} \approx 10^6\) bytes

  \begin{itemize}
  \tightlist
  \item
    Typical English book
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Gigabyte}} = \(2^{30}\) bytes = 1,000 MB

  \begin{itemize}
  \tightlist
  \item
    1/2 hour of video
  \item
    Wikipedia (compressed, no media) is 22GB
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Terabyte}} = 1 million MB

  \begin{itemize}
  \tightlist
  \item
    Human genome: \textasciitilde1 TB
  \item
    100,000 photos
  \item
    \$50 for 1TB HDD, \$23/mo on AWS S3
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Petabyte}} = 1000 TB

  \begin{itemize}
  \tightlist
  \item
    13 years of HD video
  \item
    \$250k/year on AWS S3
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{cyan}{Exabyte}} = 1M TB

  \begin{itemize}
  \tightlist
  \item
    Global yearly Internet traffic in 2004
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Zettabyte}} = 1B TB = \(10^{21}\) bytes

  \begin{itemize}
  \tightlist
  \item
    Global yearly Internet traffic in 2016
  \item
    Fill 20\% of Manhattan, New York with data centers
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Yottabytes}} = \(10^{24}\) bytes

  \begin{itemize}
  \tightlist
  \item
    Yottabyte costs \$100T
  \item
    Fill Delaware and Rhode Island with a million data centers
  \end{itemize}
\item
  \textbf{\textcolor{magenta}{Brontobytes}} = \(10^{27}\) bytes
\end{itemize}
\end{column}
\end{columns}
