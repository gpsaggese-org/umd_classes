\begin{frame}{Amazon Web Services (AWS)}
\protect\phantomsection\label{amazon-web-services-aws}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{AWS is a platform offering complete cloud solutions}}

  \begin{itemize}
  \tightlist
  \item
    Computing (e.g., EC2)
  \item
    Storage (e.g., S3)
  \item
    Networking
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Offers different levels of abstraction}}

  \begin{itemize}
  \tightlist
  \item
    IaaS, PaaS, SaaS
  \item
    From virtual hardware to applications, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Host websites
    \item
      Run enterprise software
    \item
      Run machine learning applications
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Control services in different ways}}

  \begin{itemize}
  \tightlist
  \item
    Web interface (console)
  \item
    CLI: \textcolor{blue}{\texttt{aws}} command
  \item
    Language libraries, SDKs (e.g., Python
    \textcolor{blue}{\texttt{boto3}})
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Optimizing Costs}

  \item

  \item

  \item

    Encrypting data
  \item
    Updating the OS and software
