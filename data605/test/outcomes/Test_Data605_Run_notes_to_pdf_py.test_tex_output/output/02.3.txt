\begin{frame}{Data as a Product}
\protect\phantomsection\label{data-as-a-product}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Many services today "sell" data}}

  \begin{itemize}
  \tightlist
  \item
    Services are typically powered by data and machine learning, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Personalized search engine (Google)
    \item
      Sentiment analysis on user-generated data (Facebook)
    \item
      E-commerce + recommendation engine (Amazon)
    \item
      Streaming data (Netflix, Spotify)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Several steps are required to generate data products}}

  \begin{itemize}
  \tightlist
  \item
    Data ingestion
  \item
    Data pre-processing

    \begin{itemize}
    \tightlist
    \item
      Cleaning, tokenization, feature computation
    \end{itemize}
  \item
    Model training
  \item
    Model deployment

    \begin{itemize}
    \tightlist
    \item
      MLOps
    \end{itemize}
  \item
    Model monitoring

    \begin{itemize}
    \tightlist
    \item
      Is model working?
    \item
      Is model getting slower?
    \item
      Are model performance getting worse?
    \end{itemize}
  \item
    Collect feedback from deployment

    \begin{itemize}
    \tightlist
    \item
      E.g., recommendations vs what users bought
    \item
      Ingest data from production for future versions of the model
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{ETL Paradigm: Phases}
\protect\phantomsection\label{etl-paradigm-phases}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Extract}}

  \begin{itemize}
  \tightlist
  \item
    Gather data from various data sources, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Internal / external data warehouse
    \item
      REST API
    \item
      Data downloading from API
    \item
      Web scraping
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Transform}}

  \begin{itemize}
  \tightlist
  \item
    Raw data is combined and formatted to become useful for analysis
    step
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Load}}

  \begin{itemize}
  \tightlist
  \item
    Move data into the final destination, e.g.,

    \begin{itemize}
    \tightlist
    \item
      Data warehouse
    \item
      Data lake
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Data ingestion pipeline = E + L}}

  \begin{itemize}
  \tightlist
  \item
    Move data from one point to another
  \item
    Format the data
  \item
    Make a copy
  \item
    Have different tools to operate on the data
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{ETL Paradigm: Example}

    \end{itemize}

  \begin{itemize}

  \end{itemize}

  \item
    Organize data in a format optimized for data analysis

\begin{frame}{ELT Paradigm}
\protect\phantomsection\label{elt-paradigm}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{ETL}} has been the standard approach for long
  time

  \begin{itemize}
  \tightlist
  \item
    Extract \(\to\) Transform \(\to\) Load
  \item
    \textbf{\textcolor{orange}{Cons}}

    \begin{itemize}
    \tightlist
    \item
      Need to understand the data at ingestion time
    \item
      Need to know how the data will be used
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{itemize}
\tightlist
\item
  Today \textbf{\textcolor{darkgreen}{ELT}} is becoming the pattern of
  choice

  \begin{itemize}
  \tightlist
  \item
    Extract \(\to\) Load \(\to\) Transform
  \item
    \textbf{\textcolor{teal}{Pro}}:

    \begin{itemize}
    \tightlist
    \item
      No need to know how the data will be used
    \item
      Separate data engineers and data scientists / analysts
    \item
      Data engineers focus on data ingestion (E + L)
    \item
      Data scientists focus on transform (T)
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\linewidth}

\end{columns}

\begin{itemize}
\tightlist
\item
  ETL \(\to\) ELT \textbf{\textcolor{cyan}{enabled by new technologies}}

  \begin{itemize}
  \tightlist
  \item
    Large storage to save all the raw data (cloud computing)
  \item
    Distributed data storage and querying (e.g., HDFS)
  \item
    Columnar DBs
  \item
    Data compression
  \end{itemize}

\begin{frame}{EtLT}
\protect\phantomsection\label{etlt}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{ETL}}

  \begin{itemize}
  \tightlist
  \item
    Extract \(\to\) Transform \(\to\) Load
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{ELT}}

  \begin{itemize}
  \tightlist
  \item
    Extract \(\to\) Load \(\to\) Transform
  \item
    Transformation / data modeling (``T'') according to business logic
  \end{itemize}
\item
  \textbf{\textcolor{blue}{EtLT}}

  \begin{itemize}
  \tightlist
  \item
    Sometimes transformations with limited scope (``t'') are needed

    \begin{itemize}
    \tightlist
    \item
      De-duplicate records
    \item
      Parse URLs into individual components
    \item
      Obfuscate sensitive data (for legal or security reasons)
    \end{itemize}
  \item
    Then implement rest of ``LT'' pipeline
  \end{itemize}
\end{itemize}
\end{frame}
