  \textbf{\textcolor{red}{Small datasets (< 1 GB)}}

  \begin{itemize}
  \tightlist
  \item

  \item

  \textbf{\textcolor{darkgreen}{Medium dataset (< 1TB)}}

  \begin{itemize}
  \tightlist
  \item
    Doesn't fit into RAM
  \item
    Fits into local disk

    \begin{itemize}

\begin{frame}{Scaling Up vs Scaling Out}

  \textbf{\textcolor{red}{Scaling up}}

  \begin{itemize}
  \tightlist
  \item
    Replace equipment with larger, faster options

    \begin{itemize}
    \tightlist
    \item
      E.g., buy a larger pot, replace knife with food processor
    \end{itemize}
  \item
    \textbf{\textcolor{orange}{Pros}}

    \end{itemize}
  \item
    \textbf{\textcolor{darkgreen}{Cons}}

    \begin{itemize}

      Exceed current machine capacity eventually
    \item

    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{frame}[fragile]{Dask: Computation}
\protect\phantomsection\label{dask-computation}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Lazy computations}}

  \begin{itemize}
  \tightlist
  \item
    Define transformations on data
  \item
    Define next computation without waiting
  \item
    Operate in chunks to avoid loading entire data in memory
  \item
    E.g.,

    \begin{itemize}
    \tightlist
    \item
      Split 2GB file into 32 64MB chunks
    \item
      Operate on 8 chunks per server
    \item
      Max memory use: 512MB = (8 x 64MB)
    \end{itemize}
  \item
    Track object dimensions and data types

    \begin{itemize}
    \tightlist
    \item
      No code execution
    \end{itemize}
  \end{itemize}
\item
  \textcolor{blue}{\texttt{compute()}}

  \begin{itemize}
  \item
    Run computation (materialize)

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{missing\_count\_pct }\OperatorTok{=}\NormalTok{ missing\_count.compute()}
\end{Highlighting}
\end{Shaded}
  \end{itemize}
\item
  \textcolor{blue}{\texttt{persist()}}

  \begin{itemize}
  \tightlist
  \item
    Discard intermediate work to minimize memory
  \item
    Re-run graph for additional computation on intermediate nodes
  \item
    Keep intermediate result in memory
  \item
    Speed up large, complex DAGs for reuse
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Task Scheduling}
\protect\phantomsection\label{task-scheduling-1}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Dask task scheduler orchestrates work dynamically}}

  \begin{itemize}
  \tightlist
  \item
    Not static scheduling like a relational DB
  \item
    During computation, Dask dynamically assesses:

    \begin{itemize}
    \tightlist
    \item
      Completed tasks
    \item
      Remaining tasks
    \item
      Free resources (CPUs)
    \item
      Data location
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Dynamic approach handles various issues}}

  \begin{itemize}
  \tightlist
  \item
    Worker failure

    \begin{itemize}
    \tightlist
    \item
      Re-run tasks
    \end{itemize}
  \item
    Workers completing at different speeds due to:

    \begin{itemize}
    \tightlist
    \item
      Different computation
    \item
      Different hardware
    \item
      Varying server workloads
    \item
      Slower data access
    \end{itemize}
  \item
    Network unreliability

    \begin{itemize}
    \tightlist
    \item
      Re-run or remove isolated nodes
    \end{itemize}
  \end{itemize}
\end{itemize}

\begin{frame}{Dask vs Spark}
\protect\phantomsection\label{dask-vs-spark}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Popular framework for large datasets
  \item
    In-memory alternative to MapReduce/Hadoop
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Java library, supports Python via PySpark API

    \begin{itemize}
    \tightlist
    \item
      Python code runs on JVM
    \item
      Debugging is difficult as execution is outside Python
    \end{itemize}
  \item
    Different DataFrame API than Pandas

    \begin{itemize}
    \tightlist
    \item
      Learn ``the Spark way''
    \item
      May need to implement twice for exploratory analysis and
      production
    \end{itemize}
  \item
    Optimized for MapReduce operations
  \item
    Difficult to set up and configure
  \end{itemize}
\end{itemize}
\end{frame}
