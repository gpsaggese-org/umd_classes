  \end{itemize}

  \begin{itemize}
  \tightlist
  \item
    More data is digitally available
  \item

  \item
    Cloud computing (e.g., AWS) reduces costs
  \item
    Large-scale data + simple algorithms often outperform small data +
    complex algorithms
  \end{itemize}
\end{itemize}

\begin{frame}{Key Shifts Before/After Big-Data}
\protect\phantomsection\label{key-shifts-beforeafter-big-data}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Datasets: small, curated, clean}} \(\to\)
  \textbf{\textcolor{darkgreen}{large, uncurated, messy}}

  \begin{itemize}
  \tightlist
  \item
    Before:

    \begin{itemize}
    \tightlist
    \item
      Statistics based on small, carefully collected random samples
    \item
      Costly and careful planning for experiments
    \item
      Hard to do fine-grained analysis
    \end{itemize}
  \item
    Today:

    \begin{itemize}
    \tightlist
    \item
      Easily collect huge data volumes
    \item
      Feed into algorithms
    \item
      Strong signal overcomes noise
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Causation $\to$ Correlation}}

  \begin{itemize}
  \tightlist
  \item
    Goal: determine cause and effect
  \item
    Causation hard to determine \(\to\) focus on correlation

    \begin{itemize}
    \tightlist
    \item
      Correlation is sometimes sufficient
    \item
      E.g., diapers and beer bought together
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{violet}{"Data-fication"}}

  \begin{itemize}
  \tightlist
  \item
    = converting abstract concepts into data
  \item
    E.g., ``sitting posture'' data-fied by sensors in your seat
  \item
    Preferences data-fied into likes
  \end{itemize}
\end{itemize}
\end{frame}
