\begin{frame}{Relational Model: Overview}
\protect\phantomsection\label{relational-model-overview}
\begin{itemize}
\item
  Introduced by Ted Codd (late 60's, early 70's)
\item
  \textbf{\textcolor{red}{First prototypes}}

  \begin{itemize}
  \tightlist
  \item
    Ingres Project at Berkeley (1970-1985)

    \begin{itemize}
    \tightlist
    \item
      Ingres (INteractive Graphics REtrieval System)
    \item
      \(\to\) PostgreSQL (=Post Ingres)
    \end{itemize}
  \item
    IBM System R (1970s) \(\to\) Oracle, IBM DB2
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Contributions from relational data model}}

  \begin{itemize}
  \tightlist
  \item
    Formal semantics for data operations
  \item
    Data independence: separation of logical and physical data models
  \item
    Declarative query languages (e.g., SQL)
  \item
    Query optimization
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Key to commercial success}}
\end{itemize}
\end{frame}

\begin{frame}{Relational Model: Key Definitions}
\protect\phantomsection\label{relational-model-key-definitions}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\item
  Relational DB is a collection of
  \textbf{\textcolor{red}{tables / relations}}

  \begin{itemize}
  \tightlist
  \item
    Unique name and schema for each table
  \item
    E.g., \textcolor{blue}{\texttt{instructor}} and
    \textcolor{blue}{\texttt{course}} relations
  \end{itemize}
\item
  \textbf{\textcolor{orange}{Row / tuple / record}}: Represents a
  relationship among values
\item
  \textbf{\textcolor{darkgreen}{Element}}: Corresponds to a
  \textbf{\textcolor{teal}{column / field / attribute}}

  \begin{itemize}
  \tightlist
  \item
    Atomic elements (e.g., phone number as a single object)
  \item
    \textcolor{blue}{\texttt{NULL}} for unknown or non-existent values
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Schema of a relation}}

  \begin{itemize}
  \tightlist
  \item
    List of attributes and their domains
  \item
    Like type definition in programming languages
  \item
    E.g., domain of \textcolor{blue}{\texttt{salary}} is integers
    \textgreater= 0
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Instance of relation}}

  \begin{itemize}
  \tightlist
  \item
    Specific instantiation with actual values
  \item
    Changes over time
  \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Primary Key}
\protect\phantomsection\label{primary-key}
\begin{columns}[T]
\begin{column}{0.65\linewidth}
\begin{itemize}
\tightlist
\item
  \emph{R} is a set of attributes of a relation \emph{r}

  \begin{itemize}
  \tightlist
  \item
    E.g., \textcolor{blue}{\texttt{ID, name, dept\_name, salary}} are
    attributes of \textcolor{blue}{\texttt{instructor}}
  \end{itemize}
\item
  \emph{K} is a superkey of \emph{R} if values for \emph{K} identify a
  unique tuple of each relation \emph{r(R)}

  \begin{itemize}
  \tightlist
  \item
    E.g., \textcolor{blue}{\texttt{(ID)}} and
    \textcolor{blue}{\texttt{(ID, name)}} are superkeys of
    \textcolor{blue}{\texttt{instructor}}
  \item
    \textcolor{blue}{\texttt{(name)}} is not a superkey of
    \textcolor{blue}{\texttt{instructor}}
  \end{itemize}
\item
  \textbf{\textcolor{red}{Primary key}}: minimal set of attributes that
  uniquely identify each row

  \begin{itemize}
  \tightlist
  \item
    Typically small and immutable
  \item
    Would \textcolor{blue}{\texttt{SSN}} be a primary key? Yes and no
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Primary key constraint}}: rows can't have the
  same primary key
\end{itemize}
\end{column}

\begin{frame}{Foreign Key}
\protect\phantomsection\label{foreign-key}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Foreign key}} = primary key of another
  relation

  \begin{itemize}
  \tightlist
  \item
    E.g., \textcolor{blue}{\texttt{(ID)}} from
    \textcolor{blue}{\texttt{student}} in
    \textcolor{blue}{\texttt{takes}}, \textcolor{blue}{\texttt{advisor}}
  \item
    \textcolor{blue}{\texttt{takes}} is the ``referencing relation'',
    has the foreign key
  \item
    \textcolor{blue}{\texttt{student}} is the ``referenced relation'',
    has the primary key
  \item
    Shown by an arrow from referencing \(\to\) referenced
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_4_1/lec_4_1_slide_10_image_1.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{blue}{Foreign key constraint}}: for each row, the
  primary key tuple must exist

  \begin{itemize}
  \tightlist
  \item
    Aka referential integrity constraint
  \item
    If \textcolor{blue}{\texttt{(student101, DATA605)}} is in
    \textcolor{blue}{\texttt{takes}}, there must be
    \textcolor{blue}{\texttt{student101}} in
    \textcolor{blue}{\texttt{student}}
  \end{itemize}
\item
  The key referenced as a foreign key must exist as a primary key
\end{itemize}
\end{frame}

\begin{frame}{Relational Algebra: 1/4}
\protect\phantomsection\label{relational-algebra-14}
\begin{columns}[T]
\begin{column}{0.7\linewidth}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Relation}}: set of tuples
\item
  \textbf{\textcolor{orange}{Relational algebra}}: operations on
  relations producing a new relation

  \begin{itemize}
  \tightlist
  \item
    Unary: selection, projection, rename
  \item
    Binary: union, set difference, intersection, Cartesian product, join
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Selection $\Sigma$}}: select tuples
  satisfying a predicate

  \begin{itemize}
  \tightlist
  \item
    E.g., select \textcolor{blue}{\texttt{instructor}} tuples where
    \textcolor{blue}{\texttt{dept\_name = "Physics"}}
  \end{itemize}
\item
  \textbf{\textcolor{teal}{Projection $\pi$}}: return tuples with subset
  of attributes

  \begin{itemize}
  \tightlist
  \item
    E.g., project \textcolor{blue}{\texttt{instructor}} tuples with
    \textcolor{blue}{\texttt{(name, salary)}}
  \end{itemize}
\item
  \textbf{\textcolor{cyan}{Set operations}}: union, intersection, set
  difference

  \begin{itemize}
  \tightlist
  \item
    Must be compatible (same attributes)
  \end{itemize}
\end{itemize}
\end{column}
