\begin{frame}{Queries vs Analysis Tasks}
\protect\phantomsection\label{queries-vs-analysis-tasks}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Queries}}

  \begin{itemize}
  \tightlist
  \item
    Explore data
  \item
    Result is small graph portion (often a node)
  \item
    Challenges

    \begin{itemize}
    \tightlist
    \item
      Minimize explored graph portion
    \item
      Use indexes (auxiliary data structures)
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Analysis tasks}}

  \begin{itemize}
  \tightlist
  \item
    Process entire graph
  \item
    Challenges

    \begin{itemize}
    \tightlist
    \item
      Handle large data efficiently
    \item
      Parallelize if data doesn't fit in memory/disk
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Graph Algorithms}
\protect\phantomsection\label{graph-algorithms}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Graph algorithms}} are general algorithms
  applicable to various graph types

  \begin{itemize}
  \tightlist
  \item
    Network flows (e.g., max flow, min cut)
  \item
    Spanning trees (e.g., minimal connectivity)
  \item
    \ldots{}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Logistics and supply chain optimization
  \item
    Electric grid design
  \item
    Telecommunications (e.g., bandwidth allocation)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Subgraph Matching}
\protect\phantomsection\label{subgraph-matching}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Subgraph Matching}} = find matching instances
  of a small pattern in a larger graph

  \begin{itemize}
  \tightlist
  \item
    Patterns are typically small and fixed
  \item
    Matching can be \emph{approximate}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Fraud detection (e.g., detecting fraudulent transaction patterns)
  \item
    Bioinformatics (e.g., finding protein interaction motifs)
  \item
    Social network analysis (e.g., community pattern detection)
  \end{itemize}
\end{itemize}

\begin{frame}{Shortest Path Queries}
\protect\phantomsection\label{shortest-path-queries}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Shortest Path Queries}} = find the shortest or
  optimal path between two nodes

  \begin{itemize}
  \tightlist
  \item
    May consider edge weights (e.g., distance, cost)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    GPS navigation (e.g., Google Maps)
  \item
    Network routing (e.g., Internet traffic routing)
  \item
    Robotics (e.g., path planning)
  \end{itemize}
\end{itemize}

\begin{frame}{Reachability}
\protect\phantomsection\label{reachability}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Reachability}} = determine if a path exists
  between two nodes

  \begin{itemize}
  \tightlist
  \item
    May include constraints (e.g., edge types, direction)
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Applications}}

  \begin{itemize}
  \tightlist
  \item
    Access control (e.g., ``can user X reach resource Y?'')
  \item
    Dependency analysis (e.g., package installation)
  \item
    Workflow engines (e.g., task progression)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Bulk Synchronous Parallel Model}
\protect\phantomsection\label{bulk-synchronous-parallel-model}
\begin{itemize}
\tightlist
\item
  BSP is a model for parallel computation using synchronized steps
\end{itemize}

\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Computation is organized into supersteps}}

  \begin{itemize}
  \tightlist
  \item
    \emph{Local computation}
  \item
    \emph{Message passing}

    \begin{itemize}
    \tightlist
    \item
      Parallel processors to exchange data
    \item
      Messages are sent during a superstep but delivered in the next
    \end{itemize}
  \item
    \emph{Synchronization barriers}

    \begin{itemize}
    \tightlist
    \item
      Ensure all processors complete a superstep before moving on
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.35\linewidth}
\pandocbounded{\includegraphics[keepaspectratio]{data605/lectures_source/images/lecture_12_2/lec_12_2_slide_40_image_4.png}}
\end{column}
\end{columns}

\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{darkgreen}{Pros}}

  \begin{itemize}
  \tightlist
  \item
    Deterministic and predictable behavior
  \item
    Simple abstraction for reasoning about parallelism
  \item
    Suitable for graph computations with regular communication patterns
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Cons}}

  \begin{itemize}
  \tightlist
  \item
    Global synchronization can cause idle time (stragglers)
  \item
    Inefficient for highly irregular workloads or dynamic computation
    patterns
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Pregel}
\protect\phantomsection\label{pregel}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Pregel}} is a graph processing framework
  inspired by the BSP model

  \begin{itemize}
  \tightlist
  \item
    Pregel = Parallel + Graph + Google
  \item
    Designed for large-scale distributed graph computations
  \item
    Uses a vertex-centric programming model

    \begin{itemize}
    \tightlist
    \item
      Each vertex performs computation independently in supersteps
    \item
      Motto: \emph{``Think like a vertex''}
    \end{itemize}
  \item
    Vertices

    \begin{itemize}
    \tightlist
    \item
      Send messages to neighbors
    \item
      Update state based on received messages
    \item
      Can vote to halt; computation ends when all are inactive
    \end{itemize}
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{PageRank example}}

  \begin{itemize}
  \tightlist
  \item
    Each vertex divides its rank among neighbors
  \item
    In each superstep, vertices update rank from incoming messages
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Connected components example}}

  \begin{itemize}
  \tightlist
  \item
    Each vertex propagates smallest ID seen to neighbors
  \item
    Repeats until no changes occur
  \end{itemize}
\end{itemize}
\end{frame}
