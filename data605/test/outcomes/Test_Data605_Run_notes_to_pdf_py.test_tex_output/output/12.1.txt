\begin{frame}{Storing Graph Data}
\protect\phantomsection\label{storing-graph-data}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{File systems}}

  \begin{itemize}
  \tightlist
  \item
    Simple
  \item
    No transactions, ACID compliance
  \item
    Minimal functionality (build analysis/querying on top)
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Relational databases}}

  \begin{itemize}
  \tightlist
  \item
    Mature technology
  \item
    SQL, transactions, ACID compliance, toolchains
  \item
    Minimal functionality for graph data
  \end{itemize}
\item
  \textbf{\textcolor{blue}{NoSQL key-value stores}}

  \begin{itemize}
  \tightlist
  \item
    Handle large datasets efficiently in a distributed fashion
  \item
    Minimal native functionality for graph data
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Graph databases}}

  \begin{itemize}
  \tightlist
  \item
    Efficiently support complex queries/tasks (e.g., graph traversals)
  \item
    Less mature than RDBMSs
  \item
    Often lack declarative language like SQL

    \begin{itemize}
    \tightlist
    \item
      Write custom programs
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Graph Databases}
\protect\phantomsection\label{graph-databases}
\begin{itemize}
\tightlist
\item
  Many specialized graph DB systems

  \begin{itemize}
  \tightlist
  \item
    E.g., Neo4j, Titan, OrientDB, AllegroGraph
  \end{itemize}
\item
  Key distinctions from relational / NoSQL databases

  \begin{itemize}
  \tightlist
  \item
    Store graph structure with pointers

    \begin{itemize}
    \tightlist
    \item
      Avoid joins
    \item
      Simplify graph traversals
    \end{itemize}
  \item
    Manage and query graph-structured data

    \begin{itemize}
    \tightlist
    \item
      Write queries and graph algorithms (reachability, shortest paths)
    \end{itemize}
  \item
    Support graph query languages: SPARQL, Cypher, Gremlin
  \item
    Declarative interfaces
  \item
    Provide programmatic API for arbitrary graph algorithms
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Query Languages: Cypher}
\protect\phantomsection\label{query-languages-cypher}
\begin{itemize}
\item
  \textbf{\textcolor{red}{Purpose-Built for Property Graphs}}

  \begin{itemize}
  \tightlist
  \item
    Nodes and relationships with key-value properties
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Declarative Syntax}}

  \begin{itemize}
  \tightlist
  \item
    Express \emph{what} to retrieve, not \emph{how}
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Optimized for Pattern Matching}}

  \begin{itemize}
  \tightlist
  \item
    Easily finds subgraph structures (e.g., friends of friends)
  \item
    Limited in Graph Analytics
  \item
    Not ideal for reachability, shortest paths, or centrality
  \end{itemize}
\item
  \textbf{\textcolor{violet}{Neo4j Native}}
\item
  Example:

  \begin{itemize}
  \item
    \emph{Find the names of people who know someone named ``Alice''}
    \small

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{MATCH (person)}\OperatorTok{{-}}\NormalTok{[}\CharTok{:KNOWS}\NormalTok{]}\OperatorTok{{-}\textgreater{}}\NormalTok{(friend \{name: }\OtherTok{"Alice"}\NormalTok{\})}
  \KeywordTok{RETURN}\NormalTok{ person.name}
\end{Highlighting}
\end{Shaded}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Query Languages: Gremlin}
\protect\phantomsection\label{query-languages-gremlin}
\begin{itemize}
\tightlist
\item
  \textbf{\textcolor{red}{Supports Multiple Models}}

  \begin{itemize}
  \tightlist
  \item
    Compatible with both Property Graphs and RDF
  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Imperative Style}}

  \begin{itemize}
  \tightlist
  \item
    Describes \emph{how} to traverse the graph step by step
  \end{itemize}
\item
  \textbf{\textcolor{blue}{Traversal-Based Semantics}}

  \begin{itemize}
  \tightlist
  \item
    Expresses computation as a flow of operations across vertices and
    edges
  \end{itemize}
\item
  Example:

  \begin{itemize}
  \item
    \emph{Find the names of people who know someone named ``Alice''}
    \small

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{g.V().hasLabel(}\StringTok{\textquotesingle{}Person\textquotesingle{}}\NormalTok{)}
\NormalTok{  .where(out(}\StringTok{\textquotesingle{}KNOWS\textquotesingle{}}\NormalTok{).has(}\StringTok{\textquotesingle{}name\textquotesingle{}}\NormalTok{, }\StringTok{\textquotesingle{}Alice\textquotesingle{}}\NormalTok{))}
\NormalTok{  .values(}\StringTok{\textquotesingle{}name\textquotesingle{}}\NormalTok{)}
\end{Highlighting}
\end{Shaded}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Query Languages: SPARQL}

  \textbf{\textcolor{red}{SQL-Like Syntax}}

    \textcolor{blue}{\texttt{WHERE}}, \textcolor{blue}{\texttt{FILTER}},

  \end{itemize}
\item
  \textbf{\textcolor{darkgreen}{Built for RDF Data}}

  \begin{itemize}
  \tightlist
  \item
    Queries triples: subject--predicate--object
  \end{itemize}
\item
  \textbf{\textcolor{blue}{W3C Standard}}

  \begin{itemize}
  \tightlist
  \item
    Core to Semantic Web and Linked Data applications
  \end{itemize}

  Example:

  \begin{itemize}
  \item
    \emph{Find the names of people who know someone named ``Alice''}

\NormalTok{\}}
\end{Highlighting}
